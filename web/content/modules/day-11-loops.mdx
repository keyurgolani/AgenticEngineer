## 4.2 Building a Minimal Agent

Let's build a complete agent in ~100 lines of Python:

````python
import json
from openai import OpenAI

class SimpleAgent:
    """A minimal ReAct agent implementation."""

    def __init__(self, model: str = "gpt-4o-mini"):
        self.client = OpenAI()
        self.model = model
        self.tools = {}
        self.max_iterations = 10

    def register_tool(self, name: str, func, description: str):
        """Register a tool the agent can use."""
        self.tools[name] = {
            "function": func,
            "description": description
        }

    def _build_system_prompt(self) -> str:
        """Build the system prompt with available tools."""
        tool_docs = "\n".join([
            f"- {name}: {info['description']}"
            for name, info in self.tools.items()
        ])

        return f"""You are a helpful assistant with access to tools.

Available Tools:
{tool_docs}

Respond in this JSON format:
{{
  "thought": "Your reasoning about what to do",
  "action": "tool_name OR final_answer",
  "action_input": {{"param": "value"}} OR "your final response"
}}

Always think before acting. Use tools to gather information.
When you have enough information, use action: "final_answer"."""

    def _parse_response(self, content: str) -> dict:
        """Extract JSON from LLM response."""
        if "```json" in content:
            content = content.split("```json")[1].split("```")[0]
        elif "```" in content:
            content = content.split("```")[1].split("```")[0]
        return json.loads(content.strip())

    def _execute_tool(self, name: str, input_data) -> str:
        """Execute a registered tool."""
        if name not in self.tools:
            return f"Error: Unknown tool '{name}'"
        try:
            if isinstance(input_data, dict):
                result = self.tools[name]["function"](**input_data)
            else:
                result = self.tools[name]["function"](input_data)
            return str(result)
        except Exception as e:
            return f"Error executing {name}: {str(e)}"

    def run(self, user_input: str) -> str:
        """Run the agent loop."""
        messages = [
            {"role": "system", "content": self._build_system_prompt()},
            {"role": "user", "content": user_input}
        ]

        for iteration in range(self.max_iterations):
            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                temperature=0.1
            )

            content = response.choices[0].message.content
            messages.append({"role": "assistant", "content": content})

            try:
                parsed = self._parse_response(content)
            except json.JSONDecodeError:
                messages.append({
                    "role": "user",
                    "content": "Please respond with valid JSON."
                })
                continue

            thought = parsed.get("thought", "")
            action = parsed.get("action", "")
            action_input = parsed.get("action_input", {})

            print(f"💭 Thought: {thought}")
            print(f"🎯 Action: {action}")

            if action == "final_answer":
                return action_input if isinstance(action_input, str) else json.dumps(action_input)

            observation = self._execute_tool(action, action_input)
            print(f"👁️ Observation: {observation[:200]}...")

            messages.append({
                "role": "user",
                "content": f"Observation: {observation}"
            })

        return "Max iterations reached. Please try a simpler request."
````

<Callout type="tip">
  **Connection to Day 03:** Notice how this manual loop mirrors what LangGraph
  does automatically. The `messages` list is our state, the `for` loop is our
  graph execution, and the `if action == "final_answer"` is our conditional edge
  to END.
</Callout>

## 4.3 Using Structured Outputs (Production Pattern)

The JSON parsing in our simple agent is fragile. Production agents use **structured outputs** to guarantee valid responses. Here's the modern approach using Pydantic:

<ConceptVisual
  image="/images/day_11_03_fragile_vs_structured_1768989039062.png"
  caption="Fragile Parsing vs. Structured Outputs: Moving from error-prone text matching to type-safe guarantees."
  alt="Comparison between fragile JSON parsing and structured Pydantic outputs"
/>

```python
from pydantic import BaseModel, Field
from typing import Literal, Union
from openai import OpenAI

class ToolCall(BaseModel):
    """Represents a tool the agent wants to call."""
    thought: str = Field(description="Agent's reasoning about what to do")
    action: str = Field(description="Name of the tool to call")
    action_input: dict = Field(description="Arguments for the tool")

class FinalAnswer(BaseModel):
    """Represents the agent's final response."""
    thought: str = Field(description="Agent's reasoning about the answer")
    answer: str = Field(description="The final answer to return")

class AgentResponse(BaseModel):
    """Union type for agent responses."""
    response_type: Literal["tool_call", "final_answer"]
    tool_call: ToolCall | None = None
    final_answer: FinalAnswer | None = None


class StructuredAgent:
    """Agent using structured outputs for reliable parsing."""

    def __init__(self, model: str = "gpt-4o-mini"):
        self.client = OpenAI()
        self.model = model
        self.tools = {}

    def _get_structured_response(self, messages: list) -> AgentResponse:
        """Get a structured response from the LLM."""
        response = self.client.beta.chat.completions.parse(
            model=self.model,
            messages=messages,
            response_format=AgentResponse,
        )
        return response.choices[0].message.parsed

    def run(self, user_input: str) -> str:
        """Run agent with guaranteed structured outputs."""
        messages = [
            {"role": "system", "content": self._build_system_prompt()},
            {"role": "user", "content": user_input}
        ]

        for _ in range(self.max_iterations):
            result = self._get_structured_response(messages)

            print(f"💭 {result.tool_call.thought if result.tool_call else result.final_answer.thought}")

            if result.response_type == "final_answer":
                return result.final_answer.answer

            # Execute tool call
            tool_result = self._execute_tool(
                result.tool_call.action,
                result.tool_call.action_input
            )
            messages.append({
                "role": "assistant",
                "content": f"Calling {result.tool_call.action}"
            })
            messages.append({
                "role": "user",
                "content": f"Observation: {tool_result}"
            })

        return "Max iterations reached."
```

<Callout type="success">
  **Why Structured Outputs Matter:** - **No parsing errors**: The SDK guarantees
  valid JSON matching your schema - **Type safety**: Pydantic validates all
  fields automatically - **Better prompting**: The schema itself guides the
  LLM's output format - **Easier debugging**: Clear separation between tool
  calls and final answers
</Callout>

<Callout type="info">
  **Local Model Alternative:** For Ollama, you can use the `instructor` library
  to get structured outputs from local models: ```python import instructor from
  openai import OpenAI client = instructor.from_openai(
  OpenAI(base_url="http://localhost:11434/v1", api_key="ollama") ) ```
</Callout>

import os
import subprocess

def read_file(path: str) -> str:
    """Read contents of a file."""
    try:
        with open(path, 'r') as f:
            return f.read()
    except FileNotFoundError:
        return f"File not found: {path}"
    except PermissionError:
        return f"Permission denied: {path}"

def list_directory(path: str = ".") -> str:
    """List files in a directory."""
    try:
        files = os.listdir(path)
        return "\n".join(files)

        return f"Error: {str(e)}"

def run_python(code: str) -> str:
    """Execute Python code and return output."""
    try:
        result = subprocess.run(
            ["python", "-c", code],
            capture_output=True,
            text=True,
            timeout=30
        )
        return result.stdout or result.stderr
    except subprocess.TimeoutExpired:
        return "Error: Code execution timed out"

## 4.6 Adding Memory

For multi-turn conversations, persist state across runs:

```python
class AgentWithMemory(SimpleAgent):
    """Agent with working memory."""

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.conversation_history = []
        self.working_memory = {}

    def remember(self, key: str, value: str):
        """Store a fact in working memory."""
        self.working_memory[key] = value

    def recall(self, key: str) -> str:
        """Retrieve a fact from working memory."""
        return self.working_memory.get(key, "Not found")

    def _build_system_prompt(self) -> str:
        """Include memory context in the system prompt."""
        base_prompt = super()._build_system_prompt()
        if self.working_memory:
            memory_str = "\n".join([
                f"- {k}: {v}" for k, v in self.working_memory.items()
            ])
            base_prompt += f"\n\nWorking Memory:\n{memory_str}"
        return base_prompt

    def run(self, user_input: str) -> str:
        """Run with conversation history tracking."""
        self.conversation_history.append({"role": "user", "content": user_input})
        result = super().run(user_input)
        self.conversation_history.append({"role": "assistant", "content": result})
        return result
```

## 4.7 Error Handling and Retries

Production agents need robust error handling:

<ConceptVisual
  image="/images/day_11_05_retry_mechanism_animated_1768989398374.png"
  caption="Exponential Backoff: A robust retry strategy for handling transient errors in production."
  alt="Visualization of exponential backoff retry mechanism"
/>

```python
import time
from typing import Optional

class RobustAgent(AgentWithMemory):
    """Agent with retry logic."""

    def __init__(self, *args, max_retries: int = 3, **kwargs):
        super().__init__(*args, **kwargs)
        self.max_retries = max_retries

    def _execute_tool_with_retry(self, name: str, input_data) -> str:
        """Execute tool with exponential backoff."""
        last_error = None
        for attempt in range(self.max_retries):
            try:
                result = self._execute_tool(name, input_data)
                if not result.startswith("Error"):
                    return result
                last_error = result
            except Exception as e:
                last_error = str(e)

            if attempt < self.max_retries - 1:
                wait_time = 2 ** attempt
                print(f"⚠️ Retry {attempt + 1}/{self.max_retries} in {wait_time}s...")
                time.sleep(wait_time)

        return f"Failed after {self.max_retries} attempts: {last_error}"
```

## 4.8 Streaming Responses

For better UX, stream the agent's thinking:

def run_streaming(self, user_input: str):
    """Run agent with streaming output."""
    messages = [
        {"role": "system", "content": self._build_system_prompt()},
        {"role": "user", "content": user_input}
    ]

    for iteration in range(self.max_iterations):
        print(f"\n--- Iteration {iteration + 1} ---")

        stream = self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            temperature=0.1,
            stream=True
        )

        content = ""
        print("🤖 ", end="", flush=True)

        for chunk in stream:
            if chunk.choices[0].delta.content:
                token = chunk.choices[0].delta.content
                content += token
                print(token, end="", flush=True)

        print()
        messages.append({"role": "assistant", "content": content})

        try:
            parsed = self._parse_response(content)
            if parsed.get("action") == "final_answer":
                return parsed.get("action_input")

            observation = self._execute_tool(
                parsed.get("action"),
                parsed.get("action_input")
            )
            print(f"👁️ {observation[:100]}...")
            messages.append({"role": "user", "content": f"Observation: {observation}"})
        except json.JSONDecodeError:
            messages.append({"role": "user", "content": "Please respond with valid JSON."})

## 4.10 Using Native Function Calling

Modern LLMs support **native function calling**, which is more reliable than JSON parsing. Here's how to use OpenAI's tool calling API:

<ConceptVisual
  image="/images/day_11_06_manual_parsing_vs_native_1768989535644.png"
  caption="Native Function Calling: A cleaner, more reliable pipeline for tool execution."
  alt="Comparison of manual JSON parsing vs native function calling"
/>

```python
from openai import OpenAI

class FunctionCallingAgent:
    """Agent using native function calling."""

    def __init__(self, model: str = "gpt-4o-mini"):
        self.client = OpenAI()
        self.model = model
        self.tools = []
        self.tool_functions = {}

    def register_tool(self, name: str, func, description: str, parameters: dict):
        """Register a tool with its JSON schema."""
        self.tools.append({
            "type": "function",
            "function": {
                "name": name,
                "description": description,
                "parameters": parameters
            }
        })
        self.tool_functions[name] = func

    def run(self, user_input: str) -> str:
        """Run agent with native function calling."""
        messages = [{"role": "user", "content": user_input}]

        for _ in range(10):
            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                tools=self.tools if self.tools else None,
                tool_choice="auto"
            )

            message = response.choices[0].message

            # Check if model wants to call tools
            if message.tool_calls:
                messages.append(message)

                for tool_call in message.tool_calls:
                    func = self.tool_functions[tool_call.function.name]
                    args = json.loads(tool_call.function.arguments)
                    result = func(**args)

                    messages.append({
                        "role": "tool",
                        "tool_call_id": tool_call.id,
                        "content": str(result)
                    })
            else:
                # No tool calls = final answer
                return message.content

        return "Max iterations reached."


# Usage with proper JSON schema
agent = FunctionCallingAgent()
agent.register_tool(
    name="read_file",
    func=read_file,
    description="Read the contents of a file",
    parameters={
        "type": "object",
        "properties": {
            "path": {"type": "string", "description": "Path to the file"}
        },
        "required": ["path"]
    }
)
```

<Callout type="success">
  **Native Function Calling Advantages:** - **Higher accuracy**: The model is
  trained specifically for tool use - **Parallel calls**: Can request multiple
  tools in one response - **Structured arguments**: JSON schema validation
  built-in - **Works with Ollama**: Ollama v0.8+ supports the same API
</Callout>

## 4.11 Parallel Tool Calling

Modern LLMs can request multiple tools in a single response, dramatically improving efficiency for independent operations:

<ConceptVisual
  image="/images/day_11_07_sequential_vs_parallel_perf_1768989768403.png"
  caption="Parallel Execution: Reducing latency by running independent operations concurrently."
  alt="Performance comparison of sequential vs parallel tool execution"
/>

```python
class ParallelToolAgent:
    """Agent that handles parallel tool calls."""

    def run(self, user_input: str) -> str:
        messages = [{"role": "user", "content": user_input}]

        for _ in range(10):
            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                tools=self.tools,
                tool_choice="auto",
                parallel_tool_calls=True  # Enable parallel calls
            )

            message = response.choices[0].message

            if message.tool_calls:
                messages.append(message)

                # Execute all tool calls (can be parallelized with asyncio)
                for tool_call in message.tool_calls:
                    func = self.tool_functions[tool_call.function.name]
                    args = json.loads(tool_call.function.arguments)
                    result = func(**args)

                    messages.append({
                        "role": "tool",
                        "tool_call_id": tool_call.id,
                        "content": str(result)
                    })
            else:
                return message.content

        return "Max iterations reached."
```

<Callout type="success">
  **Parallel Execution Benefits:** When the model requests multiple independent
  tools (e.g., fetching weather for 3 cities), you can execute them concurrently
  using `asyncio.gather()` to reduce total latency.
</Callout>

## 4.12 Thinking Models and Extended Reasoning

Some models like **DeepSeek-R1** and **Qwen 3** support explicit "thinking" modes where they show their reasoning process:

<ConceptVisual
  image="/images/day_11_08_thinking_models_visualized_1768989880297.png"
  caption="Thinking Models: Showing the hidden reasoning process (Chain of Thought) transparently."
  alt="Visualization of standard vs thinking models with reasoning chains"

```python
def run_with_thinking(self, user_input: str) -> dict:
    """Run agent with thinking model that shows reasoning."""
    response = self.client.chat.completions.create(
        model="deepseek-r1",  # or "qwen3:32b"
        messages=[
            {"role": "system", "content": "Think step by step before answering."},
            {"role": "user", "content": user_input}
        ],
        temperature=0.6,  # Slightly higher for reasoning diversity
    )

    content = response.choices[0].message.content

    # DeepSeek-R1 wraps thinking in <think> tags
    if "<think>" in content:
        thinking = content.split("<think>")[1].split("</think>")[0]
        answer = content.split("</think>")[-1].strip()
        return {"thinking": thinking, "answer": answer}

    return {"thinking": None, "answer": content}
```

<Callout type="info">
  **When to Use Thinking Models:** - Complex multi-step reasoning tasks - Math
  and logic problems - Code debugging and architecture decisions - When you need
  to audit the model's reasoning process **Trade-off:** Thinking models use more
  tokens but often produce more accurate results for complex tasks.
</Callout>

### Self-Correction Patterns

Effective agentic systems separate concerns into distinct phases:

<ConceptVisual
  image="/images/day_11_11_self_correction_cycle_1768990754829.png"
  caption="Self-Correction: The Plan-Execute-Verify-Revise cycle ensures reliability."
  alt="Circular diagram showing the self-correction workflow"
/>

1. **Planning** — Decompose the task into steps
2. **Execution** — Run each step with appropriate tools
3. **Verification** — Check if the output meets requirements
4. **Revision** — If verification fails, retry with feedback

```python
def run_with_self_correction(self, user_input: str, max_corrections: int = 2) -> str:
    """Agent loop with built-in self-correction."""
    result = self.run(user_input)

    for attempt in range(max_corrections):
        # Verify the result
        verification = self._verify_result(result, user_input)

        if verification["is_valid"]:
            return result

        # Revise based on feedback
        revision_prompt = f"""
        Original task: {user_input}
        Previous result: {result}
        Issue found: {verification["feedback"]}

        Please fix the issue and provide a corrected response.
        """
        result = self.run(revision_prompt)

    return result  # Return best effort after max corrections
```

### When to Use Agent Loops vs. Simpler Approaches

| Use Case                            | Recommended Approach      |
| :---------------------------------- | :------------------------ |
| Single-step tasks                   | Direct LLM call           |
| Fixed multi-step workflows          | Prompt chains             |
| Dynamic tool selection              | ReAct agent loop          |
| Complex reasoning with verification | Self-correcting agent     |
| Multi-domain tasks                  | Multi-agent orchestration |
