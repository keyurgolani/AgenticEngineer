```python
# services/voice_agent.py
from livekit.agents import AutoSubscribe, JobContext, WorkerOptions, cli
from livekit.agents.voice_assistant import VoiceAssistant
from livekit.plugins import deepgram, openai, silero
from services.intent_parser import IntentParser
from services.task_service import TaskService

class TaskmasterVoiceAgent:
    """Voice agent that processes task commands."""

    def __init__(self, task_service: TaskService):
        self.task_service = task_service
        self.intent_parser = IntentParser()

    async def process_transcript(self, transcript: str, user_id: str) -> str:
        """Process transcribed speech and return response."""
        # Parse intent from transcript
        intent = await self.intent_parser.parse(transcript)

        if intent is None:
            return "I didn't understand that. Try saying 'Add task' followed by a description."

        # Execute the appropriate action
        result = await self._execute_intent(intent, user_id)
        return result

    async def _execute_intent(self, intent: dict, user_id: str) -> str:
        """Execute parsed intent and return confirmation message."""
        action = intent.get("action")

        match action:
            case "add":
                task = await self.task_service.create_task(
                    user_id=user_id,
                    title=intent["title"],
                    due_date=intent.get("due_date"),
                    priority=intent.get("priority", "medium")
                )
                return f"Added task: {task.title}"

            case "complete":
                task = await self.task_service.complete_task(
                    user_id=user_id,
                    task_identifier=intent["task_identifier"]
                )
                return f"Completed: {task.title}"

            case "delete":
                task = await self.task_service.delete_task(
                    user_id=user_id,
                    task_identifier=intent["task_identifier"]
                )
                return f"Deleted: {task.title}"

            case "undo":
                result = await self.task_service.undo(user_id)
                return result or "Nothing to undo"

            case "redo":
                result = await self.task_service.redo(user_id)
                return result or "Nothing to redo"

            case "list":
                tasks = await self.task_service.list_tasks(
                    user_id=user_id,
                    filter_priority=intent.get("priority"),
                    filter_due=intent.get("due_filter")
                )
                if not tasks:
                    return "You have no tasks"
                task_list = ", ".join([t.title for t in tasks[:5]])
                return f"Your tasks: {task_list}"

            case _:
                return "I'm not sure how to help with that"


async def entrypoint(ctx: JobContext):
    """LiveKit agent entrypoint."""
    await ctx.connect(auto_subscribe=AutoSubscribe.AUDIO_ONLY)

    participant = await ctx.wait_for_participant()
    user_id = participant.identity

    # Initialize services
    task_service = TaskService()
    agent = TaskmasterVoiceAgent(task_service)

    # Create voice assistant with custom processing
    assistant = VoiceAssistant(
        vad=silero.VAD.load(),
        stt=deepgram.STT(model="nova-2"),
        llm=None,  # We handle LLM ourselves
        tts=openai.TTS(voice="nova"),
        will_synthesize_assistant_reply=False,
    )

    @assistant.on("user_speech_committed")
    async def on_speech(transcript: str):
        response = await agent.process_transcript(transcript, user_id)
        await assistant.say(response)

    assistant.start(ctx.room, participant)
    await assistant.say("Voice Taskmaster ready. What would you like to do?")


if __name__ == "__main__":
    cli.run_app(WorkerOptions(entrypoint_fnc=entrypoint))
```

### Step 3: Integrate Speech Recognition (45 minutes)

For applications not using LiveKit, you can integrate Deepgram directly for streaming STT.

```python
# services/stt_service.py
from deepgram import DeepgramClient, LiveTranscriptionEvents, LiveOptions
import asyncio
from typing import Callable, Optional

class StreamingSTT:
    """Real-time speech-to-text using Deepgram."""

    def __init__(self, api_key: str):
        self.client = DeepgramClient(api_key)
        self.connection = None
        self.on_transcript: Optional[Callable[[str, bool], None]] = None

    async def start(self, on_transcript: Callable[[str, bool], None]):
        """Start streaming transcription."""
        self.on_transcript = on_transcript
        self.connection = self.client.listen.live.v("1")

        def handle_transcript(self, result, **kwargs):
            transcript = result.channel.alternatives[0].transcript
            is_final = result.is_final
            if transcript and self.on_transcript:
                asyncio.create_task(
                    self._notify_transcript(transcript, is_final)
                )

        self.connection.on(LiveTranscriptionEvents.Transcript, handle_transcript)

        options = LiveOptions(
            model="nova-2",
            language="en-US",
            smart_format=True,
            interim_results=True,  # Get partial results for responsiveness
            utterance_end_ms=1000,  # Detect end of utterance
            vad_events=True,  # Voice activity detection
        )

        await self.connection.start(options)

    async def _notify_transcript(self, transcript: str, is_final: bool):
        """Notify callback of new transcript."""
        if self.on_transcript:
            await self.on_transcript(transcript, is_final)

    async def send_audio(self, audio_chunk: bytes):
        """Send audio chunk for transcription."""
        if self.connection:
            await self.connection.send(audio_chunk)

    async def stop(self):
        """Stop transcription."""
        if self.connection:
            await self.connection.finish()
```

<Callout type="tip">
  **Latency Optimization:** Enable `interim_results` to get partial transcripts
  while the user is still speaking. This allows you to start processing before
  the utterance is complete, reducing perceived latency.
</Callout>

#### Alternative: Local Whisper for Privacy

```python
# services/local_stt.py
from faster_whisper import WhisperModel
import numpy as np
from typing import Generator

class LocalWhisperSTT:
    """Local speech-to-text using faster-whisper."""

    def __init__(self, model_size: str = "base"):
        self.model = WhisperModel(
            model_size,
            device="cuda",  # or "cpu"
            compute_type="float16"
        )

    def transcribe(self, audio_array: np.ndarray) -> str:
        """Transcribe audio array to text."""
        segments, info = self.model.transcribe(
            audio_array,
            beam_size=5,
            language="en",
            vad_filter=True,  # Filter out non-speech
        )
        return " ".join([segment.text for segment in segments])

    def transcribe_stream(
        self,
        audio_chunks: Generator[np.ndarray, None, None]
    ) -> Generator[str, None, None]:
        """Stream transcription for real-time use."""
        buffer = np.array([], dtype=np.float32)

        for chunk in audio_chunks:
            buffer = np.concatenate([buffer, chunk])

            # Process when we have enough audio (e.g., 1 second)
            if len(buffer) >= 16000:  # 1 second at 16kHz
                text = self.transcribe(buffer)
                if text.strip():
                    yield text
                buffer = np.array([], dtype=np.float32)
```

The Command Pattern enables reversible operations—essential for voice interfaces where mistakes are common.

```python
# services/action_history.py
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import List, Optional, Any, Dict
from datetime import datetime

class Action(ABC):
    """Base class for undoable actions."""

    @abstractmethod
    async def execute(self) -> Any:
        """Execute the action."""
        pass

    @abstractmethod
    async def undo(self) -> Any:
        """Reverse the action."""
        pass

    @abstractmethod
    def describe(self) -> str:
        """Human-readable description of the action."""
        pass

@dataclass
class CreateTaskAction(Action):
    """Action for creating a task."""
    task_service: Any
    user_id: str
    title: str
    priority: str = "medium"
    due_date: Optional[datetime] = None
    created_task_id: Optional[int] = None

    async def execute(self) -> Any:
        task = await self.task_service._create_task_internal(
            user_id=self.user_id,
            title=self.title,
            priority=self.priority,
            due_date=self.due_date
        )
        self.created_task_id = task.id
        return task

    async def undo(self) -> Any:
        if self.created_task_id:
            await self.task_service._delete_task_internal(self.created_task_id)
        return f"Undid: Add {self.title}"

    def describe(self) -> str:
        return f"Add task: {self.title}"

@dataclass
class CompleteTaskAction(Action):
    """Action for completing a task."""
    task_service: Any
    task_id: int
    task_title: str = ""
    previous_state: bool = False

    async def execute(self) -> Any:
        task = await self.task_service._get_task_by_id(self.task_id)
        self.previous_state = task.completed
        self.task_title = task.title
        await self.task_service._update_task_internal(
            self.task_id,
            completed=True
        )
        return task

    async def undo(self) -> Any:
        await self.task_service._update_task_internal(
            self.task_id,
            completed=self.previous_state
        )
        return f"Undid: Complete {self.task_title}"

    def describe(self) -> str:
        return f"Complete task: {self.task_title}"

@dataclass
class DeleteTaskAction(Action):
    """Action for deleting a task (soft delete for undo support)."""
    task_service: Any
    task_id: int
    deleted_task_data: Optional[Dict] = None

    async def execute(self) -> Any:
        task = await self.task_service._get_task_by_id(self.task_id)
        self.deleted_task_data = {
            "title": task.title,
            "priority": task.priority,
            "due_date": task.due_date,
            "completed": task.completed,
            "user_id": task.user_id,
        }
        await self.task_service._delete_task_internal(self.task_id)
        return task

    async def undo(self) -> Any:
        if self.deleted_task_data:
            task = await self.task_service._create_task_internal(**self.deleted_task_data)
            return f"Restored: {self.deleted_task_data['title']}"
        return "Could not restore task"

    def describe(self) -> str:
        title = self.deleted_task_data.get("title", "Unknown") if self.deleted_task_data else "Unknown"
        return f"Delete task: {title}"


class ActionHistory:
    """Manages undo/redo stack for a user session."""

    def __init__(self, max_size: int = 50):
        self.max_size = max_size
        self._undo_stack: List[Action] = []
        self._redo_stack: List[Action] = []

    async def execute(self, action: Action) -> Any:
        """Execute an action and add to history."""
        result = await action.execute()

        self._undo_stack.append(action)
        self._redo_stack.clear()  # Clear redo on new action

        # Trim if exceeds max size
        if len(self._undo_stack) > self.max_size:
            self._undo_stack.pop(0)

        return result

    async def undo(self) -> Optional[str]:
        """Undo the last action."""
        if not self._undo_stack:
            return None

        action = self._undo_stack.pop()
        result = await action.undo()
        self._redo_stack.append(action)

        return result

    async def redo(self) -> Optional[str]:
        """Redo the last undone action."""
        if not self._redo_stack:
            return None

        action = self._redo_stack.pop()
        await action.execute()
        self._undo_stack.append(action)

        return f"Redid: {action.describe()}"

    def can_undo(self) -> bool:
        return len(self._undo_stack) > 0

    def can_redo(self) -> bool:
        return len(self._redo_stack) > 0
```

```python
# routers/websocket.py
from fastapi import WebSocket, WebSocketDisconnect
from typing import Dict, Set
import json
import asyncio

class ConnectionManager:
    """Manages WebSocket connections for real-time updates."""

    def __init__(self):
        # user_id -> set of WebSocket connections
        self._connections: Dict[str, Set[WebSocket]] = {}
        self._lock = asyncio.Lock()

    async def connect(self, websocket: WebSocket, user_id: str):
        """Accept and register a new connection."""
        await websocket.accept()
        async with self._lock:
            if user_id not in self._connections:
                self._connections[user_id] = set()
            self._connections[user_id].add(websocket)

    async def disconnect(self, websocket: WebSocket, user_id: str):
        """Remove a connection."""
        async with self._lock:
            if user_id in self._connections:
                self._connections[user_id].discard(websocket)
                if not self._connections[user_id]:
                    del self._connections[user_id]

    async def broadcast_to_user(self, user_id: str, message: dict):
        """Send message to all connections for a user."""
        async with self._lock:
            connections = self._connections.get(user_id, set()).copy()

        dead_connections = []
        for connection in connections:
            try:
                await connection.send_json(message)
            except Exception:
                dead_connections.append(connection)

        # Clean up dead connections
        for conn in dead_connections:
            await self.disconnect(conn, user_id)

    async def broadcast_task_event(
        self,
        user_id: str,
        event_type: str,
        task_data: dict
    ):
        """Broadcast a task-related event."""
        await self.broadcast_to_user(user_id, {
            "type": event_type,
            "task": task_data,
            "timestamp": datetime.utcnow().isoformat()
        })


# Global connection manager
manager = ConnectionManager()

from fastapi import APIRouter
router = APIRouter()

@router.websocket("/ws/{user_id}")

            message = json.loads(data)

            # Handle ping/pong for connection health
            if message.get("type") == "ping":
                await websocket.send_json({"type": "pong"})

    except WebSocketDisconnect:
        await manager.disconnect(websocket, user_id)
```

```typescript
class TaskWebSocket {

    private userId: string,

      console.log("Connected to task updates");
      this.reconnectAttempts = 0;
      this.startHeartbeat();
    };

    this.ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      if (data.type !== "pong") {
        this.onTaskEvent(data);
      }
    };

    this.ws.onclose = () => {
      this.attemptReconnect();
    };
  }

  private startHeartbeat() {
    setInterval(() => {
      if (this.ws?.readyState === WebSocket.OPEN) {
        this.ws.send(JSON.stringify({ type: "ping" }));
      }
    }, 30000);
  }

  private attemptReconnect() {
    if (this.reconnectAttempts < this.maxReconnectAttempts) {
      this.reconnectAttempts++;
      setTimeout(() => this.connect(), 1000 * this.reconnectAttempts);
    }
  }
}
```

### Step 8: Natural Language Task Parsing (60 minutes)

The intent parser converts natural speech into structured task operations.

```python
# services/intent_parser.py
from openai import OpenAI
from typing import Optional, Dict, Any
from datetime import datetime, timedelta
from dateutil import parser as date_parser
import json
import re

class IntentParser:
    """Parse natural language into task intents."""

    def __init__(self, api_key: str = None):
        self.client = OpenAI(api_key=api_key) if api_key else OpenAI()
        self.system_prompt = """You are a task intent parser. Extract the intent and entities from user speech.

Return JSON with this structure:
{
  "action": "add" | "complete" | "delete" | "list" | "update" | "undo" | "redo",
  "title": "task title if applicable",
  "task_identifier": "task number or title for existing tasks",
  "priority": "low" | "medium" | "high" | null,
  "due_date": "ISO date string or relative like 'tomorrow'" | null,
  "due_filter": "today" | "tomorrow" | "this_week" | null
}

Examples:
- "Add buy groceries" -> {"action": "add", "title": "buy groceries"}
- "Complete the milk task" -> {"action": "complete", "task_identifier": "milk"}
- "Delete task 3" -> {"action": "delete", "task_identifier": "3"}
- "Add call mom tomorrow high priority" -> {"action": "add", "title": "call mom", "due_date": "tomorrow", "priority": "high"}
- "What's due today" -> {"action": "list", "due_filter": "today"}
- "Undo" -> {"action": "undo"}

Only return valid JSON, no explanation."""

    async def parse(self, transcript: str) -> Optional[Dict[str, Any]]:
        """Parse transcript into structured intent."""
        # Quick pattern matching for common commands
        quick_result = self._quick_parse(transcript)
        if quick_result:
            return self._resolve_dates(quick_result)

        # Fall back to LLM for complex commands
        try:
            response = self.client.chat.completions.create(
                model="gpt-4o-mini",
                messages=[
                    {"role": "system", "content": self.system_prompt},
                    {"role": "user", "content": transcript}
                ],
                temperature=0,
                max_tokens=200,
                response_format={"type": "json_object"}
            )

            result = json.loads(response.choices[0].message.content)
            return self._resolve_dates(result)

        except Exception as e:
            print(f"Intent parsing error: {e}")
            return None

    def _quick_parse(self, transcript: str) -> Optional[Dict[str, Any]]:
        """Fast regex-based parsing for common patterns."""
        text = transcript.lower().strip()

        # Undo/Redo
        if text in ["undo", "undo that", "undo last action"]:
            return {"action": "undo"}
        if text in ["redo", "redo that", "redo last action"]:
            return {"action": "redo"}

        # Add task patterns
        add_patterns = [
            r"^add\s+(?:task\s+)?(.+)$",
            r"^create\s+(?:task\s+)?(.+)$",
            r"^new\s+task\s+(.+)$",
            r"^remind\s+me\s+to\s+(.+)$",
        ]
        for pattern in add_patterns:
            match = re.match(pattern, text)
            if match:
                return self._parse_add_details(match.group(1))

        # Complete task patterns
        complete_patterns = [
            r"^(?:complete|done|finish|mark)\s+(?:task\s+)?(.+?)(?:\s+as\s+done)?$",
            r"^done\s+with\s+(.+)$",
        ]
        for pattern in complete_patterns:
            match = re.match(pattern, text)
            if match:
                return {"action": "complete", "task_identifier": match.group(1)}

        # Delete task patterns
        delete_patterns = [
            r"^(?:delete|remove|cancel)\s+(?:task\s+)?(.+)$",
        ]
        for pattern in delete_patterns:
            match = re.match(pattern, text)
            if match:
                return {"action": "delete", "task_identifier": match.group(1)}

        # List patterns
        if any(phrase in text for phrase in ["what are my tasks", "list tasks", "show tasks", "my tasks"]):
            return {"action": "list"}
        if "due today" in text:
            return {"action": "list", "due_filter": "today"}
        if "due tomorrow" in text:
            return {"action": "list", "due_filter": "tomorrow"}
        if "high priority" in text:
            return {"action": "list", "priority": "high"}

        return None

    def _parse_add_details(self, text: str) -> Dict[str, Any]:
        """Extract priority and due date from add command."""
        result = {"action": "add", "title": text}

        # Extract priority
        if "high priority" in text:
            result["priority"] = "high"
            result["title"] = text.replace("high priority", "").strip()
        elif "low priority" in text:
            result["priority"] = "low"
            result["title"] = text.replace("low priority", "").strip()

        # Extract common date phrases
        date_phrases = {
            "tomorrow": 1,
            "today": 0,
            "next week": 7,
            "in a week": 7,
            "in 3 days": 3,
            "in three days": 3,
        }

        for phrase, days in date_phrases.items():
            if phrase in result["title"]:
                result["due_date"] = phrase
                result["title"] = result["title"].replace(phrase, "").strip()
                break

        # Clean up extra words
        result["title"] = re.sub(r"\s+for\s*$", "", result["title"])
        result["title"] = re.sub(r"\s+with\s*$", "", result["title"])

        return result

    def _resolve_dates(self, intent: Dict[str, Any]) -> Dict[str, Any]:
        """Convert relative dates to absolute dates."""
        if not intent.get("due_date"):
            return intent

        due = intent["due_date"].lower()
        today = datetime.now().replace(hour=23, minute=59, second=59)

        relative_dates = {
            "today": today,
            "tomorrow": today + timedelta(days=1),
            "next week": today + timedelta(days=7),
            "in a week": today + timedelta(days=7),
            "in 3 days": today + timedelta(days=3),
            "in three days": today + timedelta(days=3),
        }

        if due in relative_dates:
            intent["due_date"] = relative_dates[due].isoformat()
        else:
            # Try parsing with dateutil
            try:
                parsed = date_parser.parse(due, fuzzy=True)
                intent["due_date"] = parsed.isoformat()
            except:
                intent["due_date"] = None

        return intent
```

### Complete Task Service

Bringing it all together—the task service orchestrates operations, history, and real-time updates.

```python
# services/task_service.py
from sqlalchemy.orm import Session
from models.database import Task, Priority, get_db
from models.schemas import TaskCreate, TaskResponse, TaskUpdate
from services.action_history import (
    ActionHistory, CreateTaskAction, CompleteTaskAction, DeleteTaskAction
)
from routers.websocket import manager
from typing import Optional, List, Dict
from datetime import datetime

class TaskService:
    """Core task management service with undo/redo support."""

    def __init__(self):
        # Per-user action histories
        self._histories: Dict[str, ActionHistory] = {}

    def _get_history(self, user_id: str) -> ActionHistory:
        """Get or create action history for user."""
        if user_id not in self._histories:
            self._histories[user_id] = ActionHistory(max_size=50)
        return self._histories[user_id]

    async def create_task(
        self,
        user_id: str,
        title: str,
        due_date: Optional[datetime] = None,
        priority: str = "medium"
    ) -> TaskResponse:
        """Create a task with undo support."""
        history = self._get_history(user_id)

        action = CreateTaskAction(
            task_service=self,
            user_id=user_id,
            title=title,
            due_date=due_date,
            priority=priority
        )

        task = await history.execute(action)

        # Broadcast to connected clients
        await manager.broadcast_task_event(
            user_id,
            "task_created",
            TaskResponse.from_orm(task).dict()
        )

        return TaskResponse.from_orm(task)

    async def complete_task(
        self,
        user_id: str,
        task_identifier: str
    ) -> TaskResponse:
        """Complete a task by ID or title match."""
        task = await self._find_task(user_id, task_identifier)
        if not task:
            raise ValueError(f"Task not found: {task_identifier}")

        history = self._get_history(user_id)
        action = CompleteTaskAction(
            task_service=self,
            task_id=task.id
        )

        await history.execute(action)

        # Refresh and broadcast
        task = await self._get_task_by_id(task.id)
        await manager.broadcast_task_event(
            user_id,
            "task_completed",
            TaskResponse.from_orm(task).dict()
        )

        return TaskResponse.from_orm(task)

    async def delete_task(
        self,
        user_id: str,
        task_identifier: str
    ) -> TaskResponse:
        """Delete a task with undo support."""
        task = await self._find_task(user_id, task_identifier)
        if not task:
            raise ValueError(f"Task not found: {task_identifier}")

        task_response = TaskResponse.from_orm(task)

        history = self._get_history(user_id)
        action = DeleteTaskAction(
            task_service=self,
            task_id=task.id
        )

        await history.execute(action)

        await manager.broadcast_task_event(
            user_id,
            "task_deleted",
            task_response.dict()
        )

        return task_response

    async def list_tasks(
        self,
        user_id: str,
        filter_priority: Optional[str] = None,
        filter_due: Optional[str] = None,
        include_completed: bool = False
    ) -> List[TaskResponse]:
        """List tasks with optional filters."""
        db = next(get_db())
        try:
            query = db.query(Task).filter(Task.user_id == user_id)

            if not include_completed:
                query = query.filter(Task.completed == False)

            if filter_priority:
                query = query.filter(Task.priority == Priority(filter_priority))

            if filter_due == "today":
                today = datetime.now().date()
                query = query.filter(
                    Task.due_date >= today,
                    Task.due_date < today + timedelta(days=1)
                )
            elif filter_due == "tomorrow":
                tomorrow = datetime.now().date() + timedelta(days=1)
                query = query.filter(
                    Task.due_date >= tomorrow,
                    Task.due_date < tomorrow + timedelta(days=1)
                )

            tasks = query.order_by(Task.created_at.desc()).all()
            return [TaskResponse.from_orm(t) for t in tasks]
        finally:
            db.close()

    async def undo(self, user_id: str) -> Optional[str]:
        """Undo last action for user."""
        history = self._get_history(user_id)
        result = await history.undo()

        if result:
            # Broadcast full task list refresh
            await manager.broadcast_to_user(user_id, {
                "type": "tasks_refresh",
                "timestamp": datetime.utcnow().isoformat()
            })

        return result

    async def redo(self, user_id: str) -> Optional[str]:
        """Redo last undone action for user."""
        history = self._get_history(user_id)
        result = await history.redo()

        if result:
            await manager.broadcast_to_user(user_id, {
                "type": "tasks_refresh",
                "timestamp": datetime.utcnow().isoformat()
            })

        return result

    # Internal methods (used by actions)
    async def _create_task_internal(self, **kwargs) -> Task:
        db = next(get_db())
        try:
            task = Task(**kwargs)
            db.add(task)
            db.commit()
            db.refresh(task)
            return task
        finally:
            db.close()

    async def _delete_task_internal(self, task_id: int):
        db = next(get_db())
        try:
            db.query(Task).filter(Task.id == task_id).delete()
            db.commit()
        finally:
            db.close()

    async def _update_task_internal(self, task_id: int, **kwargs):
        db = next(get_db())
        try:
            db.query(Task).filter(Task.id == task_id).update(kwargs)
            db.commit()
        finally:
            db.close()

    async def _get_task_by_id(self, task_id: int) -> Optional[Task]:
        db = next(get_db())
        try:
            return db.query(Task).filter(Task.id == task_id).first()
        finally:
            db.close()

    async def _find_task(self, user_id: str, identifier: str) -> Optional[Task]:
        """Find task by ID number or title match."""
        db = next(get_db())
        try:
            # Try as numeric ID first
            if identifier.isdigit():
                return db.query(Task).filter(
                    Task.user_id == user_id,
                    Task.id == int(identifier)
                ).first()

            # Try title match (case-insensitive, partial)
            return db.query(Task).filter(
                Task.user_id == user_id,
                Task.title.ilike(f"%{identifier}%")
            ).first()
        finally:
            db.close()
```

```python
# services/query_engine.py
class NaturalQueryEngine:
    """Handle complex natural language queries about tasks."""

    async def execute_query(self, query: str, user_id: str) -> str:
        """Execute a natural language query."""

        # Parse query intent
        query_intent = await self._parse_query(query)

        match query_intent["type"]:
            case "completed_range":
                tasks = await self._get_completed_in_range(
                    user_id,
                    query_intent["start"],
                    query_intent["end"]
                )
                return self._format_task_list(tasks, "completed")

            case "overdue":
                tasks = await self._get_overdue(user_id)
                return self._format_task_list(tasks, "overdue")

            case "procrastinated":
                tasks = await self._get_old_incomplete(user_id, days=7)
                return self._format_task_list(tasks, "waiting")

            case "statistics":
                stats = await self._get_stats(user_id)
                return self._format_stats(stats)

    async def _get_old_incomplete(self, user_id: str, days: int) -> List[Task]:
        """Find tasks created more than N days ago that aren't complete."""
        cutoff = datetime.now() - timedelta(days=days)
        db = next(get_db())
        return db.query(Task).filter(
            Task.user_id == user_id,
            Task.completed == False,
            Task.created_at < cutoff
        ).all()
```

```python
# routers/tasks.py
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.orm import Session
from typing import List, Optional
from models.database import get_db, Task, Priority
from models.schemas import TaskCreate, TaskResponse, TaskUpdate
from services.task_service import TaskService
from datetime import datetime

router = APIRouter()
task_service = TaskService()

@router.post("/", response_model=TaskResponse, status_code=201)
async def create_task(
    task: TaskCreate,
    user_id: str = Query(..., description="User identifier"),
    db: Session = Depends(get_db)
):
    """Create a new task."""
    return await task_service.create_task(
        user_id=user_id,
        title=task.title,
        due_date=task.due_date,
        priority=task.priority.value if task.priority else "medium"
    )

@router.get("/", response_model=List[TaskResponse])
async def list_tasks(
    user_id: str = Query(..., description="User identifier"),
    priority: Optional[str] = Query(None, description="Filter by priority"),
    due: Optional[str] = Query(None, description="Filter by due date: today, tomorrow, this_week"),
    include_completed: bool = Query(False, description="Include completed tasks"),
    db: Session = Depends(get_db)
):
    """List all tasks for a user with optional filters."""
    return await task_service.list_tasks(
        user_id=user_id,
        filter_priority=priority,
        filter_due=due,
        include_completed=include_completed
    )

@router.get("/{task_id}", response_model=TaskResponse)
async def get_task(
    task_id: int,
    user_id: str = Query(..., description="User identifier"),
    db: Session = Depends(get_db)
):
    """Get a specific task by ID."""
    task = await task_service._get_task_by_id(task_id)
    if not task or task.user_id != user_id:
        raise HTTPException(status_code=404, detail="Task not found")
    return TaskResponse.from_orm(task)

@router.patch("/{task_id}", response_model=TaskResponse)
async def update_task(
    task_id: int,
    update: TaskUpdate,
    user_id: str = Query(..., description="User identifier"),
    db: Session = Depends(get_db)
):
    """Update a task."""
    task = await task_service._get_task_by_id(task_id)
    if not task or task.user_id != user_id:
        raise HTTPException(status_code=404, detail="Task not found")

    update_data = update.dict(exclude_unset=True)
    if update_data:
        await task_service._update_task_internal(task_id, **update_data)

    return TaskResponse.from_orm(await task_service._get_task_by_id(task_id))

@router.post("/{task_id}/complete", response_model=TaskResponse)
async def complete_task(
    task_id: int,
    user_id: str = Query(..., description="User identifier"),
    db: Session = Depends(get_db)
):
    """Mark a task as complete."""
    return await task_service.complete_task(user_id=user_id, task_identifier=str(task_id))

@router.delete("/{task_id}", response_model=TaskResponse)
async def delete_task(
    task_id: int,
    user_id: str = Query(..., description="User identifier"),
    db: Session = Depends(get_db)
):
    """Delete a task."""
    return await task_service.delete_task(user_id=user_id, task_identifier=str(task_id))

@router.post("/undo")
async def undo_action(
    user_id: str = Query(..., description="User identifier")
):
    """Undo the last action."""
    result = await task_service.undo(user_id)
    if not result:
        raise HTTPException(status_code=400, detail="Nothing to undo")
    return {"message": result}

@router.post("/redo")
async def redo_action(
    user_id: str = Query(..., description="User identifier")
):
    """Redo the last undone action."""
    result = await task_service.redo(user_id)
    if not result:
        raise HTTPException(status_code=400, detail="Nothing to redo")
    return {"message": result}
```

### WebSocket Hook

```tsx
// src/hooks/useTaskWebSocket.ts
import { useEffect, useRef, useCallback } from "react";

export function useTaskWebSocket(userId: string, onUpdate: () => void) {
  const wsRef = useRef<WebSocket | null>(null);
  const reconnectAttempts = useRef(0);
  const maxReconnectAttempts = 5;

  const connect = useCallback(() => {
    const wsUrl = `${process.env.REACT_APP_WS_URL || "ws://localhost:8000"}/ws/${userId}`;
    const ws = new WebSocket(wsUrl);

    ws.onopen = () => {
      console.log("WebSocket connected");
      reconnectAttempts.current = 0;
    };

    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      if (data.type !== "pong") {
        console.log("Task event:", data.type);
        onUpdate();
      }
    };

    ws.onclose = () => {
      console.log("WebSocket disconnected");
      if (reconnectAttempts.current < maxReconnectAttempts) {
        reconnectAttempts.current++;
        setTimeout(connect, 1000 * reconnectAttempts.current);
      }
    };

    ws.onerror = (error) => {
      console.error("WebSocket error:", error);
    };

    wsRef.current = ws;

    // Heartbeat
    const heartbeat = setInterval(() => {
      if (ws.readyState === WebSocket.OPEN) {
        ws.send(JSON.stringify({ type: "ping" }));
      }
    }, 30000);

    return () => {
      clearInterval(heartbeat);
      ws.close();
    };
  }, [userId, onUpdate]);

  useEffect(() => {
    const cleanup = connect();
    return cleanup;
  }, [connect]);
}
```

```tsx
import axios from "axios";

const API_BASE = process.env.REACT_APP_API_URL || "http://localhost:8000";

export interface Task {
  id: number;
  user_id: string;
  title: string;
  description?: string;
  priority: "low" | "medium" | "high";
  due_date?: string;
  completed: boolean;
  completed_at?: string;
  created_at: string;
}

export const taskApi = {
  async listTasks(
    userId: string,
    filters?: { priority?: string; due?: string },
  ): Promise<Task[]> {
    const params = new URLSearchParams({ user_id: userId, ...filters });
    const response = await axios.get(`${API_BASE}/api/tasks?${params}`);
    return response.data;
  },

  async createTask(userId: string, task: Partial<Task>): Promise<Task> {
    const response = await axios.post(
      `${API_BASE}/api/tasks?user_id=${userId}`,
      task,
    );
    return response.data;
  },

  async completeTask(userId: string, taskId: number): Promise<Task> {
    const response = await axios.post(
      `${API_BASE}/api/tasks/${taskId}/complete?user_id=${userId}`,
    );
    return response.data;
  },

  async deleteTask(userId: string, taskId: number): Promise<Task> {
    const response = await axios.delete(
      `${API_BASE}/api/tasks/${taskId}?user_id=${userId}`,
    );
    return response.data;
  },

  async undo(userId: string): Promise<{ message: string }> {
    const response = await axios.post(
      `${API_BASE}/api/tasks/undo?user_id=${userId}`,
    );
    return response.data;
  },

  async redo(userId: string): Promise<{ message: string }> {
    const response = await axios.post(
      `${API_BASE}/api/tasks/redo?user_id=${userId}`,
    );
    return response.data;
  },
};
```

```python
    def test_complete_task(self):

        create_response = client.post(

        # Complete it
        response = client.post(f"/api/tasks/{task_id}/complete?user_id=test-user")

    def test_undo_action(self):
```
