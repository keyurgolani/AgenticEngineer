## Project Overview

**Objective**: Build an intelligent GitHub bot that automatically reviews pull requests by analyzing code changes, detecting security vulnerabilities, suggesting performance improvements, and posting contextual review comments—all powered by AST analysis and LLM reasoning.

**Why This Matters**: Code review is a critical bottleneck in software development. Human reviewers often miss subtle bugs, security issues, or performance problems due to time constraints. An AI-powered review bot provides consistent, thorough analysis on every PR, catching issues before they reach production while freeing human reviewers to focus on architecture and design decisions.

## Module Summary

In this advanced project, we build a comprehensive code review automation system:

1. **GitHub Webhook Handler**: We receive and validate webhook events for pull request creation and updates, ensuring secure communication with GitHub.

2. **Diff Analysis Pipeline**: We parse PR diffs to extract changed files, identify the specific lines modified, and prepare code for analysis.

3. **AST-Based Code Analysis**: We parse Python code into ASTs to detect complexity issues, code smells, and structural problems that simple regex can't catch.

4. **Security Scanning**: We integrate Bandit and custom security rules to identify vulnerabilities like SQL injection, hardcoded secrets, and unsafe deserialization.

5. **LLM-Powered Review**: We use GPT-4o with structured outputs to generate contextual, actionable review comments that understand code intent.

6. **GitHub Comment Threading**: We post review comments on specific lines, create review threads, and suggest fixes with proper GitHub API integration.

```python
# webhook/handler.py
import hmac
import hashlib
from fastapi import APIRouter, Request, HTTPException, Header, BackgroundTasks
from typing import Optional
import logging

from config import get_settings
from models import WebhookPayload, ReviewTask, ReviewStatus
from queue.task_queue import task_queue
from review.orchestrator import ReviewOrchestrator

logger = logging.getLogger(__name__)
router = APIRouter()

class WebhookSignatureValidator:
    """Validates GitHub webhook signatures using HMAC-SHA256."""

    def __init__(self, secret: str):
        self.secret = secret.encode('utf-8')

    def validate(self, payload: bytes, signature: str) -> bool:
        """
        Validate webhook signature.

        Args:
            payload: Raw request body bytes
            signature: X-Hub-Signature-256 header value

        Returns:
            True if signature is valid, False otherwise
        """
        if not signature or not signature.startswith('sha256='):
            return False

        expected_signature = signature[7:]  # Remove 'sha256=' prefix

        computed_signature = hmac.new(
            self.secret,
            payload,
            hashlib.sha256
        ).hexdigest()

        # Use constant-time comparison to prevent timing attacks
        return hmac.compare_digest(computed_signature, expected_signature)

def parse_pr_webhook(payload: dict) -> Optional[WebhookPayload]:
    """Parse pull_request webhook payload into structured format."""
    try:
        pr = payload.get("pull_request", {})
        repo = payload.get("repository", {})
        installation = payload.get("installation", {})

        return WebhookPayload(
            action=payload.get("action", ""),
            pr_number=pr.get("number", 0),
            pr_title=pr.get("title", ""),
            head_sha=pr.get("head", {}).get("sha", ""),
            base_sha=pr.get("base", {}).get("sha", ""),
            head_ref=pr.get("head", {}).get("ref", ""),
            base_ref=pr.get("base", {}).get("ref", ""),
            repository=repo.get("full_name", ""),
            installation_id=installation.get("id", 0),
            sender=payload.get("sender", {}).get("login", ""),
            diff_url=pr.get("diff_url", "")
        )
    except Exception as e:
        logger.error(f"Failed to parse webhook payload: {e}")
        return None

@router.post("/webhook/github")
async def handle_github_webhook(
    request: Request,
    background_tasks: BackgroundTasks,
    x_github_event: str = Header(..., alias="X-GitHub-Event"),
    x_hub_signature_256: str = Header(..., alias="X-Hub-Signature-256"),
    x_github_delivery: str = Header(..., alias="X-GitHub-Delivery")
):
    """
    Handle incoming GitHub webhook events.

    Validates signature, parses payload, and queues review tasks
    for pull_request events.
    """
    settings = get_settings()

    # Read raw body for signature validation
    body = await request.body()

    # Validate webhook signature
    validator = WebhookSignatureValidator(settings.github_webhook_secret)
    if not validator.validate(body, x_hub_signature_256):
        logger.warning(f"Invalid webhook signature for delivery {x_github_delivery}")
        raise HTTPException(status_code=401, detail="Invalid webhook signature")

    # Only process pull_request events
    if x_github_event != "pull_request":
        logger.info(f"Ignoring event type: {x_github_event}")
        return {"status": "ignored", "event": x_github_event}

    # Parse payload
    payload = await request.json()
    webhook_data = parse_pr_webhook(payload)

    if not webhook_data:
        raise HTTPException(status_code=400, detail="Failed to parse webhook payload")

    # Only review on opened, synchronize (new commits), or reopened
    reviewable_actions = {"opened", "synchronize", "reopened"}
    if webhook_data.action not in reviewable_actions:
        logger.info(f"Ignoring PR action: {webhook_data.action}")
        return {"status": "ignored", "action": webhook_data.action}

    # Create review task
    task = ReviewTask(
        pr_number=webhook_data.pr_number,
        repository=webhook_data.repository,
        head_sha=webhook_data.head_sha,
        base_sha=webhook_data.base_sha,
        installation_id=webhook_data.installation_id,
        status=ReviewStatus.PENDING
    )

    # Queue for background processing
    await task_queue.enqueue(task)

    # Start background processing
    background_tasks.add_task(process_review, task.id)

    logger.info(f"Queued review {task.id} for PR #{webhook_data.pr_number}")

    return {
        "status": "queued",
        "review_id": task.id,
        "message": "Review queued for processing"
    }

async def process_review(task_id: str):
    """Background task to process a review."""
    orchestrator = ReviewOrchestrator()
    await orchestrator.run_review(task_id)
```

**Security Considerations**:

- **Constant-time comparison**: We use `hmac.compare_digest()` to prevent timing attacks
- **Raw body validation**: We validate against the raw bytes, not parsed JSON
- **Signature prefix check**: We verify the `sha256=` prefix before comparison
- **Logging without secrets**: We log delivery IDs but never log signatures or secrets

### Step 3: GitHub API Client with App Authentication

GitHub Apps use JWT-based authentication. We generate installation tokens for API access.

```python
# github/client.py
import jwt
import time
import httpx
from typing import Optional, List, Dict, Any
from datetime import datetime, timedelta
import logging
from dataclasses import dataclass

from config import get_settings
from models import FileDiff, ReviewComment

logger = logging.getLogger(__name__)

@dataclass
class InstallationToken:
    """Cached installation access token."""
    token: str
    expires_at: datetime

class GitHubAppClient:
    """GitHub API client with App authentication."""

    BASE_URL = "https://api.github.com"

    def __init__(self):
        self.settings = get_settings()
        self._installation_tokens: Dict[int, InstallationToken] = {}

    def _generate_jwt(self) -> str:
        """Generate a JWT for GitHub App authentication."""
        now = int(time.time())

        payload = {
            "iat": now - 60,  # Issued 60 seconds ago (clock skew)
            "exp": now + (10 * 60),  # Expires in 10 minutes
            "iss": self.settings.github_app_id
        }

        return jwt.encode(
            payload,
            self.settings.github_private_key,
            algorithm="RS256"
        )

    async def _get_installation_token(self, installation_id: int) -> str:
        """Get or refresh installation access token."""
        cached = self._installation_tokens.get(installation_id)

        # Return cached token if still valid (with 5 min buffer)
        if cached and cached.expires_at > datetime.utcnow() + timedelta(minutes=5):
            return cached.token

        # Request new installation token
        jwt_token = self._generate_jwt()

        async with httpx.AsyncClient() as client:
            response = await client.post(
                f"{self.BASE_URL}/app/installations/{installation_id}/access_tokens",
                headers={
                    "Authorization": f"Bearer {jwt_token}",
                    "Accept": "application/vnd.github+json",
                    "X-GitHub-Api-Version": "2022-11-28"
                }
            )
            response.raise_for_status()
            data = response.json()

        # Cache the token
        expires_at = datetime.fromisoformat(data["expires_at"].replace("Z", "+00:00"))
        self._installation_tokens[installation_id] = InstallationToken(
            token=data["token"],
            expires_at=expires_at.replace(tzinfo=None)
        )

        return data["token"]

    async def _request(
        self,
        method: str,
        endpoint: str,
        installation_id: int,
        **kwargs
    ) -> Any:
        """Make an authenticated request to GitHub API."""
        token = await self._get_installation_token(installation_id)

        headers = {
            "Authorization": f"Bearer {token}",
            "Accept": "application/vnd.github+json",
            "X-GitHub-Api-Version": "2022-11-28"
        }

        async with httpx.AsyncClient() as client:
            response = await client.request(
                method,
                f"{self.BASE_URL}{endpoint}",
                headers=headers,
                **kwargs
            )

            # Handle rate limiting
            if response.status_code == 429:
                retry_after = int(response.headers.get("Retry-After", 60))
                logger.warning(f"Rate limited, retry after {retry_after}s")
                raise RateLimitError(retry_after)

            response.raise_for_status()
            return response.json() if response.content else None

    async def get_pr_files(

        return [
            FileDiff(

                additions=f["additions"],

    async def get_file_content(
        self,
        owner: str,
        repo: str,
        path: str,

        """Get file content at a specific ref."""
        import base64

        endpoint = f"/repos/{owner}/{repo}/contents/{path}"

        try:
            data = await self._request(
                "GET", endpoint, installation_id,
                params={"ref": ref}
            )

            if data.get("encoding") == "base64":
                return base64.b64decode(data["content"]).decode("utf-8")
            return data.get("content")
        except httpx.HTTPStatusError as e:
            if e.response.status_code == 404:
                return None
            raise

            "side": comment.side
```

```python
# analysis/diff_parser.py
import re
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass, field

@dataclass
class DiffHunk:
    """A single hunk of changes in a diff."""
    old_start: int
    old_count: int
    new_start: int
    new_count: int
    lines: List[Tuple[str, str, int]]  # (type, content, new_line_number)

@dataclass
class ParsedFile:
    """A parsed file diff with line mappings."""
    path: str
    status: str
    hunks: List[DiffHunk] = field(default_factory=list)
    added_lines: Dict[int, str] = field(default_factory=dict)
    modified_context: Dict[int, List[str]] = field(default_factory=dict)

class DiffParser:
    """Parse unified diff format into structured data."""

    HUNK_HEADER_PATTERN = re.compile(
        r'^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@'
    )

    def parse_patch(self, patch: str, file_path: str, status: str) -> ParsedFile:
        """
        Parse a unified diff patch into structured format.

        Args:
            patch: The patch content from GitHub API
            file_path: Path to the file
            status: File status (added, modified, removed)

        Returns:
            ParsedFile with hunks and line mappings
        """
        parsed = ParsedFile(path=file_path, status=status)

        if not patch:
            return parsed

        lines = patch.split('\n')
        current_hunk: Optional[DiffHunk] = None
        new_line_num = 0

        for line in lines:
            # Check for hunk header
            header_match = self.HUNK_HEADER_PATTERN.match(line)
            if header_match:
                if current_hunk:
                    parsed.hunks.append(current_hunk)

                current_hunk = DiffHunk(
                    old_start=int(header_match.group(1)),
                    old_count=int(header_match.group(2) or 1),
                    new_start=int(header_match.group(3)),
                    new_count=int(header_match.group(4) or 1),
                    lines=[]
                )
                new_line_num = current_hunk.new_start
                continue

            if current_hunk is None:
                continue

            # Parse diff lines
            if line.startswith('+') and not line.startswith('+++'):
                # Added line
                content = line[1:]
                current_hunk.lines.append(('add', content, new_line_num))
                parsed.added_lines[new_line_num] = content
                new_line_num += 1

            elif line.startswith('-') and not line.startswith('---'):
                # Removed line (don't increment new line number)
                content = line[1:]
                current_hunk.lines.append(('remove', content, -1))

            elif line.startswith(' '):
                # Context line
                content = line[1:]
                current_hunk.lines.append(('context', content, new_line_num))
                new_line_num += 1

        # Don't forget the last hunk
        if current_hunk:
            parsed.hunks.append(current_hunk)

        # Build context for modified lines
        self._build_context(parsed)

        return parsed

    def _build_context(self, parsed: ParsedFile, context_lines: int = 3):
        """Build context around each added line for better analysis."""
        for hunk in parsed.hunks:
            all_lines = [(t, c, n) for t, c, n in hunk.lines if n > 0]

            for i, (line_type, content, line_num) in enumerate(all_lines):
                if line_type == 'add':
                    # Get surrounding context
                    start = max(0, i - context_lines)
                    end = min(len(all_lines), i + context_lines + 1)

                    context = [
                        all_lines[j][1] for j in range(start, end)
                    ]
                    parsed.modified_context[line_num] = context

    def get_reviewable_lines(self, parsed: ParsedFile) -> List[int]:
        """Get list of line numbers that can receive review comments."""
        # GitHub only allows comments on added lines or context lines
        reviewable = set()

        for hunk in parsed.hunks:
            for line_type, _, line_num in hunk.lines:
                if line_num > 0 and line_type in ('add', 'context'):
                    reviewable.add(line_num)

        return sorted(reviewable)

    def find_nearest_reviewable_line(
        self,
        parsed: ParsedFile,
        target_line: int
    ) -> Optional[int]:
        """Find the nearest line that can receive a review comment."""
        reviewable = self.get_reviewable_lines(parsed)

        if not reviewable:
            return None

        if target_line in reviewable:
            return target_line

        # Find closest reviewable line
        closest = min(reviewable, key=lambda x: abs(x - target_line))
        return closest
```

```python
# analysis/ast_analyzer.py
import ast
from typing import List, Dict, Optional, Set
from dataclasses import dataclass, field
import logging

from models import Finding, Severity, FindingCategory

logger = logging.getLogger(__name__)

@dataclass
class ComplexityMetrics:
    """Code complexity measurements."""
    cyclomatic: int = 0
    cognitive: int = 0
    nesting_depth: int = 0
    lines_of_code: int = 0

class ASTAnalyzer(ast.NodeVisitor):
    """
    Analyze Python code using AST for quality issues.

    Detects:
    - High cyclomatic complexity
    - Deep nesting
    - Long functions
    - Unused variables
    - Bare except clauses
    - Mutable default arguments
    - Global variable usage
    """

    # Thresholds for findings
    MAX_CYCLOMATIC_COMPLEXITY = 10
    MAX_COGNITIVE_COMPLEXITY = 15
    MAX_NESTING_DEPTH = 4
    MAX_FUNCTION_LINES = 50
    MAX_FUNCTION_ARGS = 5

    def __init__(self, file_path: str, source_lines: Dict[int, str]):
        self.file_path = file_path
        self.source_lines = source_lines  # Only analyze changed lines
        self.findings: List[Finding] = []

        # Tracking state
        self._current_function: Optional[str] = None
        self._nesting_depth = 0
        self._complexity = 0
        self._cognitive_complexity = 0
        self._defined_names: Set[str] = set()
        self._used_names: Set[str] = set()

    def analyze(self, source_code: str) -> List[Finding]:
        """
        Analyze source code and return findings.

        Args:
            source_code: Complete Python source code

        Returns:
            List of findings for issues detected
        """
        try:
            tree = ast.parse(source_code)
        except SyntaxError as e:
            # Syntax errors are handled elsewhere
            logger.warning(f"Syntax error in {self.file_path}: {e}")
            return []

        self.visit(tree)

        return self.findings

    def visit_FunctionDef(self, node: ast.FunctionDef):
        """Analyze function definitions."""
        self._analyze_function(node)
        self.generic_visit(node)

    def visit_AsyncFunctionDef(self, node: ast.AsyncFunctionDef):
        """Analyze async function definitions."""
        self._analyze_function(node)
        self.generic_visit(node)

    def _analyze_function(self, node):
        """Common function analysis logic."""
        # Only analyze if function touches changed lines
        if not self._touches_changed_lines(node):
            return

        # Check function length
        func_lines = node.end_lineno - node.lineno + 1
        if func_lines > self.MAX_FUNCTION_LINES:
            self._add_finding(
                line=node.lineno,
                severity=Severity.MEDIUM,
                category=FindingCategory.QUALITY,
                title="Function too long",
                description=f"Function `{node.name}` has {func_lines} lines. "
                           f"Consider breaking it into smaller functions.",
                rule_id="FUNC_LENGTH"
            )

        # Check argument count
        total_args = (
            len(node.args.args) +
            len(node.args.posonlyargs) +
            len(node.args.kwonlyargs)
        )
        if total_args > self.MAX_FUNCTION_ARGS:
            self._add_finding(
                line=node.lineno,
                severity=Severity.LOW,
                category=FindingCategory.QUALITY,
                title="Too many arguments",
                description=f"Function `{node.name}` has {total_args} arguments. "
                           f"Consider using a configuration object or dataclass.",
                rule_id="ARG_COUNT"
            )

        # Check for mutable default arguments
        self._check_mutable_defaults(node)

        # Calculate complexity
        complexity = self._calculate_complexity(node)
        if complexity > self.MAX_CYCLOMATIC_COMPLEXITY:
            self._add_finding(
                line=node.lineno,
                severity=Severity.HIGH,
                category=FindingCategory.QUALITY,
                title="High cyclomatic complexity",
                description=f"Function `{node.name}` has complexity {complexity}. "
                           f"Consider simplifying the logic or extracting methods.",
                rule_id="COMPLEXITY"
            )

    def _check_mutable_defaults(self, node):
        """Check for mutable default arguments (common Python gotcha)."""
        for default in node.args.defaults + node.args.kw_defaults:
            if default is None:
                continue
            if isinstance(default, (ast.List, ast.Dict, ast.Set)):
                self._add_finding(
                    line=node.lineno,
                    severity=Severity.HIGH,
                    category=FindingCategory.QUALITY,
                    title="Mutable default argument",
                    description=f"Function `{node.name}` uses a mutable default argument. "
                               f"This can cause unexpected behavior. Use `None` and "
                               f"initialize inside the function.",
                    suggestion=f"def {node.name}(..., param=None):\n"
                              f"    if param is None:\n"
                              f"        param = []  # or {{}} for dict",
                    rule_id="MUTABLE_DEFAULT"
                )

    def visit_ExceptHandler(self, node: ast.ExceptHandler):
        """Check for bare except clauses."""
        if not self._touches_changed_lines(node):
            self.generic_visit(node)
            return

        if node.type is None:
            self._add_finding(
                line=node.lineno,
                severity=Severity.HIGH,
                category=FindingCategory.QUALITY,
                title="Bare except clause",
                description="Bare `except:` catches all exceptions including "
                           "KeyboardInterrupt and SystemExit. Catch specific "
                           "exceptions instead.",
                suggestion="except Exception as e:  # Or more specific exception",
                rule_id="BARE_EXCEPT"
            )

        self.generic_visit(node)

    def visit_Global(self, node: ast.Global):
        """Flag global variable usage."""
        if not self._in_changed_lines(node.lineno):
            return

        self._add_finding(
            line=node.lineno,
            severity=Severity.MEDIUM,
            category=FindingCategory.QUALITY,
            title="Global variable usage",
            description=f"Using global variables ({', '.join(node.names)}) "
                       f"makes code harder to test and reason about. "
                       f"Consider passing values as parameters.",
            rule_id="GLOBAL_VAR"
        )

    def visit_Assert(self, node: ast.Assert):
        """Check for assert statements that might be stripped in production."""
        if not self._in_changed_lines(node.lineno):
            return

        self._add_finding(
            line=node.lineno,
            severity=Severity.INFO,
            category=FindingCategory.QUALITY,
            title="Assert statement in production code",
            description="Assert statements are removed when Python runs with "
                       "optimization (-O flag). Use explicit validation for "
                       "production code.",
            rule_id="ASSERT_PROD"
        )

    def visit_Compare(self, node: ast.Compare):
        """Check for common comparison issues."""
        if not self._in_changed_lines(node.lineno):
            self.generic_visit(node)
            return

        # Check for `== None` instead of `is None`
        for op, comparator in zip(node.ops, node.comparators):
            if isinstance(op, (ast.Eq, ast.NotEq)):
                if isinstance(comparator, ast.Constant) and comparator.value is None:
                    self._add_finding(
                        line=node.lineno,
                        severity=Severity.LOW,
                        category=FindingCategory.STYLE,
                        title="Use 'is None' instead of '== None'",
                        description="PEP 8 recommends using `is None` or `is not None` "
                                   "for None comparisons.",
                        suggestion="if value is None:" if isinstance(op, ast.Eq)
                                  else "if value is not None:",
                        rule_id="NONE_COMPARE"
                    )

        self.generic_visit(node)

    def _calculate_complexity(self, node) -> int:
        """Calculate cyclomatic complexity of a function."""
        complexity = 1  # Base complexity

        for child in ast.walk(node):
            # Decision points add complexity
            if isinstance(child, (ast.If, ast.While, ast.For, ast.AsyncFor)):
                complexity += 1
            elif isinstance(child, ast.ExceptHandler):
                complexity += 1
            elif isinstance(child, ast.BoolOp):
                # and/or add complexity
                complexity += len(child.values) - 1
            elif isinstance(child, ast.comprehension):
                complexity += 1
                if child.ifs:
                    complexity += len(child.ifs)

        return complexity

    def _touches_changed_lines(self, node) -> bool:
        """Check if a node spans any changed lines."""
        if not hasattr(node, 'lineno') or not hasattr(node, 'end_lineno'):
            return False

        for line in range(node.lineno, (node.end_lineno or node.lineno) + 1):
            if line in self.source_lines:
                return True
        return False

    def _in_changed_lines(self, lineno: int) -> bool:
        """Check if a specific line was changed."""
        return lineno in self.source_lines

    def _add_finding(
        self,
        line: int,
        severity: Severity,
        category: FindingCategory,
        title: str,
        description: str,
        suggestion: Optional[str] = None,
        rule_id: Optional[str] = None
    ):
        """Add a finding to the results."""
        # Get code snippet if available
        snippet = self.source_lines.get(line, "")

        self.findings.append(Finding(
            file_path=self.file_path,
            line_number=line,
            severity=severity,
            category=category,
            title=title,
            description=description,
            suggestion=suggestion,
            code_snippet=snippet,
            rule_id=rule_id
        ))
```

```python
from typing import List, Dict, Optional, Set
from dataclasses import dataclass
import subprocess
import json

import os

@dataclass
class SecurityRule:

    name: str
    pattern: re.Pattern
    severity: Severity
    description: str
    cwe: Optional[str] = None

class SecurityScanner:
    """
    Scan code for security vulnerabilities.

    Combines:
    - Bandit for Python-specific security issues
    - Custom regex patterns for secrets
    - AST-based detection for injection patterns
    """

            description="SQL query uses string formatting. Use parameterized queries.",
            cwe="CWE-89"
        ),

        self.use_bandit = use_bandit

        self,
        source_code: str,
        file_path: str,
        changed_lines: Dict[int, str]
    ) -> List[Finding]:
        """
        Scan source code for security vulnerabilities.

        Args:
            source_code: Complete source code
            file_path: Path to the file
            changed_lines: Dict of line_number -> line_content for changed lines

        Returns:
            List of security findings
        """
        findings = []

        # Run custom pattern matching
        findings.extend(self._scan_patterns(source_code, file_path, changed_lines))

        # Run Bandit if available
        if self.use_bandit:
            findings.extend(self._run_bandit(source_code, file_path, changed_lines))

        # Run AST-based security checks
        findings.extend(self._scan_ast(source_code, file_path, changed_lines))

        return findings

    def _scan_patterns(
        self,
        source_code: str,
        file_path: str,
        changed_lines: Dict[int, str]
    ) -> List[Finding]:
        """Scan using regex patterns."""
        findings = []
        lines = source_code.split('\n')

        all_patterns = self.SECRET_PATTERNS + self.DANGEROUS_PATTERNS

        for line_num, line in enumerate(lines, 1):
            # Only check changed lines
            if line_num not in changed_lines:
                continue

            for rule in all_patterns:
                if rule.pattern.search(line):
                    findings.append(Finding(
                        file_path=file_path,
                        line_number=line_num,
                        severity=rule.severity,
                        category=FindingCategory.SECURITY,
                        title=rule.name,
                        description=rule.description,
                        code_snippet=line.strip(),
                        rule_id=rule.id
                    ))

        return findings

    def _run_bandit(
        self,
        source_code: str,
        file_path: str,
        changed_lines: Dict[int, str]
    ) -> List[Finding]:
        """Run Bandit security scanner."""
        findings = []

        # Write code to temp file
        with tempfile.NamedTemporaryFile(
            mode='w',
            suffix='.py',
            delete=False
        ) as f:
            f.write(source_code)
            temp_path = f.name

        try:
            result = subprocess.run(
                [
                    'bandit',
                    '-f', 'json',
                    '-ll',  # Only medium and above
                    temp_path
                ],
                capture_output=True,
                text=True,
                timeout=30
            )

            if result.stdout:
                bandit_results = json.loads(result.stdout)

                for issue in bandit_results.get('results', []):
                    line_num = issue.get('line_number', 0)

                    # Only report issues on changed lines
                    if line_num not in changed_lines:
                        continue

                    severity_map = {
                        'HIGH': Severity.HIGH,
                        'MEDIUM': Severity.MEDIUM,
                        'LOW': Severity.LOW
                    }

                    findings.append(Finding(
                        file_path=file_path,
                        line_number=line_num,
                        severity=severity_map.get(
                            issue.get('issue_severity', 'LOW'),
                            Severity.LOW
                        ),
                        category=FindingCategory.SECURITY,
                        title=issue.get('issue_text', 'Security Issue'),
                        description=f"{issue.get('issue_text', '')} "
                                   f"(CWE: {issue.get('issue_cwe', {}).get('id', 'N/A')})",
                        code_snippet=issue.get('code', ''),
                        rule_id=issue.get('test_id', '')
                    ))

        except FileNotFoundError:
            logger.warning("Bandit not installed, skipping Bandit scan")
        except subprocess.TimeoutExpired:
            logger.warning("Bandit scan timed out")
        except json.JSONDecodeError:
            logger.warning("Failed to parse Bandit output")
        finally:
            os.unlink(temp_path)

        return findings

    def _scan_ast(
        self,
        source_code: str,
        file_path: str,
        changed_lines: Dict[int, str]
    ) -> List[Finding]:
        """AST-based security scanning for complex patterns."""
        findings = []

        try:
            tree = ast.parse(source_code)
        except SyntaxError:
            return findings

        for node in ast.walk(tree):
            # Check for unsafe YAML loading
            if isinstance(node, ast.Call):
                if self._is_unsafe_yaml_load(node):
                    if node.lineno in changed_lines:
                        findings.append(Finding(
                            file_path=file_path,
                            line_number=node.lineno,
                            severity=Severity.HIGH,
                            category=FindingCategory.SECURITY,
                            title="Unsafe YAML Loading",
                            description="yaml.load() without Loader is unsafe. "
                                       "Use yaml.safe_load() instead.",
                            suggestion="yaml.safe_load(data)",
                            rule_id="SEC020"
                        ))

                # Check for requests without timeout
                if self._is_request_without_timeout(node):
                    if node.lineno in changed_lines:
                        findings.append(Finding(
                            file_path=file_path,
                            line_number=node.lineno,
                            severity=Severity.MEDIUM,
                            category=FindingCategory.SECURITY,
                            title="HTTP Request Without Timeout",
                            description="HTTP requests without timeout can hang indefinitely. "
                                       "Always specify a timeout.",
                            suggestion="requests.get(url, timeout=30)",
                            rule_id="SEC021"
                        ))

        return findings

    def _is_unsafe_yaml_load(self, node: ast.Call) -> bool:
        """Check if this is an unsafe yaml.load() call."""
        if isinstance(node.func, ast.Attribute):
            if node.func.attr == 'load':
                # Check if Loader is specified
                for keyword in node.keywords:
                    if keyword.arg == 'Loader':
                        return False
                return True
        return False

    def _is_request_without_timeout(self, node: ast.Call) -> bool:
        """Check if this is a requests call without timeout."""
        if isinstance(node.func, ast.Attribute):
            if node.func.attr in ('get', 'post', 'put', 'delete', 'patch'):
                # Check if it's from requests module
                if isinstance(node.func.value, ast.Name):
                    if node.func.value.id == 'requests':
                        # Check for timeout keyword
                        for keyword in node.keywords:
                            if keyword.arg == 'timeout':
                                return False
                        return True
        return False
```

            end = min(len(lines), finding.line_number + context_lines)
            for i in range(start, end):
                relevant_ranges.add(i)

        context_parts = []
        sorted_lines = sorted(relevant_ranges)

        prev_line = -2
        for line_idx in sorted_lines:
            if line_idx - prev_line > 1:
                context_parts.append("...")
            context_parts.append(f"{line_idx + 1:4d} | {lines[line_idx]}")
            prev_line = line_idx

        return '\n'.join(context_parts)

        suggestion = llm_review.get('suggestion', finding.suggestion)

            body_parts.extend([

            ])

        if code_example:
            body_parts.extend([
                "**Example fix:**",
                "```python",

                "```",
                ""
            ])

            f"<sub>Rule: `{finding.rule_id or 'N/A'}` | "
            f"Severity: {finding.severity.value}</sub>"
        ])

            line=finding.line_number,
            side="RIGHT",

```python
import asyncio
from typing import List, Dict
import logging

from github.client import GitHubAppClient, RateLimitError
from models import ReviewComment, Finding

logger = logging.getLogger(__name__)

class CommentPoster:
    """Post review comments to GitHub with rate limiting."""

    def __init__(self, github_client: GitHubAppClient):
        self.github = github_client
        self._rate_limit_delay = 0

    async def post_review(
        self,
        owner: str,
        repo: str,
        pr_number: int,
        installation_id: int,
        comments: List[ReviewComment],
        summary: str
    ) -> Dict:
        """
        Post a complete review with comments to GitHub.

        Args:
            owner: Repository owner
            repo: Repository name
            pr_number: Pull request number
            installation_id: GitHub App installation ID
            comments: List of review comments
            summary: Summary comment body

        Returns:
            Dict with posting results
        """
        results = {
            "comments_posted": 0,
            "comments_failed": 0,
            "errors": []
        }

        if not comments:
            logger.info("No comments to post")
            return results

        # Respect rate limits
        if self._rate_limit_delay > 0:
            logger.info(f"Waiting {self._rate_limit_delay}s for rate limit")
            await asyncio.sleep(self._rate_limit_delay)
            self._rate_limit_delay = 0

        try:
            # Post as a single review with all comments
            # This is more efficient and creates a proper review
            await self.github.create_review(
                owner=owner,
                repo=repo,
                pr_number=pr_number,
                installation_id=installation_id,
                comments=comments,
                body=summary,
                event="COMMENT"  # or "REQUEST_CHANGES" for blocking
            )

            results["comments_posted"] = len(comments)
            logger.info(f"Posted review with {len(comments)} comments")

        except RateLimitError as e:
            self._rate_limit_delay = e.retry_after
            results["errors"].append(f"Rate limited: retry after {e.retry_after}s")
            logger.warning(f"Rate limited, will retry after {e.retry_after}s")

        except Exception as e:
            # Fall back to posting comments individually
            logger.warning(f"Batch review failed, posting individually: {e}")

            for comment in comments:
                try:
                    await self._post_single_comment(
                        owner, repo, pr_number, installation_id, comment
                    )
                    results["comments_posted"] += 1

                    # Small delay between individual posts
                    await asyncio.sleep(0.5)

                except Exception as ce:
                    results["comments_failed"] += 1
                    results["errors"].append(str(ce))
                    logger.error(f"Failed to post comment: {ce}")

        return results

    async def _post_single_comment(
        self,
        owner: str,
        repo: str,
        pr_number: int,
        installation_id: int,
        comment: ReviewComment
    ):
        """Post a single review comment."""
        await self.github.create_review_comment(
            owner=owner,
            repo=repo,
            pr_number=pr_number,
            installation_id=installation_id,
```

```python
# review/orchestrator.py
import asyncio
from typing import List, Optional
import logging
from datetime import datetime

from config import get_settings
from models import (
    ReviewTask, ReviewStatus, Finding, FileDiff, ReviewComment
)
from github.client import GitHubAppClient
from analysis.diff_parser import DiffParser
from analysis.ast_analyzer import ASTAnalyzer
from analysis.security_scanner import SecurityScanner
from review.llm_reviewer import LLMReviewer
from review.comment_formatter import CommentFormatter
from review.comment_poster import CommentPoster
from queue.task_queue import task_queue

logger = logging.getLogger(__name__)

class ReviewOrchestrator:
    """
    Orchestrate the complete code review process.

    Coordinates:
    1. Fetching PR diff and file contents
    2. Running all analyzers
    3. Generating LLM-powered comments
    4. Posting review to GitHub
    """

    SUPPORTED_EXTENSIONS = {'.py'}  # Start with Python, extend later

    def __init__(self):
        self.settings = get_settings()
        self.github = GitHubAppClient()
        self.diff_parser = DiffParser()
        self.security_scanner = SecurityScanner()
        self.llm_reviewer = LLMReviewer()
        self.comment_formatter = CommentFormatter()
        self.comment_poster = CommentPoster(self.github)

    async def run_review(self, task_id: str) -> None:
        """
        Execute a complete code review for a PR.

        Args:
            task_id: ID of the review task to process
        """
        # Get task from queue
        task = await task_queue.get(task_id)
        if not task:
            logger.error(f"Task {task_id} not found")
            return

        try:
            # Update status
            task.status = ReviewStatus.PROCESSING
            await task_queue.update(task)

            logger.info(f"Starting review for PR #{task.pr_number} in {task.repository}")

            # Parse repository info
            owner, repo = task.repository.split('/')

            # Fetch PR files
            files = await self.github.get_pr_files(
                owner, repo, task.pr_number, task.installation_id
            )

            # Filter to supported files
            reviewable_files = self._filter_reviewable_files(files)

            if not reviewable_files:
                logger.info("No reviewable files in PR")
                task.status = ReviewStatus.COMPLETED
                task.completed_at = datetime.utcnow()
                await task_queue.update(task)
                return

            # Analyze each file
            all_findings: List[Finding] = []

            for file_diff in reviewable_files:
                findings = await self._analyze_file(
                    owner, repo, task.head_sha, task.installation_id, file_diff
                )
                all_findings.extend(findings)

            # Store findings
            task.findings = all_findings

            if not all_findings:
                logger.info("No issues found in PR")
                task.status = ReviewStatus.COMPLETED
                task.completed_at = datetime.utcnow()
                await task_queue.update(task)
                return

            # Generate and post review comments
            await self._generate_and_post_review(
                owner, repo, task.pr_number, task.installation_id,
                task.head_sha, all_findings, reviewable_files, len(reviewable_files)
            )

            # Mark complete
            task.status = ReviewStatus.COMPLETED
            task.completed_at = datetime.utcnow()
            task.comments_posted = len(all_findings)
            await task_queue.update(task)

            logger.info(
                f"Review complete for PR #{task.pr_number}: "
                f"{len(all_findings)} findings posted"
            )

        except Exception as e:
            logger.exception(f"Review failed for task {task_id}: {e}")
            task.status = ReviewStatus.FAILED
            task.error_message = str(e)
            task.completed_at = datetime.utcnow()
            await task_queue.update(task)

    def _filter_reviewable_files(self, files: List[FileDiff]) -> List[FileDiff]:
        """Filter files to those we can review."""
        reviewable = []

        for file in files:
            # Skip deleted files
            if file.status == 'removed':
                continue

            # Check extension
            ext = '.' + file.path.rsplit('.', 1)[-1] if '.' in file.path else ''
            if ext not in self.SUPPORTED_EXTENSIONS:
                continue

            # Skip files that are too large
            if file.additions + file.deletions > self.settings.max_file_size_kb * 10:
                logger.info(f"Skipping large file: {file.path}")
                continue

            reviewable.append(file)

        # Limit number of files
        return reviewable[:self.settings.max_files_per_review]

    async def _analyze_file(
        self,
        owner: str,
        repo: str,
        ref: str,
        installation_id: int,
        file_diff: FileDiff
    ) -> List[Finding]:
        """Analyze a single file for issues."""
        findings = []

        # Get full file content
        content = await self.github.get_file_content(
            owner, repo, file_diff.path, ref, installation_id
        )

        if not content:
            logger.warning(f"Could not fetch content for {file_diff.path}")
            return findings

        # Parse the diff to get changed lines
        parsed = self.diff_parser.parse_patch(
            file_diff.patch or '',
            file_diff.path,
            file_diff.status
        )

        # Only analyze changed lines
        changed_lines = parsed.added_lines

        if not changed_lines:
            return findings

        # Run AST analysis
        ast_analyzer = ASTAnalyzer(file_diff.path, changed_lines)
        findings.extend(ast_analyzer.analyze(content))

        # Run security scanning
        security_findings = self.security_scanner.scan(
            content, file_diff.path, changed_lines
        )
        findings.extend(security_findings)

        # Deduplicate findings on same line
        findings = self._deduplicate_findings(findings)

        return findings

    def _deduplicate_findings(self, findings: List[Finding]) -> List[Finding]:
        """Remove duplicate findings on the same line."""
        seen = set()
        unique = []

        for finding in findings:
            key = (finding.file_path, finding.line_number, finding.rule_id)
            if key not in seen:
                seen.add(key)
                unique.append(finding)

        return unique

    async def _generate_and_post_review(
        self,
        owner: str,
        repo: str,
        pr_number: int,
        installation_id: int,
        commit_id: str,
        findings: List[Finding],
        files: List[FileDiff],
        files_reviewed: int
    ) -> None:
        """Generate LLM comments and post review."""

        # Group findings by file for context
        findings_by_file: dict = {}
        for finding in findings:
            if finding.file_path not in findings_by_file:
                findings_by_file[finding.file_path] = []
            findings_by_file[finding.file_path].append(finding)

        # Generate comments for each file
        all_comments: List[ReviewComment] = []

        for file_path, file_findings in findings_by_file.items():
            # Get file content for context
            content = await self.github.get_file_content(
                owner, repo, file_path, commit_id, installation_id
            )

            if not content:
                continue

            # Generate LLM-powered comments
            llm_reviews = await self.llm_reviewer.generate_review_comments(
                file_findings, content, file_path
            )

            # Format as GitHub comments
            for review in llm_reviews:
                comment = self.comment_formatter.format_comment(
                    review['finding'],
                    review,
                    commit_id
                )
                all_comments.append(comment)

        # Generate summary
        summary = self.comment_formatter.format_summary_comment(
            findings, files_reviewed
        )

        # Post to GitHub
        await self.comment_poster.post_review(
            owner, repo, pr_number, installation_id,
            all_comments, summary
        )
```

### Step 10: Background Task Queue

We use an in-memory queue for simplicity, but this can be replaced with Redis or a database for production.

```python
# queue/task_queue.py
import asyncio
from typing import Dict, Optional
from datetime import datetime
import logging

from models import ReviewTask, ReviewStatus

logger = logging.getLogger(__name__)

class InMemoryTaskQueue:
    """
    Simple in-memory task queue for review tasks.

    For production, replace with Redis, PostgreSQL, or a proper
    message queue like RabbitMQ or AWS SQS.
    """

    def __init__(self):
        self._tasks: Dict[str, ReviewTask] = {}
        self._lock = asyncio.Lock()

    async def enqueue(self, task: ReviewTask) -> str:
        """Add a task to the queue."""
        async with self._lock:
            self._tasks[task.id] = task
            logger.info(f"Enqueued task {task.id}")
            return task.id

    async def get(self, task_id: str) -> Optional[ReviewTask]:
        """Get a task by ID."""
        return self._tasks.get(task_id)

    async def update(self, task: ReviewTask) -> None:
        """Update a task in the queue."""
        async with self._lock:
            self._tasks[task.id] = task

    async def get_pending(self) -> list:
        """Get all pending tasks."""
        return [
            t for t in self._tasks.values()
            if t.status == ReviewStatus.PENDING
        ]

    async def cleanup_old(self, max_age_hours: int = 24) -> int:
        """Remove completed tasks older than max_age_hours."""
        async with self._lock:
            now = datetime.utcnow()
            to_remove = []

            for task_id, task in self._tasks.items():
                if task.completed_at:
                    age = (now - task.completed_at).total_seconds() / 3600
                    if age > max_age_hours:
                        to_remove.append(task_id)

            for task_id in to_remove:
                del self._tasks[task_id]

            return len(to_remove)

# Global queue instance
task_queue = InMemoryTaskQueue()
```

### Step 12: Test Coverage Analyzer

Analyze whether new code has corresponding tests.

```python
# analysis/test_analyzer.py
import ast
import re
from typing import List, Dict, Set, Optional
from dataclasses import dataclass
import logging

from models import Finding, Severity, FindingCategory

logger = logging.getLogger(__name__)

@dataclass
class FunctionInfo:
    """Information about a function definition."""
    name: str
    line_number: int
    is_public: bool
    is_test: bool
    complexity: int

class TestCoverageAnalyzer:
    """
    Analyze test coverage for new code.

    Checks:
    - New public functions have corresponding tests
    - Test file exists for new modules
    - Test naming conventions are followed
    """

    TEST_FILE_PATTERNS = [
        r'test_.*\.py$',
        r'.*_test\.py$',
        r'tests/.*\.py$',
    ]

    def __init__(self):
        self.test_patterns = [re.compile(p) for p in self.TEST_FILE_PATTERNS]

    def analyze(
        self,
        source_code: str,
        file_path: str,
        changed_lines: Dict[int, str],
        all_pr_files: List[str]
    ) -> List[Finding]:
        """
        Analyze test coverage for changed code.

        Args:
            source_code: The source code content
            file_path: Path to the file
            changed_lines: Dict of changed line numbers
            all_pr_files: List of all files in the PR

        Returns:
            List of findings about test coverage
        """
        findings = []

        # Skip if this is a test file
        if self._is_test_file(file_path):
            return findings

        # Parse the source code
        try:
            tree = ast.parse(source_code)
        except SyntaxError:
            return findings

        # Find new public functions
        new_functions = self._find_new_functions(tree, changed_lines)

        if not new_functions:
            return findings

        # Check if corresponding test file exists in PR
        test_file = self._find_test_file(file_path, all_pr_files)

        if not test_file:
            # No test file in PR - suggest adding tests
            for func in new_functions:
                if func.is_public and not func.is_test:
                    findings.append(Finding(
                        file_path=file_path,
                        line_number=func.line_number,
                        severity=Severity.MEDIUM,
                        category=FindingCategory.TEST,
                        title="New function without tests",
                        description=f"Function `{func.name}` was added but no "
                                   f"corresponding test file was found in this PR.",
                        suggestion=f"Consider adding tests in `test_{file_path.split('/')[-1]}` "
                                  f"or `{file_path.replace('.py', '_test.py')}`",
                        rule_id="TEST001"
                    ))

        return findings

    def _is_test_file(self, file_path: str) -> bool:
        """Check if a file is a test file."""
        return any(p.search(file_path) for p in self.test_patterns)

    def _find_new_functions(
        self,
        tree: ast.AST,
        changed_lines: Dict[int, str]
    ) -> List[FunctionInfo]:
        """Find functions that were added or modified."""
        functions = []

        for node in ast.walk(tree):
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                # Check if function definition line is in changed lines
                if node.lineno in changed_lines:
                    functions.append(FunctionInfo(
                        name=node.name,
                        line_number=node.lineno,
                        is_public=not node.name.startswith('_'),
                        is_test=node.name.startswith('test_'),
                        complexity=self._calculate_complexity(node)
                    ))

        return functions

    def _find_test_file(
        self,
        source_file: str,
        all_files: List[str]
    ) -> Optional[str]:
        """Find corresponding test file in PR files."""
        # Generate possible test file names
        base_name = source_file.split('/')[-1].replace('.py', '')
        possible_names = [
            f'test_{base_name}.py',
            f'{base_name}_test.py',
            f'tests/test_{base_name}.py',
            f'tests/{base_name}_test.py',
        ]

        for test_name in possible_names:
            for pr_file in all_files:
                if pr_file.endswith(test_name):
                    return pr_file

        return None

    def _calculate_complexity(self, node: ast.AST) -> int:
        """Calculate cyclomatic complexity."""
        complexity = 1
        for child in ast.walk(node):
            if isinstance(child, (ast.If, ast.While, ast.For, ast.ExceptHandler)):
                complexity += 1
            elif isinstance(child, ast.BoolOp):
                complexity += len(child.values) - 1
        return complexity
```
