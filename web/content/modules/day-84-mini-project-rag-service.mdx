## Project Overview

**Objective**: Build a document processing pipeline that chunks, embeds, and stores documents for semantic retrieval. Supports PDF, Markdown, and text files with citation tracking.

**Why This Matters**: RAG (Retrieval-Augmented Generation) is the standard for connecting LLMs to private data. A dedicated service allows multiple agents to share a common knowledge base.

**Skills Showcased**:

- Document chunking strategies
- Vector database integration (ChromaDB)
- Hybrid search (Semantic + Keyword)

## Detailed Requirements Specification

### Functional Requirements (FR)

| ID    | Requirement                     | Description                                                                                                                                 | Priority |
| ----- | ------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------------- | -------- |
| FR-01 | Multi-Format Document Ingestion | The system shall accept and parse PDF, Markdown (.md), and plain text (.txt) files                                                          | P0       |
| FR-02 | Intelligent Document Chunking   | The system shall split documents using RecursiveCharacterTextSplitter with configurable chunk size (default 1000) and overlap (default 200) | P0       |
| FR-03 | Vector Embedding Storage        | The system shall generate embeddings using sentence-transformers and store them in ChromaDB with cosine similarity metric                   | P0       |
| FR-04 | Semantic Search                 | The system shall retrieve document chunks based on semantic similarity to a query with configurable top_k results                           | P0       |
| FR-05 | Hybrid Search                   | The system shall combine semantic (vector) search with BM25 keyword search using configurable alpha weighting (default 0.7 semantic)        | P1       |
| FR-06 | Metadata Filtering              | The system shall support filtering search results by user_id and custom metadata fields                                                     | P1       |
| FR-07 | Document Deletion               | The system shall support deleting all chunks associated with a document by doc_id                                                           | P1       |
| FR-08 | Citation Tracking               | Each chunk shall maintain source attribution including filename, page number, and chunk index                                               | P0       |

```python
# services/document_processor.py
from langchain_community.document_loaders import PyPDFLoader, TextLoader, UnstructuredMarkdownLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from sentence_transformers import SentenceTransformer
import chromadb
from pathlib import Path
from typing import List, Dict
import hashlib
import uuid

# Note: For LangChain integration with ChromaDB, install langchain-chroma:
# pip install langchain-chroma
# Then use: from langchain_chroma import Chroma

class DocumentProcessor:
    """Process and store documents for RAG."""

    CHUNK_SIZE = 1000
    CHUNK_OVERLAP = 200

    def __init__(self, collection_name: str = "documents"):
        self.embedder = SentenceTransformer("all-MiniLM-L6-v2")
        self.client = chromadb.PersistentClient(path="./chroma_db")
        self.collection = self.client.get_or_create_collection(
            name=collection_name,
            metadata={"hnsw:space": "cosine"}
        )

        self.splitter = RecursiveCharacterTextSplitter(
            chunk_size=self.CHUNK_SIZE,
            chunk_overlap=self.CHUNK_OVERLAP,
            separators=["\n\n", "\n", ". ", " ", ""]
        )

    def process_document(
        self,
        file_path: str,
        user_id: str,
        metadata: Dict = None
    ) -> Dict:
        """Process and store a document."""
        path = Path(file_path)

        # Load document
        if path.suffix == ".pdf":
            loader = PyPDFLoader(str(path))
        elif path.suffix == ".md":
            loader = UnstructuredMarkdownLoader(str(path))
        else:
            loader = TextLoader(str(path))

        documents = loader.load()

        # Generate document ID
        content_hash = hashlib.md5(
            "".join(d.page_content for d in documents).encode()
        ).hexdigest()[:12]
        doc_id = f"{user_id}_{content_hash}"

        # Chunk documents
        chunks = self.splitter.split_documents(documents)

        # Prepare for storage
        ids = []
        texts = []
        metadatas = []
        embeddings = []

        for i, chunk in enumerate(chunks):
            chunk_id = f"{doc_id}_chunk_{i}"
            ids.append(chunk_id)
            texts.append(chunk.page_content)

            chunk_metadata = {
                "doc_id": doc_id,
                "user_id": user_id,
                "source": path.name,
                "chunk_index": i,
                "page": chunk.metadata.get("page", 0),
                **(metadata or {})
            }
            metadatas.append(chunk_metadata)

            embedding = self.embedder.encode(chunk.page_content).tolist()
            embeddings.append(embedding)

        # Store in ChromaDB
        self.collection.add(
            ids=ids,
            documents=texts,
            metadatas=metadatas,
            embeddings=embeddings
        )

        return {
            "doc_id": doc_id,
            "chunks": len(chunks),
            "source": path.name
        }

    def query(
        self,
        query: str,
        user_id: str = None,
        top_k: int = 5,
        filter_metadata: Dict = None
    ) -> List[Dict]:
        """Query documents with semantic search."""

        # Build filter
        where_filter = {}
        if user_id:
            where_filter["user_id"] = user_id
        if filter_metadata:
            where_filter.update(filter_metadata)

        # Generate query embedding
        query_embedding = self.embedder.encode(query).tolist()

        # Search
        results = self.collection.query(
            query_embeddings=[query_embedding],
            n_results=top_k,
            where=where_filter if where_filter else None,
            include=["documents", "metadatas", "distances"]
        )

        # Format results
        formatted = []
        for i in range(len(results["ids"][0])):
            formatted.append({
                "id": results["ids"][0][i],
                "content": results["documents"][0][i],
                "metadata": results["metadatas"][0][i],
                "score": 1 - results["distances"][0][i]  # Convert distance to similarity
            })

        return formatted

    def delete_document(self, doc_id: str):
        """Delete all chunks of a document."""
        self.collection.delete(where={"doc_id": doc_id})
```

```python
# services/hybrid_search.py
from rank_bm25 import BM25Okapi
import numpy as np
from typing import List, Dict

class HybridRetriever:
    """Combine semantic and keyword search."""

    def __init__(self, processor: DocumentProcessor, alpha: float = 0.7):
        self.processor = processor
        self.alpha = alpha  # Weight for semantic search

    def search(
        self,
        query: str,
        user_id: str,
        top_k: int = 5
    ) -> List[Dict]:
        """Hybrid search combining semantic and BM25."""

        # Semantic search
        semantic_results = self.processor.query(query, user_id, top_k=top_k * 2)

        # BM25 keyword search
        documents = [r["content"] for r in semantic_results]
        tokenized_docs = [doc.lower().split() for doc in documents]
        bm25 = BM25Okapi(tokenized_docs)

        tokenized_query = query.lower().split()
        bm25_scores = bm25.get_scores(tokenized_query)

        # Normalize scores
        semantic_scores = np.array([r["score"] for r in semantic_results])
        semantic_scores = (semantic_scores - semantic_scores.min()) / (semantic_scores.max() - semantic_scores.min() + 1e-6)

        bm25_scores = (bm25_scores - bm25_scores.min()) / (bm25_scores.max() - bm25_scores.min() + 1e-6)

        # Combine scores
        combined_scores = self.alpha * semantic_scores + (1 - self.alpha) * bm25_scores

        # Re-rank
        ranked_indices = np.argsort(combined_scores)[::-1][:top_k]

        return [
            {**semantic_results[i], "combined_score": float(combined_scores[i])}
            for i in ranked_indices
        ]
```

# routers/rag.py
from fastapi import APIRouter, UploadFile, File, Depends, HTTPException
from services.document_processor import DocumentProcessor
from services.hybrid_search import HybridRetriever
import tempfile
import os

router = APIRouter(prefix="/rag", tags=["rag"])
processor = DocumentProcessor()
retriever = HybridRetriever(processor)

@router.post("/{user_id}/documents")
async def upload_document(
    user_id: str,
    file: UploadFile = File(...)
):
    """Upload and process a document."""
    # Save temporarily
    with tempfile.NamedTemporaryFile(delete=False, suffix=file.filename) as tmp:
        content = await file.read()
        tmp.write(content)
        tmp_path = tmp.name

    try:
        result = processor.process_document(tmp_path, user_id)
        return result
    finally:
        os.unlink(tmp_path)

@router.post("/{user_id}/query")
def query_documents(
    user_id: str,
    query: str,
    top_k: int = 5,
    hybrid: bool = True
):
    """Query documents."""
    if hybrid:
        return retriever.search(query, user_id, top_k)
    return processor.query(query, user_id, top_k)

@router.delete("/{user_id}/documents/{doc_id}")
def delete_document(user_id: str, doc_id: str):
    """Delete a document."""
    processor.delete_document(doc_id)
    return {"status": "deleted"}

# Example: RAG-powered agent
class RAGAgent:
    def __init__(self, user_id: str):
        self.user_id = user_id
        self.retriever = HybridRetriever(DocumentProcessor())
        self.llm = ChatOpenAI(model="gpt-4o-mini")

    async def answer(self, question: str) -> str:
        # Retrieve relevant chunks
        chunks = self.retriever.search(question, self.user_id, top_k=5)

        # Build context with citations
        context = "\n\n".join([
            f"[{i+1}] {c['content']}\nSource: {c['metadata']['source']}, Page {c['metadata']['page']}"
            for i, c in enumerate(chunks)
        ])

        prompt = f"""Answer based on the provided context. Cite sources using [1], [2], etc.

Context:
{context}

Question: {question}

If the context doesn't contain the answer, say so."""

        response = await self.llm.ainvoke(prompt)
        return response.content
