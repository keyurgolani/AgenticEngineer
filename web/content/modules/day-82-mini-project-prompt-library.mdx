## Project Overview

**Objective**: Build a prompt library service that stores, versions, and serves prompts for your agent fleet. This becomes the "prompt brain" of your agentic ecosystem.

**Why This Matters**: Production agent systems need centralized prompt management. Hardcoded prompts become unmaintainable. This service enables A/B testing, rollbacks, and team collaboration on prompts.

## Detailed Requirements Specification

### Functional Requirements (FR)

| ID    | Requirement                    | Description                                                                                                                           | Priority |
| ----- | ------------------------------ | ------------------------------------------------------------------------------------------------------------------------------------- | -------- |
| FR-01 | Prompt CRUD Operations         | The system shall support Create, Read, Update, and Delete operations for prompts with unique name identifiers                         | P0       |
| FR-02 | Template Variable Substitution | The system shall support Jinja2-style template variables (e.g., `{{ variable }}`) with filters (`upper`, `lower`, `truncate`, `json`) | P0       |
| FR-03 | Automatic Variable Extraction  | The system shall automatically extract variable names from prompt templates when not explicitly provided                              | P0       |
| FR-04 | Prompt Versioning              | Each prompt update shall create a new version, preserving the complete history of all previous versions                               | P0       |
| FR-05 | Version Rollback               | The system shall allow retrieval of any historical version of a prompt by specifying the version number                               | P1       |
| FR-06 | Category and Tag Filtering     | The system shall support filtering prompts by category (single value) and tags (multiple values, OR logic)                            | P1       |
| FR-07 | Full-Text Search               | The system shall support searching prompts by name and description using case-insensitive partial matching                            | P1       |
| FR-08 | Usage Analytics                | The system shall record prompt usage including agent ID, tokens consumed, latency, and success/failure status                         | P2       |

### Non-Functional Requirements (NFR)

| ID     | Requirement         | Target                                                                                     | Measurement                      |
| ------ | ------------------- | ------------------------------------------------------------------------------------------ | -------------------------------- |
| NFR-01 | Response Latency    | Prompt retrieval and rendering shall complete in < 50ms (p95)                              | API response time monitoring     |
| NFR-02 | Availability        | Service shall maintain 99.9% uptime                                                        | Health check endpoint monitoring |
| NFR-03 | Data Integrity      | All prompt versions shall be immutable once created; updates create new versions           | Database constraints             |
| NFR-04 | Concurrent Access   | Service shall handle 100+ concurrent prompt render requests                                | Load testing                     |
| NFR-05 | Template Validation | Invalid Jinja2 templates shall be rejected at creation/update time with descriptive errors | Unit tests                       |

```python
# models/database.py
from sqlalchemy import create_engine, Column, Integer, String, Text, DateTime, JSON, ForeignKey, Boolean
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, relationship
from datetime import datetime
import os

DATABASE_URL = os.getenv("DATABASE_URL", "sqlite:///./prompts.db")

engine = create_engine(DATABASE_URL, connect_args={"check_same_thread": False} if "sqlite" in DATABASE_URL else {})
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

class Prompt(Base):
    """Main prompt storage."""
    __tablename__ = "prompts"

    id = Column(Integer, primary_key=True, index=True)
    name = Column(String(255), unique=True, index=True, nullable=False)
    description = Column(Text)
    category = Column(String(100), index=True)  # research, code, analysis, etc.

    # The actual prompt content
    system_prompt = Column(Text, nullable=False)
    user_prompt_template = Column(Text)  # With {variable} placeholders

    # Metadata
    variables = Column(JSON, default=list)  # List of required variables
    tags = Column(JSON, default=list)
    model_recommendation = Column(String(100))  # gpt-4o, claude-3, etc.
    temperature = Column(String(10), default="0.7")

    # Versioning
    version = Column(Integer, default=1)
    is_active = Column(Boolean, default=True)

    # Timestamps
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    created_by = Column(String(100))

    # Relationships
    versions = relationship("PromptVersion", back_populates="prompt")

class PromptVersion(Base):
    """Version history for prompts."""
    __tablename__ = "prompt_versions"

    id = Column(Integer, primary_key=True, index=True)
    prompt_id = Column(Integer, ForeignKey("prompts.id"), nullable=False)
    version = Column(Integer, nullable=False)

    system_prompt = Column(Text, nullable=False)
    user_prompt_template = Column(Text)
    variables = Column(JSON, default=list)

    change_notes = Column(Text)
    created_at = Column(DateTime, default=datetime.utcnow)
    created_by = Column(String(100))

    prompt = relationship("Prompt", back_populates="versions")

class PromptUsage(Base):
    """Track prompt usage for analytics."""
    __tablename__ = "prompt_usage"

    id = Column(Integer, primary_key=True, index=True)
    prompt_id = Column(Integer, ForeignKey("prompts.id"), nullable=False)
    prompt_version = Column(Integer)

    agent_id = Column(String(100))
    variables_used = Column(JSON)

    # Performance tracking
    tokens_used = Column(Integer)
    latency_ms = Column(Integer)
    success = Column(Boolean)

    created_at = Column(DateTime, default=datetime.utcnow)

# Create tables
Base.metadata.create_all(bind=engine)

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()
```

```python
# services/template_engine.py
from jinja2 import Environment, BaseLoader, TemplateSyntaxError, UndefinedError
from typing import Dict, Any, List, Set
import re

class PromptTemplateEngine:
    """Renders prompt templates with variables."""

    def __init__(self):
        self.env = Environment(
            loader=BaseLoader(),
            autoescape=False,  # Don't escape for prompts
            undefined=StrictUndefined
        )

        # Add custom filters
        self.env.filters['upper'] = str.upper
        self.env.filters['lower'] = str.lower
        self.env.filters['title'] = str.title
        self.env.filters['truncate'] = self._truncate
        self.env.filters['json'] = self._to_json

    def _truncate(self, text: str, length: int = 100) -> str:
        """Truncate text to specified length."""
        if len(text) <= length:
            return text
        return text[:length-3] + "..."

    def _to_json(self, obj: Any) -> str:
        """Convert object to JSON string."""
        import json
        return json.dumps(obj, indent=2)

    def extract_variables(self, template: str) -> List[str]:
        """Extract variable names from a template."""
        # Match {{ variable }} and {{ variable | filter }}
        pattern = r'\{\{\s*(\w+)(?:\s*\|[^}]+)?\s*\}\}'
        matches = re.findall(pattern, template)
        return list(set(matches))

    def validate_template(self, template: str) -> tuple[bool, str]:
        """Validate template syntax."""
        try:
            self.env.from_string(template)
            return True, "Valid"
        except TemplateSyntaxError as e:
            return False, f"Syntax error: {e.message}"

    def render(
        self,
        template: str,
        variables: Dict[str, Any],
        strict: bool = True
    ) -> str:
        """
        Render a template with variables.

        Args:
            template: The template string
            variables: Dictionary of variable values
            strict: If True, raise error for missing variables
        """
        try:
            tmpl = self.env.from_string(template)
            return tmpl.render(**variables)
        except UndefinedError as e:
            if strict:
                raise ValueError(f"Missing variable: {e}")
            # Return template with missing vars as placeholders
            return template

    def render_prompt(
        self,
        system_prompt: str,
        user_template: str | None,
        variables: Dict[str, Any]
    ) -> tuple[str, str | None]:
        """Render both system and user prompts."""
        rendered_system = self.render(system_prompt, variables)
        rendered_user = None

        if user_template:
            rendered_user = self.render(user_template, variables)

        return rendered_system, rendered_user

class StrictUndefined:
    """Raises error for undefined variables."""
    def __init__(self, name):
        self.name = name

    def __str__(self):
        raise UndefinedError(f"Variable '{self.name}' is not defined")

    def __repr__(self):
        return self.__str__()
```

```python
# services/prompt_service.py
from sqlalchemy.orm import Session
from models.database import Prompt, PromptVersion, PromptUsage
from models.schemas import PromptCreate, PromptUpdate, RenderedPrompt
from services.template_engine import PromptTemplateEngine
from typing import List, Optional, Dict, Any
from datetime import datetime

class PromptService:
    """Business logic for prompt management."""

    def __init__(self, db: Session):
        self.db = db
        self.template_engine = PromptTemplateEngine()

    def create_prompt(self, prompt_data: PromptCreate) -> Prompt:
        """Create a new prompt."""
        # Validate templates
        is_valid, error = self.template_engine.validate_template(prompt_data.system_prompt)
        if not is_valid:
            raise ValueError(f"Invalid system prompt: {error}")

        if prompt_data.user_prompt_template:
            is_valid, error = self.template_engine.validate_template(prompt_data.user_prompt_template)
            if not is_valid:
                raise ValueError(f"Invalid user prompt template: {error}")

        # Auto-extract variables if not provided
        variables = prompt_data.variables
        if not variables:
            variables = self.template_engine.extract_variables(prompt_data.system_prompt)
            if prompt_data.user_prompt_template:
                variables.extend(self.template_engine.extract_variables(prompt_data.user_prompt_template))
            variables = list(set(variables))

        prompt = Prompt(
            name=prompt_data.name,
            description=prompt_data.description,
            category=prompt_data.category,
            system_prompt=prompt_data.system_prompt,
            user_prompt_template=prompt_data.user_prompt_template,
            variables=variables,
            tags=prompt_data.tags,
            model_recommendation=prompt_data.model_recommendation,
            temperature=prompt_data.temperature,
            created_by=prompt_data.created_by,
            version=1
        )

        self.db.add(prompt)
        self.db.commit()
        self.db.refresh(prompt)

        # Create initial version
        self._create_version(prompt, "Initial version", prompt_data.created_by)

        return prompt

    def update_prompt(self, prompt_id: int, update_data: PromptUpdate) -> Prompt:
        """Update a prompt and create new version."""
        prompt = self.db.query(Prompt).filter(Prompt.id == prompt_id).first()
        if not prompt:
            raise ValueError(f"Prompt {prompt_id} not found")

        # Update fields
        update_dict = update_data.model_dump(exclude_unset=True, exclude={'change_notes', 'updated_by'})

        for field, value in update_dict.items():
            setattr(prompt, field, value)

        # Increment version
        prompt.version += 1
        prompt.updated_at = datetime.utcnow()

        self.db.commit()
        self.db.refresh(prompt)

        # Create version record
        self._create_version(prompt, update_data.change_notes, update_data.updated_by)

        return prompt

    def _create_version(self, prompt: Prompt, notes: str | None, created_by: str | None):
        """Create a version snapshot."""
        version = PromptVersion(
            prompt_id=prompt.id,
            version=prompt.version,
            system_prompt=prompt.system_prompt,
            user_prompt_template=prompt.user_prompt_template,
            variables=prompt.variables,
            change_notes=notes,
            created_by=created_by
        )
        self.db.add(version)
        self.db.commit()

    def get_prompt(self, name: str, version: int | None = None) -> Prompt | None:
        """Get a prompt by name, optionally at specific version."""
        prompt = self.db.query(Prompt).filter(Prompt.name == name).first()

        if not prompt:
            return None

        if version and version != prompt.version:
            # Get historical version
            hist = self.db.query(PromptVersion).filter(
                PromptVersion.prompt_id == prompt.id,
                PromptVersion.version == version
            ).first()

            if hist:
                # Return prompt with historical content
                prompt.system_prompt = hist.system_prompt
                prompt.user_prompt_template = hist.user_prompt_template
                prompt.variables = hist.variables
                prompt.version = hist.version

        return prompt

    def render_prompt(
        self,
        name: str,
        variables: Dict[str, Any],
        version: int | None = None
    ) -> RenderedPrompt:
        """Render a prompt with variables."""
        prompt = self.get_prompt(name, version)
        if not prompt:
            raise ValueError(f"Prompt '{name}' not found")

        # Render templates
        system, user = self.template_engine.render_prompt(
            prompt.system_prompt,
            prompt.user_prompt_template,
            variables
        )

        return RenderedPrompt(
            system_prompt=system,
            user_prompt=user,
            model_recommendation=prompt.model_recommendation or "gpt-4o",
            temperature=float(prompt.temperature),
            prompt_id=prompt.id,
            prompt_version=prompt.version
        )

    def list_prompts(
        self,
        category: str | None = None,
        tags: List[str] | None = None,
        search: str | None = None
    ) -> List[Prompt]:
        """List prompts with optional filters."""
        query = self.db.query(Prompt).filter(Prompt.is_active == True)

        if category:
            query = query.filter(Prompt.category == category)

        if search:
            query = query.filter(
                Prompt.name.ilike(f"%{search}%") |
                Prompt.description.ilike(f"%{search}%")
            )

        prompts = query.all()

        # Filter by tags (JSON field)
        if tags:
            prompts = [p for p in prompts if any(t in p.tags for t in tags)]

        return prompts

    def get_versions(self, prompt_id: int) -> List[PromptVersion]:
        """Get version history for a prompt."""
        return self.db.query(PromptVersion).filter(
            PromptVersion.prompt_id == prompt_id
        ).order_by(PromptVersion.version.desc()).all()

    def record_usage(
        self,
        prompt_id: int,
        prompt_version: int,
        agent_id: str,
        variables: Dict[str, Any],
        tokens: int | None = None,
        latency_ms: int | None = None,
        success: bool = True
    ):
        """Record prompt usage for analytics."""
        usage = PromptUsage(
            prompt_id=prompt_id,
            prompt_version=prompt_version,
            agent_id=agent_id,
            variables_used=variables,
            tokens_used=tokens,
            latency_ms=latency_ms,
            success=success
        )
        self.db.add(usage)
        self.db.commit()
```

```python
# client/prompt_client.py
import httpx
from typing import Dict, Any, Optional
from pydantic import BaseModel

class RenderedPrompt(BaseModel):
    system_prompt: str
    user_prompt: Optional[str]
    model_recommendation: str
    temperature: float
    prompt_id: int
    prompt_version: int

class PromptLibraryClient:
    """Client for the Prompt Library Service."""

    def __init__(self, base_url: str = "http://localhost:8001"):
        self.base_url = base_url
        self.client = httpx.Client(timeout=10.0)

    def get_prompt(
        self,
        name: str,
        variables: Dict[str, Any] = {},
        version: Optional[int] = None
    ) -> RenderedPrompt:
        """Get and render a prompt."""
        params = {}
        if version:
            params["version"] = version

        response = self.client.post(
            f"{self.base_url}/prompts/{name}/render",
            params=params,
            json={"variables": variables}
        )
        response.raise_for_status()
        return RenderedPrompt(**response.json())

    def list_prompts(
        self,
        category: Optional[str] = None,
        tags: Optional[list] = None
    ) -> list:
        """List available prompts."""
        params = {}
        if category:
            params["category"] = category
        if tags:
            params["tags"] = ",".join(tags)

        response = self.client.get(
            f"{self.base_url}/prompts/",
            params=params
        )
        response.raise_for_status()
        return response.json()

    def record_usage(
        self,
        prompt_id: int,
        prompt_version: int,
        agent_id: str,
        tokens: Optional[int] = None,
        latency_ms: Optional[int] = None,
        success: bool = True
    ):
        """Record prompt usage."""
        self.client.post(
            f"{self.base_url}/prompts/{prompt_id}/usage",
            params={
                "agent_id": agent_id,
                "prompt_version": prompt_version,
                "tokens": tokens,
                "latency_ms": latency_ms,
                "success": success
            }
        )
```

```python
from langchain_openai import ChatOpenAI
from client.prompt_client import PromptLibraryClient

class ResearchAgent:
    def __init__(self):
        self.prompt_client = PromptLibraryClient()

    async def research(self, topic: str) -> str:
        # Get prompt from library
        prompt = self.prompt_client.get_prompt(
            "research_analyst",
            variables={"topic": topic, "depth": "comprehensive"}
        )

        # Use the prompt
        llm = ChatOpenAI(
            model=prompt.model_recommendation,
            temperature=prompt.temperature

        messages = [
            {"role": "system", "content": prompt.system_prompt},
            {"role": "user", "content": prompt.user_prompt}

        # Record usage
        self.prompt_client.record_usage(
            prompt_id=prompt.prompt_id,
            prompt_version=prompt.prompt_version,
            agent_id="research-agent-1"
        )

        return response.content
```
