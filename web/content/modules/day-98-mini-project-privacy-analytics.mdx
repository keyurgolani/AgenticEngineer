## Project Overview

**Objective**: Build a Privacy-Preserving Analytics Agent that enables data analysis on sensitive datasets (healthcare, finance) while providing mathematical privacy guarantees through differential privacy, schema-only analysis, and comprehensive audit logging.

**Why This Matters**: Organizations hold vast amounts of sensitive data that could yield valuable insights, but privacy regulations (GDPR, HIPAA, CCPA) and ethical considerations restrict direct access. A privacy-preserving analytics system allows data scientists to extract aggregate insights without exposing individual records—enabling innovation while protecting privacy.

```python
"""Privacy budget models."""

from pydantic import BaseModel, Field
from datetime import datetime
from typing import Optional


class PrivacyBudget(BaseModel):
    """User's privacy budget state."""
    user_id: str
    total_budget: float = Field(default=10.0)
    used_budget: float = Field(default=0.0)
    queries_count: int = Field(default=0)
    created_at: datetime = Field(default_factory=datetime.utcnow)
    last_query_at: Optional[datetime] = None

    @property
    def remaining_budget(self) -> float:
        """Calculate remaining budget."""
        return max(0.0, self.total_budget - self.used_budget)

    def can_spend(self, epsilon: float) -> bool:
        """Check if budget allows spending epsilon."""
        return self.remaining_budget >= epsilon

    def spend(self, epsilon: float) -> None:
        """Deduct epsilon from budget."""
        if not self.can_spend(epsilon):
            raise ValueError(f"Insufficient budget: {self.remaining_budget} < {epsilon}")
        self.used_budget += epsilon
        self.queries_count += 1
        self.last_query_at = datetime.utcnow()


class BudgetCheckResponse(BaseModel):
    """Response for budget check endpoint."""
    user_id: str
    budget: dict
    queries_executed: int
    last_query_at: Optional[datetime]
```

```python
# services/differential_privacy.py
"""Differential privacy engine implementing the Laplace mechanism."""

import math
import logging
from typing import Tuple, Optional
from dataclasses import dataclass
import numpy as np

from models.query import QueryType, ConfidenceInterval

logger = logging.getLogger(__name__)


@dataclass
class SensitivityConfig:
    """Configuration for sensitivity calculation."""
    # For bounded queries, these define the data range
    lower_bound: Optional[float] = None
    upper_bound: Optional[float] = None
    # For COUNT queries
    max_contributions_per_user: int = 1


class DifferentialPrivacyEngine:
    """
    Implements differential privacy using the Laplace mechanism.

    The Laplace mechanism adds noise drawn from Laplace(0, sensitivity/epsilon)
    to query results, providing ε-differential privacy.
    """

    def __init__(self, seed: Optional[int] = None):
        """Initialize with optional random seed for reproducibility in tests."""
        self.rng = np.random.default_rng(seed)

    def add_noise(
        self,
        true_value: float,
        sensitivity: float,
        epsilon: float,
        lower_bound: Optional[float] = None,
        upper_bound: Optional[float] = None
    ) -> Tuple[float, ConfidenceInterval]:
        """
        Add Laplace noise to achieve ε-differential privacy.

        Args:
            true_value: The actual query result
            sensitivity: Query sensitivity (max change from one record)
            epsilon: Privacy parameter (smaller = more private)
            lower_bound: Optional lower bound for result clamping
            upper_bound: Optional upper bound for result clamping

        Returns:
            Tuple of (noisy_value, confidence_interval)
        """
        if epsilon <= 0:
            raise ValueError(f"Epsilon must be positive, got {epsilon}")
        if sensitivity < 0:
            raise ValueError(f"Sensitivity must be non-negative, got {sensitivity}")

        # Calculate scale parameter for Laplace distribution
        scale = sensitivity / epsilon

        # Sample noise from Laplace distribution
        noise = self.rng.laplace(0, scale)
        noisy_value = true_value + noise

        # Clamp to valid range if bounds provided
        if lower_bound is not None:
            noisy_value = max(lower_bound, noisy_value)
        if upper_bound is not None:
            noisy_value = min(upper_bound, noisy_value)

        # Calculate confidence interval
        confidence_interval = self._calculate_confidence_interval(
            noisy_value, scale, lower_bound, upper_bound
        )

        logger.debug(
            f"DP: true={true_value:.4f}, noise={noise:.4f}, "
            f"noisy={noisy_value:.4f}, scale={scale:.4f}"
        )

        return noisy_value, confidence_interval

    def _calculate_confidence_interval(
        self,
        noisy_value: float,
        scale: float,
        lower_bound: Optional[float],
        upper_bound: Optional[float],
        confidence: float = 0.95
    ) -> ConfidenceInterval:
        """
        Calculate confidence interval for the noisy result.

        For Laplace distribution, the 95% CI is approximately ±3*scale.
        """
        # For Laplace, P(|X| < k*scale) = 1 - exp(-k)
        # For 95% confidence: k = -ln(0.05/2) ≈ 2.996
        k = -math.log((1 - confidence) / 2)
        margin = k * scale

        lower = noisy_value - margin
        upper = noisy_value + margin

        # Apply bounds if provided
        if lower_bound is not None:
            lower = max(lower_bound, lower)
        if upper_bound is not None:
            upper = min(upper_bound, upper)

        return ConfidenceInterval(
            lower=lower,
            upper=upper,
            confidence=confidence
        )

    def calculate_sensitivity(
        self,
        query_type: QueryType,
        config: SensitivityConfig
    ) -> float:
        """
        Calculate the sensitivity for a given query type.

        Sensitivity is the maximum change in query output when
        a single record is added or removed.
        """
        if query_type == QueryType.COUNT:
            # Adding/removing one record changes count by at most 1
            return float(config.max_contributions_per_user)

        elif query_type == QueryType.SUM:
            # Sensitivity depends on the range of values
            if config.lower_bound is None or config.upper_bound is None:
                raise ValueError("SUM queries require bounded data range")
            return config.upper_bound - config.lower_bound

        elif query_type == QueryType.AVG:
            # For average, sensitivity = range / n, but we use range for safety
            # In practice, use bounded average mechanisms
            if config.lower_bound is None or config.upper_bound is None:
                raise ValueError("AVG queries require bounded data range")
            return config.upper_bound - config.lower_bound

        elif query_type in (QueryType.MIN, QueryType.MAX):
            # MIN/MAX have high sensitivity in general
            # We use the full range as a conservative estimate
            if config.lower_bound is None or config.upper_bound is None:
                raise ValueError("MIN/MAX queries require bounded data range")
            return config.upper_bound - config.lower_bound

        else:
            raise ValueError(f"Unsupported query type: {query_type}")


# Global instance
dp_engine = DifferentialPrivacyEngine()
```

**Key Implementation Details**:

- **Laplace Mechanism**: The core of differential privacy—noise is drawn from Laplace(0, sensitivity/ε)
- **Sensitivity Calculation**: Different query types have different sensitivities; COUNT has sensitivity 1, while SUM depends on data range
- **Result Clamping**: After adding noise, results are clamped to valid ranges (e.g., counts can't be negative)
- **Confidence Intervals**: Users see uncertainty bounds to understand result reliability
- **Reproducibility**: Optional seed for deterministic testing

### Privacy Budget Manager

```python
# services/privacy_budget.py
"""Privacy budget tracking and enforcement."""

import logging
from typing import Dict, Optional
from datetime import datetime, timedelta
from threading import Lock

from models.budget import PrivacyBudget
from config import get_settings

logger = logging.getLogger(__name__)


class BudgetExhaustedError(Exception):
    """Raised when privacy budget is exhausted."""
    def __init__(self, user_id: str, requested: float, remaining: float):
        self.user_id = user_id
        self.requested = requested
        self.remaining = remaining
        super().__init__(
            f"Budget exhausted for user {user_id}: "
            f"requested {requested}, remaining {remaining}"
        )


class PrivacyBudgetManager:
    """
    Manages privacy budgets for users and globally.

    Privacy budget (epsilon) is a finite resource. Each query consumes
    some epsilon, and once exhausted, no more queries are allowed.
    This prevents privacy degradation through repeated queries.
    """

    def __init__(self):
        self.settings = get_settings()
        self._user_budgets: Dict[str, PrivacyBudget] = {}
        self._global_used_today: float = 0.0
        self._global_reset_date: datetime = datetime.utcnow().date()
        self._lock = Lock()

    def get_or_create_budget(self, user_id: str) -> PrivacyBudget:
        """Get existing budget or create new one for user."""
        with self._lock:
            if user_id not in self._user_budgets:
                self._user_budgets[user_id] = PrivacyBudget(
                    user_id=user_id,
                    total_budget=self.settings.default_user_budget
                )
                logger.info(f"Created new budget for user {user_id}")
            return self._user_budgets[user_id]

    def check_budget(self, user_id: str, epsilon: float) -> bool:
        """
        Check if user has sufficient budget for query.

        Returns True if budget is available, False otherwise.
        """
        self._maybe_reset_global_budget()
        budget = self.get_or_create_budget(user_id)

        # Check user budget
        if not budget.can_spend(epsilon):
            return False

        # Check global daily budget
        with self._lock:
            if self._global_used_today + epsilon > self.settings.global_daily_budget:
                return False

        return True

    def spend_budget(self, user_id: str, epsilon: float) -> PrivacyBudget:
        """
        Deduct epsilon from user's budget.

        Raises BudgetExhaustedError if insufficient budget.
        """
        self._maybe_reset_global_budget()
        budget = self.get_or_create_budget(user_id)

        with self._lock:
            # Double-check budget availability
            if not budget.can_spend(epsilon):
                raise BudgetExhaustedError(
                    user_id, epsilon, budget.remaining_budget
                )

            if self._global_used_today + epsilon > self.settings.global_daily_budget:
                raise BudgetExhaustedError(
                    user_id, epsilon,
                    self.settings.global_daily_budget - self._global_used_today
                )

            # Deduct from both budgets
            budget.spend(epsilon)
            self._global_used_today += epsilon

            logger.info(
                f"Budget spent: user={user_id}, epsilon={epsilon}, "
                f"user_remaining={budget.remaining_budget:.4f}, "
                f"global_remaining={self.get_global_remaining():.4f}"
            )

        return budget

    def get_global_remaining(self) -> float:
        """Get remaining global daily budget."""
        self._maybe_reset_global_budget()
        return self.settings.global_daily_budget - self._global_used_today

    def _maybe_reset_global_budget(self) -> None:
        """Reset global budget if it's a new day."""
        today = datetime.utcnow().date()
        with self._lock:
            if today > self._global_reset_date:
                logger.info(f"Resetting global budget for new day: {today}")
                self._global_used_today = 0.0
                self._global_reset_date = today

    def get_budget_info(self, user_id: str) -> dict:
        """Get budget information for a user."""
        budget = self.get_or_create_budget(user_id)
        return {
            "user_id": user_id,
            "budget": {
                "total": budget.total_budget,
                "used": budget.used_budget,
                "remaining": budget.remaining_budget
            },
            "queries_executed": budget.queries_count,
            "last_query_at": budget.last_query_at
        }


# Global instance
budget_manager = PrivacyBudgetManager()
```

**Budget Management Principles**:

- **Per-User Budgets**: Each user has their own epsilon allocation (default: 10.0)
- **Global Daily Budget**: Prevents system-wide privacy degradation (default: 100.0/day)
- **Atomic Operations**: Thread-safe budget deduction prevents race conditions
- **Automatic Reset**: Global budget resets daily for ongoing analytics needs
- **Fail-Safe**: Queries are rejected before execution if budget is insufficient

```python
# services/schema_analyzer.py
"""Schema-only data analysis without raw data access."""

import logging
from typing import List, Dict, Any, Optional
from dataclasses import dataclass
from sqlalchemy import create_engine, inspect, text
from sqlalchemy.engine import Engine

from config import get_settings

logger = logging.getLogger(__name__)


@dataclass
class ColumnProfile:
    """Statistical profile of a column (computed with DP where needed)."""
    name: str
    data_type: str
    nullable: bool
    is_sensitive: bool
    # These are approximate/noisy for sensitive columns
    min_value: Optional[float] = None
    max_value: Optional[float] = None
    distinct_count_range: Optional[str] = None  # e.g., "100-1000"


@dataclass
class TableProfile:
    """Profile of a database table."""
    name: str
    columns: List[ColumnProfile]
    row_count_range: str  # Bucketed for privacy, e.g., "10K-100K"
    description: Optional[str] = None


class SchemaAnalyzer:
    """
    Analyzes database schema without exposing raw data.

    This service provides metadata and statistical profiles that help
    analysts understand data structure without seeing individual records.
    """

    # Columns with these keywords are marked as sensitive
    SENSITIVE_KEYWORDS = {
        "ssn", "social_security", "tax_id", "passport",
        "salary", "income", "wage", "compensation",
        "diagnosis", "condition", "medication", "treatment",
        "password", "secret", "token", "key",
        "phone", "email", "address", "dob", "birth"
    }

    def __init__(self):
        self.settings = get_settings()
        self._engine: Optional[Engine] = None
        self._schema_cache: Optional[Dict[str, TableProfile]] = None

    @property
    def engine(self) -> Engine:
        """Get or create database engine."""
        if self._engine is None:
            self._engine = create_engine(
                self.settings.database_url,
                echo=self.settings.debug
            )
        return self._engine

    def get_schema(self, force_refresh: bool = False) -> List[TableProfile]:
        """
        Get schema information for all tables.

        This is a privacy-free operation as it only returns metadata.
        """
        if self._schema_cache is not None and not force_refresh:
            return list(self._schema_cache.values())

        inspector = inspect(self.engine)
        tables = []

        for table_name in inspector.get_table_names():
            profile = self._profile_table(inspector, table_name)
            tables.append(profile)

        self._schema_cache = {t.name: t for t in tables}
        logger.info(f"Schema loaded: {len(tables)} tables")
        return tables

    def _profile_table(self, inspector, table_name: str) -> TableProfile:
        """Create a profile for a single table."""
        columns = []

        for col in inspector.get_columns(table_name):
            col_name = col["name"]
            is_sensitive = self._is_sensitive_column(col_name)

            profile = ColumnProfile(
                name=col_name,
                data_type=str(col["type"]),
                nullable=col.get("nullable", True),
                is_sensitive=is_sensitive
            )

            # Get bounds for numeric columns (needed for sensitivity calculation)
            if self._is_numeric_type(str(col["type"])) and not is_sensitive:
                bounds = self._get_column_bounds(table_name, col_name)
                if bounds:
                    profile.min_value, profile.max_value = bounds

            columns.append(profile)

        # Get bucketed row count for privacy
        row_count = self._get_row_count(table_name)
        row_count_range = self._bucket_count(row_count)

        return TableProfile(
            name=table_name,
            columns=columns,
            row_count_range=row_count_range
        )

    def _is_sensitive_column(self, column_name: str) -> bool:
        """Determine if a column contains sensitive data based on name."""
        name_lower = column_name.lower()
        return any(kw in name_lower for kw in self.SENSITIVE_KEYWORDS)

    def _is_numeric_type(self, type_str: str) -> bool:
        """Check if a SQL type is numeric."""
        numeric_types = {"integer", "int", "float", "real", "numeric", "decimal", "double"}
        return any(t in type_str.lower() for t in numeric_types)

    def _get_column_bounds(
        self,
        table_name: str,
        column_name: str
    ) -> Optional[tuple]:
        """Get min/max bounds for a numeric column."""
        try:
            with self.engine.connect() as conn:
                query = text(
                    f'SELECT MIN("{column_name}"), MAX("{column_name}") '
                    f'FROM "{table_name}" '
                    f'WHERE "{column_name}" IS NOT NULL'
                )
                result = conn.execute(query).fetchone()
                if result and result[0] is not None:
                    return (float(result[0]), float(result[1]))
        except Exception as e:
            logger.warning(f"Could not get bounds for {table_name}.{column_name}: {e}")
        return None

    def _get_row_count(self, table_name: str) -> int:
        """Get exact row count for a table."""
        try:
            with self.engine.connect() as conn:
                query = text(f'SELECT COUNT(*) FROM "{table_name}"')
                return conn.execute(query).scalar() or 0
        except Exception as e:
            logger.warning(f"Could not get row count for {table_name}: {e}")
            return 0

    def _bucket_count(self, count: int) -> str:
        """Convert exact count to privacy-preserving bucket."""
        if count < 100:
            return "<100"
        elif count < 1000:
            return "100-1K"
        elif count < 10000:
            return "1K-10K"
        elif count < 100000:
            return "10K-100K"
        elif count < 1000000:
            return "100K-1M"
        else:
            return ">1M"

    def get_column_bounds(
        self,
        table_name: str,
        column_name: str
    ) -> Optional[tuple]:
        """Get bounds for a column (public interface)."""
        schema = self.get_schema()
        for table in schema:
            if table.name == table_name:
                for col in table.columns:
                    if col.name == column_name:
                        if col.min_value is not None and col.max_value is not None:
                            return (col.min_value, col.max_value)
        return self._get_column_bounds(table_name, column_name)


# Global instance
schema_analyzer = SchemaAnalyzer()
```

- **Sensitive Column Detection**: Automatically identifies columns likely to contain PII based on naming patterns
- **Bucketed Counts**: Row counts are reported in ranges (e.g., "10K-100K") to prevent exact enumeration
- **Bounds Extraction**: Numeric column bounds are extracted for sensitivity calculation
- **No Raw Data**: The analyzer never returns individual records, only metadata

### Synthetic Data Generator

```python
# services/synthetic_generator.py
"""Synthetic data generation preserving statistical properties."""

import logging
from typing import List, Dict, Any, Optional
from dataclasses import dataclass
import numpy as np
from sqlalchemy import text
import uuid
from datetime import datetime

from services.schema_analyzer import schema_analyzer
from config import get_settings

logger = logging.getLogger(__name__)


@dataclass
class SyntheticDataset:
    """Generated synthetic dataset."""
    id: str
    table_name: str
    num_rows: int
    columns: List[str]
    data: List[Dict[str, Any]]
    statistics_comparison: Dict[str, Dict[str, float]]
    created_at: datetime


class SyntheticDataGenerator:
    """
    Generates synthetic data that preserves statistical properties.

    Synthetic data allows analysts to develop and test queries without
    accessing real sensitive data. The generated data maintains similar
    distributions to the original.
    """

    def __init__(self, seed: Optional[int] = None):
        self.settings = get_settings()
        self.rng = np.random.default_rng(seed)
        self._generated_datasets: Dict[str, SyntheticDataset] = {}

    def generate(
        self,
        table_name: str,
        num_rows: int = 1000,
        preserve_correlations: bool = True
    ) -> SyntheticDataset:
        """
        Generate synthetic data for a table.

        Args:
            table_name: Source table to mimic
            num_rows: Number of synthetic rows to generate
            preserve_correlations: Whether to preserve column correlations

        Returns:
            SyntheticDataset with generated data and statistics comparison
        """
        if num_rows > self.settings.max_synthetic_rows:
            raise ValueError(
                f"Requested {num_rows} rows exceeds maximum {self.settings.max_synthetic_rows}"
            )

        # Get schema and statistics for the table
        schema = schema_analyzer.get_schema()
        table_profile = next((t for t in schema if t.name == table_name), None)

        if table_profile is None:
            raise ValueError(f"Table not found: {table_name}")

        # Get column statistics from original data
        column_stats = self._get_column_statistics(table_name, table_profile)

        # Generate synthetic data
        synthetic_data = []
        for _ in range(num_rows):
            row = self._generate_row(table_profile, column_stats)
            synthetic_data.append(row)

        # Calculate statistics comparison
        stats_comparison = self._compare_statistics(
            table_name, table_profile, synthetic_data, column_stats
        )

        dataset = SyntheticDataset(
            id=str(uuid.uuid4()),
            table_name=table_name,
            num_rows=num_rows,
            columns=[col.name for col in table_profile.columns],
            data=synthetic_data,
            statistics_comparison=stats_comparison,
            created_at=datetime.utcnow()
        )

        self._generated_datasets[dataset.id] = dataset
        logger.info(f"Generated synthetic dataset {dataset.id} with {num_rows} rows")

        return dataset

    def _get_column_statistics(
        self,
        table_name: str,
        table_profile
    ) -> Dict[str, Dict[str, Any]]:
        """Get statistics for each column to guide generation."""
        stats = {}
        engine = schema_analyzer.engine

        with engine.connect() as conn:
            for col in table_profile.columns:
                col_stats = {"type": col.data_type}

                if "int" in col.data_type.lower() or "float" in col.data_type.lower():
                    # Numeric column: get mean, std, min, max
                    query = text(f'''
                        SELECT
                            AVG("{col.name}") as mean,
                            AVG("{col.name}" * "{col.name}") - AVG("{col.name}") * AVG("{col.name}") as variance,
                            MIN("{col.name}") as min_val,
                            MAX("{col.name}") as max_val
                        FROM "{table_name}"
                        WHERE "{col.name}" IS NOT NULL
                    ''')
                    result = conn.execute(query).fetchone()
                    if result and result[0] is not None:
                        col_stats["mean"] = float(result[0])
                        col_stats["std"] = float(result[1]) ** 0.5 if result[1] and result[1] > 0 else 1.0
                        col_stats["min"] = float(result[2])
                        col_stats["max"] = float(result[3])

                elif "varchar" in col.data_type.lower() or "text" in col.data_type.lower():
                    # String column: get distinct values for categorical
                    query = text(f'''
                        SELECT "{col.name}", COUNT(*) as cnt
                        FROM "{table_name}"
                        WHERE "{col.name}" IS NOT NULL
                        GROUP BY "{col.name}"
                        ORDER BY cnt DESC
                        LIMIT 100
                    ''')
                    result = conn.execute(query).fetchall()
                    if result:
                        values = [r[0] for r in result]
                        weights = [r[1] for r in result]
                        total = sum(weights)
                        col_stats["values"] = values
                        col_stats["weights"] = [w / total for w in weights]

                stats[col.name] = col_stats

        return stats

    def _generate_row(
        self,
        table_profile,
        column_stats: Dict[str, Dict[str, Any]]
    ) -> Dict[str, Any]:
        """Generate a single synthetic row."""
        row = {}

        for col in table_profile.columns:
            stats = column_stats.get(col.name, {})
            col_type = stats.get("type", col.data_type).lower()

            if "int" in col_type:
                if "mean" in stats:
                    # Generate from normal distribution, round to int
                    value = self.rng.normal(stats["mean"], stats["std"])
                    value = int(np.clip(value, stats["min"], stats["max"]))
                else:
                    value = self.rng.integers(0, 1000)
                row[col.name] = value

            elif "float" in col_type or "real" in col_type or "numeric" in col_type:
                if "mean" in stats:
                    value = self.rng.normal(stats["mean"], stats["std"])
                    value = np.clip(value, stats["min"], stats["max"])
                else:
                    value = self.rng.uniform(0, 1000)
                row[col.name] = round(float(value), 2)

            elif "values" in stats:
                # Categorical: sample from observed distribution
                row[col.name] = self.rng.choice(
                    stats["values"],
                    p=stats["weights"]
                )

            else:
                # Default: generate random string
                row[col.name] = f"synthetic_{self.rng.integers(0, 10000)}"

        return row

    def _compare_statistics(
        self,
        table_name: str,
        table_profile,
        synthetic_data: List[Dict],
        original_stats: Dict[str, Dict[str, Any]]
    ) -> Dict[str, Dict[str, float]]:
        """Compare statistics between original and synthetic data."""
        comparison = {}

        for col in table_profile.columns:
            orig = original_stats.get(col.name, {})
            if "mean" not in orig:
                continue

            # Calculate synthetic statistics
            values = [row[col.name] for row in synthetic_data if row.get(col.name) is not None]
            if not values:
                continue

            syn_mean = np.mean(values)
            syn_std = np.std(values)

            comparison[col.name] = {
                "original_mean": orig["mean"],
                "synthetic_mean": float(syn_mean),
                "original_std": orig["std"],
                "synthetic_std": float(syn_std),
                "mean_error_pct": abs(syn_mean - orig["mean"]) / (orig["mean"] + 1e-10) * 100
            }

        return comparison

    def get_dataset(self, dataset_id: str) -> Optional[SyntheticDataset]:
        """Retrieve a generated dataset by ID."""
        return self._generated_datasets.get(dataset_id)

synthetic_generator = SyntheticDataGenerator()
```

**Synthetic Data Generation Features**:

- **Distribution Preservation**: Numeric columns maintain similar mean and standard deviation
- **Categorical Sampling**: String columns sample from observed value distributions
- **Statistics Comparison**: Output includes comparison metrics to validate fidelity
- **Bounded Generation**: Values are clamped to original data ranges
- **No Privacy Cost**: Synthetic data generation doesn't consume privacy budget

```python
# services/audit_logger.py
"""Comprehensive audit logging for compliance."""

import logging
import hashlib
import json
from typing import List, Optional, Dict, Any
from datetime import datetime, timedelta
from threading import Lock
import uuid

from models.audit import AuditEntry, AuditLogResponse
from config import get_settings

logger = logging.getLogger(__name__)


class AuditLogger:
    """
    Comprehensive audit logging for privacy compliance.

    Every query is logged with full context including:
    - Who made the query (user_id)
    - What was queried (table, column, filters)
    - Privacy cost (epsilon used)
    - Result metadata (hashed true result, noisy result)
    - Request context (IP, user agent)

    This enables compliance audits and forensic analysis.
    """

    def __init__(self):
        self.settings = get_settings()
        self._entries: List[AuditEntry] = []
        self._lock = Lock()

    def log_query(
        self,
        user_id: str,
        query_type: str,
        table_name: str,
        column_name: Optional[str],
        filters: Optional[Dict[str, Any]],
        epsilon_requested: float,
        epsilon_used: float,
        sensitivity: float,
        true_result: float,
        noisy_result: float,
        success: bool,
        error_message: Optional[str] = None,
        ip_address: str = "unknown",
        user_agent: str = "unknown"
    ) -> AuditEntry:
        """
        Log a query execution for audit purposes.

        The true result is hashed (not stored in plain text) to enable
        verification without exposing actual values.
        """
        # Hash the true result for compliance (can verify but not reverse)
        true_result_hash = None
        if self.settings.hash_true_results:
            true_result_hash = self._hash_result(true_result, user_id, query_type)

        entry = AuditEntry(
            id=str(uuid.uuid4()),
            timestamp=datetime.utcnow(),
            user_id=user_id,
            query_type=query_type,
            table_name=table_name,
            column_name=column_name,
            filters=filters,
            epsilon_requested=epsilon_requested,
            epsilon_used=epsilon_used,
            sensitivity=sensitivity,
            true_result_hash=true_result_hash,
            noisy_result=noisy_result,
            success=success,
            error_message=error_message,
            ip_address=ip_address,
            user_agent=user_agent
        )

        with self._lock:
            self._entries.append(entry)

        logger.info(
            f"Audit: user={user_id}, type={query_type}, table={table_name}, "
            f"epsilon={epsilon_used}, success={success}"
        )

        return entry

    def _hash_result(self, result: float, user_id: str, query_type: str) -> str:
        """Create a salted hash of the true result."""
        # Salt with user_id and query_type to prevent rainbow table attacks
        salt = f"{user_id}:{query_type}:{datetime.utcnow().date()}"
        data = f"{salt}:{result:.10f}"
        return hashlib.sha256(data.encode()).hexdigest()[:16]

    def get_entries(
        self,
        user_id: Optional[str] = None,
        start_date: Optional[datetime] = None,
        end_date: Optional[datetime] = None,
        limit: int = 100
    ) -> AuditLogResponse:
        """
        Query audit log entries with optional filters.

        Args:
            user_id: Filter by specific user
            start_date: Start of date range
            end_date: End of date range
            limit: Maximum entries to return
        """
        with self._lock:
            filtered = self._entries.copy()

        # Apply filters
        if user_id:
            filtered = [e for e in filtered if e.user_id == user_id]

        if start_date:
            filtered = [e for e in filtered if e.timestamp >= start_date]

        if end_date:
            filtered = [e for e in filtered if e.timestamp <= end_date]

        # Sort by timestamp descending (most recent first)
        filtered.sort(key=lambda e: e.timestamp, reverse=True)

        total = len(filtered)
        entries = filtered[:limit]

        return AuditLogResponse(
            entries=entries,
            total=total,
            limit=limit
        )

    def get_user_activity_summary(self, user_id: str) -> Dict[str, Any]:
        """Get activity summary for a specific user."""
        with self._lock:
            user_entries = [e for e in self._entries if e.user_id == user_id]

        if not user_entries:
            return {"user_id": user_id, "total_queries": 0}

        total_epsilon = sum(e.epsilon_used for e in user_entries)
        success_count = sum(1 for e in user_entries if e.success)

        return {
            "user_id": user_id,
            "total_queries": len(user_entries),
            "successful_queries": success_count,
            "failed_queries": len(user_entries) - success_count,
            "total_epsilon_used": total_epsilon,
            "first_query": min(e.timestamp for e in user_entries),
            "last_query": max(e.timestamp for e in user_entries),
            "tables_accessed": list(set(e.table_name for e in user_entries))
        }

    def cleanup_old_entries(self) -> int:
        """Remove entries older than retention period."""
        cutoff = datetime.utcnow() - timedelta(days=self.settings.audit_log_retention_days)

        with self._lock:
            original_count = len(self._entries)
            self._entries = [e for e in self._entries if e.timestamp >= cutoff]
            removed = original_count - len(self._entries)

        if removed > 0:
            logger.info(f"Cleaned up {removed} audit entries older than {cutoff}")

        return removed


# Global instance
audit_logger = AuditLogger()
```

**Audit Logging Features**:

- **Complete Context**: Every query logged with user, query details, privacy cost, and results
- **Hashed True Results**: Original values are hashed (not stored plaintext) for verification without exposure
- **Append-Only**: Logs cannot be modified, only appended (in production, use immutable storage)
- **Retention Policy**: Automatic cleanup of old entries per compliance requirements
- **Activity Summaries**: Aggregate views of user activity for compliance reporting

### Query Executor Service

```python
# services/query_executor.py
"""Privacy-preserving query execution."""

import logging
import time
from typing import Optional, Dict, Any, Tuple
from sqlalchemy import text

from models.query import (
    QueryRequest, QueryResponse, QueryResult, QueryType,
    BudgetInfo, QueryMetadata, ConfidenceInterval
)
from services.differential_privacy import dp_engine, SensitivityConfig
from services.privacy_budget import budget_manager, BudgetExhaustedError
from services.schema_analyzer import schema_analyzer
from services.audit_logger import audit_logger
from config import get_settings

logger = logging.getLogger(__name__)


class QueryExecutor:
    """
    Executes privacy-preserving queries with differential privacy.

    This is the main orchestrator that:
    1. Validates the query and checks budget
    2. Calculates sensitivity for the query type
    3. Executes the aggregate query
    4. Adds calibrated Laplace noise
    5. Logs the query for audit
    6. Returns the noisy result
    """

    def __init__(self):
        self.settings = get_settings()

    async def execute(
        self,
        request: QueryRequest,
        ip_address: str = "unknown",
        user_agent: str = "unknown"
    ) -> QueryResponse:
        """
        Execute a privacy-preserving query.

        Args:
            request: The query request
            ip_address: Client IP for audit logging
            user_agent: Client user agent for audit logging

        Returns:
            QueryResponse with noisy result and metadata
        """
        start_time = time.time()

        # Validate query type
        if request.query_type is None:
            raise ValueError("query_type is required")

        if request.table is None:
            raise ValueError("table is required")

        # Check privacy budget
        if not budget_manager.check_budget(request.user_id, request.epsilon):
            budget = budget_manager.get_or_create_budget(request.user_id)
            raise BudgetExhaustedError(
                request.user_id,
                request.epsilon,
                budget.remaining_budget
            )

        # Get column bounds for sensitivity calculation
        bounds = self._get_bounds(request)

        # Calculate sensitivity
        sensitivity_config = SensitivityConfig(
            lower_bound=bounds[0] if bounds else None,
            upper_bound=bounds[1] if bounds else None
        )
        sensitivity = dp_engine.calculate_sensitivity(
            request.query_type,
            sensitivity_config
        )

        # Execute the true query
        true_result = self._execute_aggregate(request)

        # Add differential privacy noise
        noisy_result, confidence_interval = dp_engine.add_noise(
            true_value=true_result,
            sensitivity=sensitivity,
            epsilon=request.epsilon,
            lower_bound=0.0 if request.query_type == QueryType.COUNT else bounds[0] if bounds else None,
            upper_bound=bounds[1] if bounds else None
        )

        # Deduct from privacy budget
        budget = budget_manager.spend_budget(request.user_id, request.epsilon)

        # Log for audit
        audit_logger.log_query(
            user_id=request.user_id,
            query_type=request.query_type.value,
            table_name=request.table,
            column_name=request.column,
            filters=request.filters,
            epsilon_requested=request.epsilon,
            epsilon_used=request.epsilon,
            sensitivity=sensitivity,
            true_result=true_result,
            noisy_result=noisy_result,
            success=True,
            ip_address=ip_address,
            user_agent=user_agent
        )

        execution_time_ms = int((time.time() - start_time) * 1000)

        return QueryResponse(
            result=QueryResult(
                value=noisy_result,
                noise_added=True,
                epsilon_used=request.epsilon,
                confidence_interval=confidence_interval
            ),
            budget=BudgetInfo(
                user_remaining=budget.remaining_budget,
                user_total=budget.total_budget,
                global_remaining=budget_manager.get_global_remaining()
            ),
            metadata=QueryMetadata(
                query_type=request.query_type,
                table=request.table,
                column=request.column,
                sensitivity=sensitivity,
                execution_time_ms=execution_time_ms
            )
        )

    def _get_bounds(self, request: QueryRequest) -> Optional[Tuple[float, float]]:
        """Get data bounds for the query column."""
        if request.query_type == QueryType.COUNT:
            return None  # COUNT doesn't need bounds

        if request.column is None:
            raise ValueError(f"{request.query_type.value} queries require a column")

        bounds = schema_analyzer.get_column_bounds(request.table, request.column)
        if bounds is None:
            raise ValueError(
                f"Cannot determine bounds for {request.table}.{request.column}. "
                "Bounds are required for differential privacy."
            )
        return bounds

    def _execute_aggregate(self, request: QueryRequest) -> float:
        """Execute the aggregate query and return true result."""
        engine = schema_analyzer.engine

        # Build the aggregate function
        if request.query_type == QueryType.COUNT:
            agg_func = "COUNT(*)"
        elif request.query_type == QueryType.SUM:
            agg_func = f'SUM("{request.column}")'
        elif request.query_type == QueryType.AVG:
            agg_func = f'AVG("{request.column}")'
        elif request.query_type == QueryType.MIN:
            agg_func = f'MIN("{request.column}")'
        elif request.query_type == QueryType.MAX:
            agg_func = f'MAX("{request.column}")'
        else:
            raise ValueError(f"Unsupported query type: {request.query_type}")

        # Build WHERE clause from filters
        where_clause = ""
        params = {}
        if request.filters:
            conditions = []
            for i, (col, val) in enumerate(request.filters.items()):
                param_name = f"p{i}"
                conditions.append(f'"{col}" = :{param_name}')
                params[param_name] = val
            where_clause = "WHERE " + " AND ".join(conditions)

        query = text(f'SELECT {agg_func} FROM "{request.table}" {where_clause}')

        with engine.connect() as conn:
            result = conn.execute(query, params).scalar()

        return float(result) if result is not None else 0.0


# Global instance
query_executor = QueryExecutor()
```

```python
# services/llm_interface.py
"""LLM integration for natural language query translation."""

import logging
from typing import Optional
from openai import AsyncOpenAI
from pydantic import BaseModel, Field

from models.query import QueryType
from services.schema_analyzer import schema_analyzer
from config import get_settings

logger = logging.getLogger(__name__)


class ParsedQuery(BaseModel):
    """Structured output from LLM query parsing."""
    query_type: QueryType
    table: str
    column: Optional[str] = None
    filters: Optional[dict] = None
    confidence: float = Field(ge=0.0, le=1.0)
    explanation: str


class LLMInterface:
    """
    Translates natural language questions to structured queries.

    IMPORTANT: The LLM never sees raw data. It only receives:
    - Schema information (table names, column names, types)
    - The user's question

    This prevents data leakage through LLM prompts.
    """

    SYSTEM_PROMPT = """You are a privacy-aware query translator. Your job is to convert natural language questions into structured aggregate queries.

RULES:
1. You can ONLY generate aggregate queries: COUNT, SUM, AVG, MIN, MAX
2. You NEVER have access to individual records - only aggregates
3. You must identify the correct table and column from the schema
4. If the question cannot be answered with an aggregate, explain why

SUPPORTED QUERY TYPES:
- COUNT: "How many...", "Number of...", "Total count..."
- SUM: "Total...", "Sum of...", "Combined..."
- AVG: "Average...", "Mean...", "Typical..."
- MIN: "Minimum...", "Lowest...", "Smallest..."
- MAX: "Maximum...", "Highest...", "Largest..."

OUTPUT FORMAT:
- query_type: One of COUNT, SUM, AVG, MIN, MAX
- table: The table name from the schema
- column: The column to aggregate (null for COUNT)
- filters: Optional WHERE conditions as key-value pairs
- confidence: Your confidence in the translation (0-1)
- explanation: Brief explanation of your interpretation"""

    def __init__(self):
        self.settings = get_settings()
        self.client = AsyncOpenAI(api_key=self.settings.openai_api_key)

    async def parse_question(self, question: str) -> ParsedQuery:
        """
        Parse a natural language question into a structured query.

        Args:
            question: Natural language question from user

        Returns:
            ParsedQuery with query type, table, column, and filters
        """
        # Get schema context (no raw data, just structure)
        schema = schema_analyzer.get_schema()
        schema_context = self._format_schema_for_llm(schema)

        user_prompt = f"""DATABASE SCHEMA:
{schema_context}

USER QUESTION:
{question}

Translate this question into a structured aggregate query. If the question cannot be answered with the available schema or aggregate functions, set confidence to 0 and explain why."""

        response = await self.client.beta.chat.completions.parse(
            model=self.settings.openai_model,
            temperature=self.settings.llm_temperature,
            messages=[
                {"role": "system", "content": self.SYSTEM_PROMPT},
                {"role": "user", "content": user_prompt}
            ],
            response_format=ParsedQuery
        )

        result = response.choices[0].message.parsed

        if result is None:
            raise ValueError("Failed to parse LLM response")

        logger.info(
            f"Parsed query: type={result.query_type}, table={result.table}, "
            f"column={result.column}, confidence={result.confidence}"
        )

        return result

    def _format_schema_for_llm(self, schema) -> str:
        """Format schema for LLM context (no sensitive data)."""
        lines = []
        for table in schema:
            lines.append(f"\nTable: {table.name} ({table.row_count_range} rows)")
            for col in table.columns:
                sensitive_marker = " [SENSITIVE]" if col.is_sensitive else ""
                lines.append(f"  - {col.name}: {col.data_type}{sensitive_marker}")
        return "\n".join(lines)


# Global instance
llm_interface = LLMInterface()
```

**LLM Integration Security**:

- **Schema-Only Context**: LLM receives only table/column names, never actual data
- **Aggregate-Only Output**: LLM can only generate aggregate query types
- **Confidence Scoring**: Low confidence queries can be flagged for human review
- **Structured Output**: Pydantic models ensure valid query structure

## Understanding the Privacy Guarantees

### What Does ε-Differential Privacy Mean?

<Mermaid
  chart={`

    subgraph "Database D"
        D1[Alice's Record]
        D2[Bob's Record]
        D3[Carol's Record]
    end

    R1 --> Similar[Results are<br/>statistically similar]
    R2 --> Similar
    
    style D1 fill:#ffcdd2,stroke:#c62828
    style Similar fill:#c8e6c9,stroke:#2e7d32

  caption="Differential Privacy: Adding or removing one person doesn't significantly change the output."
/>

**Formal Definition**: A mechanism M provides ε-differential privacy if for any two databases D and D' that differ by one record, and for any output S:

```
P(M(D) ∈ S) ≤ e^ε × P(M(D') ∈ S)
```

**Intuition**: An attacker who sees the query result cannot determine whether any specific individual's data was included in the database. The smaller ε, the stronger this guarantee.

### Privacy Budget Composition

<Mermaid
  chart={`
graph TD
    Budget[Total Budget: ε = 10.0]
    
    Q1[Query 1: ε = 0.5] --> Used1[Used: 0.5]
    Q2[Query 2: ε = 0.3] --> Used2[Used: 0.8]
    Q3[Query 3: ε = 1.0] --> Used3[Used: 1.8]
    Q4[Query 4: ε = 0.2] --> Used4[Used: 2.0]
    
    Budget --> Q1
    Used1 --> Q2
    Used2 --> Q3
    Used3 --> Q4
    
    Used4 --> Remaining[Remaining: 8.0]
    
    style Budget fill:#e3f2fd,stroke:#0277bd
    style Remaining fill:#c8e6c9,stroke:#2e7d32
`}
  caption="Budget Composition: Each query consumes epsilon, and they add up."
/>

**Sequential Composition Theorem**: If you run k queries with privacy parameters ε₁, ε₂, ..., εₖ, the total privacy loss is at most ε₁ + ε₂ + ... + εₖ.

This is why we track and limit the total epsilon spent—to prevent privacy degradation through many queries.

### Choosing Epsilon Values

| Epsilon | Privacy Level | Use Case                                | Noise Level |
| ------- | ------------- | --------------------------------------- | ----------- |
| 0.01    | Very Strong   | Highly sensitive data (medical records) | Very High   |
| 0.1     | Strong        | Sensitive analytics (salary data)       | High        |
| 0.5     | Moderate      | General business analytics              | Moderate    |
| 1.0     | Weak          | Non-sensitive aggregates                | Low         |
| > 1.0   | Minimal       | Testing only                            | Very Low    |

**Rule of Thumb**: Start with ε = 0.1 for sensitive data. Increase only if the noise makes results unusable, and document the privacy tradeoff.

```python
        self,
        true_answer: bool,
```

```python
class DPModelTrainer:
    """Train ML models with differential privacy."""

    def dp_sgd_step(
        self,
        gradients: np.ndarray,
        clip_norm: float,
        noise_multiplier: float,
        batch_size: int
    ) -> np.ndarray:
        """
        Perform one step of DP-SGD.

        1. Clip per-example gradients to bound sensitivity
        2. Average clipped gradients
        3. Add Gaussian noise calibrated to sensitivity
        """
        # Clip gradients
        clipped = self._clip_gradients(gradients, clip_norm)
        # Average
        avg_gradient = np.mean(clipped, axis=0)
        # Add noise
        noise = np.random.normal(0, clip_norm * noise_multiplier / batch_size, avg_gradient.shape)
        return avg_gradient + noise
```
