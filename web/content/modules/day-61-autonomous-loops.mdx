### The Basic Loop Structure

```bash
#!/bin/bash
# The "Ralph Wiggum" technique - simple but powerful

MAX_ITERATIONS=20
ITERATION=0

while [ $ITERATION -lt $MAX_ITERATIONS ]; do
    echo "=== Iteration $ITERATION ==="

    # Feed the prompt to the agent
    # The agent reads project files to understand current state
    claude --prompt "
        Read the project files to understand current state.
        Your goal: Achieve 80% test coverage.

        Current coverage: $(npm run coverage --silent | grep 'All files')

        If coverage >= 80%, respond with 'DONE'.
        Otherwise, identify untested code and write tests.
    " --allowedTools "read,write,bash"

    # Check if done
    if grep -q "DONE" /tmp/agent_output.txt; then
        echo "Goal achieved!"
        exit 0
    fi

    ITERATION=$((ITERATION + 1))
done

echo "Max iterations reached"
exit 1
```

## 61.3 File System as Memory

The key insight: **the file system is the agent's external memory**.

<Callout type="tip">
  **Why This Works:** By reading project files at the start of each loop
  iteration, the agent sees its own previous work. This bypasses the limitation
  of finite conversation history—the agent's "memory" is the actual state of the
  codebase.
</Callout>

```python
# agent_loop.py
import subprocess
import json
from pathlib import Path

class AutonomousLoop:
    def __init__(self, goal: str, max_iterations: int = 20):
        self.goal = goal
        self.max_iterations = max_iterations
        self.state_file = Path(".agent_state.json")

    def load_state(self) -> dict:
        """File system as memory - load previous state."""
        if self.state_file.exists():
            return json.loads(self.state_file.read_text())
        return {"iteration": 0, "history": []}

    def save_state(self, state: dict):
        """Persist state to file system."""
        self.state_file.write_text(json.dumps(state, indent=2))

    def check_end_condition(self) -> bool:
        """Define what 'done' looks like."""
        # Example: Check test coverage
        result = subprocess.run(
            ["npm", "run", "coverage", "--", "--json"],
            capture_output=True, text=True
        )
        coverage = self.parse_coverage(result.stdout)
        return coverage >= 80

    def run(self):
        state = self.load_state()

        while state["iteration"] < self.max_iterations:
            print(f"Iteration {state['iteration']}")

            # Check if we're done
            if self.check_end_condition():
                print("✅ Goal achieved!")
                return True

            # Run agent with current project state as context
            result = self.run_agent_iteration(state)

            # Update state
            state["iteration"] += 1
            state["history"].append(result)
            self.save_state(state)

        print("❌ Max iterations reached")
        return False
```

```python
class CostControlledLoop:
    def __init__(
        self,
        max_iterations: int = 20,
        max_cost_usd: float = 10.0,
        cost_per_iteration: float = 0.15
    ):
        self.max_iterations = max_iterations
        self.max_cost_usd = max_cost_usd
        self.cost_per_iteration = cost_per_iteration
        self.total_cost = 0.0

    def can_continue(self, iteration: int) -> tuple[bool, str]:
        """Check all cost controls."""
        if iteration >= self.max_iterations:
            return False, f"Max iterations ({self.max_iterations}) reached"

        projected_cost = self.total_cost + self.cost_per_iteration
        if projected_cost > self.max_cost_usd:
            return False, f"Cost budget (${self.max_cost_usd}) would be exceeded"

        # Alert at 50% budget
        if self.total_cost > self.max_cost_usd * 0.5:
            print(f"⚠️ Warning: 50% of cost budget consumed (${self.total_cost:.2f})")

        return True, "OK"

    def record_iteration(self, tokens_used: int, model: str):
        """Track actual costs."""
        cost = self.calculate_cost(tokens_used, model)
        self.total_cost += cost
        print(f"💰 Iteration cost: ${cost:.4f} | Total: ${self.total_cost:.2f}")
```

### Prompt Caching for Cost Reduction (2024-2025)

Both Anthropic and OpenAI now offer prompt caching, which dramatically reduces costs for autonomous loops where the system prompt and context remain stable across iterations.

```python
import anthropic

class CachedAutonomousLoop:
    """Leverage prompt caching for 90% cost reduction on repeated context."""

    def __init__(self, system_prompt: str, project_context: str):
        self.client = anthropic.Anthropic()
        # Mark large, stable content for caching
        self.cached_system = {
            "type": "text",
            "text": system_prompt,
            "cache_control": {"type": "ephemeral"}  # Cache for 5 minutes
        }
        self.cached_context = {
            "type": "text",
            "text": project_context,
            "cache_control": {"type": "ephemeral"}
        }

    def run_iteration(self, current_state: str) -> dict:
        response = self.client.messages.create(
            model="claude-sonnet-4-20250514",
            max_tokens=4096,
            system=[self.cached_system],
            messages=[{
                "role": "user",
                "content": [
                    self.cached_context,  # Cached - 90% cheaper after first call
                    {"type": "text", "text": f"Current state:\n{current_state}"}
                ]
            }]
        )

        # Check cache performance
        usage = response.usage
        cache_savings = (usage.cache_read_input_tokens * 0.9) / 1_000_000 * 3  # ~$3/M tokens saved
        print(f"💾 Cache hit: {usage.cache_read_input_tokens} tokens (saved ~${cache_savings:.4f})")

        return response
```

### Model Routing for Cost Optimization

Use cheaper models for routine iterations, expensive models for complex decisions:

```python
class ModelRouter:
    """Route to appropriate model based on task complexity."""

    MODELS = {
        "fast": {"name": "claude-3-5-haiku-20241022", "cost_per_1k": 0.001},
        "balanced": {"name": "claude-sonnet-4-20250514", "cost_per_1k": 0.003},
        "powerful": {"name": "claude-sonnet-4-20250514", "cost_per_1k": 0.015}

    def select_model(self, task_type: str, iteration: int, confidence: float) -> str:

        # Use fast model for routine checks
        if task_type == "status_check":

        # Use powerful model for final review or low confidence
        if task_type == "final_review" or confidence < 0.5:
            return self.MODELS["powerful"]["name"]

        # Use balanced model for most iterations
        return self.MODELS["balanced"]["name"]

### Token Budget Strategies

```python
class TokenBudgetManager:
    """Manage token budgets across loop iterations."""

    def __init__(self, total_budget: int = 100_000):
        self.total_budget = total_budget
        self.used_tokens = 0
        self.iteration_history = []

    def allocate_for_iteration(self, iteration: int, remaining_iterations: int) -> int:
        """Dynamically allocate tokens based on remaining budget."""
        remaining_budget = self.total_budget - self.used_tokens

        if remaining_iterations <= 0:
            return min(remaining_budget, 4000)

        # Reserve tokens for future iterations
        reserved = remaining_iterations * 2000  # Minimum per iteration
        available = remaining_budget - reserved

        # Allow more tokens early, taper off
        allocation = min(
            available * 0.3,  # Max 30% of available per iteration
            8000,             # Hard cap per iteration
            remaining_budget  # Never exceed remaining
        )

        return int(max(allocation, 1000))  # Minimum 1000 tokens

    def record_usage(self, tokens: int):
        self.used_tokens += tokens
        self.iteration_history.append(tokens)

        if self.used_tokens > self.total_budget * 0.8:
            print(f"⚠️ Token budget 80% consumed: {self.used_tokens}/{self.total_budget}")
```

## 61.6 Human-in-the-Loop Patterns

Pure autonomy isn't always the answer. The most robust production systems blend autonomous execution with strategic human oversight.

### Approval Gates for High-Risk Actions

```python
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.postgres import PostgresSaver
from enum import Enum

class RiskLevel(Enum):
    LOW = "low"           # Auto-approve
    MEDIUM = "medium"     # Log and continue
    HIGH = "high"         # Require human approval
    CRITICAL = "critical" # Block until explicit approval

def assess_risk(action: dict) -> RiskLevel:
    """Classify action risk level."""
    high_risk_patterns = [
        "delete", "drop", "truncate",  # Destructive operations
        "production", "prod",           # Production environment
        "payment", "billing",           # Financial operations
        "password", "secret", "key"     # Security-sensitive
    ]

    action_str = str(action).lower()

    if any(p in action_str for p in ["drop database", "rm -rf", "format"]):
        return RiskLevel.CRITICAL

    if any(p in action_str for p in high_risk_patterns):
        return RiskLevel.HIGH

    if "write" in action_str or "update" in action_str:
        return RiskLevel.MEDIUM

    return RiskLevel.LOW

class HumanInTheLoopAgent:
    def __init__(self, approval_callback):
        self.approval_callback = approval_callback
        self.pending_approvals = {}

    async def execute_with_approval(self, action: dict, state: dict) -> dict:
        risk = assess_risk(action)

        if risk == RiskLevel.CRITICAL:
            # Block and wait for explicit human approval
            approval_id = self.request_approval(action, "CRITICAL action requires approval")
            approved = await self.wait_for_approval(approval_id, timeout=3600)
            if not approved:
                return {"status": "blocked", "reason": "Human rejected critical action"}

        elif risk == RiskLevel.HIGH:
            # Request approval with timeout, auto-reject if no response
            approval_id = self.request_approval(action, "High-risk action")
            approved = await self.wait_for_approval(approval_id, timeout=300)
            if not approved:
                return {"status": "blocked", "reason": "No approval received"}

        # Execute the action
        return await self.execute_action(action)

    def request_approval(self, action: dict, reason: str) -> str:
        """Send approval request via configured channel (Slack, email, etc.)."""
        approval_id = generate_id()
        self.pending_approvals[approval_id] = {
            "action": action,
            "reason": reason,
            "status": "pending",
            "requested_at": datetime.now()
        }
        # Notify via Slack, email, or dashboard
        self.approval_callback(approval_id, action, reason)
        return approval_id
```

### Confidence-Based Escalation

from pydantic import BaseModel, Field

class AgentDecision(BaseModel):
    action: str
    reasoning: str
    confidence: float = Field(ge=0, le=1, description="0-1 confidence score")
    uncertainties: list[str] = Field(default_factory=list)

def confidence_based_routing(decision: AgentDecision) -> str:
    """Route based on agent's self-reported confidence."""
    if decision.confidence >= 0.9:
        return "auto_execute"
    elif decision.confidence >= 0.7:
        return "execute_with_logging"
    elif decision.confidence >= 0.5:
        return "request_review"

### Async Human Review Workflows

For non-blocking human oversight, implement async review queues:

```python
import asyncio
from datetime import datetime, timedelta

class AsyncReviewQueue:
    """Non-blocking human review that doesn't stop the agent."""

    def __init__(self, review_callback):
        self.queue = asyncio.Queue()
        self.review_callback = review_callback

    async def submit_for_review(self, action: dict, result: dict):
        """Submit completed action for async human review."""
        review_item = {
            "action": action,
            "result": result,
            "submitted_at": datetime.now(),
            "status": "pending_review"
        }
        await self.queue.put(review_item)
        # Don't wait - agent continues immediately
        return review_item

    async def process_reviews(self):
        """Background task to process review queue."""
        while True:
            item = await self.queue.get()
            # Notify reviewer (Slack, email, dashboard)

### Recovery Patterns

```python
class RecoveryManager:
    """Handle various failure scenarios in autonomous loops."""

    def __init__(self, max_retries: int = 3):
        self.max_retries = max_retries
        self.retry_delays = [1, 5, 30]  # Exponential backoff (seconds)

    async def with_recovery(self, operation, state: dict) -> dict:
        """Execute operation with comprehensive recovery."""
        for attempt in range(self.max_retries):
            try:
                return await operation(state)

            except RateLimitError as e:
                # Wait for rate limit reset
                wait_time = e.retry_after or self.retry_delays[attempt]
                print(f"⏳ Rate limited. Waiting {wait_time}s...")
                await asyncio.sleep(wait_time)

            except APIConnectionError:
                # Network issue - retry with backoff
                if attempt < self.max_retries - 1:
                    await asyncio.sleep(self.retry_delays[attempt])
                else:
                    # Save state for manual resume
                    await self.save_recovery_state(state)
                    raise

            except TokenLimitExceeded:
                # Reduce context and retry
                state = self.reduce_context(state)
                continue

        raise MaxRetriesExceeded(f"Failed after {self.max_retries} attempts")

    async def save_recovery_state(self, state: dict):
        """Persist state for manual recovery."""
        recovery_file = f"recovery_{state['thread_id']}_{datetime.now().isoformat()}.json"
        Path(recovery_file).write_text(json.dumps(state, indent=2))
        print(f"💾 Recovery state saved to {recovery_file}")
```

### Detecting Stuck Loops

```python
from datetime import datetime, timedelta
from dataclasses import dataclass
import asyncio

@dataclass
class LoopMetrics:
    iteration: int
    started_at: datetime
    last_progress_at: datetime
    goal_distance: float  # How far from goal (0-1)
    tokens_used: int
    actions_taken: list[str]

class StuckLoopDetector:
    """Detect when autonomous loops are making no progress."""

    def __init__(
        self,
        max_time_without_progress: timedelta = timedelta(minutes=10),
        min_goal_improvement: float = 0.01,
        max_repeated_actions: int = 3
    ):
        self.max_time_without_progress = max_time_without_progress
        self.min_goal_improvement = min_goal_improvement
        self.max_repeated_actions = max_repeated_actions
        self.metrics_history: list[LoopMetrics] = []

    def record_iteration(self, metrics: LoopMetrics):
        self.metrics_history.append(metrics)

    def is_stuck(self) -> tuple[bool, str]:
        if len(self.metrics_history) < 2:
            return False, "Not enough data"

        current = self.metrics_history[-1]
        previous = self.metrics_history[-2]

        # Check 1: Time without progress
        time_since_progress = datetime.now() - current.last_progress_at
        if time_since_progress > self.max_time_without_progress:
            return True, f"No progress for {time_since_progress}"

        # Check 2: Goal distance not improving
        if len(self.metrics_history) >= 5:
            recent_distances = [m.goal_distance for m in self.metrics_history[-5:]]
            improvement = recent_distances[0] - recent_distances[-1]
            if improvement < self.min_goal_improvement:
                return True, f"Goal distance not improving: {recent_distances}"

        # Check 3: Repeated actions (loop detected)
        recent_actions = [m.actions_taken[-1] if m.actions_taken else "" for m in self.metrics_history[-5:]]
        if len(set(recent_actions)) == 1 and recent_actions[0]:
            return True, f"Repeating same action: {recent_actions[0]}"

        return False, "Loop progressing normally"

# Integration with monitoring
detector = StuckLoopDetector()

async def monitored_loop(goal: str):
    for iteration in range(max_iterations):
        result = await execute_iteration()

        metrics = LoopMetrics(
            iteration=iteration,
            started_at=start_time,
            last_progress_at=datetime.now() if result.made_progress else last_progress,
            goal_distance=result.distance_to_goal,
            tokens_used=result.tokens,
            actions_taken=result.actions
        )
        detector.record_iteration(metrics)

        is_stuck, reason = detector.is_stuck()
        if is_stuck:
            await alert_and_pause(reason)
            break

### Kill Switches and Circuit Breakers

```python
import asyncio
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"      # Normal operation
    OPEN = "open"          # Blocked - too many failures
    HALF_OPEN = "half_open"  # Testing if recovered

class LoopCircuitBreaker:
    """Circuit breaker pattern for autonomous loops."""

    def __init__(
        self,
        failure_threshold: int = 5,
        recovery_timeout: float = 60.0,
        half_open_max_calls: int = 3
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_max_calls = half_open_max_calls

        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = None
        self.half_open_calls = 0

    async def call(self, operation):
        """Execute operation with circuit breaker protection."""
        if self.state == CircuitState.OPEN:
            if self._should_attempt_reset():
                self.state = CircuitState.HALF_OPEN
                self.half_open_calls = 0
            else:
                raise CircuitOpenError("Circuit breaker is OPEN")

        try:
            result = await operation()

            if self.state == CircuitState.HALF_OPEN:
                self.half_open_calls += 1
                if self.half_open_calls >= self.half_open_max_calls:
                    self.state = CircuitState.CLOSED
                    self.failure_count = 0

            return result

        except Exception as e:
            self._record_failure()
            raise

    def _record_failure(self):
        self.failure_count += 1
        self.last_failure_time = asyncio.get_event_loop().time()

        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
            print(f"🔴 Circuit breaker OPEN after {self.failure_count} failures")

    def _should_attempt_reset(self) -> bool:
        if self.last_failure_time is None:
            return True
        elapsed = asyncio.get_event_loop().time() - self.last_failure_time
        return elapsed >= self.recovery_timeout

class KillSwitch:
    """Emergency stop for autonomous loops."""

    def __init__(self):
        self.killed = False
        self.kill_reason = None

    def kill(self, reason: str):
        """Immediately stop all loops."""
        self.killed = True
        self.kill_reason = reason
        print(f"🛑 KILL SWITCH ACTIVATED: {reason}")

    def check(self):
        """Check if killed - call at start of each iteration."""
        if self.killed:
            raise LoopKilledException(self.kill_reason)

    def reset(self):
        """Reset kill switch after investigation."""
        self.killed = False
        self.kill_reason = None

# Global kill switch (can be triggered via API endpoint)
global_kill_switch = KillSwitch()

# Usage in loop
async def safe_autonomous_loop():
    circuit_breaker = LoopCircuitBreaker()

    for iteration in range(max_iterations):
        global_kill_switch.check()  # Check kill switch

        try:
            await circuit_breaker.call(lambda: execute_iteration())
        except CircuitOpenError:
            print("Circuit open - waiting for recovery...")
            await asyncio.sleep(60)
```
