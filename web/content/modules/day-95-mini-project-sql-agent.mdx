## Project Overview

**Objective**: Build an intelligent SQL Query Agent that translates natural language questions into SQL queries, executes them safely against a database, and presents results with visualizations and plain-English explanations.

**Why This Matters**: Databases contain valuable insights, but SQL expertise is a barrier for most users. A SQL agent democratizes data access—product managers can ask "What were our top 10 products last quarter?" without writing a single line of SQL. This is one of the most practical applications of LLMs in enterprise settings.

**Architecture Insights**:

- **Schema-First Design**: The schema loader runs at startup and caches database structure, providing rich context for SQL generation
- **Multi-Layer Validation**: Queries pass through syntax validation, injection detection, and permission checking before execution
- **Read-Only Enforcement**: Database connections are configured with read-only permissions at the connection level, not just query level
- **Graceful Degradation**: If visualization fails, the system falls back to formatted tables; if explanation fails, raw results are returned

```python
# models/schema.py
"""Database schema data models."""

from pydantic import BaseModel, Field
from typing import List, Optional, Any
from enum import Enum


class DataType(str, Enum):
    """Common SQL data types."""
    INTEGER = "INTEGER"
    BIGINT = "BIGINT"
    FLOAT = "FLOAT"
    DECIMAL = "DECIMAL"
    VARCHAR = "VARCHAR"
    TEXT = "TEXT"
    BOOLEAN = "BOOLEAN"
    DATE = "DATE"
    DATETIME = "DATETIME"
    TIMESTAMP = "TIMESTAMP"
    JSON = "JSON"
    BLOB = "BLOB"


class ForeignKeyRef(BaseModel):
    """Foreign key reference."""
    table: str
    column: str


class ColumnSchema(BaseModel):
    """Schema for a database column."""
    name: str
    data_type: str
    nullable: bool = True
    default_value: Optional[str] = None
    is_primary_key: bool = False
    is_foreign_key: bool = False
    references: Optional[ForeignKeyRef] = None
    sample_values: List[Any] = Field(default_factory=list)


class IndexSchema(BaseModel):
    """Schema for a database index."""
    name: str
    columns: List[str]
    unique: bool = False


class TableSchema(BaseModel):
    """Schema for a database table."""
    name: str
    columns: List[ColumnSchema]
    primary_key: List[str] = Field(default_factory=list)
    indexes: List[IndexSchema] = Field(default_factory=list)
    row_count: int = 0
    description: Optional[str] = None


class Relationship(BaseModel):
    """Relationship between tables."""
    from_table: str
    from_column: str
    to_table: str
    to_column: str
    relationship_type: str = "many-to-one"


class DatabaseSchema(BaseModel):
    """Complete database schema."""
    database_name: str
    database_type: str
    tables: List[TableSchema]
    relationships: List[Relationship] = Field(default_factory=list)

    def to_prompt_context(self) -> str:
        """Convert schema to LLM-friendly context string."""
        lines = [f"Database: {self.database_name} ({self.database_type})\n"]
        lines.append("Tables:\n")

        for table in self.tables:
            lines.append(f"\n{table.name} (~{table.row_count:,} rows):")
            for col in table.columns:
                pk = " [PK]" if col.is_primary_key else ""
                fk = f" [FK -> {col.references.table}.{col.references.column}]" if col.references else ""
                nullable = " NULL" if col.nullable else " NOT NULL"
                lines.append(f"  - {col.name}: {col.data_type}{pk}{fk}{nullable}")
                if col.sample_values:
                    samples = ", ".join(str(v) for v in col.sample_values[:3])
                    lines.append(f"    Examples: {samples}")

        if self.relationships:
            lines.append("\n\nRelationships:")
            for rel in self.relationships:
                lines.append(f"  {rel.from_table}.{rel.from_column} -> {rel.to_table}.{rel.to_column}")

        return "\n".join(lines)
```

```python
# models/query.py
"""Query request and response models."""

from pydantic import BaseModel, Field, field_validator
from typing import List, Optional, Any, Dict
from datetime import datetime
from enum import Enum
import uuid


class VisualizationType(str, Enum):
    """Supported visualization types."""
    BAR = "bar"
    LINE = "line"
    PIE = "pie"
    TABLE = "table"
    SCATTER = "scatter"
    AREA = "area"


class QueryRequest(BaseModel):
    """Request model for natural language query."""
    question: str = Field(..., min_length=3, max_length=1000)
    max_rows: int = Field(default=100, ge=1, le=10000)
    timeout_seconds: int = Field(default=30, ge=1, le=60)
    include_visualization: bool = True
    include_explanation: bool = True
    include_optimization: bool = False

    @field_validator("question")
    @classmethod
    def validate_question(cls, v: str) -> str:
        """Validate and clean the question."""
        # Remove control characters
        cleaned = "".join(char for char in v if ord(char) >= 32 or char in "\n\t")
        return cleaned.strip()


class QueryResult(BaseModel):
    """Query execution result."""
    columns: List[str]
    rows: List[List[Any]]
    row_count: int
    truncated: bool = False


class VisualizationConfig(BaseModel):
    """Configuration for visualization."""
    type: VisualizationType
    x_column: Optional[str] = None
    y_columns: List[str] = Field(default_factory=list)
    title: str = ""
    x_label: str = ""
    y_label: str = ""


class Visualization(BaseModel):
    """Generated visualization."""
    type: VisualizationType
    html: str
    config: VisualizationConfig


class OptimizationSuggestion(BaseModel):
    """Query optimization suggestion."""
    suggestions: List[str] = Field(default_factory=list)
    estimated_improvement: str = ""
    optimized_sql: Optional[str] = None


class QueryResponse(BaseModel):
    """Complete response for a query."""
    id: str = Field(default_factory=lambda: str(uuid.uuid4()))
    question: str
    sql: str
    results: QueryResult
    visualization: Optional[Visualization] = None
    explanation: Optional[str] = None
    optimization: Optional[OptimizationSuggestion] = None
    execution_time_ms: int
    cached: bool = False
    timestamp: datetime = Field(default_factory=datetime.utcnow)


class QueryError(BaseModel):
    """Error response model."""
    detail: str
    reason: Optional[str] = None
    sql: Optional[str] = None
    suggestion: Optional[str] = None


class QueryHistoryItem(BaseModel):
    """Query history entry."""
    id: str
    question: str
    sql: str
    executed_at: datetime
    execution_time_ms: int
    row_count: int
    success: bool


class QueryHistoryResponse(BaseModel):
    """Query history response."""
    queries: List[QueryHistoryItem]
    total: int
    limit: int
    offset: int
```

```python
# services/schema_loader.py
"""Database schema extraction and caching."""

import asyncio
from typing import List, Optional, Dict, Any
from sqlalchemy import create_engine, inspect, text
from sqlalchemy.engine import Engine
import logging

from models.schema import (
    DatabaseSchema, TableSchema, ColumnSchema,
    IndexSchema, Relationship, ForeignKeyRef
)
from config import get_settings

logger = logging.getLogger(__name__)


class SchemaLoader:
    """Extracts and caches database schema for LLM context."""

    def __init__(self):
        self.settings = get_settings()
        self._engine: Optional[Engine] = None
        self._schema: Optional[DatabaseSchema] = None
        self._schema_context: Optional[str] = None

    @property
    def engine(self) -> Engine:
        """Get or create database engine."""
        if self._engine is None:
            connect_args = {}
            if self.settings.database_type == "sqlite":
                # SQLite read-only mode
                connect_args["check_same_thread"] = False
                if self.settings.database_read_only:
                    connect_args["uri"] = True

            self._engine = create_engine(
                self.settings.database_url,
                connect_args=connect_args,
                echo=self.settings.debug
            )
        return self._engine

    async def load_schema(self) -> DatabaseSchema:
        """Load complete database schema."""
        if self._schema is not None:
            return self._schema

        logger.info("Loading database schema...")

        inspector = inspect(self.engine)
        tables: List[TableSchema] = []
        relationships: List[Relationship] = []

        table_names = inspector.get_table_names()
        logger.info(f"Found {len(table_names)} tables")

        for table_name in table_names:
            table_schema = await self._extract_table_schema(
                inspector, table_name
            )
            tables.append(table_schema)

            # Extract relationships from foreign keys
            for col in table_schema.columns:
                if col.references:
                    relationships.append(Relationship(
                        from_table=table_name,
                        from_column=col.name,
                        to_table=col.references.table,
                        to_column=col.references.column
                    ))

        self._schema = DatabaseSchema(
            database_name=self._get_database_name(),
            database_type=self.settings.database_type,
            tables=tables,
            relationships=relationships
        )

        # Pre-compute context string
        self._schema_context = self._schema.to_prompt_context()

        logger.info(f"Schema loaded: {len(tables)} tables, {len(relationships)} relationships")
        return self._schema

    async def _extract_table_schema(
        self,
        inspector,
        table_name: str
    ) -> TableSchema:
        """Extract schema for a single table."""
        columns: List[ColumnSchema] = []

        # Get column information
        column_info = inspector.get_columns(table_name)
        pk_columns = inspector.get_pk_constraint(table_name).get("constrained_columns", [])
        fk_info = inspector.get_foreign_keys(table_name)

        # Build foreign key lookup
        fk_lookup: Dict[str, ForeignKeyRef] = {}
        for fk in fk_info:
            for i, col in enumerate(fk.get("constrained_columns", [])):
                ref_cols = fk.get("referred_columns", [])
                if i < len(ref_cols):
                    fk_lookup[col] = ForeignKeyRef(
                        table=fk.get("referred_table", ""),
                        column=ref_cols[i]
                    )

        for col in column_info:
            col_name = col["name"]
            columns.append(ColumnSchema(
                name=col_name,
                data_type=str(col["type"]),
                nullable=col.get("nullable", True),
                default_value=str(col.get("default")) if col.get("default") else None,
                is_primary_key=col_name in pk_columns,
                is_foreign_key=col_name in fk_lookup,
                references=fk_lookup.get(col_name),
                sample_values=await self._get_sample_values(table_name, col_name)
            ))

        # Get indexes
        indexes = [
            IndexSchema(
                name=idx["name"] or f"{table_name}_idx",
                columns=idx["column_names"],
                unique=idx.get("unique", False)
            )
            for idx in inspector.get_indexes(table_name)
        ]

        # Get approximate row count
        row_count = await self._get_row_count(table_name)

        return TableSchema(
            name=table_name,
            columns=columns,
            primary_key=pk_columns,
            indexes=indexes,
            row_count=row_count
        )

    async def _get_sample_values(
        self,
        table_name: str,
        column_name: str,
        limit: int = 5
    ) -> List[Any]:
        """Get sample values for a column."""
        try:
            with self.engine.connect() as conn:
                # Use parameterized query for safety
                query = text(
                    f"SELECT DISTINCT \"{column_name}\" FROM \"{table_name}\" "
                    f"WHERE \"{column_name}\" IS NOT NULL LIMIT :limit"
                )
                result = conn.execute(query, {"limit": limit})
                return [row[0] for row in result.fetchall()]
        except Exception as e:
            logger.warning(f"Could not get samples for {table_name}.{column_name}: {e}")
            return []

    async def _get_row_count(self, table_name: str) -> int:
        """Get approximate row count for a table."""
        try:
            with self.engine.connect() as conn:
                # For large tables, use approximate count if available
                if self.settings.database_type == "postgresql":
                    query = text(
                        "SELECT reltuples::bigint FROM pg_class "
                        "WHERE relname = :table_name"
                    )
                    result = conn.execute(query, {"table_name": table_name})
                    row = result.fetchone()
                    if row and row[0] > 0:
                        return int(row[0])

                # Fallback to exact count (may be slow for large tables)
                query = text(f"SELECT COUNT(*) FROM \"{table_name}\"")
                result = conn.execute(query)
                return result.scalar() or 0
        except Exception as e:
            logger.warning(f"Could not get row count for {table_name}: {e}")
            return 0

    def _get_database_name(self) -> str:
        """Extract database name from URL."""
        url = self.settings.database_url
        if "sqlite" in url:
            return url.split("/")[-1].replace(".db", "")
        elif "postgresql" in url:
            return url.split("/")[-1].split("?")[0]
        return "database"

    def get_schema_context(self) -> str:
        """Get pre-computed schema context for LLM."""
        if self._schema_context is None:
            raise RuntimeError("Schema not loaded. Call load_schema() first.")
        return self._schema_context

    def get_schema(self) -> DatabaseSchema:
        """Get cached schema."""
        if self._schema is None:
            raise RuntimeError("Schema not loaded. Call load_schema() first.")
        return self._schema


# Global instance
schema_loader = SchemaLoader()
```

- **Lazy Loading**: Schema is loaded once at startup and cached for all subsequent requests
- **Sample Values**: Collecting sample values helps the LLM understand data patterns (e.g., knowing a `status` column contains "active", "pending", "closed")
- **Relationship Detection**: Foreign keys are automatically detected and included in the context
- **Approximate Counts**: For PostgreSQL, uses `pg_class` for fast approximate counts on large tables

```python
# services/query_validator.py
"""SQL query validation and security checking."""

import re
import logging
from typing import Tuple, Optional, Set
from enum import Enum
import sqlparse
from sqlparse.sql import Statement, Token
from sqlparse.tokens import Keyword, DML

from services.schema_loader import schema_loader

logger = logging.getLogger(__name__)


class ValidationError(Enum):
    """Types of validation errors."""
    INJECTION_DETECTED = "injection_detected"
    WRITE_OPERATION = "write_operation"
    SYNTAX_ERROR = "syntax_error"
    INVALID_TABLE = "invalid_table"
    INVALID_COLUMN = "invalid_column"
    MULTIPLE_STATEMENTS = "multiple_statements"
    DANGEROUS_FUNCTION = "dangerous_function"


class QueryValidator:
    """Multi-layer SQL query validation."""

    # Patterns that indicate SQL injection attempts
    INJECTION_PATTERNS = [
        r";\s*--",                          # Statement termination with comment
        r";\s*(DROP|DELETE|UPDATE|INSERT)", # Statement chaining with DML/DDL
        r"UNION\s+ALL\s+SELECT",            # UNION-based injection
        r"OR\s+['\"]?\d+['\"]?\s*=\s*['\"]?\d+['\"]?",  # OR 1=1 pattern
        r"AND\s+['\"]?\d+['\"]?\s*=\s*['\"]?\d+['\"]?", # AND 1=1 pattern
        r"'\s*OR\s*'",                      # String-based OR injection
        r"--\s*$",                          # Trailing comment
        r"/\*.*\*/",                        # Block comments (potential bypass)
        r"EXEC\s*\(",                       # EXEC calls
        r"xp_\w+",                          # Extended stored procedures
        r"WAITFOR\s+DELAY",                 # Time-based injection
        r"BENCHMARK\s*\(",                  # MySQL benchmark
        r"SLEEP\s*\(",                      # Sleep function
        r"LOAD_FILE\s*\(",                  # File access
        r"INTO\s+(OUTFILE|DUMPFILE)",       # File writing
    ]

    # Dangerous SQL functions that should be blocked
    DANGEROUS_FUNCTIONS = {
        "load_file", "into outfile", "into dumpfile",
        "benchmark", "sleep", "waitfor",
        "exec", "execute", "xp_cmdshell",
        "pg_read_file", "pg_write_file",
        "dbms_pipe", "utl_http"
    }

    # Allowed statement types
    ALLOWED_STATEMENTS = {"SELECT"}

    # Blocked keywords that indicate write operations
    BLOCKED_KEYWORDS = {
        "INSERT", "UPDATE", "DELETE", "DROP", "CREATE", "ALTER",
        "TRUNCATE", "REPLACE", "MERGE", "GRANT", "REVOKE",
        "COMMIT", "ROLLBACK", "SAVEPOINT"
    }

    def __init__(self):
        self._compiled_patterns = [
            re.compile(pattern, re.IGNORECASE)
            for pattern in self.INJECTION_PATTERNS
        ]

    def validate(self, sql: str) -> Tuple[bool, Optional[ValidationError], Optional[str]]:
        """
        Validate SQL query through multiple security layers.

        Returns:
            Tuple of (is_valid, error_type, error_message)
        """
        # Layer 1: Pattern-based injection detection
        is_safe, error = self._check_injection_patterns(sql)
        if not is_safe:
            return False, ValidationError.INJECTION_DETECTED, error

        # Layer 2: Parse and validate syntax
        try:
            parsed = sqlparse.parse(sql)
        except Exception as e:
            return False, ValidationError.SYNTAX_ERROR, f"Failed to parse SQL: {e}"

        if not parsed:
            return False, ValidationError.SYNTAX_ERROR, "Empty or invalid SQL"

        # Layer 3: Check for multiple statements
        if len(parsed) > 1:
            return False, ValidationError.MULTIPLE_STATEMENTS, "Multiple statements not allowed"

        statement = parsed[0]

        # Layer 4: Validate statement type
        is_select, error = self._validate_statement_type(statement)
        if not is_select:
            return False, ValidationError.WRITE_OPERATION, error

        # Layer 5: Check for dangerous functions
        has_dangerous, func = self._check_dangerous_functions(sql)
        if has_dangerous:
            return False, ValidationError.DANGEROUS_FUNCTION, f"Dangerous function detected: {func}"

        # Layer 6: Validate table and column references
        is_valid, error = self._validate_references(sql)
        if not is_valid:
            return False, ValidationError.INVALID_TABLE, error

        logger.info("Query validation passed all checks")
        return True, None, None

    def _check_injection_patterns(self, sql: str) -> Tuple[bool, Optional[str]]:
        """Check for common SQL injection patterns."""
        for pattern in self._compiled_patterns:
            if pattern.search(sql):
                return False, f"Suspicious pattern detected: {pattern.pattern}"
        return True, None

    def _validate_statement_type(self, statement: Statement) -> Tuple[bool, Optional[str]]:
        """Ensure the statement is a SELECT query."""
        # Get the first meaningful token
        first_token = statement.token_first(skip_cm=True, skip_ws=True)

        if first_token is None:
            return False, "Empty statement"

        # Check if it's a SELECT statement
        if first_token.ttype is DML and first_token.value.upper() == "SELECT":
            # Additional check: ensure no blocked keywords appear
            sql_upper = statement.value.upper()
            for keyword in self.BLOCKED_KEYWORDS:
                # Check for keyword as a standalone word (not part of column name)
                if re.search(rf'\b{keyword}\b', sql_upper):
                    return False, f"Blocked keyword detected: {keyword}"
            return True, None

        return False, f"Only SELECT statements allowed, got: {first_token.value}"

    def _check_dangerous_functions(self, sql: str) -> Tuple[bool, Optional[str]]:
        """Check for dangerous SQL functions."""
        sql_lower = sql.lower()
        for func in self.DANGEROUS_FUNCTIONS:
            if func in sql_lower:
                return True, func
        return False, None

    def _validate_references(self, sql: str) -> Tuple[bool, Optional[str]]:
        """Validate that referenced tables exist in schema."""
        try:
            schema = schema_loader.get_schema()
            valid_tables = {t.name.lower() for t in schema.tables}

            # Extract table names from query (simplified extraction)
            # This catches most common patterns: FROM table, JOIN table
            table_pattern = r'\b(?:FROM|JOIN)\s+["\']?(\w+)["\']?'
            referenced_tables = re.findall(table_pattern, sql, re.IGNORECASE)

            for table in referenced_tables:
                if table.lower() not in valid_tables:
                    return False, f"Unknown table: {table}"

            return True, None

        except Exception as e:
            logger.warning(f"Could not validate table references: {e}")
            # Don't fail validation if schema check fails
            return True, None

    def sanitize_for_logging(self, sql: str) -> str:
        """Sanitize SQL for safe logging (remove potential sensitive data)."""
        # Replace string literals with placeholders
        sanitized = re.sub(r"'[^']*'", "'***'", sql)
        sanitized = re.sub(r'"[^"]*"', '"***"', sanitized)
        return sanitized
```

- **Dangerous Function Blocking**: Blocks functions that could access files or cause delays
- **Schema Validation**: Ensures referenced tables actually exist in the database
- **Logging Sanitization**: Removes sensitive data before logging queries

```python
# services/query_executor.py
"""Safe SQL query execution with timeouts and limits."""

import asyncio
import logging
from typing import List, Any, Optional
from datetime import datetime
from contextlib import asynccontextmanager
from sqlalchemy import create_engine, text
from sqlalchemy.engine import Engine
from sqlalchemy.exc import SQLAlchemyError

from config import get_settings
from models.query import QueryResult

logger = logging.getLogger(__name__)


class QueryExecutionError(Exception):
    """Custom exception for query execution failures."""
    def __init__(self, message: str, sql: Optional[str] = None):
        super().__init__(message)
        self.sql = sql


class QueryExecutor:
    """Executes SQL queries safely with read-only connections."""

    def __init__(self):
        self.settings = get_settings()
        self._engine: Optional[Engine] = None

    @property
    def engine(self) -> Engine:
        """Get or create read-only database engine."""
        if self._engine is None:
            connect_args = {}

            if self.settings.database_type == "sqlite":
                connect_args["check_same_thread"] = False
                # SQLite read-only mode via URI
                if self.settings.database_read_only:
                    url = self.settings.database_url
                    if "?" not in url:
                        url += "?mode=ro"
                    else:
                        url += "&mode=ro"
                    connect_args["uri"] = True

            elif self.settings.database_type == "postgresql":
                # PostgreSQL read-only transaction mode
                connect_args["options"] = "-c default_transaction_read_only=on"

            self._engine = create_engine(
                self.settings.database_url,
                connect_args=connect_args,
                pool_pre_ping=True,
                pool_recycle=3600,
                echo=self.settings.debug
            )

        return self._engine

    async def execute(
        self,
        sql: str,
        max_rows: int = 100,
        timeout_seconds: int = 30
    ) -> QueryResult:
        """
        Execute SQL query with safety constraints.

        Args:
            sql: The SQL query to execute
            max_rows: Maximum rows to return
            timeout_seconds: Query timeout in seconds

        Returns:
            QueryResult with columns, rows, and metadata
        """
        # Enforce row limit by adding/modifying LIMIT clause
        limited_sql = self._enforce_limit(sql, max_rows)

        start_time = datetime.utcnow()

        try:
            # Execute with timeout
            result = await asyncio.wait_for(
                self._execute_query(limited_sql),
                timeout=timeout_seconds
            )

            execution_time = (datetime.utcnow() - start_time).total_seconds() * 1000
            logger.info(f"Query executed in {execution_time:.0f}ms, {len(result['rows'])} rows")

            return QueryResult(
                columns=result["columns"],
                rows=result["rows"],
                row_count=len(result["rows"]),
                truncated=len(result["rows"]) >= max_rows
            )

        except asyncio.TimeoutError:
            raise QueryExecutionError(
                f"Query timed out after {timeout_seconds} seconds",
                sql=sql
            )
        except SQLAlchemyError as e:
            raise QueryExecutionError(f"Database error: {str(e)}", sql=sql)
        except Exception as e:
            raise QueryExecutionError(f"Execution failed: {str(e)}", sql=sql)

    async def _execute_query(self, sql: str) -> dict:
        """Execute query in thread pool to avoid blocking."""
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(None, self._sync_execute, sql)

    def _sync_execute(self, sql: str) -> dict:
        """Synchronous query execution."""
        with self.engine.connect() as conn:
            # Set statement timeout for PostgreSQL
            if self.settings.database_type == "postgresql":
                conn.execute(text(
                    f"SET statement_timeout = '{self.settings.query_timeout_seconds * 1000}'"
                ))

            result = conn.execute(text(sql))

            columns = list(result.keys())
            rows = [list(row) for row in result.fetchall()]

            return {"columns": columns, "rows": rows}

    def _enforce_limit(self, sql: str, max_rows: int) -> str:
        """Ensure query has a LIMIT clause."""
        sql_upper = sql.upper().strip()

        # Check if LIMIT already exists
        if "LIMIT" in sql_upper:
            # Extract existing limit and enforce maximum
            import re
            match = re.search(r'LIMIT\s+(\d+)', sql_upper)
            if match:
                existing_limit = int(match.group(1))
                if existing_limit > max_rows:
                    # Replace with enforced limit
                    sql = re.sub(
                        r'LIMIT\s+\d+',
                        f'LIMIT {max_rows}',
                        sql,
                        flags=re.IGNORECASE
                    )
            return sql

        # Add LIMIT clause
        # Handle queries ending with semicolon
```

```python
# services/visualizer.py
"""Automatic visualization generation for query results."""

import json
import logging
from typing import List, Any, Optional
from datetime import datetime, date

from models.query import (
    QueryResult, Visualization, VisualizationConfig, VisualizationType
)
from config import get_settings

logger = logging.getLogger(__name__)


class Visualizer:
    """Generates appropriate visualizations for query results."""

    # Plotly CDN for rendering
    PLOTLY_CDN = "https://cdn.plot.ly/plotly-2.27.0.min.js"

    def __init__(self):
        self.settings = get_settings()

    def visualize(
        self,
        result: QueryResult,
        title: str = "Query Results"
    ) -> Optional[Visualization]:
        """
        Generate appropriate visualization for query results.

        Args:
            result: Query execution result
            title: Chart title

        Returns:
            Visualization object with HTML and config
        """
        if not result.rows:
            return None

        # Determine best visualization type
        viz_type = self._select_visualization_type(result)
        config = self._build_config(result, viz_type, title)

        # Generate HTML based on type
        if viz_type == VisualizationType.TABLE:
            html = self._render_table(result, config)
        else:
            html = self._render_plotly_chart(result, config)

        return Visualization(
            type=viz_type,
            html=html,
            config=config
        )

    def _select_visualization_type(self, result: QueryResult) -> VisualizationType:
        """Select the most appropriate visualization type."""
        columns = result.columns
        rows = result.rows

        if not rows or not columns:
            return VisualizationType.TABLE

        num_columns = len(columns)
        num_rows = len(rows)

        # Analyze column types
        numeric_cols = []
        date_cols = []
        text_cols = []

        for i, col in enumerate(columns):
            sample_values = [row[i] for row in rows[:10] if row[i] is not None]
            if not sample_values:
                continue

            col_type = self._infer_column_type(sample_values)
            if col_type == "numeric":
                numeric_cols.append(i)
            elif col_type == "date":
                date_cols.append(i)
            else:
                text_cols.append(i)

        # Decision logic
        # Time series: date column + numeric column
        if date_cols and numeric_cols:
            return VisualizationType.LINE

        # Pie chart: 2 columns, one text, one numeric, few rows
        if num_columns == 2 and text_cols and numeric_cols and num_rows <= 10:
            return VisualizationType.PIE

        # Bar chart: text column + numeric column(s)
        if text_cols and numeric_cols and num_rows <= 20:
            return VisualizationType.BAR

        # Scatter: 2+ numeric columns
        if len(numeric_cols) >= 2:
            return VisualizationType.SCATTER

        # Default to table
        return VisualizationType.TABLE

    def _infer_column_type(self, values: List[Any]) -> str:
        """Infer column type from sample values."""
        for val in values:
            if val is None:
                continue
            if isinstance(val, (int, float)):
                return "numeric"
            if isinstance(val, (datetime, date)):
                return "date"
            if isinstance(val, str):
                # Try to parse as date
                try:
                    datetime.fromisoformat(val.replace('Z', '+00:00'))
                    return "date"
                except (ValueError, AttributeError):
                    pass
                # Try to parse as number
                try:
                    float(val.replace(',', ''))
                    return "numeric"
                except (ValueError, AttributeError):
                    pass
        return "text"

    def _build_config(
        self,
        result: QueryResult,
        viz_type: VisualizationType,
        title: str
    ) -> VisualizationConfig:
        """Build visualization configuration."""
        columns = result.columns

        # Find appropriate x and y columns
        x_col = columns[0] if columns else None
        y_cols = columns[1:] if len(columns) > 1 else []

        return VisualizationConfig(
            type=viz_type,
            x_column=x_col,
            y_columns=y_cols,
            title=title,
            x_label=x_col or "",
            y_label=y_cols[0] if y_cols else ""
        )

    def _render_plotly_chart(
        self,
        result: QueryResult,
        config: VisualizationConfig
    ) -> str:
        """Render Plotly chart as HTML."""
        x_idx = result.columns.index(config.x_column) if config.x_column in result.columns else 0
        x_values = [row[x_idx] for row in result.rows]

        traces = []

        for y_col in config.y_columns:
            if y_col not in result.columns:
                continue
            y_idx = result.columns.index(y_col)
            y_values = [row[y_idx] for row in result.rows]

            trace = {
                "x": x_values,
                "y": y_values,
                "name": y_col,
                "type": self._plotly_trace_type(config.type)
            }

            if config.type == VisualizationType.PIE:
                trace = {
                    "labels": x_values,
                    "values": y_values,
                    "type": "pie"
                }

            traces.append(trace)

        layout = {
            "title": config.title,
            "xaxis": {"title": config.x_label},
            "yaxis": {"title": config.y_label},
            "width": self.settings.default_chart_width,
            "height": self.settings.default_chart_height
        }

        chart_id = f"chart_{id(result)}"

        return f"""
<div id="{chart_id}"></div>
<script src="{self.PLOTLY_CDN}"></script>
<script>
    Plotly.newPlot('{chart_id}', {json.dumps(traces)}, {json.dumps(layout)});
</script>
"""

    def _plotly_trace_type(self, viz_type: VisualizationType) -> str:
        """Map visualization type to Plotly trace type."""
        mapping = {
            VisualizationType.BAR: "bar",
            VisualizationType.LINE: "scatter",
            VisualizationType.SCATTER: "scatter",
            VisualizationType.AREA: "scatter",
            VisualizationType.PIE: "pie"
        }
        return mapping.get(viz_type, "scatter")

    def _render_table(self, result: QueryResult, config: VisualizationConfig) -> str:
        """Render results as HTML table."""
        html = ['<table class="sql-results" style="border-collapse: collapse; width: 100%;">']

        # Header
        html.append('<thead><tr>')
        for col in result.columns:
            html.append(f'<th style="border: 1px solid #ddd; padding: 8px; background: #f4f4f4;">{col}</th>')
        html.append('</tr></thead>')

        # Body
        html.append('<tbody>')
        for row in result.rows:
            html.append('<tr>')
            for val in row:
                display_val = "" if val is None else str(val)
                html.append(f'<td style="border: 1px solid #ddd; padding: 8px;">{display_val}</td>')
            html.append('</tr>')
        html.append('</tbody>')

        html.append('</table>')

        if result.truncated:
            html.append(f'<p style="color: #666; font-style: italic;">Showing {result.row_count} rows (results truncated)</p>')

        return '\n'.join(html)
```

```python
"""SQL query optimization suggestions."""

import re
import logging

from openai import AsyncOpenAI

from config import get_settings

logger = logging.getLogger(__name__)


class QueryOptimizer:
    """Analyzes queries and suggests optimizations."""

    SYSTEM_PROMPT = """You are a database performance expert. Analyze SQL queries and suggest optimizations.

Focus on:
1. Missing indexes that could speed up the query
2. Inefficient JOIN patterns
3. Suboptimal WHERE clause ordering
4. Unnecessary columns in SELECT
5. Missing LIMIT clauses for large result sets
6. N+1 query patterns
7. Opportunities for query rewriting

Provide specific, actionable suggestions. If the query is already optimal, say so.

Format your response as a JSON object:
{
    "suggestions": ["suggestion 1", "suggestion 2"],
    "estimated_improvement": "description of potential improvement",
    "optimized_sql": "rewritten query if applicable, or null"
}"""

    def __init__(self):
        self.settings = get_settings()
        self.client = AsyncOpenAI(api_key=self.settings.openai_api_key)

    async def analyze(self, sql: str) -> OptimizationSuggestion:
        """
        Analyze query and suggest optimizations.

        Args:
            sql: The SQL query to analyze

        Returns:
            OptimizationSuggestion with recommendations
        """
        # First, do rule-based analysis
        rule_suggestions = self._rule_based_analysis(sql)

        # Then, get LLM suggestions
        llm_suggestions = await self._llm_analysis(sql)

        # Combine suggestions
        all_suggestions = list(set(rule_suggestions + llm_suggestions.suggestions))

        return OptimizationSuggestion(
            suggestions=all_suggestions,
            estimated_improvement=llm_suggestions.estimated_improvement,
            optimized_sql=llm_suggestions.optimized_sql
        )

    def _rule_based_analysis(self, sql: str) -> List[str]:
        """Apply rule-based optimization checks."""
        suggestions = []
        sql_upper = sql.upper()

        # Check for SELECT *
        if "SELECT *" in sql_upper or "SELECT  *" in sql_upper:
            suggestions.append(
                "Avoid SELECT * - specify only the columns you need to reduce data transfer"
            )

        # Check for missing LIMIT
        if "LIMIT" not in sql_upper and "COUNT(" not in sql_upper:
            suggestions.append(
                "Consider adding a LIMIT clause to prevent returning too many rows"
            )

        # Check for LIKE with leading wildcard
        if re.search(r"LIKE\s+['\"]%", sql, re.IGNORECASE):
            suggestions.append(
                "LIKE patterns starting with % cannot use indexes - consider full-text search"
            )

        # Check for functions on indexed columns in WHERE
        if re.search(r"WHERE.*\b(UPPER|LOWER|DATE|YEAR|MONTH)\s*\(", sql, re.IGNORECASE):
            suggestions.append(
                "Functions on columns in WHERE clause prevent index usage - consider computed columns"
            )

        # Check for OR conditions
        if " OR " in sql_upper and "WHERE" in sql_upper:
            suggestions.append(
                "OR conditions can prevent index usage - consider UNION for better performance"
            )

        # Check for implicit type conversion
        if re.search(r"=\s*['\"]?\d+['\"]", sql):
            suggestions.append(
                "Potential implicit type conversion detected - ensure data types match"
            )

        # Check for NOT IN with subquery
        if "NOT IN" in sql_upper and "SELECT" in sql_upper.split("NOT IN")[1]:
            suggestions.append(
                "NOT IN with subquery can be slow - consider NOT EXISTS or LEFT JOIN"
            )

        return suggestions

    async def _llm_analysis(self, sql: str) -> OptimizationSuggestion:
        """Get LLM-based optimization suggestions."""
        schema_context = schema_loader.get_schema_context()

        prompt = f"""Database Schema:
{schema_context}

SQL Query to Optimize:
{sql}

Analyze this query and provide optimization suggestions."""

        try:
            response = await self.client.chat.completions.create(
                model=self.settings.openai_model,
                temperature=0.2,
                max_tokens=500,
                messages=[
                    {"role": "system", "content": self.SYSTEM_PROMPT},
                    {"role": "user", "content": prompt}
                ],
                response_format={"type": "json_object"}
            )

            import json
            result = json.loads(response.choices[0].message.content)

            return OptimizationSuggestion(
                suggestions=result.get("suggestions", []),
                estimated_improvement=result.get("estimated_improvement", ""),
                optimized_sql=result.get("optimized_sql")
            )

        except Exception as e:
            logger.error(f"LLM optimization analysis failed: {e}")
            return OptimizationSuggestion(
                suggestions=[],
                estimated_improvement="Unable to analyze query"
            )
```

```python
@router.post("/query", response_model=QueryResponse)
async def execute_query(request: QueryRequest):
    """
    Execute a natural language query against the database.

    Translates the question to SQL, validates, executes, and returns
    results with optional visualization and explanation.
    """
    start_time = datetime.utcnow()

    try:
        # Step 1: Generate SQL from natural language
        logger.info(f"Processing question: {request.question[:100]}...")
        generated = await sql_generator.generate(request.question)
        sql = generated.sql

        # Step 2: Validate the generated SQL
        is_valid, error_type, error_msg = query_validator.validate(sql)

        if not is_valid:
            logger.warning(f"Query validation failed: {error_msg}")
            raise HTTPException(
                status_code=400,
                detail=QueryError(
                    detail="Query validation failed",
                    reason=error_type.value if error_type else "unknown",
                    sql=sql
                ).model_dump()
            )

        # Step 3: Execute the query
        result = await query_executor.execute(
            sql=sql,
            max_rows=request.max_rows,
            timeout_seconds=request.timeout_seconds
        )

        # Step 4: Generate visualization (optional)
        viz = None
        if request.include_visualization and result.rows:
            viz = visualizer.visualize(result, title=request.question[:50])

        # Step 5: Generate explanation (optional)
        explanation = None
        if request.include_explanation:
            explanation = await query_explainer.explain(sql, request.question)

        # Step 6: Generate optimization suggestions (optional)
        optimization = None
        if request.include_optimization:
            optimization = await query_optimizer.analyze(sql)

        # Calculate execution time
        execution_time_ms = int(
            (datetime.utcnow() - start_time).total_seconds() * 1000
        )

        # Record in history
        history_item = QueryHistoryItem(
            id=str(len(query_history) + 1),
            question=request.question,
            sql=sql,
            executed_at=datetime.utcnow(),
            execution_time_ms=execution_time_ms,
            row_count=result.row_count,
            success=True
        )
        query_history.append(history_item)

        return QueryResponse(
            question=request.question,
            sql=sql,
            results=result,
            visualization=viz,
            explanation=explanation,
            optimization=optimization,
            execution_time_ms=execution_time_ms,
            cached=False
        )

    except QueryExecutionError as e:
        logger.error(f"Query execution failed: {e}")
        raise HTTPException(
            status_code=500,
            detail=QueryError(
                detail="Query execution failed",
                reason=str(e),
                sql=e.sql
            ).model_dump()
        )

    except HTTPException:
        raise

    except Exception as e:
        logger.exception(f"Unexpected error: {e}")
        raise HTTPException(
            status_code=500,
            detail=QueryError(
                detail="An unexpected error occurred",
```

### Challenge 1: Query Caching (Intermediate)

Implement intelligent query caching to avoid re-executing identical queries:

```python
# services/cache.py
"""Query result caching with TTL."""

import hashlib
import json
from datetime import datetime, timedelta
from typing import Optional, Dict, Any
from dataclasses import dataclass


@dataclass
class CacheEntry:
    result: Dict[str, Any]
    created_at: datetime
    ttl_seconds: int

    def is_expired(self) -> bool:
        return datetime.utcnow() > self.created_at + timedelta(seconds=self.ttl_seconds)


class QueryCache:
    """LRU cache for query results."""

    def __init__(self, max_size: int = 100, default_ttl: int = 300):
        self.max_size = max_size
        self.default_ttl = default_ttl
        self._cache: Dict[str, CacheEntry] = {}

    def _hash_query(self, sql: str, params: Optional[Dict] = None) -> str:
        """Generate cache key from SQL and parameters."""
        key_data = {"sql": sql.strip().lower(), "params": params or {}}
        return hashlib.sha256(json.dumps(key_data, sort_keys=True).encode()).hexdigest()

    def get(self, sql: str, params: Optional[Dict] = None) -> Optional[Dict[str, Any]]:
        """Get cached result if available and not expired."""
        key = self._hash_query(sql, params)
        entry = self._cache.get(key)

        if entry is None:
            return None

        if entry.is_expired():
            del self._cache[key]
            return None

        return entry.result

    def set(
        self,
        sql: str,
        result: Dict[str, Any],
        params: Optional[Dict] = None,
        ttl: Optional[int] = None
    ) -> None:
        """Cache query result."""
        # Evict oldest entries if at capacity
        if len(self._cache) >= self.max_size:
            oldest_key = min(self._cache.keys(), key=lambda k: self._cache[k].created_at)
            del self._cache[oldest_key]

        key = self._hash_query(sql, params)
        self._cache[key] = CacheEntry(
            result=result,
            created_at=datetime.utcnow(),
            ttl_seconds=ttl or self.default_ttl
        )

    def invalidate(self, pattern: Optional[str] = None) -> int:
        """Invalidate cache entries. If pattern provided, only matching entries."""
        if pattern is None:
            count = len(self._cache)
            self._cache.clear()
            return count

        # Simple pattern matching on SQL
        keys_to_delete = [
            k for k, v in self._cache.items()
            if pattern.lower() in str(v.result.get("sql", "")).lower()
        ]
        for key in keys_to_delete:
            del self._cache[key]
        return len(keys_to_delete)
```

```python
from typing import Dict, List, Optional

@dataclass
class DatabaseConfig:
    name: str
    url: str
    db_type: str
    description: str


class MultiDatabaseManager:
    """Manage connections to multiple databases."""

    def __init__(self):
        self.databases: Dict[str, DatabaseConfig] = {}
        self.loaders: Dict[str, SchemaLoader] = {}

    def register_database(self, config: DatabaseConfig) -> None:
        """Register a new database."""
        self.databases[config.name] = config
        # Create schema loader for this database
        loader = SchemaLoader()
        loader.settings.database_url = config.url
        loader.settings.database_type = config.db_type
        self.loaders[config.name] = loader

    def get_combined_schema_context(self) -> str:
        """Get schema context for all databases."""
        contexts = []
        for name, loader in self.loaders.items():
            db_config = self.databases[name]
            contexts.append(f"=== Database: {name} ({db_config.description}) ===")
            contexts.append(loader.get_schema_context())
        return "\n\n".join(contexts)

    def route_query(self, sql: str) -> str:
        """Determine which database a query should run against."""
        # Simple routing based on table names
        sql_lower = sql.lower()

        for name, loader in self.loaders.items():
            schema = loader.get_schema()
            for table in schema.tables:
                if table.name.lower() in sql_lower:
                    return name

        # Default to first database
        return list(self.databases.keys())[0]
```

### Challenge 3: Natural Language Filters (Advanced)

Allow users to refine results with follow-up natural language:

```python
# services/filter_parser.py
"""Parse natural language filters and apply to results."""

from typing import List, Any, Dict, Optional
from openai import AsyncOpenAI
from pydantic import BaseModel


class FilterCondition(BaseModel):
    column: str
    operator: str  # eq, ne, gt, lt, gte, lte, contains, startswith
    value: Any


class ParsedFilter(BaseModel):
    conditions: List[FilterCondition]
    sort_column: Optional[str] = None
    sort_direction: str = "asc"
    limit: Optional[int] = None


class FilterParser:
    """Parse natural language into filter conditions."""

    SYSTEM_PROMPT = """Parse natural language filter requests into structured conditions.

Given a filter request and available columns, extract:
1. Filter conditions (column, operator, value)
2. Sort preferences
3. Limit preferences

Operators: eq (equals), ne (not equals), gt (greater than), lt (less than),
           gte (>=), lte (<=), contains, startswith

Example:
Input: "only show amounts over 1000, sorted by date"
Columns: ["id", "amount", "date", "status"]
Output: {
    "conditions": [{"column": "amount", "operator": "gt", "value": 1000}],
    "sort_column": "date",
    "sort_direction": "asc"
}"""

    def __init__(self):
        self.client = AsyncOpenAI()

    async def parse(
        self,
        filter_text: str,
        columns: List[str]
    ) -> ParsedFilter:
        """Parse natural language filter into structured format."""
        response = await self.client.beta.chat.completions.parse(
            model="gpt-4o",
            messages=[
                {"role": "system", "content": self.SYSTEM_PROMPT},
                {"role": "user", "content": f"Columns: {columns}\nFilter: {filter_text}"}
            ],
            response_format=ParsedFilter
        )
        return response.choices[0].message.parsed

    def apply_filter(
        self,
        rows: List[List[Any]],
        columns: List[str],
        parsed: ParsedFilter
    ) -> List[List[Any]]:
        """Apply parsed filter to result rows."""
        filtered = rows

        # Apply conditions
        for condition in parsed.conditions:
            if condition.column not in columns:
                continue
            col_idx = columns.index(condition.column)
            filtered = [
                row for row in filtered
                if self._matches_condition(row[col_idx], condition)
            ]

        # Apply sort
        if parsed.sort_column and parsed.sort_column in columns:
            sort_idx = columns.index(parsed.sort_column)
            reverse = parsed.sort_direction == "desc"
            filtered = sorted(filtered, key=lambda r: r[sort_idx] or "", reverse=reverse)

        # Apply limit
        if parsed.limit:
            filtered = filtered[:parsed.limit]

        return filtered

    def _matches_condition(self, value: Any, condition: FilterCondition) -> bool:
        """Check if value matches condition."""
        target = condition.value

        if condition.operator == "eq":
            return value == target
        elif condition.operator == "ne":
            return value != target
        elif condition.operator == "gt":
            return value > target
        elif condition.operator == "lt":
            return value < target
        elif condition.operator == "gte":
            return value >= target
        elif condition.operator == "lte":
            return value <= target
        elif condition.operator == "contains":
            return str(target).lower() in str(value).lower()
        elif condition.operator == "startswith":
            return str(value).lower().startswith(str(target).lower())

        return True
```

1. **Schema is Context**: The quality of SQL generation depends heavily on providing rich schema context to the LLM. Include table relationships, sample values, and row counts.

2. **Defense in Depth**: Never trust a single security layer. Combine pattern matching, AST parsing, statement type validation, and read-only connections.

3. **Structured Outputs**: Use LLM structured outputs (JSON schema) to get predictable, parseable responses. This eliminates regex parsing of free-form text.

4. **Graceful Degradation**: If visualization fails, return tables. If explanation fails, return raw results. Always provide value to the user.

5. **Transparency**: Show users the generated SQL, explain what it does, and document assumptions. This builds trust and helps users learn.
