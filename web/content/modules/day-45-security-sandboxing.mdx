## The Danger of Code Execution

When your agent can execute code, it can do anything—including `rm -rf /`. Today we learn how to contain the blast radius.

<Callout type="warning">
  **Real-World Analogy:** Think of code execution like giving someone the keys
  to your house. Without sandboxing, you're handing over the master key to your
  entire property. With proper sandboxing, you're giving them a key to a single
  room with reinforced walls, no windows, and a time-limited lock—even if they
  try to break out, the damage is contained.
</Callout>

[E2B](https://e2b.dev) provides Firecracker-based sandboxes with an AI-friendly SDK. It's an open-source runtime designed specifically for executing AI-generated code securely in cloud sandboxes.

```python
from e2b_code_interpreter import Sandbox

def execute_safely(code: str) -> dict:
    """Execute code in a secure sandbox."""

    with Sandbox() as sandbox:
        # Sandbox starts in ~200-400ms

        # Execute the code
        execution = sandbox.run_code(code)

        # Handle results
        if execution.error:
            return {
                "success": False,
                "error": f"{execution.error.name}: {execution.error.value}"
            }

        # Collect outputs
        results = []
        for result in execution.results:
            if result.text:
                results.append({"type": "text", "content": result.text})
            if result.png:
                results.append({"type": "image", "content": result.png})

        return {
            "success": True,
            "results": results,
            "stdout": execution.logs.stdout,
            "stderr": execution.logs.stderr
        }

# Usage
result = execute_safely("""
import pandas as pd
import matplotlib.pyplot as plt

df = pd.DataFrame({'x': [1,2,3], 'y': [4,5,6]})
plt.plot(df['x'], df['y'])
plt.savefig('plot.png')
print("Plot created!")
""")
```

For full control, build a Docker-based sandbox:

```python
import docker
import tempfile
import os
from pathlib import Path

class DockerSandbox:
    """Docker-based code execution sandbox."""

    def __init__(self, image: str = "python:3.11-slim"):
        self.client = docker.from_env()
        self.image = image

        # Resource limits
        self.mem_limit = "256m"
        self.cpu_period = 100000
        self.cpu_quota = 50000  # 50% of one CPU
        self.timeout = 30

    def execute(self, code: str, files: dict = None) -> dict:
        """Execute code in isolated container."""

        # Create temp directory for code and files
        with tempfile.TemporaryDirectory() as tmpdir:
            # Write code to file
            code_path = Path(tmpdir) / "code.py"
            code_path.write_text(code)

            # Write any additional files
            if files:
                for name, content in files.items():
                    (Path(tmpdir) / name).write_text(content)

            try:
                # Run container with strict limits
                result = self.client.containers.run(
                    self.image,
                    command=["python", "/sandbox/code.py"],
                    volumes={tmpdir: {"bind": "/sandbox", "mode": "ro"}},
                    mem_limit=self.mem_limit,
                    cpu_period=self.cpu_period,
                    cpu_quota=self.cpu_quota,
                    network_disabled=True,  # No network access!
                    read_only=True,  # Read-only filesystem
                    remove=True,
                    timeout=self.timeout,
                    user="nobody",  # Non-root user
                    security_opt=["no-new-privileges"],
                )

                return {
                    "success": True,
                    "output": result.decode("utf-8")
                }

            except docker.errors.ContainerError as e:
                return {
                    "success": False,
                    "error": e.stderr.decode("utf-8") if e.stderr else str(e)
                }
            except Exception as e:
                return {
                    "success": False,
                    "error": str(e)
                }

# Usage
sandbox = DockerSandbox()
result = sandbox.execute("""
print("Hello from sandbox!")
print(2 + 2)
""")
```

```python
# Using wasmtime-py (2024)
from wasmtime import Store, Module, Instance, Func, FuncType, ValType

def create_wasm_sandbox():
    """Create a WASM sandbox with limited capabilities."""
    store = Store()

    # Define allowed imports (capabilities)
    def log_message(msg_ptr: int, msg_len: int):
        # Only allow logging, no file/network access
        print(f"[SANDBOX LOG]: {msg_ptr}, {msg_len}")

    # Load and instantiate WASM module
    module = Module.from_file(store.engine, "sandbox.wasm")

    # Only expose safe functions
    imports = [
        Func(store, FuncType([ValType.i32(), ValType.i32()], []), log_message)
    ]

    instance = Instance(store, module, imports)
    return instance

# Wasmer alternative with WASI
import wasmer
from wasmer import Store, Module, Instance
from wasmer_compiler_cranelift import Compiler

store = Store(Compiler())
module = Module(store, open("sandbox.wasm", "rb").read())

# WASI with restricted capabilities
from wasmer import wasi

wasi_config = wasi.StateBuilder("sandbox").finalize()
instance = Instance(module, wasi_config.generate_imports(store, module))
```

### When to Use WASM vs Containers

<Callout type="tip">
  **Use WASM when:** Sub-millisecond startup is critical (real-time agents),

Add a pre-execution validation layer:

```python
import ast
import re

class CodeValidator:
    """Validate code before execution."""

    # Dangerous patterns
    BLOCKED_IMPORTS = {
        'os', 'subprocess', 'sys', 'shutil',
        'socket', 'requests', 'urllib', 'http'
    }

    BLOCKED_FUNCTIONS = {
        'eval', 'exec', 'compile', '__import__',
        'open', 'input', 'breakpoint'
    }

    BLOCKED_PATTERNS = [
        r'__\w+__',  # Dunder methods
        r'\.system\s*\(',  # os.system
        r'\.popen\s*\(',  # os.popen
        r'subprocess\.',  # subprocess module
    ]

    def validate(self, code: str) -> tuple[bool, str]:
        """Validate code. Returns (is_valid, error_message)."""

        # Check for blocked patterns
        for pattern in self.BLOCKED_PATTERNS:
            if re.search(pattern, code):
                return False, f"Blocked pattern detected: {pattern}"

        # Parse AST
        try:
            tree = ast.parse(code)
        except SyntaxError as e:
            return False, f"Syntax error: {e}"

        # Check imports
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                for alias in node.names:
                    if alias.name.split('.')[0] in self.BLOCKED_IMPORTS:
                        return False, f"Blocked import: {alias.name}"

            elif isinstance(node, ast.ImportFrom):
                if node.module and node.module.split('.')[0] in self.BLOCKED_IMPORTS:
                    return False, f"Blocked import: {node.module}"

            elif isinstance(node, ast.Call):
                if isinstance(node.func, ast.Name):
                    if node.func.id in self.BLOCKED_FUNCTIONS:
                        return False, f"Blocked function: {node.func.id}"

        return True, "Code validated successfully"

# Usage
validator = CodeValidator()
is_valid, message = validator.validate("""
import os
os.system('rm -rf /')
""")
# Returns: (False, "Blocked import: os")
```

```python
import subprocess
import json
import tempfile
from pathlib import Path

class AdvancedCodeValidator:
    """Production-grade code validation with Semgrep and Bandit."""

    def __init__(self):
        # Custom Semgrep rules for agent security
        self.semgrep_rules = """
rules:
  - id: dangerous-exec
    patterns:
      - pattern-either:
          - pattern: exec(...)
          - pattern: eval(...)
          - pattern: compile(..., ..., "exec")
    message: "Dangerous code execution detected"
    severity: ERROR
    languages: [python]

  - id: network-access
    patterns:
      - pattern-either:
          - pattern: requests.$METHOD(...)
          - pattern: urllib.request.urlopen(...)
          - pattern: httpx.$METHOD(...)
    message: "Network access attempt detected"
    severity: WARNING
    languages: [python]

  - id: file-system-write
    patterns:
      - pattern-either:
          - pattern: open($PATH, "w", ...)
          - pattern: open($PATH, "a", ...)
          - pattern: Path(...).write_text(...)
    message: "File write attempt detected"
    severity: WARNING
    languages: [python]
"""

    def validate_with_semgrep(self, code: str) -> dict:
        """Run Semgrep security scan."""
        with tempfile.TemporaryDirectory() as tmpdir:
            # Write code and rules
            code_path = Path(tmpdir) / "code.py"
            rules_path = Path(tmpdir) / "rules.yaml"
            code_path.write_text(code)
            rules_path.write_text(self.semgrep_rules)

            # Run Semgrep
            result = subprocess.run(
                ["semgrep", "--config", str(rules_path),
                 "--json", str(code_path)],
                capture_output=True, text=True, timeout=30
            )

            findings = json.loads(result.stdout)
            return {
                "valid": len(findings.get("results", [])) == 0,
                "findings": findings.get("results", [])
            }

    def validate_with_bandit(self, code: str) -> dict:
        """Run Bandit Python security scanner."""
        with tempfile.NamedTemporaryFile(suffix=".py", mode="w") as f:
            f.write(code)
            f.flush()

            result = subprocess.run(
                ["bandit", "-f", "json", "-ll", f.name],
                capture_output=True, text=True, timeout=30
            )

            findings = json.loads(result.stdout) if result.stdout else {}
            issues = findings.get("results", [])

            return {
                "valid": len(issues) == 0,
                "findings": issues,
                "severity_counts": findings.get("metrics", {})
            }

    def validate(self, code: str) -> dict:
        """Run all validators."""
        semgrep_result = self.validate_with_semgrep(code)
        bandit_result = self.validate_with_bandit(code)

        return {
            "valid": semgrep_result["valid"] and bandit_result["valid"],
            "semgrep": semgrep_result,
            "bandit": bandit_result
        }
```

### Real-Time Validation During Agent Execution

Integrate validation into your agent's tool execution loop:

```python
from typing import Callable
import asyncio

class ValidatedCodeTool:
    """Code execution tool with real-time validation."""

    def __init__(self, sandbox, validator):
        self.sandbox = sandbox
        self.validator = validator
        self.validation_cache = {}  # Cache validation results

    async def execute(
        self,
        code: str,
        on_validation_start: Callable = None,
        on_validation_complete: Callable = None
    ) -> dict:
        """Execute code with streaming validation feedback."""

        # Check cache first
        code_hash = hash(code)
        if code_hash in self.validation_cache:
            validation = self.validation_cache[code_hash]
        else:
            if on_validation_start:
                await on_validation_start()

            # Run validation (can be async for Semgrep/Bandit)
            validation = await asyncio.to_thread(
                self.validator.validate, code
            )
            self.validation_cache[code_hash] = validation

            if on_validation_complete:
                await on_validation_complete(validation)

        if not validation["valid"]:
            return {
                "success": False,
                "stage": "validation",
                "errors": validation.get("findings", [])
            }

        # Execute in sandbox
        return await asyncio.to_thread(self.sandbox.execute, code)

# Usage in agent loop
async def agent_code_execution(code: str):
    tool = ValidatedCodeTool(sandbox, validator)

    result = await tool.execute(
        code,
        on_validation_start=lambda: print("🔍 Validating code..."),
        on_validation_complete=lambda v: print(
            f"✅ Validation passed" if v["valid"]
            else f"❌ Found {len(v.get('findings', []))} issues"
        )
    )
    return result
```

```python
from enum import Enum
from dataclasses import dataclass
from typing import Set, Optional, Callable
import logging

class Permission(Enum):
    """Tool permission levels."""
    READ_FILES = "read_files"
    WRITE_FILES = "write_files"
    EXECUTE_CODE = "execute_code"
    NETWORK_ACCESS = "network_access"
    SHELL_ACCESS = "shell_access"
    DATABASE_READ = "database_read"
    DATABASE_WRITE = "database_write"
    SEND_EMAIL = "send_email"
    API_CALLS = "api_calls"

@dataclass
class ToolDefinition:
    """Tool with required permissions."""
    name: str
    function: Callable
    required_permissions: Set[Permission]
    description: str
    risk_level: str = "low"  # low, medium, high, critical

class PermissionManager:
    """Capability-based permission system for agents."""

    def __init__(self):
        self.granted_permissions: Set[Permission] = set()
        self.denied_permissions: Set[Permission] = set()
        self.escalation_callback: Optional[Callable] = None
        self.audit_log = []

    def grant(self, *permissions: Permission):
        """Grant permissions to the agent."""
        for perm in permissions:
            self.granted_permissions.add(perm)
            self.denied_permissions.discard(perm)

    def deny(self, *permissions: Permission):
        """Explicitly deny permissions."""
        for perm in permissions:
            self.denied_permissions.add(perm)
            self.granted_permissions.discard(perm)

    def set_escalation_handler(self, callback: Callable):
        """Set callback for permission escalation requests."""
        self.escalation_callback = callback

    def check_permission(
        self,
        tool: ToolDefinition,
        context: dict = None
    ) -> tuple[bool, str]:
        """Check if tool execution is permitted."""

        missing = tool.required_permissions - self.granted_permissions
        denied = tool.required_permissions & self.denied_permissions

        # Log the check
        self.audit_log.append({
            "tool": tool.name,
            "required": [p.value for p in tool.required_permissions],
            "granted": [p.value for p in self.granted_permissions],
            "context": context
        })

        if denied:
            return False, f"Explicitly denied: {[p.value for p in denied]}"

        if missing:
            # Attempt escalation for high-risk tools
            if tool.risk_level in ("high", "critical") and self.escalation_callback:
                approved = self.escalation_callback(tool, missing, context)
                if approved:
                    self.grant(*missing)
                    return True, "Approved via escalation"
            return False, f"Missing permissions: {[p.value for p in missing]}"

        return True, "Permitted"

    def execute_tool(
        self,
        tool: ToolDefinition,
        *args,
        context: dict = None,
        **kwargs
    ):
        """Execute tool with permission check."""
        allowed, reason = self.check_permission(tool, context)

        if not allowed:
            logging.warning(f"Tool {tool.name} blocked: {reason}")
            raise PermissionError(f"Tool '{tool.name}' not permitted: {reason}")

        logging.info(f"Executing tool {tool.name}")
        return tool.function(*args, **kwargs)

# Usage
permissions = PermissionManager()

# Grant minimal permissions (least privilege)
permissions.grant(
    Permission.READ_FILES,
    Permission.EXECUTE_CODE
)

# Define tools with required permissions
read_file_tool = ToolDefinition(
    name="read_file",
    function=lambda path: open(path).read(),
    required_permissions={Permission.READ_FILES},
    description="Read a file from disk",
    risk_level="low"
)

shell_tool = ToolDefinition(
    name="run_shell",
    function=lambda cmd: subprocess.run(cmd, shell=True),
    required_permissions={Permission.SHELL_ACCESS, Permission.EXECUTE_CODE},
    description="Run shell command",
    risk_level="critical"
)

# This works
content = permissions.execute_tool(read_file_tool, "readme.md")

# This raises PermissionError
# permissions.execute_tool(shell_tool, "ls -la")
```

### Runtime Permission Escalation

For dynamic workflows, implement human-in-the-loop escalation:

```python
import asyncio
from typing import Awaitable

class EscalationManager:
    """Handle runtime permission escalation requests."""

    def __init__(self, timeout: int = 300):
        self.timeout = timeout
        self.pending_requests = {}

    async def request_escalation(
        self,
        tool: ToolDefinition,
        missing_permissions: Set[Permission],
        context: dict,
        notify_callback: Callable[[dict], Awaitable[None]]
    ) -> bool:
        """Request human approval for permission escalation."""

        request_id = f"{tool.name}_{hash(frozenset(missing_permissions))}"

        # Notify human (e.g., Slack, email, UI)
        await notify_callback({
            "request_id": request_id,
            "tool": tool.name,
            "description": tool.description,
            "risk_level": tool.risk_level,
            "permissions_needed": [p.value for p in missing_permissions],
            "context": context,
            "timeout": self.timeout
        })

        # Wait for approval
        approval_event = asyncio.Event()
        self.pending_requests[request_id] = {
            "event": approval_event,
            "approved": False
        }

        try:
            await asyncio.wait_for(approval_event.wait(), self.timeout)
            return self.pending_requests[request_id]["approved"]
        except asyncio.TimeoutError:
            return False
        finally:
            del self.pending_requests[request_id]

    def approve(self, request_id: str):
        """Approve a pending escalation request."""
        if request_id in self.pending_requests:
            self.pending_requests[request_id]["approved"] = True
            self.pending_requests[request_id]["event"].set()

    def deny(self, request_id: str):
        """Deny a pending escalation request."""
        if request_id in self.pending_requests:
            self.pending_requests[request_id]["approved"] = False
            self.pending_requests[request_id]["event"].set()
```

### Least Privilege Patterns

<Callout type="warning">
  **The Principle of Least Privilege for Agents:** (1) Start with zero
  permissions - agents earn trust. (2) Grant per-task - permissions expire after
  task completion. (3) Scope narrowly - use `read_file:/data/*` not
  `read_file:*`. (4) Log everything - audit trail for all permission checks. (5)
  Fail closed - deny by default, require explicit grants.
</Callout>

```python
from contextlib import contextmanager

class ScopedPermissions:
    """Context manager for temporary permission grants."""

    def __init__(self, manager: PermissionManager):
        self.manager = manager
        self.original_permissions = set()

    @contextmanager
    def temporary_grant(self, *permissions: Permission):
        """Grant permissions for the duration of a context."""
        self.original_permissions = self.manager.granted_permissions.copy()

        try:
            self.manager.grant(*permissions)
            yield
        finally:
            # Restore original permissions
            self.manager.granted_permissions = self.original_permissions

# Usage
scoped = ScopedPermissions(permissions)

# Temporarily grant network access for API call
with scoped.temporary_grant(Permission.NETWORK_ACCESS):
    result = permissions.execute_tool(api_call_tool, "https://api.example.com")
# Network access automatically revoked
```

```python
from azure.mgmt.containerinstance import ContainerInstanceManagementClient

            self.resource_group, container_name, container_name
        )

        # Cleanup
        self.client.container_groups.begin_delete(
            self.resource_group, container_name
        )

        return {"output": logs.content}
```

```python
class SecureCodeExecutor:
    """Complete secure code execution pipeline."""

    def __init__(self):
        self.validator = CodeValidator()
        self.sandbox = DockerSandbox()

    def execute(self, code: str) -> dict:
        """Validate and execute code securely."""

        # Layer 1: Static validation
        is_valid, message = self.validator.validate(code)
        if not is_valid:
            return {
                "success": False,
                "stage": "validation",
                "error": message
            }

        # Layer 2: Sandbox execution
        result = self.sandbox.execute(code)

        # Layer 3: Output sanitization
        if result["success"]:
            result["output"] = self._sanitize_output(result["output"])

        return result

    def _sanitize_output(self, output: str) -> str:
        """Remove sensitive information from output."""
        # Remove potential secrets
        patterns = [
            (r'[A-Za-z0-9+/]{40,}', '[REDACTED_KEY]'),  # Base64 keys
            (r'sk-[A-Za-z0-9]{48}', '[REDACTED_API_KEY]'),  # OpenAI keys
            (r'ghp_[A-Za-z0-9]{36}', '[REDACTED_GITHUB_TOKEN]'),  # GitHub tokens
        ]

        for pattern, replacement in patterns:
            output = re.sub(pattern, replacement, output)

        return output
```
