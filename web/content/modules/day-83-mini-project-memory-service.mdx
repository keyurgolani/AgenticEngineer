## Project Overview

**Objective**: Build a memory service that provides short-term (conversation), long-term (facts), and episodic (experiences) memory for agents. This enables agents to remember user preferences, learn from past interactions, and maintain context.

**Why This Matters**: Stateless agents forget everything between calls. Memory transforms agents from tools into assistants that truly understand users over time.

**Skills Showcased**:

- Memory architecture patterns
- Vector embeddings for semantic search
- Redis for fast access
- PostgreSQL for persistence
- Memory consolidation strategies

## Module Summary

In this mini-project, we built a comprehensive memory service that gives agents a "brain" with three distinct memory types:

1. **Short-Term Memory (Redis)**: For immediate context and conversation history.
2. **Long-Term Memory (PostgreSQL + Vectors)**: For persistent facts, preferences, and knowledge.
3. **Episodic Memory**: For remembering past interactions and outcomes.

We also implemented a consolidation background job to process finished sessions and distill them into long-term knowledge, mimicking human memory consolidation during sleep.

```python
# services/embeddings.py
from sentence_transformers import SentenceTransformer
import numpy as np
from typing import List, Union
import json

class EmbeddingService:
    """Generate and compare embeddings for semantic memory search."""

    def __init__(self, model_name: str = "all-MiniLM-L6-v2"):
        self.model = SentenceTransformer(model_name)
        self.dimension = self.model.get_sentence_embedding_dimension()

    def embed(self, text: Union[str, List[str]]) -> np.ndarray:
        """Generate embedding(s) for text."""
        if isinstance(text, str):
            text = [text]

        embeddings = self.model.encode(text, convert_to_numpy=True)
        return embeddings

    def embed_to_json(self, text: str) -> str:
        """Generate embedding and return as JSON string (for SQLite)."""
        embedding = self.embed(text)[0]
        return json.dumps(embedding.tolist())

    def json_to_embedding(self, json_str: str) -> np.ndarray:
        """Convert JSON string back to numpy array."""
        return np.array(json.loads(json_str))

    def cosine_similarity(self, a: np.ndarray, b: np.ndarray) -> float:
        """Calculate cosine similarity between two embeddings."""
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))

    def find_similar(
        self,
        query_embedding: np.ndarray,
        candidates: List[tuple],  # [(id, embedding_json), ...]
        top_k: int = 5,
        threshold: float = 0.5
    ) -> List[tuple]:
        """Find most similar memories."""
        results = []

        for memory_id, embedding_json in candidates:
            if not embedding_json:
                continue

            candidate_embedding = self.json_to_embedding(embedding_json)
            similarity = self.cosine_similarity(query_embedding, candidate_embedding)

            if similarity >= threshold:
                results.append((memory_id, similarity))

        # Sort by similarity descending
        results.sort(key=lambda x: x[1], reverse=True)
        return results[:top_k]

# Singleton instance
_embedding_service = None

def get_embedding_service() -> EmbeddingService:
    global _embedding_service
    if _embedding_service is None:
        _embedding_service = EmbeddingService()
    return _embedding_service
```

```python
# services/short_term.py
import redis
import json
from typing import List, Dict, Any, Optional
from datetime import datetime, timedelta
import os

class ShortTermMemory:
    """Fast, session-based memory using Redis."""

    def __init__(self, redis_url: str = None):
        self.redis_url = redis_url or os.getenv("REDIS_URL", "redis://localhost:6379")
        self.client = redis.from_url(self.redis_url, decode_responses=True)
        self.default_ttl = 3600 * 4  # 4 hours

    def _key(self, user_id: str, session_id: str, suffix: str = "") -> str:
        """Generate Redis key."""
        base = f"stm:{user_id}:{session_id}"
        return f"{base}:{suffix}" if suffix else base

    # Conversation History
    def add_message(
        self,
        user_id: str,
        session_id: str,
        role: str,
        content: str,
        metadata: Dict[str, Any] = None
    ):
        """Add a message to conversation history."""
        key = self._key(user_id, session_id, "messages")

        message = {
            "role": role,
            "content": content,
            "timestamp": datetime.utcnow().isoformat(),
            "metadata": metadata or {}
        }

        self.client.rpush(key, json.dumps(message))
        self.client.expire(key, self.default_ttl)

    def get_messages(
        self,
        user_id: str,
        session_id: str,
        limit: int = 20
    ) -> List[Dict]:
        """Get recent conversation messages."""
        key = self._key(user_id, session_id, "messages")

        # Get last N messages
        messages = self.client.lrange(key, -limit, -1)
        return [json.loads(m) for m in messages]

    def get_context_window(
        self,
        user_id: str,
        session_id: str,
        max_tokens: int = 4000
    ) -> List[Dict]:
        """Get messages that fit within token limit."""
        messages = self.get_messages(user_id, session_id, limit=50)

        # Rough token estimation (4 chars per token)
        result = []
        total_chars = 0
        max_chars = max_tokens * 4

        for msg in reversed(messages):
            msg_chars = len(msg["content"])
            if total_chars + msg_chars > max_chars:
                break
            result.insert(0, msg)
            total_chars += msg_chars

        return result

    # Working Memory (current task context)
    def set_working_memory(
        self,
        user_id: str,
        session_id: str,
        key: str,
        value: Any
    ):
        """Store working memory item."""
        redis_key = self._key(user_id, session_id, f"working:{key}")
        self.client.set(redis_key, json.dumps(value), ex=self.default_ttl)

    def get_working_memory(
        self,
        user_id: str,
        session_id: str,
        key: str
    ) -> Any:
        """Retrieve working memory item."""
        redis_key = self._key(user_id, session_id, f"working:{key}")
        value = self.client.get(redis_key)
        return json.loads(value) if value else None

    def get_all_working_memory(
        self,
        user_id: str,
        session_id: str
    ) -> Dict[str, Any]:
        """Get all working memory items."""
        pattern = self._key(user_id, session_id, "working:*")
        keys = self.client.keys(pattern)

        result = {}
        for key in keys:
            item_key = key.split(":")[-1]
            value = self.client.get(key)
            if value:
                result[item_key] = json.loads(value)

        return result

    # Session Summary
    def set_session_summary(
        self,
        user_id: str,
        session_id: str,
        summary: str
    ):
        """Store session summary for consolidation."""
        key = self._key(user_id, session_id, "summary")
        self.client.set(key, summary, ex=self.default_ttl * 2)

    def get_session_summary(
        self,
        user_id: str,
        session_id: str
    ) -> Optional[str]:
        """Get session summary."""
        key = self._key(user_id, session_id, "summary")
        return self.client.get(key)

    # Session Management
    def end_session(self, user_id: str, session_id: str) -> Dict:
        """End session and return data for consolidation."""
        messages = self.get_messages(user_id, session_id, limit=100)
        working_memory = self.get_all_working_memory(user_id, session_id)
        summary = self.get_session_summary(user_id, session_id)

        # Clean up (optional - let TTL handle it)
        # pattern = self._key(user_id, session_id, "*")
        # for key in self.client.keys(pattern):
        #     self.client.delete(key)

        return {
            "messages": messages,
            "working_memory": working_memory,
            "summary": summary,
            "ended_at": datetime.utcnow().isoformat()
        }

    def list_active_sessions(self, user_id: str) -> List[str]:
        """List active sessions for a user."""
        pattern = f"stm:{user_id}:*:messages"
        keys = self.client.keys(pattern)

        sessions = set()
        for key in keys:
            parts = key.split(":")
            if len(parts) >= 3:
                sessions.add(parts[2])

        return list(sessions)
```

```python
from sqlalchemy.orm import Session
from models.database import LongTermMemory
from services.embeddings import get_embedding_service
from typing import List, Dict, Any, Optional
from datetime import datetime
import json

class LongTermMemoryService:
    """Persistent memory with semantic search."""

    def __init__(self, db: Session):
        self.db = db
        self.embeddings = get_embedding_service()

    def store(
        self,
        user_id: str,
        content: str,
        memory_type: str = "fact",
        agent_id: str = None,
        importance: float = 0.5,
        source_type: str = "conversation",
        source_id: str = None,
        summary: str = None
    ) -> LongTermMemory:
        """Store a new long-term memory."""

        # Generate embedding
        embedding_json = self.embeddings.embed_to_json(content)

        memory = LongTermMemory(
            user_id=user_id,
            agent_id=agent_id,
            memory_type=memory_type,
            content=content,
            summary=summary or content[:200],
            embedding_json=embedding_json,
            importance=importance,
            source_type=source_type,
            source_id=source_id
        )

        self.db.add(memory)
        self.db.commit()
        self.db.refresh(memory)

        return memory

    def recall(
        self,
        user_id: str,
        query: str,
        memory_type: str = None,
        top_k: int = 5,
        threshold: float = 0.5
    ) -> List[Dict]:
        """Recall memories semantically similar to query."""

        # Generate query embedding
        query_embedding = self.embeddings.embed(query)[0]

        # Get candidate memories
        query_obj = self.db.query(
            LongTermMemory.id,
            LongTermMemory.embedding_json
        ).filter(LongTermMemory.user_id == user_id)

        if memory_type:
            query_obj = query_obj.filter(LongTermMemory.memory_type == memory_type)

        candidates = query_obj.all()

        # Find similar
        similar = self.embeddings.find_similar(
            query_embedding,
            candidates,
            top_k=top_k,
            threshold=threshold
        )

        # Fetch full memories
        results = []
        for memory_id, similarity in similar:
            memory = self.db.query(LongTermMemory).filter(
                LongTermMemory.id == memory_id
            ).first()

            if memory:
                # Update access stats
                memory.access_count += 1
                memory.last_accessed = datetime.utcnow()

                results.append({
                    "id": memory.id,
                    "content": memory.content,
                    "summary": memory.summary,
                    "memory_type": memory.memory_type,
                    "importance": memory.importance,
                    "similarity": similarity,
                    "created_at": memory.created_at.isoformat()
                })

        self.db.commit()
        return results

    def get_by_type(
        self,
        user_id: str,
        memory_type: str,
        limit: int = 10
    ) -> List[LongTermMemory]:
        """Get memories by type, ordered by importance."""
        return self.db.query(LongTermMemory).filter(
            LongTermMemory.user_id == user_id,
            LongTermMemory.memory_type == memory_type
        ).order_by(
            LongTermMemory.importance.desc()
        ).limit(limit).all()

    def get_recent(
        self,
        user_id: str,
        limit: int = 10
    ) -> List[LongTermMemory]:
        """Get most recently accessed memories."""
        return self.db.query(LongTermMemory).filter(
            LongTermMemory.user_id == user_id
        ).order_by(
            LongTermMemory.last_accessed.desc()
        ).limit(limit).all()

    def update_importance(
        self,
        memory_id: int,
        importance: float
    ):
        """Update memory importance score."""
        memory = self.db.query(LongTermMemory).filter(
            LongTermMemory.id == memory_id
        ).first()

        if memory:
            memory.importance = max(0, min(1, importance))
            self.db.commit()

    def merge_memories(
        self,
        memory_ids: List[int],
        merged_content: str,
        merged_summary: str
    ) -> LongTermMemory:
        """Merge multiple memories into one."""
        memories = self.db.query(LongTermMemory).filter(
            LongTermMemory.id.in_(memory_ids)
        ).all()

        if not memories:
            raise ValueError("No memories found to merge")

        # Create merged memory
        merged = LongTermMemory(
            user_id=memories[0].user_id,
            agent_id=memories[0].agent_id,
            memory_type=memories[0].memory_type,
            content=merged_content,
            summary=merged_summary,
            embedding_json=self.embeddings.embed_to_json(merged_content),
            importance=max(m.importance for m in memories),
            source_type="consolidation",
            source_id=",".join(str(m.id) for m in memories)
        )

        self.db.add(merged)

        # Mark old memories as merged (or delete)
        for memory in memories:
            self.db.delete(memory)

        self.db.commit()
        self.db.refresh(merged)

        return merged

    def forget(
        self,
        user_id: str,
        memory_id: int = None,
        memory_type: str = None,
        older_than_days: int = None
    ):
        """Remove memories (selective forgetting)."""
        query = self.db.query(LongTermMemory).filter(
            LongTermMemory.user_id == user_id
        )

        if memory_id:
            query = query.filter(LongTermMemory.id == memory_id)

        if memory_type:
            query = query.filter(LongTermMemory.memory_type == memory_type)

        if older_than_days:
            from datetime import timedelta
            cutoff = datetime.utcnow() - timedelta(days=older_than_days)
            query = query.filter(LongTermMemory.last_accessed < cutoff)

        deleted = query.delete()
        self.db.commit()

        return deleted
```

```python
# services/consolidator.py
from sqlalchemy.orm import Session
from models.database import LongTermMemory, EpisodicMemory, MemoryConsolidation
from services.short_term import ShortTermMemory
from services.long_term import LongTermMemoryService
from services.episodic import EpisodicMemoryService
from services.embeddings import get_embedding_service
from langchain_openai import ChatOpenAI
from datetime import datetime
from typing import List, Dict

class MemoryConsolidator:
    """Consolidate short-term memories into long-term storage."""

    def __init__(self, db: Session):
        self.db = db
        self.stm = ShortTermMemory()
        self.ltm = LongTermMemoryService(db)
        self.episodic = EpisodicMemoryService(db)
        self.llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.1)

    async def consolidate_session(
        self,
        user_id: str,
        session_id: str
    ) -> Dict:
        """Consolidate a session's memories."""

        # Start consolidation record
        consolidation = MemoryConsolidation(
            user_id=user_id,
            source_type="short_term",
            status="running"
        )
        self.db.add(consolidation)
        self.db.commit()

        try:
            # Get session data
            session_data = self.stm.end_session(user_id, session_id)
            messages = session_data["messages"]

            if not messages:
                consolidation.status = "completed"
                consolidation.memories_processed = 0
                self.db.commit()
                return {"status": "no_messages"}

            # Extract facts and preferences
            facts = await self._extract_facts(messages)
            preferences = await self._extract_preferences(messages)

            # Store long-term memories
            memories_created = 0

            for fact in facts:
                self.ltm.store(
                    user_id=user_id,
                    content=fact["content"],
                    memory_type="fact",
                    importance=fact.get("importance", 0.5),
                    source_type="consolidation",
                    source_id=session_id
                )
                memories_created += 1

            for pref in preferences:
                self.ltm.store(
                    user_id=user_id,
                    content=pref["content"],
                    memory_type="preference",
                    importance=pref.get("importance", 0.7),
                    source_type="consolidation",
                    source_id=session_id
                )
                memories_created += 1

            # Create episode record
            episode_summary = await self._summarize_session(messages)
            self.episodic.record_episode(
                user_id=user_id,
                session_id=session_id,
                episode_type="conversation",
                summary=episode_summary,
                details={"message_count": len(messages)},
                outcome="completed",
                started_at=datetime.fromisoformat(messages[0]["timestamp"]),
                ended_at=datetime.fromisoformat(messages[-1]["timestamp"])
            )

            # Update consolidation record
            consolidation.memories_processed = len(messages)
            consolidation.memories_created = memories_created
            consolidation.completed_at = datetime.utcnow()
            consolidation.status = "completed"
            self.db.commit()

            return {
                "status": "completed",
                "facts_extracted": len(facts),
                "preferences_extracted": len(preferences),
                "episode_created": True
            }

        except Exception as e:
            consolidation.status = "failed"
            self.db.commit()
            raise e

    async def _extract_facts(self, messages: List[Dict]) -> List[Dict]:
        """Extract factual information from conversation."""
        conversation = "\n".join([
            f"{m['role']}: {m['content']}" for m in messages
        ])

        prompt = f"""Extract factual information about the user from this conversation.
Return only concrete facts (name, location, job, preferences stated as facts).

Conversation:
{conversation[:3000]}

Return as JSON array:
[{{"content": "fact text", "importance": 0.0-1.0}}]

Only include clear facts, not inferences."""

        response = await self.llm.ainvoke(prompt)

        try:
            import json
            return json.loads(response.content)
        except:
            return []

    async def _extract_preferences(self, messages: List[Dict]) -> List[Dict]:
        """Extract user preferences from conversation."""
        conversation = "\n".join([
            f"{m['role']}: {m['content']}" for m in messages
        ])

        prompt = f"""Extract user preferences from this conversation.
Look for likes, dislikes, preferred styles, communication preferences.

Conversation:
{conversation[:3000]}

Return as JSON array:
[{{"content": "preference description", "importance": 0.0-1.0}}]"""

        response = await self.llm.ainvoke(prompt)

        try:
            import json
            return json.loads(response.content)
        except:
            return []

    async def _summarize_session(self, messages: List[Dict]) -> str:
        """Create a summary of the session."""
        conversation = "\n".join([
            f"{m['role']}: {m['content']}" for m in messages[-20:]  # Last 20 messages
        ])

        prompt = f"""Summarize this conversation in 2-3 sentences.
Focus on what was discussed and any outcomes.

Conversation:
{conversation}"""

        response = await self.llm.ainvoke(prompt)
        return response.content

    async def merge_similar_memories(
        self,
        user_id: str,
        similarity_threshold: float = 0.85
    ) -> int:
        """Find and merge highly similar memories."""
        embeddings = get_embedding_service()

        # Get all memories
        memories = self.db.query(LongTermMemory).filter(
            LongTermMemory.user_id == user_id
        ).all()

        merged_count = 0
        processed_ids = set()

        for memory in memories:
            if memory.id in processed_ids:
                continue

            # Find similar memories
            query_embedding = embeddings.json_to_embedding(memory.embedding_json)
            candidates = [
                (m.id, m.embedding_json)
                for m in memories
                if m.id != memory.id and m.id not in processed_ids
            ]

            similar = embeddings.find_similar(
                query_embedding,
                candidates,
                top_k=5,
                threshold=similarity_threshold
            )

            if similar:
                # Merge memories
                similar_ids = [memory.id] + [s[0] for s in similar]
                similar_memories = [m for m in memories if m.id in similar_ids]

                # Create merged content
                merged_content = await self._merge_content(
                    [m.content for m in similar_memories]
                )

                self.ltm.merge_memories(
                    similar_ids,
                    merged_content,
                    merged_content[:200]
                )

                processed_ids.update(similar_ids)
                merged_count += 1

        return merged_count

    async def _merge_content(self, contents: List[str]) -> str:
        """Merge multiple memory contents into one."""
        prompt = f"""Merge these related memories into a single, comprehensive memory:

{chr(10).join(f'- {c}' for c in contents)}

Create a single, clear statement that captures all the information."""

        response = await self.llm.ainvoke(prompt)
        return response.content
```

```python
    return [{"id": e.id, "summary": e.summary, "outcome": e.outcome} for e in episodes]

@router.post("/{user_id}/episodes/recall")
def recall_episodes(
    user_id: str,
    request: RecallRequest,
    db: Session = Depends(get_db)
):
    """Recall similar past episodes."""
    episodic = EpisodicMemoryService(db)
    return episodic.recall_similar(user_id, request.query, top_k=request.top_k)

# Consolidation endpoints
@router.post("/{user_id}/sessions/{session_id}/consolidate")
async def consolidate_session(
    user_id: str,
    session_id: str,
    db: Session = Depends(get_db)
):
    """Consolidate session memories."""
    consolidator = MemoryConsolidator(db)
    return await consolidator.consolidate_session(user_id, session_id)
```

```python
# Example: Agent using memory service
from services.short_term import ShortTermMemory
from services.long_term import LongTermMemoryService
from services.episodic import EpisodicMemoryService

class MemoryAwareAgent:
    def __init__(self, user_id: str, session_id: str, db):
        self.user_id = user_id
        self.session_id = session_id
        self.stm = ShortTermMemory()
        self.ltm = LongTermMemoryService(db)
        self.episodic = EpisodicMemoryService(db)

    async def process_message(self, user_message: str) -> str:
        # Store user message
        self.stm.add_message(self.user_id, self.session_id, "user", user_message)

        # Recall relevant long-term memories
        memories = self.ltm.recall(self.user_id, user_message, top_k=3)

        # Recall similar past episodes
        episodes = self.episodic.recall_similar(self.user_id, user_message, top_k=2)

        # Build context
        context = self._build_context(memories, episodes)

        # Get conversation history
        history = self.stm.get_context_window(self.user_id, self.session_id)

        # Generate response (your LLM call here)
        response = await self._generate_response(user_message, context, history)

        # Store assistant response
        self.stm.add_message(self.user_id, self.session_id, "assistant", response)

        return response

    def _build_context(self, memories, episodes) -> str:
        context_parts = []

        if memories:
            context_parts.append("Relevant memories:")
            for m in memories:
                context_parts.append(f"- {m['content']}")

        if episodes:
            context_parts.append("\nSimilar past interactions:")
            for e in episodes:
                context_parts.append(f"- {e['summary']} (outcome: {e['outcome']})")

        return "\n".join(context_parts)
```

## Key Takeaways

1. **Three memory types** serve different purposes: STM for immediate context, LTM for persistent knowledge, and Episodic for learning from past experiences
2. **Semantic search with embeddings** enables intelligent recall beyond keyword matching—agents can find relevant memories by meaning, not just exact words
3. **Memory consolidation** mimics human sleep—background processing that transforms raw conversations into structured, lasting knowledge
4. **Memory architecture patterns** from this project apply to any stateful agent system, whether you're building chatbots, autonomous agents, or multi-agent systems
5. **Performance matters**: Redis for < 10ms STM access, optimized vector search for < 200ms LTM recall—these targets ensure memory doesn't become a bottleneck
