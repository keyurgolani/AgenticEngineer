## 80.1 The Monitor Agent

The Monitor Agent tracks domain status using RDAP (Registration Data Access Protocol).

<Mermaid

    participant Parser as JSON Parser

        Parser-->>Agent: Calculate Days Until Available

`}
  caption="RDAP Query Flow: The standardized way to check domain lifecycle status."
/>

<Callout type="concept">
  **Understanding RDAP:** Think of RDAP as the "credit report" for domain names.
  Just as a credit report shows the status and history of a person's credit,
  RDAP provides standardized information about a domain's registration status,
  expiry date, and registrar. It replaced the older WHOIS protocol with a
  modern, JSON-based API that's easier to parse and more reliable.
</Callout>

```python
import httpx
from datetime import datetime, timedelta
from typing import Optional
from pydantic import BaseModel
from enum import Enum

class DomainStatus(Enum):
    REGISTERED = "registered"
    PENDING_DELETE = "pending_delete"
    REDEMPTION = "redemption"
    AVAILABLE = "available"
    UNKNOWN = "unknown"

class DomainInfo(BaseModel):
    domain: str
    status: DomainStatus
    expiry_date: Optional[datetime]
    registrar: Optional[str]
    last_checked: datetime
    days_until_available: Optional[int]

class MonitorAgent:
    """Agent 1: Monitor domain status via RDAP."""

    RDAP_SERVERS = {
        "com": "https://rdap.verisign.com/com/v1/domain/",
        "net": "https://rdap.verisign.com/net/v1/domain/",
        "org": "https://rdap.publicinterestregistry.org/rdap/domain/",
    }

    def __init__(self, watch_list: list[str]):
        self.watch_list = watch_list
        self.client = httpx.AsyncClient(timeout=10.0)
        self.status_cache: dict[str, DomainInfo] = {}

    async def check_domain(self, domain: str) -> DomainInfo:
        """Query RDAP for domain status."""
        tld = domain.split(".")[-1]
        rdap_url = self.RDAP_SERVERS.get(tld)

        if not rdap_url:
            return DomainInfo(
                domain=domain,
                status=DomainStatus.UNKNOWN,
                expiry_date=None,
                registrar=None,
                last_checked=datetime.now(),
                days_until_available=None
            )

        try:
            response = await self.client.get(f"{rdap_url}{domain}")

            if response.status_code == 404:
                # Domain not found = available
                return DomainInfo(
                    domain=domain,
                    status=DomainStatus.AVAILABLE,
                    expiry_date=None,
                    registrar=None,
                    last_checked=datetime.now(),
                    days_until_available=0
                )

            data = response.json()
            return self._parse_rdap_response(domain, data)

        except Exception as e:
            print(f"Error checking {domain}: {e}")
            return self.status_cache.get(domain, DomainInfo(
                domain=domain,
                status=DomainStatus.UNKNOWN,
                expiry_date=None,
                registrar=None,
                last_checked=datetime.now(),
                days_until_available=None
            ))

    def _parse_rdap_response(self, domain: str, data: dict) -> DomainInfo:
        """Parse RDAP response into DomainInfo."""
        status_list = data.get("status", [])

        # Determine status from RDAP status codes
        if "pendingDelete" in status_list:
            status = DomainStatus.PENDING_DELETE
        elif "redemptionPeriod" in status_list:
            status = DomainStatus.REDEMPTION
        else:
            status = DomainStatus.REGISTERED

        # Extract expiry date
        expiry_date = None
        for event in data.get("events", []):
            if event.get("eventAction") == "expiration":
                expiry_date = datetime.fromisoformat(
                    event["eventDate"].replace("Z", "+00:00")
                )
                break

        # Calculate days until potentially available
        days_until = None
        if expiry_date:
            # Domains typically available ~75 days after expiry
            available_date = expiry_date + timedelta(days=75)
            days_until = (available_date - datetime.now()).days

        return DomainInfo(
            domain=domain,
            status=status,
            expiry_date=expiry_date,
            registrar=data.get("entities", [{}])[0].get("handle"),
            last_checked=datetime.now(),
            days_until_available=days_until
        )

    async def run_check_cycle(self) -> list[DomainInfo]:
        """Check all domains in watch list."""
        results = []

        for domain in self.watch_list:
            info = await self.check_domain(domain)
            self.status_cache[domain] = info
            results.append(info)

            # Emit event if domain status changed to actionable
            if info.status in [DomainStatus.AVAILABLE, DomainStatus.PENDING_DELETE]:
                await self.emit_event("domain_actionable", info)

        return results

    async def emit_event(self, event_type: str, data: DomainInfo):
        """Emit event to trigger downstream agents."""
        print(f"📡 Event: {event_type} for {data.domain}")
        # In production, this would publish to a message queue
```

<Callout type="concept">
  **Domain Valuation Analogy:** Evaluating a domain is like appraising real
  estate. You consider location (TLD), neighborhood history (previous use,
  backlinks), property condition (blacklist status), and comparable sales
  (similar domains). Just as a house appraiser uses multiple data sources, our
  Evaluator Agent combines LLM intelligence with API data to estimate value and
  risk.
</Callout>

```python
from langchain_openai import ChatOpenAI
from pydantic import BaseModel, Field

class DomainEvaluation(BaseModel):
    domain: str
    approved: bool
    confidence: float = Field(ge=0, le=1)
    estimated_value: float
    risk_factors: list[str]
    recommendation: str

class EvaluatorAgent:
    """Agent 2: Evaluate domain value and risk."""

    BLACKLIST_APIS = [
        "https://api.spamhaus.org/check/",
        "https://api.surbl.org/check/",
    ]

    def __init__(self, max_price: float = 100.0, min_confidence: float = 0.7):
        self.max_price = max_price
        self.min_confidence = min_confidence
        self.llm = ChatOpenAI(model="gpt-4o-mini")
        self.client = httpx.AsyncClient(timeout=5.0)

    async def evaluate(self, domain_info: DomainInfo) -> DomainEvaluation:
        """Comprehensive domain evaluation."""

        # Check blacklists
        blacklist_results = await self._check_blacklists(domain_info.domain)

        # Get historical data (Wayback Machine, etc.)
        history = await self._get_domain_history(domain_info.domain)

        # LLM-based value assessment
        value_assessment = await self._assess_value(
            domain_info, blacklist_results, history
        )

        # Make final decision
        approved = (
            value_assessment["confidence"] >= self.min_confidence and
            value_assessment["estimated_value"] <= self.max_price and
            len(blacklist_results["hits"]) == 0
        )

        return DomainEvaluation(
            domain=domain_info.domain,
            approved=approved,
            confidence=value_assessment["confidence"],
            estimated_value=value_assessment["estimated_value"],
            risk_factors=blacklist_results["hits"] + value_assessment["risks"],
            recommendation=value_assessment["recommendation"]
        )

    async def _check_blacklists(self, domain: str) -> dict:
        """Check domain against spam/malware blacklists."""
        hits = []

        # Simplified blacklist check (production would use real APIs)
        # Check if domain was previously used for spam/malware
        suspicious_patterns = ["free", "cheap", "casino", "pharma"]
        for pattern in suspicious_patterns:
            if pattern in domain.lower():
                hits.append(f"Suspicious keyword: {pattern}")

        return {"hits": hits, "checked": len(self.BLACKLIST_APIS)}

    async def _get_domain_history(self, domain: str) -> dict:
        """Fetch domain history from Wayback Machine."""
        try:
            response = await self.client.get(
                f"https://archive.org/wayback/available?url={domain}"
            )
            data = response.json()

            if data.get("archived_snapshots", {}).get("closest"):
                return {
                    "has_history": True,
                    "oldest_snapshot": data["archived_snapshots"]["closest"]["timestamp"],
                    "url": data["archived_snapshots"]["closest"]["url"]
                }
        except Exception:
            pass

        return {"has_history": False}

    async def _assess_value(
        self,
        domain_info: DomainInfo,
        blacklist: dict,
        history: dict
    ) -> dict:
        """LLM-based domain value assessment."""

        prompt = f"""Evaluate this domain for acquisition:

Domain: {domain_info.domain}
Status: {domain_info.status.value}
Previous Registrar: {domain_info.registrar}
Has Web History: {history.get('has_history', False)}
Blacklist Hits: {len(blacklist['hits'])}

Assess:
1. Estimated market value (USD)
2. Confidence in assessment (0-1)
3. Risk factors
4. Recommendation (buy/pass)

Respond in JSON format:
{{"estimated_value": float, "confidence": float, "risks": [str], "recommendation": str}}
"""

        response = await self.llm.ainvoke(prompt)

        # Parse JSON response (with fallback)
        try:
            import json
            return json.loads(response.content)
        except:
            return {
                "estimated_value": 50.0,
                "confidence": 0.5,
                "risks": ["Could not parse LLM response"],
                "recommendation": "Manual review required"
            }
```

```python
class PurchaseResult(BaseModel):
    domain: str
    success: bool
    transaction_id: Optional[str]
    price_paid: Optional[float]
    error: Optional[str]

class PurchaserAgent:
    """Agent 3: Execute domain purchase."""

    def __init__(self, registrar_api_key: str, max_price: float = 100.0):
        self.api_key = registrar_api_key
        self.max_price = max_price
        self.client = httpx.AsyncClient(timeout=30.0)

    async def purchase(self, evaluation: DomainEvaluation) -> PurchaseResult:
        """Execute domain purchase if approved."""

        if not evaluation.approved:
            return PurchaseResult(
                domain=evaluation.domain,
                success=False,
                transaction_id=None,
                price_paid=None,
                error="Domain not approved by evaluator"
            )

        # Double-check price
        if evaluation.estimated_value > self.max_price:
            return PurchaseResult(
                domain=evaluation.domain,
                success=False,
                transaction_id=None,
                price_paid=None,

            )

        # Execute purchase (example with Namecheap-style API)
        try:
            result = await self._execute_purchase(evaluation.domain)

                domain=evaluation.domain,
```

## 80.4 Orchestration with LangGraph

<Mermaid
  chart={`
stateDiagram-v2
    [*] --> Monitoring
    
    Monitoring --> Evaluating: Domain Available
    Monitoring --> Monitoring: Domain Taken
    
    Evaluating --> Purchasing: Approved
    Evaluating --> Complete: Rejected
    
    Purchasing --> Complete: Success/Fail
    
    Complete --> Monitoring: Reset / Next Cycle
`}
  caption="Autonomous Loop: The agent continuously cycles through states based on external triggers."
/>

<Callout type="concept">
  **LangGraph for Agent Coordination:** LangGraph is like a GPS navigation
  system for your agents. It defines the route (workflow), handles decision
  points (conditional edges), maintains state (where you are and where you've
  been), and can reroute if needed (error handling). Each node is a checkpoint
  where the agent performs work and updates the shared state.
</Callout>

```python
from langgraph.graph import StateGraph, START, END
from typing import TypedDict, Literal

class WatchdogState(TypedDict):
    watch_list: list[str]
    current_domain: Optional[str]
    domain_info: Optional[dict]
    evaluation: Optional[dict]
    purchase_result: Optional[dict]
    notifications: list[str]
    status: Literal["monitoring", "evaluating", "purchasing", "complete", "error"]

async def monitor_node(state: WatchdogState) -> dict:
    """Monitor agent checks domain status."""
    monitor = MonitorAgent(state["watch_list"])
    results = await monitor.run_check_cycle()

    # Find actionable domains
    actionable = [r for r in results if r.status == DomainStatus.AVAILABLE]

    if actionable:
        return {
            "current_domain": actionable[0].domain,
            "domain_info": actionable[0].model_dump(),
            "status": "evaluating"
        }

    return {"status": "complete", "notifications": ["No actionable domains found"]}

async def evaluate_node(state: WatchdogState) -> dict:
    """Evaluator agent assesses domain."""
    evaluator = EvaluatorAgent(max_price=100.0)
    domain_info = DomainInfo(**state["domain_info"])

    evaluation = await evaluator.evaluate(domain_info)

    if evaluation.approved:
        return {
            "evaluation": evaluation.model_dump(),
            "status": "purchasing"
        }

    return {
        "evaluation": evaluation.model_dump(),
        "status": "complete",
        "notifications": [f"Domain {domain_info.domain} rejected: {evaluation.recommendation}"]
    }

async def purchase_node(state: WatchdogState) -> dict:
    """Purchaser agent executes acquisition."""
    purchaser = PurchaserAgent(
        registrar_api_key="your-api-key",
        max_price=100.0
    )

    evaluation = DomainEvaluation(**state["evaluation"])
    result = await purchaser.purchase(evaluation)

    notification = (
        f"✅ Purchased {result.domain} for ${result.price_paid}"
        if result.success
        else f"❌ Failed to purchase {result.domain}: {result.error}"
    )

    return {
        "purchase_result": result.model_dump(),
        "status": "complete",
        "notifications": state["notifications"] + [notification]
    }

def route_after_monitor(state: WatchdogState) -> str:
    if state["status"] == "evaluating":
        return "evaluate"
    return "end"

def route_after_evaluate(state: WatchdogState) -> str:
    if state["status"] == "purchasing":
        return "purchase"
    return "end"

# Build the graph
workflow = StateGraph(WatchdogState)

workflow.add_node("monitor", monitor_node)
workflow.add_node("evaluate", evaluate_node)
workflow.add_node("purchase", purchase_node)

workflow.add_edge(START, "monitor")
workflow.add_conditional_edges("monitor", route_after_monitor, {
    "evaluate": "evaluate",
    "end": END
})
workflow.add_conditional_edges("evaluate", route_after_evaluate, {
    "purchase": "purchase",
    "end": END
})
workflow.add_edge("purchase", END)

app = workflow.compile()

# Run the swarm
async def run_watchdog():
    result = await app.ainvoke({
        "watch_list": [
            "coolstartup.com",
            "aiagents.io",
            "techbrand.net"
        ],
        "notifications": [],
        "status": "monitoring"
    })

    print("Notifications:")
    for n in result["notifications"]:
        print(f"  {n}")
```

```python
                    expiry_date=None,
                    registrar=None,
                    last_checked=datetime.now(),
                    days_until_available=0
                )

        except Exception as e:
```

```python
from dataclasses import dataclass
from typing import Optional
import httpx

@dataclass
class ValuationResult:
    source: str
    estimated_value: float
    confidence: float
    factors: dict

class DomainValuationService:
    """Multi-source domain valuation service."""

    def __init__(
        self,
        godaddy_api_key: Optional[str] = None,
        estibot_api_key: Optional[str] = None
    ):
        self.godaddy_key = godaddy_api_key
        self.estibot_key = estibot_api_key
        self.client = httpx.AsyncClient(timeout=30.0)

    async def get_valuations(self, domain: str) -> list[ValuationResult]:
        """Get valuations from multiple sources."""
        valuations = []

        # GoDaddy GoValue API
        if self.godaddy_key:
            try:
                gd_value = await self._godaddy_valuation(domain)
                if gd_value:
                    valuations.append(gd_value)
            except Exception as e:
                print(f"GoDaddy valuation failed: {e}")

        # EstiBot API
        if self.estibot_key:
            try:
                eb_value = await self._estibot_valuation(domain)
                if eb_value:
                    valuations.append(eb_value)
            except Exception as e:
                print(f"EstiBot valuation failed: {e}")

        # NameBio comparable sales (free)
        try:
            nb_value = await self._namebio_comparables(domain)
            if nb_value:
                valuations.append(nb_value)
        except Exception:
            pass

        return valuations

    async def _godaddy_valuation(self, domain: str) -> ValuationResult:
        """GoDaddy GoValue API valuation."""
        response = await self.client.get(
            f"https://api.godaddy.com/v1/appraisal/{domain}",
            headers={
                "Authorization": f"sso-key {self.godaddy_key}",
                "Accept": "application/json"
            }
        )
        data = response.json()

        return ValuationResult(
            source="godaddy",
            estimated_value=data.get("govalue", 0),
            confidence=data.get("confidence", 0) / 100,
            factors={
                "comparable_sales": data.get("comparables", []),
                "keyword_value": data.get("keyword_value", 0)
            }
        )

    async def _estibot_valuation(self, domain: str) -> ValuationResult:
        """EstiBot API valuation."""
        response = await self.client.get(
            "https://www.estibot.com/api/appraisal",
            params={
                "domain": domain,
                "apikey": self.estibot_key
            }
        )
        data = response.json()

        return ValuationResult(
            source="estibot",
            estimated_value=float(data.get("appraised_value", 0)),
            confidence=0.7,  # EstiBot doesn't provide confidence
            factors={
                "search_volume": data.get("search_volume", 0),
                "cpc": data.get("cpc", 0),
                "length": data.get("length", 0)
            }
        )

    def aggregate_valuations(
        self,
        valuations: list[ValuationResult]
    ) -> tuple[float, float]:
        """Weighted average of valuations."""
        if not valuations:
            return 0.0, 0.0

        # Weight by confidence
        total_weight = sum(v.confidence for v in valuations)
        weighted_value = sum(
            v.estimated_value * v.confidence for v in valuations
        )

        avg_value = weighted_value / total_weight if total_weight > 0 else 0
        avg_confidence = total_weight / len(valuations)

        return avg_value, avg_confidence
```

### Confidence Thresholds and Budget Management

```python
from dataclasses import dataclass
from datetime import datetime, timedelta
from typing import Optional
import json

@dataclass
class BudgetConfig:
    daily_limit: float = 500.0
    weekly_limit: float = 2000.0
    monthly_limit: float = 5000.0
    max_per_domain: float = 200.0
    reserve_percentage: float = 0.2  # Keep 20% reserve

@dataclass
class ConfidenceThresholds:
    auto_purchase: float = 0.85      # Auto-buy above this
    human_review: float = 0.6        # Request review between these
    auto_reject: float = 0.4         # Auto-reject below this
    high_value_threshold: float = 500.0  # Always review above this price

@dataclass
class PurchaseDecision:
    domain: str
    action: str  # "auto_purchase", "request_review", "auto_reject"
    confidence: float
    estimated_value: float
    reason: str
    requires_approval: bool = False

class AutonomousDecisionEngine:
    """Engine for autonomous purchase decisions with safeguards."""

    def __init__(
        self,
        budget: BudgetConfig,
        thresholds: ConfidenceThresholds,
        audit_log_path: str = "audit_log.jsonl"
    ):
        self.budget = budget
        self.thresholds = thresholds
        self.audit_log_path = audit_log_path
        self.spending_tracker = SpendingTracker()

    async def make_decision(
        self,
        domain: str,
        evaluation: DomainEvaluation,
        valuations: list[ValuationResult]
    ) -> PurchaseDecision:
        """Make autonomous purchase decision with safeguards."""

        # Aggregate valuations
        avg_value, avg_confidence = self._aggregate_valuations(valuations)

        # Check budget constraints
        budget_available = await self._check_budget(avg_value)

        # Risk assessment
        risk_score = self._assess_risk(domain, evaluation, valuations)

        # Decision logic
        decision = self._apply_decision_rules(
            domain=domain,
            confidence=evaluation.confidence,
            estimated_value=avg_value,
            budget_available=budget_available,
            risk_score=risk_score
        )

        # Log decision for audit trail
        await self._log_decision(decision, evaluation, valuations)

        return decision

    def _apply_decision_rules(
        self,
        domain: str,
        confidence: float,
        estimated_value: float,
        budget_available: bool,
        risk_score: float
    ) -> PurchaseDecision:
        """Apply decision rules with multiple safeguards."""

        # Rule 1: Budget check
        if not budget_available:
            return PurchaseDecision(
                domain=domain,
                action="auto_reject",
                confidence=confidence,
                estimated_value=estimated_value,
                reason="Insufficient budget"
            )

        # Rule 2: High-value domains always need review
        if estimated_value > self.thresholds.high_value_threshold:
            return PurchaseDecision(
                domain=domain,
                action="request_review",
                confidence=confidence,
                estimated_value=estimated_value,
                reason=f"High value domain (>${self.thresholds.high_value_threshold})",
                requires_approval=True
            )

        # Rule 3: High risk domains need review
        if risk_score > 0.7:
            return PurchaseDecision(
                domain=domain,
                action="request_review",
                confidence=confidence,
                estimated_value=estimated_value,
                reason=f"High risk score ({risk_score:.2f})",
                requires_approval=True
            )

        # Rule 4: Confidence-based decisions
        if confidence >= self.thresholds.auto_purchase:
            return PurchaseDecision(
                domain=domain,
                action="auto_purchase",
                confidence=confidence,
                estimated_value=estimated_value,
                reason=f"High confidence ({confidence:.2f})"
            )
        elif confidence >= self.thresholds.human_review:
            return PurchaseDecision(
                domain=domain,
                action="request_review",
                confidence=confidence,
                estimated_value=estimated_value,
                reason=f"Medium confidence ({confidence:.2f})",
                requires_approval=True
            )
        else:
            return PurchaseDecision(
                domain=domain,
                action="auto_reject",
                confidence=confidence,
                estimated_value=estimated_value,
                reason=f"Low confidence ({confidence:.2f})"
            )

    def _assess_risk(
        self,
        domain: str,
        evaluation: DomainEvaluation,
        valuations: list[ValuationResult]
    ) -> float:
        """Calculate risk score (0-1)."""
        risk_factors = []

        # Valuation variance risk
        if len(valuations) > 1:
            values = [v.estimated_value for v in valuations]
            variance = max(values) / min(values) if min(values) > 0 else 10
            if variance > 3:
                risk_factors.append(0.3)

        # Blacklist risk
        if evaluation.risk_factors:
            risk_factors.append(0.2 * len(evaluation.risk_factors))

        # Domain pattern risk
        suspicious_patterns = ["free", "cheap", "buy", "sale", "xxx"]
        if any(p in domain.lower() for p in suspicious_patterns):
            risk_factors.append(0.2)

        # TLD risk
        risky_tlds = ["xyz", "top", "click", "link"]
        if domain.split(".")[-1] in risky_tlds:
            risk_factors.append(0.15)

        return min(sum(risk_factors), 1.0)

    async def _check_budget(self, amount: float) -> bool:
        """Check if purchase fits within budget constraints."""
        spending = await self.spending_tracker.get_spending()

        if amount > self.budget.max_per_domain:
            return False
        if spending["daily"] + amount > self.budget.daily_limit:
            return False
        if spending["weekly"] + amount > self.budget.weekly_limit:
            return False
        if spending["monthly"] + amount > self.budget.monthly_limit:
            return False

        # Check reserve
        remaining = self.budget.monthly_limit - spending["monthly"]
        reserve = self.budget.monthly_limit * self.budget.reserve_percentage
        if remaining - amount < reserve:
            return False

        return True

    async def _log_decision(
        self,
        decision: PurchaseDecision,
        evaluation: DomainEvaluation,
        valuations: list[ValuationResult]
    ):
        """Log decision for audit trail."""
        audit_entry = {
            "timestamp": datetime.now().isoformat(),
            "domain": decision.domain,
            "action": decision.action,
            "confidence": decision.confidence,
            "estimated_value": decision.estimated_value,
            "reason": decision.reason,
            "evaluation": evaluation.model_dump(),
            "valuations": [v.__dict__ for v in valuations]
        }

        with open(self.audit_log_path, "a") as f:
            f.write(json.dumps(audit_entry) + "\n")
```

```python
class ApprovalWorkflow:
    """Human-in-the-loop approval workflow."""

    def __init__(self, slack_webhook: str, approval_timeout_hours: int = 4):
        self.slack_webhook = slack_webhook
        self.timeout = timedelta(hours=approval_timeout_hours)
        self.pending_approvals: dict[str, dict] = {}

    async def request_approval(
        self,
        decision: PurchaseDecision,
        evaluation: DomainEvaluation
    ) -> str:
        """Send approval request and return request ID."""
        request_id = f"approval-{decision.domain}-{datetime.now().timestamp()}"

        self.pending_approvals[request_id] = {
            "decision": decision,
            "evaluation": evaluation,
            "requested_at": datetime.now(),
            "status": "pending"
        }

        await self._send_slack_approval_request(request_id, decision, evaluation)
        return request_id

    async def _send_slack_approval_request(
        self,
        request_id: str,
        decision: PurchaseDecision,
        evaluation: DomainEvaluation
    ):
        """Send interactive Slack message for approval."""
        message = {
            "blocks": [
                {
                    "type": "header",
                    "text": {
                        "type": "plain_text",
                        "text": "🔔 Domain Purchase Approval Required"
                    }
                },
                {
                    "type": "section",
                    "fields": [
                        {"type": "mrkdwn", "text": f"*Domain:*\n{decision.domain}"},
                        {"type": "mrkdwn", "text": f"*Est. Value:*\n${decision.estimated_value:.2f}"},
                        {"type": "mrkdwn", "text": f"*Confidence:*\n{decision.confidence:.1%}"},
                        {"type": "mrkdwn", "text": f"*Reason:*\n{decision.reason}"}
                    ]
                },
                {
                    "type": "actions",
                    "elements": [
                        {
                            "type": "button",
                            "text": {"type": "plain_text", "text": "✅ Approve"},
                            "style": "primary",
                            "action_id": f"approve_{request_id}"
                        },
                        {
                            "type": "button",
                            "text": {"type": "plain_text", "text": "❌ Reject"},
                            "style": "danger",
                            "action_id": f"reject_{request_id}"
                        }
                    ]
                }
            ]
        }

        async with httpx.AsyncClient() as client:
            await client.post(self.slack_webhook, json=message)

    async def wait_for_approval(
        self,
        request_id: str
    ) -> tuple[bool, Optional[str]]:
        """Wait for approval with timeout."""
        start_time = datetime.now()

        while datetime.now() - start_time < self.timeout:
            approval = self.pending_approvals.get(request_id)

            if approval and approval["status"] != "pending":
                return (
                    approval["status"] == "approved",
                    approval.get("approved_by")
                )

            await asyncio.sleep(30)

        self.pending_approvals[request_id]["status"] = "timeout"
        return False, None
```

```python
class CostManager:
    """Monitor and manage operational costs."""

    def __init__(self, monthly_budget: float = 100.0):
        self.monthly_budget = monthly_budget
        self.cost_per_check = 0.001
        self.cost_per_valuation = 0.01

    def estimate_monthly_cost(
        self,
        watch_list_size: int,
        checks_per_day: int = 4
    ) -> dict:
        """Estimate monthly operational costs."""
        checks_per_month = watch_list_size * checks_per_day * 30

        costs = {
            "lambda_compute": checks_per_month * self.cost_per_check,
            "valuation_apis": checks_per_month * self.cost_per_valuation * 0.1,
            "dynamodb": 5.0,
            "secrets_manager": 0.40,
            "cloudwatch": 3.0,
        }

        costs["total"] = sum(costs.values())
        costs["within_budget"] = costs["total"] <= self.monthly_budget

        return costs

    def optimize_check_frequency(
        self,
        domain_info: DomainInfo
    ) -> int:
        """Optimize check frequency based on domain status."""
        if domain_info.days_until_available is None:
            return 24  # Check once daily

        days = domain_info.days_until_available

        if days > 60:
            return 168  # Weekly
        elif days > 30:
            return 72   # Every 3 days
        elif days > 7:
            return 24   # Daily
        elif days > 1:
            return 6    # Every 6 hours
        else:
            return 1    # Hourly (domain dropping soon!)
```

```python
from dotenv import load_dotenv

load_dotenv()

    domain: str
    status: Literal["available", "taken", "unknown"]
    valuation: dict
    decision: str     # "buy", "pass"
    purchase_result: str

# 2. Nodes
async def monitor_node(state: WatchdogState):
    print(f"🔍 Monitoring: {state['domain']}")
    # Mock result for demo
    return {"status": "available"}

async def evaluator_node(state: WatchdogState):
    print("📊 Evaluating...")
    # Mock valuation
    return {
        "valuation": {"value": 500, "risk": "low"},
        "decision": "buy"
    }

async def purchaser_node(state: WatchdogState):
    print("💳 Purchasing...")
    return {"purchase_result": "Success: TXN-12345"}

# 3. Graph
workflow = StateGraph(WatchdogState)
workflow.add_node("monitor", monitor_node)
workflow.add_node("evaluator", evaluator_node)
workflow.add_node("purchaser", purchaser_node)

workflow.add_edge(START, "monitor")
workflow.add_edge("monitor", "evaluator")
workflow.add_edge("evaluator", "purchaser")
workflow.add_edge("purchaser", END)

app = workflow.compile()
```
