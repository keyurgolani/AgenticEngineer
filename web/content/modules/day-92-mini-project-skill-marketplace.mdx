## Project Overview

- **Skill Reuse**: Don't reinvent the wheel—use community-built skills
- **Quality Control**: Validated skills with reviews and ratings
- **Version Management**: Safe updates with dependency tracking
- **Discovery**: Find the right skill for any task via semantic search

### Step 1: SKILL.md Schema and Validation

The SKILL.md file is the heart of every skill. We need a robust validator that ensures all published skills meet the standard.

```typescript
// src/validators/skill-schema.ts
import { z } from "zod";

/**
 * SKILL.md Frontmatter Schema
 * Validates the YAML frontmatter in SKILL.md files
 */
export const SkillFrontmatterSchema = z.object({
  // Required fields
  name: z
    .string()
    .min(1)
    .max(100)
    .regex(/^[a-z0-9-]+$/, "Name must be kebab-case"),
  description: z.string().min(10).max(500),

  // Optional metadata
  "argument-hint": z.string().max(200).optional(),
  "disable-model-invocation": z.boolean().optional(),
  "user-invocable": z.boolean().optional(),

  // Tool restrictions
  "allowed-tools": z.array(z.string()).optional(),

  // Model configuration
  model: z.string().optional(),

  // Subagent configuration
  context: z.enum(["fork"]).optional(),
  agent: z
    .enum(["Explore", "Plan", "general-purpose"])
    .or(z.string())
    .optional(),

  // Marketplace-specific fields
  version: z
    .string()
    .regex(/^\d+\.\d+\.\d+$/, "Version must be semver (X.Y.Z)"),
  author: z.string().min(1).max(100),
  license: z.string().default("MIT"),
  repository: z.string().url().optional(),
  categories: z.array(z.string()).min(1).max(5),
  dependencies: z
    .array(
      z.object({
        name: z.string(),
        version: z.string(),
      }),
    )
    .optional(),
});

export type SkillFrontmatter = z.infer<typeof SkillFrontmatterSchema>;
```

```typescript
// src/validators/skill-validator.ts
import matter from "gray-matter";
import { SkillFrontmatterSchema, type SkillFrontmatter } from "./skill-schema";

export interface ValidationResult {
  valid: boolean;
  frontmatter?: SkillFrontmatter;
  content?: string;
  errors: ValidationError[];
  warnings: ValidationWarning[];
}

export interface ValidationError {
  field: string;
  message: string;
  line?: number;
}

export interface ValidationWarning {
  field: string;
  message: string;
  suggestion?: string;
}

export class SkillValidator {
  /**
   * Validate a SKILL.md file content
   */
  validate(skillMdContent: string): ValidationResult {
    const errors: ValidationError[] = [];
    const warnings: ValidationWarning[] = [];

    // Parse frontmatter
    let parsed: matter.GrayMatterFile<string>;
    try {
      parsed = matter(skillMdContent);
    } catch (e) {
      return {
        valid: false,
        errors: [{ field: "frontmatter", message: "Invalid YAML frontmatter" }],
        warnings: [],
      };
    }

    // Validate frontmatter schema
    const schemaResult = SkillFrontmatterSchema.safeParse(parsed.data);

    if (!schemaResult.success) {
      for (const issue of schemaResult.error.issues) {
        errors.push({
          field: issue.path.join("."),
          message: issue.message,
        });
      }
    }

    // Validate content structure
    const contentWarnings = this.validateContent(parsed.content);
    warnings.push(...contentWarnings);

    // Check for best practices
    const practiceWarnings = this.checkBestPractices(
      parsed.data as SkillFrontmatter,
      parsed.content,
    );
    warnings.push(...practiceWarnings);

    return {
      valid: errors.length === 0,
      frontmatter: schemaResult.success ? schemaResult.data : undefined,
      content: parsed.content,
      errors,
      warnings,
    };
  }

  private validateContent(content: string): ValidationWarning[] {
    const warnings: ValidationWarning[] = [];

    // Check for required sections
    if (!content.includes("# ")) {
      warnings.push({
        field: "content",
        message: "Missing main heading",
        suggestion: "Add a # heading describing the skill",
      });
    }

    // Check for process/instructions section
    if (!content.toLowerCase().includes("## ")) {
      warnings.push({
        field: "content",
        message: "No subsections found",
        suggestion: "Add ## sections for Process, Output Format, etc.",
      });
    }

    return warnings;
  }

  private checkBestPractices(
    frontmatter: SkillFrontmatter,
    content: string,
  ): ValidationWarning[] {
    const warnings: ValidationWarning[] = [];

    // Check description quality
    if (frontmatter?.description && frontmatter.description.length < 50) {
      warnings.push({
        field: "description",
        message: "Description is short",
        suggestion:
          "Provide a more detailed description for better discoverability",
      });
    }

    // Check for output format section
    if (!content.toLowerCase().includes("output")) {
      warnings.push({
        field: "content",
        message: "No output format specified",
        suggestion: "Add an '## Output Format' section",
      });
    }

    return warnings;
  }
}
```

```typescript
// src/packager/skill-packager.ts
import * as tar from "tar";
import * as crypto from "crypto";
import * as fs from "fs/promises";
import * as path from "path";
import { SkillValidator } from "../validators/skill-validator";

export interface PackageResult {
  success: boolean;
  packagePath?: string;
  hash?: string;
  size?: number;
  errors: string[];
}

export interface PackageManifest {
  name: string;
  version: string;
  description: string;
  author: string;
  license: string;
  files: string[];
  hash: string;
  created_at: string;
}

export class SkillPackager {
  private validator = new SkillValidator();
  private maxPackageSize = 10 * 1024 * 1024; // 10MB

  /**
   * Create a skill package from a directory
   */
  async createPackage(
    skillDir: string,
    outputDir: string,
  ): Promise<PackageResult> {
    const errors: string[] = [];

    // Verify directory structure
    const structureValid = await this.verifyStructure(skillDir);
    if (!structureValid.valid) {
      return { success: false, errors: structureValid.errors };
    }

    // Read and validate SKILL.md
    const skillMdPath = path.join(skillDir, "SKILL.md");
    const skillMdContent = await fs.readFile(skillMdPath, "utf-8");
    const validation = this.validator.validate(skillMdContent);

    if (!validation.valid) {
      return {
        success: false,
        errors: validation.errors.map((e) => `${e.field}: ${e.message}`),
      };
    }

    const { name, version } = validation.frontmatter!;
    const packageName = `${name}-${version}.tar.gz`;
    const packagePath = path.join(outputDir, packageName);

    // Collect files to package
    const files = await this.collectFiles(skillDir);

    // Create package manifest
    const manifest: PackageManifest = {
      name,
      version,
      description: validation.frontmatter!.description,
      author: validation.frontmatter!.author,
      license: validation.frontmatter!.license,
      files,
      hash: "", // Will be set after packaging
      created_at: new Date().toISOString(),
    };

    // Write manifest
    const manifestPath = path.join(skillDir, "package.json");
    await fs.writeFile(manifestPath, JSON.stringify(manifest, null, 2));

    // Create tar.gz package
    await tar.create(
      {
        gzip: true,
        file: packagePath,
        cwd: path.dirname(skillDir),
      },
      [path.basename(skillDir)],
    );

    // Calculate hash
    const packageBuffer = await fs.readFile(packagePath);
    const hash = crypto
      .createHash("sha256")
      .update(packageBuffer)
      .digest("hex");

    // Check size
    const stats = await fs.stat(packagePath);
    if (stats.size > this.maxPackageSize) {
      await fs.unlink(packagePath);
      return {
        success: false,
        errors: [
          `Package size ${stats.size} exceeds limit of ${this.maxPackageSize}`,
        ],
      };
    }

    // Update manifest with hash
    manifest.hash = hash;
    await fs.writeFile(manifestPath, JSON.stringify(manifest, null, 2));

    return {
      success: true,
      packagePath,
      hash,
      size: stats.size,
      errors: [],
    };
  }

  private async verifyStructure(
    skillDir: string,
  ): Promise<{ valid: boolean; errors: string[] }> {
    const errors: string[] = [];
    const requiredFiles = ["SKILL.md"];
    const allowedDirs = ["scripts", "references", "assets"];

    // Check required files
    for (const file of requiredFiles) {
      const filePath = path.join(skillDir, file);
      try {
        await fs.access(filePath);
      } catch {
        errors.push(`Missing required file: ${file}`);
      }
    }

    // Check for unexpected files/directories
    const entries = await fs.readdir(skillDir, { withFileTypes: true });
    for (const entry of entries) {
      if (entry.isDirectory() && !allowedDirs.includes(entry.name)) {
        errors.push(`Unexpected directory: ${entry.name}`);
      }
    }

    return { valid: errors.length === 0, errors };
  }

  private async collectFiles(skillDir: string): Promise<string[]> {
    const files: string[] = [];

    async function walk(dir: string, base: string) {
      const entries = await fs.readdir(dir, { withFileTypes: true });
      for (const entry of entries) {
        const fullPath = path.join(dir, entry.name);
        const relativePath = path.join(base, entry.name);

        if (entry.isDirectory()) {
          await walk(fullPath, relativePath);
        } else {
          files.push(relativePath);
        }
      }
    }

    await walk(skillDir, "");
    return files;
  }
}
```

### Step 3: Publishing API and Workflow

The publishing API handles skill uploads, validation, and storage.

```typescript
// src/api/routes/skills.ts
import { Router } from "express";
import multer from "multer";
import { SkillService } from "../services/skill-service";
import { authMiddleware } from "../middleware/auth";

const router = Router();
const upload = multer({
  limits: { fileSize: 10 * 1024 * 1024 },
  storage: multer.memoryStorage(),
});
const skillService = new SkillService();

/**
 * POST /api/skills - Publish a new skill or version
 */
router.post("/", authMiddleware, upload.single("package"), async (req, res) => {
  try {
    const { release_notes } = req.body;
    const publisherId = req.user!.id;

    if (!req.file) {
      return res.status(400).json({ detail: "Package file required" });
    }

    const result = await skillService.publish({
      packageBuffer: req.file.buffer,
      publisherId,
      releaseNotes: release_notes,
    });

    if (!result.success) {
      return res.status(400).json({
        detail: "Validation failed",
        errors: result.errors,
      });
    }

    res.status(201).json(result.skill);
  } catch (error) {
    console.error("Publish error:", error);
    res.status(500).json({ detail: "Internal server error" });
  }
});

/**
 * GET /api/skills/search - Search skills
 */
router.get("/search", async (req, res) => {
  try {
    const { q, category, limit = "20", offset = "0" } = req.query;

    if (!q || typeof q !== "string") {
      return res.status(400).json({ detail: "Query parameter 'q' required" });
    }

    const results = await skillService.search({
      query: q,
      category: category as string | undefined,
      limit: parseInt(limit as string, 10),
      offset: parseInt(offset as string, 10),
    });

    res.json(results);
  } catch (error) {
    console.error("Search error:", error);
    res.status(500).json({ detail: "Internal server error" });
  }
});

/**
 * GET /api/skills/:name - Get skill details
 */
router.get("/:name", async (req, res) => {
  try {
    const { name } = req.params;
    const { version } = req.query;

    const skill = await skillService.getByName(
      name,
      version as string | undefined,
```

```typescript
import {
  S3Client,
  PutObjectCommand,
  GetObjectCommand,
} from "@aws-sdk/client-s3";
import { getSignedUrl } from "@aws-sdk/s3-request-presigner";
import * as tar from "tar";
import * as crypto from "crypto";
import { Readable } from "stream";
import { SkillValidator } from "../validators/skill-validator";
import { SearchService } from "./search-service";

const prisma = new PrismaClient();
const s3 = new S3Client({ region: process.env.AWS_REGION });

export interface PublishInput {
  packageBuffer: Buffer;
  publisherId: string;
  releaseNotes?: string;
}

export interface PublishResult {
  success: boolean;
  skill?: any;
  errors: string[];
}

export class SkillService {
  private validator = new SkillValidator();
  private searchService = new SearchService();

  async publish(input: PublishInput): Promise<PublishResult> {
    const { packageBuffer, publisherId, releaseNotes } = input;

    // Extract and validate package
    const extracted = await this.extractPackage(packageBuffer);
    if (!extracted.skillMd) {
      return { success: false, errors: ["SKILL.md not found in package"] };
    }

    const validation = this.validator.validate(extracted.skillMd);
    if (!validation.valid) {
      return {
        success: false,
        errors: validation.errors.map((e) => `${e.field}: ${e.message}`),
      };
    }

    const { name, version, description, categories, dependencies } =
      validation.frontmatter!;

    // Check for duplicate version
    const existing = await prisma.skillVersion.findFirst({
      where: {
        skill: { name },
        version,
      },
    });

    if (existing) {
      return {
        success: false,
        errors: [`Version ${version} already published for ${name}`],
      };
    }

    // Calculate package hash
    const hash = crypto
      .createHash("sha256")
      .update(packageBuffer)
      .digest("hex");

    // Upload to S3
    const s3Key = `skills/${name}/${version}/${name}-${version}.tar.gz`;
    await s3.send(
      new PutObjectCommand({
        Bucket: process.env.S3_BUCKET!,
        Key: s3Key,
        Body: packageBuffer,
        ContentType: "application/gzip",
      }),
    );

    // Create or update skill in database
    const skill = await prisma.skill.upsert({
      where: { name },
      create: {
        name,
        description,
        currentVersion: version,
        publisherId,
        categories,
        tags: [],
      },
      update: {
        description,
        currentVersion: version,
        updatedAt: new Date(),
      },
    });

    // Create version record
    await prisma.skillVersion.create({
      data: {
        skillId: skill.id,
        version,
        packageUrl: s3Key,
        packageHash: hash,
        skillMdContent: extracted.skillMd,
        metadata: validation.frontmatter,
        dependencies: dependencies || [],
        releaseNotes,
      },
    });

    // Index in search
    await this.searchService.indexSkill({
      id: skill.id,
      name,
      description,
      version,
      categories,
      publisher: publisherId,
    });

    return {
      success: true,
      skill: {
        id: skill.id,
        name,
        version,
        description,
        downloads: skill.downloads,
        created_at: skill.createdAt,
      },
    };
  }

  private async extractPackage(
    buffer: Buffer,
  ): Promise<{ skillMd?: string; files: string[] }> {
    const files: string[] = [];
    let skillMd: string | undefined;

    // Create readable stream from buffer
    const stream = Readable.from(buffer);

    await new Promise<void>((resolve, reject) => {
      stream
        .pipe(tar.t())
        .on("entry", (entry) => {
          files.push(entry.path);
          if (entry.path.endsWith("SKILL.md")) {
            let content = "";
            entry.on("data", (chunk: Buffer) => {
              content += chunk.toString();
            });
            entry.on("end", () => {
              skillMd = content;
            });
          }
        })
        .on("end", resolve)
        .on("error", reject);
    });

    return { skillMd, files };
  }

  async getDownloadUrl(name: string, version?: string): Promise<string | null> {
    const skill = await prisma.skill.findUnique({
      where: { name },
      include: { versions: true },
    });

    if (!skill) return null;

    const targetVersion = version || skill.currentVersion;
    const versionRecord = skill.versions.find(
      (v) => v.version === targetVersion,
    );

    if (!versionRecord) return null;

    const command = new GetObjectCommand({
      Bucket: process.env.S3_BUCKET!,
      Key: versionRecord.packageUrl,
    });

    return getSignedUrl(s3, command, { expiresIn: 3600 });
  }

  async recordDownload(name: string): Promise<void> {
    await prisma.skill.update({
      where: { name },
      data: { downloads: { increment: 1 } },
    });
  }
}
```

### Step 5: Version Management

Proper version management ensures safe updates and dependency resolution.

```typescript
// src/services/version-service.ts
import { PrismaClient } from "@prisma/client";
import * as semver from "semver";

const prisma = new PrismaClient();

export interface VersionInfo {
  version: string;
  releaseNotes?: string;
  createdAt: Date;
  downloads: number;
}

export interface DependencyResolution {
  resolved: Array<{ name: string; version: string }>;
  missing: string[];
  conflicts: Array<{ name: string; required: string; available: string[] }>;
}

export class VersionService {
  /**
   * Get all versions of a skill
   */
  async getVersions(skillName: string): Promise<VersionInfo[]> {
    const skill = await prisma.skill.findUnique({
      where: { name: skillName },
      include: {
        versions: {
          orderBy: { createdAt: "desc" },
        },
      },
    });

    if (!skill) return [];

    return skill.versions.map((v) => ({
      version: v.version,
      releaseNotes: v.releaseNotes || undefined,
      createdAt: v.createdAt,
      downloads: 0, // Would need per-version tracking
    }));
  }

  /**
   * Check if a version satisfies a semver range
   */
  satisfies(version: string, range: string): boolean {
    return semver.satisfies(version, range);
  }

  /**
   * Get the latest version matching a range
   */
  async getLatestMatching(
    skillName: string,
    range: string,
  ): Promise<string | null> {
    const versions = await this.getVersions(skillName);
    const versionStrings = versions.map((v) => v.version);

    return semver.maxSatisfying(versionStrings, range);
  }

  /**
   * Resolve dependencies for a skill
   */
  async resolveDependencies(
    skillName: string,
    version: string,
  ): Promise<DependencyResolution> {
    const resolved: Array<{ name: string; version: string }> = [];
    const missing: string[] = [];
    const conflicts: Array<{
      name: string;
      required: string;
      available: string[];
    }> = [];

    // Get skill version with dependencies
    const skillVersion = await prisma.skillVersion.findFirst({
      where: {
        skill: { name: skillName },
        version,
      },
    });

    if (!skillVersion) {
      return { resolved, missing: [skillName], conflicts };
    }

    const dependencies = skillVersion.dependencies as Array<{
      name: string;
      version: string;
    }>;

    // Resolve each dependency
    for (const dep of dependencies) {
      const depVersions = await this.getVersions(dep.name);

      if (depVersions.length === 0) {
        missing.push(dep.name);
        continue;
      }

      const matchingVersion = await this.getLatestMatching(
        dep.name,
        dep.version,
      );

      if (matchingVersion) {
        resolved.push({ name: dep.name, version: matchingVersion });
      } else {
        conflicts.push({
          name: dep.name,
          required: dep.version,
          available: depVersions.map((v) => v.version),
        });
      }
    }

    return { resolved, missing, conflicts };
  }

  /**
   * Validate version bump
   */
  validateVersionBump(
    currentVersion: string,
    newVersion: string,
  ): { valid: boolean; error?: string } {
    if (!semver.valid(newVersion)) {
      return { valid: false, error: "Invalid semver format" };
    }

    if (!semver.gt(newVersion, currentVersion)) {
      return {
        valid: false,
        error: `New version ${newVersion} must be greater than ${currentVersion}`,
      };
    }

    return { valid: true };
  }
}
```

```typescript
// cli/src/installer.ts
import * as fs from "fs/promises";
import * as path from "path";
import * as os from "os";
import * as tar from "tar";
import { SkillClient } from "./client";

export interface InstallOptions {
  version?: string;
  global?: boolean;
  installDeps?: boolean;
}

export interface InstallResult {
  success: boolean;
  version?: string;
  path?: string;
  dependencies: string[];
  error?: string;
}

export interface InstalledSkill {
  name: string;
  version: string;
  path: string;
}

export class Installer {
  private client: SkillClient;

  constructor(client: SkillClient) {
    this.client = client;
  }

  /**
   * Get the skills installation directory
   */
  private getSkillsDir(global: boolean): string {
    if (global) {
      return path.join(os.homedir(), ".gemini", "skills");
    }
    return path.join(process.cwd(), ".gemini", "skills");
  }

  /**
   * Install a skill from the marketplace
   */
  async install(
    name: string,
    options: InstallOptions = {},
  ): Promise<InstallResult> {
    const { version, global = false, installDeps = true } = options;
    const installedDeps: string[] = [];

    try {
      // Get skill info
      const skill = await this.client.getSkill(name, version);
      if (!skill) {
        return {
          success: false,
          error: `Skill '${name}' not found`,
          dependencies: [],
        };
      }

      // Resolve dependencies first
      if (installDeps && skill.dependencies?.length > 0) {
        for (const dep of skill.dependencies) {
          const depResult = await this.install(dep.name, {
            version: dep.version,
            global,
            installDeps: true,
          });

          if (depResult.success) {
            installedDeps.push(`${dep.name}@${depResult.version}`);
          }
        }
      }

      // Get download URL
      const downloadUrl = await this.client.getDownloadUrl(name, skill.version);
      if (!downloadUrl) {
        return {
          success: false,
          error: "Failed to get download URL",
          dependencies: [],
        };
      }

      // Download package
      const packageBuffer = await this.client.downloadPackage(downloadUrl);

      // Extract to skills directory
      const skillsDir = this.getSkillsDir(global);
      const skillDir = path.join(skillsDir, name);

      // Remove existing installation
      await fs.rm(skillDir, { recursive: true, force: true });
      await fs.mkdir(skillDir, { recursive: true });

      // Extract package
      await tar.extract({
        file: packageBuffer,
        cwd: skillsDir,
        strip: 1, // Remove top-level directory from archive
      });

      return {
        success: true,
        version: skill.version,
        path: skillDir,
        dependencies: installedDeps,
      };
    } catch (error) {
      return {
        success: false,
        error: (error as Error).message,
        dependencies: installedDeps,
      };
    }
  }

  /**
   * List installed skills
   */
  async listInstalled(global: boolean = false): Promise<InstalledSkill[]> {
    const skillsDir = this.getSkillsDir(global);
    const skills: InstalledSkill[] = [];

    try {
      const entries = await fs.readdir(skillsDir, { withFileTypes: true });

      for (const entry of entries) {
        if (!entry.isDirectory()) continue;

        const skillMdPath = path.join(skillsDir, entry.name, "SKILL.md");
        try {
          const content = await fs.readFile(skillMdPath, "utf-8");
          const versionMatch = content.match(
            /version:\s*["']?(\d+\.\d+\.\d+)["']?/,
          );

          skills.push({
            name: entry.name,
            version: versionMatch?.[1] || "unknown",
            path: path.join(skillsDir, entry.name),
          });
        } catch {
          // Skip directories without valid SKILL.md
        }
      }
    } catch {
      // Skills directory doesn't exist
    }

    return skills;
  }

  /**
   * Uninstall a skill
   */
  async uninstall(name: string, global: boolean = false): Promise<void> {
    const skillsDir = this.getSkillsDir(global);
    const skillDir = path.join(skillsDir, name);

    try {
      await fs.access(skillDir);
      await fs.rm(skillDir, { recursive: true });
    } catch {
      throw new Error(`Skill '${name}' is not installed`);
    }
  }
}
```

````typescript
// src/loader/progressive-loader.ts
import * as fs from "fs/promises";
import * as path from "path";
import matter from "gray-matter";

export interface SkillMetadata {
  name: string;
  description: string;
  version: string;
  categories: string[];
  argumentHint?: string;
  tokenEstimate: number;
}

export interface ActivatedSkill {
  name: string;
  instructions: string;
  metadata: SkillMetadata;
  resources: {
    scripts: string[];
    references: string[];
    assets: string[];
  };
  fileTree: string;
}

export class ProgressiveSkillLoader {
  private skillsDir: string;
  private metadataCache: Map<string, SkillMetadata> = new Map();
  private contentCache: Map<string, ActivatedSkill> = new Map();

  constructor(skillsDir: string) {
    this.skillsDir = skillsDir;
  }

  /**
   * Get lightweight catalog of all available skills
   * This is what goes into the agent's initial context
   */
  async getCatalog(): Promise<string> {
    const skills = await this.discoverSkills();

    let catalog = "## Available Skills\n\n";
    catalog += "Use `activate_skill(name)` to load full instructions.\n\n";

    for (const skill of skills) {
      catalog += `- **${skill.name}**: ${skill.description}`;
      if (skill.argumentHint) {
        catalog += ` (hint: ${skill.argumentHint})`;
      }
      catalog += "\n";
    }

    // Add token estimate
    const totalTokens = skills.reduce((sum, s) => sum + 35, 0); // ~35 tokens per entry
    catalog += `\n_Catalog: ~${totalTokens} tokens | Full skills: 500-3000 tokens each_\n`;

    return catalog;
  }

  /**
   * Discover all skills and extract metadata only
   */
  async discoverSkills(): Promise<SkillMetadata[]> {
    if (this.metadataCache.size > 0) {
      return Array.from(this.metadataCache.values());
    }

    const skills: SkillMetadata[] = [];

    try {
      const entries = await fs.readdir(this.skillsDir, { withFileTypes: true });

      for (const entry of entries) {
        if (!entry.isDirectory()) continue;

        const skillMdPath = path.join(this.skillsDir, entry.name, "SKILL.md");

        try {
          const content = await fs.readFile(skillMdPath, "utf-8");
          const { data } = matter(content);

          const metadata: SkillMetadata = {
            name: data.name || entry.name,
            description: data.description || "No description",
            version: data.version || "0.0.0",
            categories: data.categories || [],
            argumentHint: data["argument-hint"],
            tokenEstimate: this.estimateTokens(content),
          };

          this.metadataCache.set(metadata.name, metadata);
          skills.push(metadata);
        } catch {
          // Skip invalid skills
        }
      }
    } catch {
      // Skills directory doesn't exist
    }

    return skills;
  }

  /**
   * Activate a skill - load full content
   */
  async activate(skillName: string): Promise<ActivatedSkill | null> {
    // Check cache first
    if (this.contentCache.has(skillName)) {
      return this.contentCache.get(skillName)!;
    }

    const skillDir = path.join(this.skillsDir, skillName);
    const skillMdPath = path.join(skillDir, "SKILL.md");

    try {
      const content = await fs.readFile(skillMdPath, "utf-8");
      const { data, content: instructions } = matter(content);

      // Collect resources
      const resources = {
        scripts: await this.listDir(path.join(skillDir, "scripts")),
        references: await this.listDir(path.join(skillDir, "references")),
        assets: await this.listDir(path.join(skillDir, "assets")),
      };

      // Build file tree
      const fileTree = await this.buildFileTree(skillDir);

      const activated: ActivatedSkill = {
        name: skillName,
        instructions,
        metadata: {
          name: data.name || skillName,
          description: data.description || "",
          version: data.version || "0.0.0",
          categories: data.categories || [],
          argumentHint: data["argument-hint"],
          tokenEstimate: this.estimateTokens(content),
        },
        resources,
        fileTree,
      };

      this.contentCache.set(skillName, activated);
      return activated;
    } catch {
      return null;
    }
  }

  /**
   * Format activated skill for injection into agent context
   */
  formatForContext(skill: ActivatedSkill): string {
    let context = `## Activated Skill: ${skill.name}\n\n`;
    context += skill.instructions;
    context += "\n\n### Available Resources\n\n";
    context += "```\n" + skill.fileTree + "```\n";

    if (skill.resources.scripts.length > 0) {
      context += "\n**Scripts:** " + skill.resources.scripts.join(", ");
    }
    if (skill.resources.references.length > 0) {
      context += "\n**References:** " + skill.resources.references.join(", ");
    }

    return context;
  }

  private async listDir(dirPath: string): Promise<string[]> {
    try {
      const entries = await fs.readdir(dirPath);
      return entries;
    } catch {
      return [];
    }
  }

  private async buildFileTree(dirPath: string, prefix = ""): Promise<string> {
    let tree = "";

    try {
      const entries = await fs.readdir(dirPath, { withFileTypes: true });
      const sorted = entries.sort((a, b) => {
        if (a.isDirectory() !== b.isDirectory()) {
          return a.isDirectory() ? -1 : 1;
        }
        return a.name.localeCompare(b.name);
      });

      for (let i = 0; i < sorted.length; i++) {
        const entry = sorted[i];
        const isLast = i === sorted.length - 1;
        const connector = isLast ? "└── " : "├── ";

        tree += prefix + connector + entry.name + "\n";

        if (entry.isDirectory()) {
          const newPrefix = prefix + (isLast ? "    " : "│   ");
          tree += await this.buildFileTree(
            path.join(dirPath, entry.name),
            newPrefix,
          );
        }
      }
    } catch {
      // Directory doesn't exist
    }

    return tree;
  }

  private estimateTokens(content: string): number {
    // Rough estimate: ~4 characters per token
    return Math.ceil(content.length / 4);
  }
}
````

```typescript
import { ProgressiveSkillLoader, ActivatedSkill } from "./progressive-loader";

/**
 * Example integration with an agent framework
 */
export class SkillEnabledAgent {
  private loader: ProgressiveSkillLoader;
  private activeSkills: Map<string, ActivatedSkill> = new Map();

  constructor(skillsDir: string) {
    this.loader = new ProgressiveSkillLoader(skillsDir);
  }

  /**
   * Build initial system prompt with skill catalog
   */
  async buildSystemPrompt(basePrompt: string): Promise<string> {
    const catalog = await this.loader.getCatalog();

    return `${basePrompt}

${catalog}

To use a skill:
1. Call activate_skill(skill_name) to load full instructions
2. Follow the skill's process and output format
3. Use run_skill_script(skill_name, script_name, args) if needed
`;
  }

  /**
   * Tool: Activate a skill
   */
  async activateSkill(skillName: string): Promise<string> {
    const skill = await this.loader.activate(skillName);

    if (!skill) {
      return `Error: Skill '${skillName}' not found`;
    }

    this.activeSkills.set(skillName, skill);

    // Return formatted instructions for context injection
    return this.loader.formatForContext(skill);
  }

  /**
   * Tool: Run a skill script
   */
  async runSkillScript(
    skillName: string,
    scriptName: string,
    args: string[] = [],
  ): Promise<string> {
    const skill = this.activeSkills.get(skillName);

    if (!skill) {
      return `Error: Skill '${skillName}' not activated. Call activate_skill first.`;
    }

    if (!skill.resources.scripts.includes(scriptName)) {
      return `Error: Script '${scriptName}' not found in skill '${skillName}'`;
    }

    // Execute script (simplified - would use child_process in real implementation)
    const scriptPath = `${skillName}/scripts/${scriptName}`;
    return `[Would execute: ${scriptPath} ${args.join(" ")}]`;
  }

  /**
   * Tool: Read a skill reference
   */
  async readSkillReference(
    skillName: string,
    referenceName: string,
  ): Promise<string> {
    const skill = this.activeSkills.get(skillName);

    if (!skill) {
      return `Error: Skill '${skillName}' not activated`;
    }

    if (!skill.resources.references.includes(referenceName)) {
      return `Error: Reference '${referenceName}' not found`;
    }

    // Read reference file (simplified)
    return `[Would read: ${skillName}/references/${referenceName}]`;
  }

  /**
   * Get context window usage stats
   */
  getContextStats(): { catalog: number; active: number; total: number } {
    const catalogTokens = this.loader["metadataCache"].size * 35;
    const activeTokens = Array.from(this.activeSkills.values()).reduce(
      (sum, skill) => sum + skill.metadata.tokenEstimate,
      0,
    );

    return {
      catalog: catalogTokens,
      active: activeTokens,
      total: catalogTokens + activeTokens,
    };
  }
}
```

## Key Takeaways

<Callout type="success">

**What You Built**:

1. **SKILL.md Validator**: Schema-based validation ensuring all published skills meet the standard
2. **Package System**: Compressed, versioned skill packages with integrity verification
3. **Semantic Search**: Fast, typo-tolerant discovery using Typesense
4. **Publishing API**: Complete workflow from upload to indexed, downloadable skill
5. **CLI Tool**: Developer-friendly commands for search, install, publish, and management
6. **Progressive Disclosure**: Efficient loading that prevents context window bloat

</Callout>

<Callout type="info">

**Architecture Decisions**:

- **Typesense over Elasticsearch**: Simpler setup, excellent typo tolerance, fast enough for this scale
- **S3 for packages**: Reliable, scalable, cost-effective storage with presigned URLs
- **PostgreSQL for metadata**: ACID compliance, JSON support, familiar tooling
- **Commander.js for CLI**: Mature, well-documented, excellent developer experience

</Callout>

## Production Considerations

Before deploying to production, consider:

1. **Authentication**: Implement OAuth (GitHub) for publisher accounts
2. **Rate Limiting**: Protect API endpoints from abuse
3. **CDN**: Cache package downloads at edge locations
4. **Monitoring**: Add observability with Prometheus/Grafana
5. **Backup**: Regular PostgreSQL and Typesense backups
6. **Security**: Scan uploaded packages for malicious content
