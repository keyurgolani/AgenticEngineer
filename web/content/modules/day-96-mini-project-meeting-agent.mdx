## Project Overview

**Objective**: Build an intelligent Meeting Intelligence Agent that processes audio recordings, transcribes conversations with speaker identification, extracts action items with assignees, generates comprehensive summaries, and maintains a searchable meeting archive.

**Why This Matters**: Meetings consume significant organizational time, yet valuable insights often get lost. A Meeting Intelligence Agent transforms passive recordings into active knowledge—automatically capturing who said what, what was decided, and what needs to happen next. This is one of the highest-impact applications of AI in enterprise productivity.

## Module Summary

In this project, you'll build a complete Meeting Intelligence Agent:

1. **Audio Processing**: Handle various audio formats (WAV, MP3, M4A) with quality normalization
2. **Transcription**: Use Whisper for accurate speech-to-text with word-level timestamps
3. **Speaker Diarization**: Identify and label different speakers throughout the meeting
4. **Action Item Extraction**: Use GPT-4o to identify tasks, owners, and deadlines
5. **Summary Generation**: Create structured summaries with key topics and decisions
6. **Meeting Archive**: Store meetings with vector embeddings for semantic search
7. **Follow-up Tasks**: Generate and track action items from meetings

**Architecture Insights**:

- **Pipeline Design**: Each stage processes data independently, enabling parallel processing and easy debugging
- **Speaker Diarization**: Uses audio features to cluster speech segments by speaker, then merges with transcript
- **Dual Storage**: SQLite for structured data (meetings, action items), vector store for semantic search
- **Graceful Degradation**: If diarization fails, falls back to single-speaker transcript; if extraction fails, returns raw transcript

### Error Handling Matrix

| Error Code           | HTTP Status | Cause                          | Response Body                                 | Recovery Action                    |
| -------------------- | ----------- | ------------------------------ | --------------------------------------------- | ---------------------------------- |
| INVALID_FORMAT       | 400         | Unsupported audio format       | `{"detail": "Unsupported audio format"}`      | Convert to supported format        |
| FILE_TOO_LARGE       | 400         | File exceeds 500MB limit       | `{"detail": "File size exceeds 500MB limit"}` | Split or compress audio            |
| TRANSCRIPTION_FAILED | 500         | Whisper API error              | `{"detail": "Transcription failed"}`          | Retry or check audio quality       |
| DIARIZATION_FAILED   | 500         | Speaker detection failed       | `{"detail": "Speaker diarization failed"}`    | Falls back to single speaker       |
| EXTRACTION_FAILED    | 500         | LLM extraction error           | `{"detail": "Action extraction failed"}`      | Returns transcript without actions |
| MEETING_NOT_FOUND    | 404         | Meeting ID doesn't exist       | `{"detail": "Meeting not found"}`             | Check meeting ID                   |
| PROCESSING_TIMEOUT   | 500         | Processing exceeded time limit | `{"detail": "Processing timed out"}`          | Retry with shorter audio           |

```python
# models/meeting.py
"""Meeting data models."""

from pydantic import BaseModel, Field
from typing import List, Optional
from datetime import datetime, date
from enum import Enum
import uuid


class ProcessingStatus(str, Enum):
    """Meeting processing status."""
    PENDING = "pending"
    PROCESSING = "processing"
    COMPLETED = "completed"
    FAILED = "failed"


class Priority(str, Enum):
    """Action item priority levels."""
    HIGH = "high"
    MEDIUM = "medium"
    LOW = "low"


class ActionItemStatus(str, Enum):
    """Action item completion status."""
    PENDING = "pending"
    COMPLETED = "completed"


class Sentiment(str, Enum):
    """Meeting sentiment classification."""
    POSITIVE = "positive"
    NEUTRAL = "neutral"
    NEGATIVE = "negative"


class TranscriptSegment(BaseModel):
    """A segment of transcribed speech."""
    id: str = Field(default_factory=lambda: str(uuid.uuid4()))
    speaker_id: str
    speaker_name: Optional[str] = None
    text: str
    start_time: float
    end_time: float
    confidence: float = 1.0


class Transcript(BaseModel):
    """Complete meeting transcript."""
    segments: List[TranscriptSegment]
    full_text: str = ""
    language: str = "en"
    duration_seconds: float = 0.0

    def model_post_init(self, __context) -> None:
        """Generate full text from segments if not provided."""
        if not self.full_text and self.segments:
            self.full_text = " ".join(seg.text for seg in self.segments)
        if not self.duration_seconds and self.segments:
            self.duration_seconds = max(seg.end_time for seg in self.segments)


class ActionItem(BaseModel):
    """An action item extracted from the meeting."""
    id: str = Field(default_factory=lambda: str(uuid.uuid4()))
    description: str
    assignee: Optional[str] = None
    deadline: Optional[date] = None
    priority: Priority = Priority.MEDIUM
    status: ActionItemStatus = ActionItemStatus.PENDING
    context: str = ""  # Relevant transcript excerpt
    created_at: datetime = Field(default_factory=datetime.utcnow)


class MeetingSummary(BaseModel):
    """Structured meeting summary."""
    overview: str
    key_topics: List[str] = Field(default_factory=list)
    decisions: List[str] = Field(default_factory=list)
    next_steps: List[str] = Field(default_factory=list)
    sentiment: Sentiment = Sentiment.NEUTRAL


class Meeting(BaseModel):
    """Complete meeting record."""
    id: str = Field(default_factory=lambda: str(uuid.uuid4()))
    title: str
    date: datetime
    duration_seconds: int = 0
    status: ProcessingStatus = ProcessingStatus.PENDING
    participants: List[str] = Field(default_factory=list)
    audio_path: Optional[str] = None
    language: str = "en"
    transcript: Optional[Transcript] = None
    action_items: List[ActionItem] = Field(default_factory=list)
    summary: Optional[MeetingSummary] = None
    created_at: datetime = Field(default_factory=datetime.utcnow)
    processed_at: Optional[datetime] = None
    error_message: Optional[str] = None


class MeetingUploadRequest(BaseModel):
    """Request model for meeting upload."""
    title: Optional[str] = None
    date: Optional[datetime] = None
    participants: List[str] = Field(default_factory=list)
    language: str = "en"


class MeetingListItem(BaseModel):
    """Simplified meeting for list views."""
    id: str
    title: str
    date: datetime
    duration_seconds: int
    status: ProcessingStatus
    action_item_count: int
    participant_count: int


class MeetingListResponse(BaseModel):
    """Paginated meeting list response."""
    meetings: List[MeetingListItem]
    total: int
    limit: int
    offset: int
```

```python
# models/search.py
"""Search-related data models."""

from pydantic import BaseModel, Field
from typing import List, Optional
from datetime import datetime


class SearchQuery(BaseModel):
    """Search request model."""
    query: str = Field(..., min_length=2, max_length=500)
    limit: int = Field(default=10, ge=1, le=50)
    include_context: bool = True
    from_date: Optional[datetime] = None
    to_date: Optional[datetime] = None


class MatchingSegment(BaseModel):
    """A transcript segment matching the search query."""
    speaker: str
    text: str
    timestamp: float


class SearchResult(BaseModel):
    """A single search result."""
    meeting_id: str
    title: str
    date: datetime
    relevance_score: float
    matching_segments: List[MatchingSegment] = Field(default_factory=list)


class SearchResponse(BaseModel):
    """Search response with results."""
    results: List[SearchResult]
    query: str
    total_results: int
```

```python
class AudioProcessingError(Exception):

class AudioProcessor:
    """Handles audio file validation, conversion, and preprocessing."""

    def __init__(self):
        self.settings = get_settings()
        self._ensure_storage_dirs()

    def _ensure_storage_dirs(self) -> None:
        """Create storage directories if they don't exist."""
        self.settings.audio_storage_path.mkdir(parents=True, exist_ok=True)

    def validate_file(self, file_path: Path) -> Tuple[bool, Optional[str]]:
        """
        Validate audio file format and size.

        Returns:
            Tuple of (is_valid, error_message)
        """
        # Check file exists
        if not file_path.exists():
            return False, "File does not exist"

        # Check file extension
        suffix = file_path.suffix.lower()
        if suffix not in self.settings.supported_audio_formats:
            return False, f"Unsupported format: {suffix}. Supported: {self.settings.supported_audio_formats}"

        # Check file size
        file_size_mb = file_path.stat().st_size / (1024 * 1024)
        if file_size_mb > self.settings.max_file_size_mb:
            return False, f"File too large: {file_size_mb:.1f}MB. Maximum: {self.settings.max_file_size_mb}MB"

        # Verify file is readable audio
        try:
            duration = self._get_duration(file_path)
            max_duration = self.settings.max_audio_duration_hours * 3600
            if duration > max_duration:
                return False, f"Audio too long: {duration/3600:.1f}h. Maximum: {self.settings.max_audio_duration_hours}h"
        except Exception as e:
            return False, f"Cannot read audio file: {str(e)}"

        return True, None

    def _get_duration(self, file_path: Path) -> float:
        """Get audio duration in seconds using ffprobe."""
        try:
            result = subprocess.run(
                [
                    "ffprobe", "-v", "error",
                    "-show_entries", "format=duration",
                    "-of", "default=noprint_wrappers=1:nokey=1",
                    str(file_path)
                ],
                capture_output=True,
                text=True,
                timeout=30
            )
            return float(result.stdout.strip())
        except (subprocess.TimeoutExpired, ValueError) as e:
            raise AudioProcessingError(f"Failed to get audio duration: {e}")

    def convert_to_wav(
        self,
        input_path: Path,
        output_path: Optional[Path] = None,
        sample_rate: int = 16000,
        channels: int = 1
    ) -> Path:
        """
        Convert audio file to WAV format optimized for speech recognition.

        Args:
            input_path: Path to input audio file
            output_path: Optional output path (generates temp file if not provided)
            sample_rate: Target sample rate (16kHz optimal for Whisper)
            channels: Number of audio channels (mono for speech)

        Returns:
            Path to converted WAV file
        """

            output_path = input_path.with_suffix(".wav")

        try:
            subprocess.run(
                [
                    "ffmpeg", "-y",
                    "-i", str(input_path),
                    "-ar", str(sample_rate),
                    "-ac", str(channels),
                    "-c:a", "pcm_s16le",
                    str(output_path)
                ],
                capture_output=True,
                check=True,
                timeout=300  # 5 minute timeout
            )

            logger.info(f"Converted {input_path.name} to WAV: {output_path}")
            return output_path

        except subprocess.CalledProcessError as e:
            raise AudioProcessingError(f"FFmpeg conversion failed: {e.stderr.decode()}")
        except subprocess.TimeoutExpired:
            raise AudioProcessingError("Audio conversion timed out")

    def get_audio_info(self, file_path: Path) -> dict:
        """Get audio file metadata."""
        try:
            result = subprocess.run(
                [
                    "ffprobe", "-v", "quiet",
                    "-print_format", "json",
                    "-show_format", "-show_streams",
                    str(file_path)
                ],
                capture_output=True,
                text=True,
                timeout=30
            )
            import json
            return json.loads(result.stdout)
        except Exception as e:

            Path to saved file

        # Preserve original extension
        suffix = Path(filename).suffix.lower()
        output_path = meeting_dir / f"original{suffix}"

        # Write file
        output_path.write_bytes(file_content)
        logger.info(f"Saved audio file: {output_path} ({len(file_content) / 1024 / 1024:.1f}MB)")

        return output_path
```

```python
"""Speech-to-text transcription using OpenAI Whisper."""

import logging
from pathlib import Path
from typing import List, Optional
from openai import AsyncOpenAI
import asyncio

from config import get_settings
from models.meeting import TranscriptSegment, Transcript

logger = logging.getLogger(__name__)


class TranscriptionError(Exception):
    """Custom exception for transcription failures."""
    pass


class Transcriber:
    """Transcribes audio files using OpenAI Whisper API."""

    # Whisper API file size limit
    MAX_FILE_SIZE_MB = 25

    def __init__(self):
        self.settings = get_settings()
        self.client = AsyncOpenAI(api_key=self.settings.openai_api_key)

    async def transcribe(
        self,
        audio_path: Path,
        language: Optional[str] = None
    ) -> Transcript:
        """
        Transcribe audio file to text with timestamps.

        Args:
            audio_path: Path to audio file (WAV format preferred)
            language: Optional language code (auto-detected if not provided)

        Returns:
            Transcript with word-level segments
        """
        if not audio_path.exists():
            raise TranscriptionError(f"Audio file not found: {audio_path}")

        file_size_mb = audio_path.stat().st_size / (1024 * 1024)

        # Handle large files by chunking
        if file_size_mb > self.MAX_FILE_SIZE_MB:
            logger.info(f"Large file ({file_size_mb:.1f}MB), processing in chunks")
            return await self._transcribe_chunked(audio_path, language)

        return await self._transcribe_single(audio_path, language)

    async def _transcribe_single(
        self,
        audio_path: Path,
        language: Optional[str] = None
    ) -> Transcript:
        """Transcribe a single audio file."""
        try:
            with open(audio_path, "rb") as audio_file:
                # Request verbose JSON for word-level timestamps
                response = await self.client.audio.transcriptions.create(
                    model=self.settings.whisper_model,
                    file=audio_file,
                    language=language or self.settings.default_language,
                    response_format="verbose_json",
                    timestamp_granularities=["word", "segment"]
                )

            # Parse response into segments
            segments = self._parse_whisper_response(response)

            return Transcript(
                segments=segments,
                language=response.language or language or "en",
                duration_seconds=response.duration or 0.0
            )

        except Exception as e:
            logger.error(f"Transcription failed: {e}")
            raise TranscriptionError(f"Whisper API error: {str(e)}")

    async def _transcribe_chunked(
        self,
        audio_path: Path,
        language: Optional[str] = None
    ) -> Transcript:
        """
        Transcribe large audio file by splitting into chunks.

        Uses ffmpeg to split audio into segments under the API limit.
        """
        import subprocess
        import tempfile

        # Get audio duration
        duration = self._get_duration(audio_path)
        chunk_duration = 600  # 10 minutes per chunk

        all_segments: List[TranscriptSegment] = []
        time_offset = 0.0

        with tempfile.TemporaryDirectory() as temp_dir:
            chunk_num = 0
            current_time = 0.0

            while current_time < duration:
                chunk_path = Path(temp_dir) / f"chunk_{chunk_num}.wav"

                # Extract chunk using ffmpeg
                subprocess.run(
                    [
                        "ffmpeg", "-y",
                        "-i", str(audio_path),
                        "-ss", str(current_time),
                        "-t", str(chunk_duration),
                        "-c:a", "pcm_s16le",
                        str(chunk_path)
                    ],
                    capture_output=True,
                    check=True
                )

                # Transcribe chunk
                chunk_transcript = await self._transcribe_single(chunk_path, language)

                # Adjust timestamps and add to results
                for segment in chunk_transcript.segments:
                    segment.start_time += time_offset
                    segment.end_time += time_offset
                    all_segments.append(segment)

                time_offset = current_time + chunk_duration
                current_time += chunk_duration
                chunk_num += 1

                logger.info(f"Processed chunk {chunk_num}, {len(all_segments)} segments total")

        return Transcript(
            segments=all_segments,
            language=language or "en",
            duration_seconds=duration
        )

    def _parse_whisper_response(self, response) -> List[TranscriptSegment]:
        """Parse Whisper API response into transcript segments."""
        segments: List[TranscriptSegment] = []

        # Use segment-level data (more reliable than word-level for speaker grouping)
        if hasattr(response, 'segments') and response.segments:
            for seg in response.segments:
                segments.append(TranscriptSegment(
                    speaker_id="speaker_0",  # Will be updated by diarization
                    text=seg.get("text", "").strip(),
                    start_time=seg.get("start", 0.0),
                    end_time=seg.get("end", 0.0),
                    confidence=seg.get("avg_logprob", 0.0) if "avg_logprob" in seg else 1.0
                ))
        elif hasattr(response, 'text'):
            # Fallback: single segment for entire transcript
            segments.append(TranscriptSegment(
                speaker_id="speaker_0",
                text=response.text.strip(),
                start_time=0.0,
                end_time=response.duration or 0.0,
                confidence=1.0
            ))

        return segments

    def _get_duration(self, audio_path: Path) -> float:
        """Get audio duration using ffprobe."""
        import subprocess
        result = subprocess.run(
            [
                "ffprobe", "-v", "error",
                "-show_entries", "format=duration",
                "-of", "default=noprint_wrappers=1:nokey=1",
                str(audio_path)
            ],
            capture_output=True,
            text=True
        )
        return float(result.stdout.strip())
```

```python
# services/diarizer.py
"""Speaker diarization for multi-speaker identification."""

import logging
from pathlib import Path
from typing import List, Tuple, Optional
from dataclasses import dataclass
import numpy as np

from config import get_settings
from models.meeting import TranscriptSegment, Transcript

logger = logging.getLogger(__name__)


@dataclass
class SpeakerSegment:
    """A segment of speech attributed to a speaker."""
    speaker_id: str
    start_time: float
    end_time: float


class DiarizationError(Exception):
    """Custom exception for diarization failures."""
    pass


class SimpleDiarizer:
    """
    Simplified speaker diarization using audio energy and clustering.

    For production use, consider pyannote.audio or similar specialized libraries.
    This implementation provides a basic approach suitable for learning.
    """

    def __init__(self):
        self.settings = get_settings()

    async def diarize(
        self,
        audio_path: Path,
        min_speakers: int = 1,
        max_speakers: int = 10
    ) -> List[SpeakerSegment]:
        """
        Perform speaker diarization on audio file.

        Args:
            audio_path: Path to WAV audio file
            min_speakers: Minimum expected speakers
            max_speakers: Maximum expected speakers

        Returns:
            List of speaker segments with timestamps
        """
        try:
            # Load audio
            audio_data, sample_rate = self._load_audio(audio_path)

            # Detect speech segments using energy-based VAD
            speech_segments = self._detect_speech(audio_data, sample_rate)

            if not speech_segments:
                logger.warning("No speech detected in audio")
                return [SpeakerSegment(
                    speaker_id="speaker_0",
                    start_time=0.0,
                    end_time=len(audio_data) / sample_rate
                )]

            # Extract features for each segment
            features = self._extract_features(audio_data, sample_rate, speech_segments)

            # Cluster segments by speaker
            speaker_labels = self._cluster_speakers(
                features,
                min_speakers,
                max_speakers
            )

            # Create speaker segments
            speaker_segments = []
            for i, (start, end) in enumerate(speech_segments):
                speaker_segments.append(SpeakerSegment(
                    speaker_id=f"speaker_{speaker_labels[i]}",
                    start_time=start,
                    end_time=end
                ))

            # Merge consecutive segments from same speaker
            merged_segments = self._merge_consecutive(speaker_segments)

            logger.info(f"Diarization complete: {len(set(s.speaker_id for s in merged_segments))} speakers detected")
            return merged_segments

        except Exception as e:
            logger.error(f"Diarization failed: {e}")
            raise DiarizationError(f"Speaker diarization failed: {str(e)}")

    def _load_audio(self, audio_path: Path) -> Tuple[np.ndarray, int]:
        """Load audio file as numpy array."""
        import wave

        with wave.open(str(audio_path), 'rb') as wav:
            sample_rate = wav.getframerate()
            n_frames = wav.getnframes()
            audio_bytes = wav.readframes(n_frames)

            # Convert to numpy array
            audio_data = np.frombuffer(audio_bytes, dtype=np.int16).astype(np.float32)
            audio_data /= 32768.0  # Normalize to [-1, 1]

        return audio_data, sample_rate

    def _detect_speech(
        self,
        audio_data: np.ndarray,
        sample_rate: int,
        frame_duration: float = 0.03,
        energy_threshold: float = 0.01
    ) -> List[Tuple[float, float]]:
        """
        Detect speech segments using energy-based voice activity detection.

        Args:
            audio_data: Audio samples
            sample_rate: Sample rate in Hz
            frame_duration: Frame size in seconds
            energy_threshold: Minimum energy for speech

        Returns:
            List of (start_time, end_time) tuples
        """
        frame_size = int(sample_rate * frame_duration)
        n_frames = len(audio_data) // frame_size

        # Calculate energy per frame
        energies = []
        for i in range(n_frames):
            frame = audio_data[i * frame_size:(i + 1) * frame_size]
            energy = np.sqrt(np.mean(frame ** 2))
            energies.append(energy)

        # Adaptive threshold
        if energies:
            threshold = max(energy_threshold, np.percentile(energies, 30))
        else:
            threshold = energy_threshold

        # Find speech segments
        speech_frames = [e > threshold for e in energies]

        # Convert to time segments
        segments = []
        in_speech = False
        start_frame = 0

        for i, is_speech in enumerate(speech_frames):
            if is_speech and not in_speech:
                start_frame = i
                in_speech = True
            elif not is_speech and in_speech:
                # Require minimum segment duration
                duration = (i - start_frame) * frame_duration
                if duration >= self.settings.min_segment_duration:
                    segments.append((
                        start_frame * frame_duration,
                        i * frame_duration
                    ))
                in_speech = False

        # Handle segment at end
        if in_speech:
            segments.append((
                start_frame * frame_duration,
                n_frames * frame_duration
            ))

        return segments

    def _extract_features(
        self,
        audio_data: np.ndarray,
        sample_rate: int,
        segments: List[Tuple[float, float]]
    ) -> np.ndarray:
        """
        Extract audio features for speaker clustering.

        Uses simple spectral features (MFCCs would be better for production).
        """
        features = []

        for start, end in segments:
            start_sample = int(start * sample_rate)
            end_sample = int(end * sample_rate)
            segment_audio = audio_data[start_sample:end_sample]

            if len(segment_audio) < 100:
                # Too short, use zeros
                features.append(np.zeros(13))
                continue

            # Simple spectral features
            # In production, use librosa.feature.mfcc or similar
            fft = np.abs(np.fft.rfft(segment_audio))

            # Divide spectrum into bands and compute energy
            n_bands = 13
            band_size = len(fft) // n_bands
            band_energies = []

            for i in range(n_bands):
                band = fft[i * band_size:(i + 1) * band_size]
                band_energies.append(np.mean(band) if len(band) > 0 else 0)

            features.append(np.array(band_energies))

        return np.array(features)

    def _cluster_speakers(
        self,
        features: np.ndarray,
        min_speakers: int,
        max_speakers: int
    ) -> List[int]:
        """
        Cluster audio segments by speaker using K-means.

        Automatically determines optimal number of speakers.
        """
        from sklearn.cluster import KMeans
        from sklearn.metrics import silhouette_score

        if len(features) < 2:
            return [0] * len(features)

        # Normalize features
        features_norm = (features - features.mean(axis=0)) / (features.std(axis=0) + 1e-8)

        # Try different numbers of clusters
        best_score = -1
        best_labels = [0] * len(features)
        best_k = 1

        for k in range(min_speakers, min(max_speakers + 1, len(features))):
            if k < 2:
                continue

            kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)
            labels = kmeans.fit_predict(features_norm)

            if k < len(features):
                score = silhouette_score(features_norm, labels)
                if score > best_score:
                    best_score = score
                    best_labels = labels.tolist()
                    best_k = k

        logger.info(f"Optimal speaker count: {best_k} (silhouette: {best_score:.3f})")
        return best_labels

    def _merge_consecutive(
        self,
        segments: List[SpeakerSegment],
        gap_threshold: float = 0.5
    ) -> List[SpeakerSegment]:
        """Merge consecutive segments from the same speaker."""
        if not segments:
            return segments

        merged = [segments[0]]

        for segment in segments[1:]:
            last = merged[-1]

            # Merge if same speaker and small gap
            if (segment.speaker_id == last.speaker_id and
                segment.start_time - last.end_time < gap_threshold):
                merged[-1] = SpeakerSegment(
                    speaker_id=last.speaker_id,
                    start_time=last.start_time,
                    end_time=segment.end_time
                )
            else:
                merged.append(segment)

        return merged

    def merge_with_transcript(
        self,
        transcript: Transcript,
        speaker_segments: List[SpeakerSegment]
    ) -> Transcript:
        """
        Merge speaker labels with transcript segments.

        Assigns speaker IDs to transcript segments based on timestamp overlap.
        """
        for trans_seg in transcript.segments:
            # Find overlapping speaker segment
            best_overlap = 0
            best_speaker = "speaker_0"

            for spk_seg in speaker_segments:
                # Calculate overlap
                overlap_start = max(trans_seg.start_time, spk_seg.start_time)
                overlap_end = min(trans_seg.end_time, spk_seg.end_time)
                overlap = max(0, overlap_end - overlap_start)

                if overlap > best_overlap:
                    best_overlap = overlap
                    best_speaker = spk_seg.speaker_id

            trans_seg.speaker_id = best_speaker

        return transcript
```

```python
class ActionExtractor:

Your task is to identify ALL action items, tasks, commitments, and follow-ups mentioned in the transcript.

RULES FOR EXTRACTION:
1. An action item must be a specific, actionable task (not a general discussion topic)
2. Look for phrases like: "will do", "need to", "should", "action item", "follow up", "by [date]", "responsible for"
3. Identify the assignee when mentioned (person's name or role like "engineering team")
4. Note any deadlines or timeframes mentioned

EXAMPLES OF ACTION ITEMS:

                    description=extracted.description,
                    assignee=extracted.assignee,
                    deadline=self._parse_deadline(extracted.deadline_description),
                    priority=self._parse_priority(extracted.priority),
                    context=extracted.context_quote
                ))

            logger.info(f"Extracted {len(action_items)} action items")
            return action_items

        except Exception as e:
            logger.error(f"Action extraction failed: {e}")
            return []

    def _format_transcript(self, transcript: Transcript) -> str:
        """Format transcript for LLM processing."""
        lines = []
        for segment in transcript.segments:
            speaker = segment.speaker_name or segment.speaker_id
            timestamp = f"[{segment.start_time:.1f}s]"
            lines.append(f"{timestamp} {speaker}: {segment.text}")
        return "\n".join(lines)

    def _parse_deadline(self, deadline_desc: Optional[str]) -> Optional[date]:
        """Parse deadline description into a date."""
        if not deadline_desc:
            return None

        deadline_lower = deadline_desc.lower()
        today = date.today()

        # Common patterns
        if "today" in deadline_lower:
            return today
        elif "tomorrow" in deadline_lower:
            return today + timedelta(days=1)
        elif "next week" in deadline_lower:
            return today + timedelta(weeks=1)
        elif "end of week" in deadline_lower or "friday" in deadline_lower:
            days_until_friday = (4 - today.weekday()) % 7
            return today + timedelta(days=days_until_friday or 7)
        elif "end of month" in deadline_lower:
            next_month = today.replace(day=28) + timedelta(days=4)
            return next_month.replace(day=1) - timedelta(days=1)
        elif "asap" in deadline_lower or "urgent" in deadline_lower:
            return today + timedelta(days=2)

        # Could add more sophisticated date parsing here
        return None

    def _parse_priority(self, priority_str: str) -> Priority:
        """Parse priority string to enum."""
        priority_lower = priority_str.lower()
        if priority_lower in ("high", "urgent", "critical"):
            return Priority.HIGH
        elif priority_lower in ("low", "minor", "eventually"):
            return Priority.LOW
        return Priority.MEDIUM
```

```python
        Generate comprehensive meeting summary.

        Args:
            transcript: Complete meeting transcript
            action_items: Previously extracted action items (for context)
            meeting_title: Meeting title for context

            Structured meeting summary
        """

            return MeetingSummary(

                key_topics=["No content"],
                sentiment=Sentiment.NEUTRAL
            )

        # Format transcript
        formatted_transcript = self._format_transcript(transcript)

        # Include action items for context
        action_context = ""
        if action_items:
            action_context = "\n\nPreviously extracted action items:\n" + "\n".join(
                f"- {item.description}" + (f" (assigned to {item.assignee})" if item.assignee else "")
                for item in action_items

                model=self.settings.openai_model,
                temperature=self.settings.llm_temperature,
                messages=[
                    {"role": "system", "content": self.SYSTEM_PROMPT},
                    {"role": "user", "content": f"Summarize this meeting titled '{meeting_title}':\n\n{formatted_transcript}{action_context}"}
                ],
                response_format=GeneratedSummary
            )

            result = response.choices[0].message.parsed

            if result is None:
                raise ValueError("Failed to parse summary")

            return MeetingSummary(
                overview=result.overview,
                key_topics=result.key_topics,
                decisions=result.decisions,
                next_steps=result.next_steps,
                sentiment=self._parse_sentiment(result.sentiment)
            )

            return MeetingSummary(

                key_topics=["Transcript available"],
                sentiment=Sentiment.NEUTRAL

        full_text = "\n".join(lines)
        max_chars = 12000  # Leave room for system prompt

        if len(full_text) > max_chars:
            half = max_chars // 2
            full_text = full_text[:half] + "\n\n[... transcript truncated ...]\n\n" + full_text[-half:]

        return full_text

        elif "negative" in sentiment_lower or "contentious" in sentiment_lower:
            return Sentiment.NEGATIVE
        return Sentiment.NEUTRAL
```

```python
# services/embedder.py
"""Generate and manage vector embeddings for semantic search."""

import logging
from typing import List, Optional, Tuple
from pathlib import Path
import json
import numpy as np
from openai import AsyncOpenAI

from config import get_settings
from models.meeting import Meeting, Transcript

logger = logging.getLogger(__name__)


class EmbeddingChunk:
    """A chunk of text with its embedding."""
    def __init__(
        self,
        meeting_id: str,
        text: str,
        embedding: List[float],
        start_time: float,
        end_time: float,
        speaker: str
    ):
        self.meeting_id = meeting_id
        self.text = text
        self.embedding = embedding
        self.start_time = start_time
        self.end_time = end_time
        self.speaker = speaker


class VectorStore:
    """Simple file-based vector store for meeting embeddings."""

    def __init__(self, storage_path: Path):
        self.storage_path = storage_path
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self._index: List[EmbeddingChunk] = []
        self._load_index()

    def _load_index(self) -> None:
        """Load existing embeddings from storage."""
        index_file = self.storage_path / "index.json"
        if index_file.exists():
            try:
                with open(index_file, "r") as f:
                    data = json.load(f)
                    for item in data:
                        self._index.append(EmbeddingChunk(
                            meeting_id=item["meeting_id"],
                            text=item["text"],
                            embedding=item["embedding"],
                            start_time=item["start_time"],
                            end_time=item["end_time"],
                            speaker=item["speaker"]
                        ))
                logger.info(f"Loaded {len(self._index)} embeddings from storage")
            except Exception as e:
                logger.warning(f"Could not load embedding index: {e}")

    def _save_index(self) -> None:
        """Save embeddings to storage."""
        index_file = self.storage_path / "index.json"
        data = [
            {
                "meeting_id": chunk.meeting_id,
                "text": chunk.text,
                "embedding": chunk.embedding,
                "start_time": chunk.start_time,
                "end_time": chunk.end_time,
                "speaker": chunk.speaker
            }
            for chunk in self._index
        ]
        with open(index_file, "w") as f:
            json.dump(data, f)

    def add(self, chunks: List[EmbeddingChunk]) -> None:
        """Add embedding chunks to the store."""
        self._index.extend(chunks)
        self._save_index()

    def search(
        self,
        query_embedding: List[float],
        limit: int = 10,
        meeting_id: Optional[str] = None
    ) -> List[Tuple[EmbeddingChunk, float]]:
        """
        Search for similar chunks using cosine similarity.

        Returns:
            List of (chunk, similarity_score) tuples
        """
        if not self._index:
            return []

        query_vec = np.array(query_embedding)

        results = []
        for chunk in self._index:
            # Filter by meeting if specified
            if meeting_id and chunk.meeting_id != meeting_id:
                continue

            # Cosine similarity
            chunk_vec = np.array(chunk.embedding)
            similarity = np.dot(query_vec, chunk_vec) / (
                np.linalg.norm(query_vec) * np.linalg.norm(chunk_vec) + 1e-8
            )
            results.append((chunk, float(similarity)))

        # Sort by similarity descending
        results.sort(key=lambda x: x[1], reverse=True)
        return results[:limit]

    def delete_meeting(self, meeting_id: str) -> int:
        """Delete all embeddings for a meeting."""
        original_count = len(self._index)
        self._index = [c for c in self._index if c.meeting_id != meeting_id]
        deleted = original_count - len(self._index)
        if deleted > 0:
            self._save_index()
        return deleted


class Embedder:
    """Generate embeddings for meeting content."""

    CHUNK_SIZE = 500  # Characters per chunk
    CHUNK_OVERLAP = 50  # Overlap between chunks

    def __init__(self):
        self.settings = get_settings()
        self.client = AsyncOpenAI(api_key=self.settings.openai_api_key)
        self.vector_store = VectorStore(self.settings.vector_store_path)

    async def embed_meeting(self, meeting: Meeting) -> int:
        """
        Generate and store embeddings for a meeting.

        Args:
            meeting: Processed meeting with transcript

        Returns:
            Number of chunks embedded
        """
        if not meeting.transcript or not meeting.transcript.segments:
            return 0

        # Create chunks from transcript
        chunks = self._create_chunks(meeting.id, meeting.transcript)

        if not chunks:
            return 0

        # Generate embeddings in batches
        texts = [c["text"] for c in chunks]
        embeddings = await self._generate_embeddings(texts)

        # Create embedding chunks
        embedding_chunks = []
        for chunk, embedding in zip(chunks, embeddings):
            embedding_chunks.append(EmbeddingChunk(
                meeting_id=chunk["meeting_id"],
                text=chunk["text"],
                embedding=embedding,
                start_time=chunk["start_time"],
                end_time=chunk["end_time"],
                speaker=chunk["speaker"]
            ))

        # Store embeddings
        self.vector_store.add(embedding_chunks)

        logger.info(f"Embedded {len(embedding_chunks)} chunks for meeting {meeting.id}")
        return len(embedding_chunks)

    def _create_chunks(
        self,
        meeting_id: str,
        transcript: Transcript
    ) -> List[dict]:
        """Create text chunks from transcript segments."""
        chunks = []
        current_chunk = {
            "meeting_id": meeting_id,
            "text": "",
            "start_time": 0.0,
            "end_time": 0.0,
            "speaker": ""
        }

        for segment in transcript.segments:
            segment_text = f"{segment.speaker_id}: {segment.text}"

            # Start new chunk if current is too long
            if len(current_chunk["text"]) + len(segment_text) > self.CHUNK_SIZE:
                if current_chunk["text"]:
                    chunks.append(current_chunk.copy())

                # Start new chunk with overlap
                overlap_text = current_chunk["text"][-self.CHUNK_OVERLAP:] if current_chunk["text"] else ""
                current_chunk = {
                    "meeting_id": meeting_id,
                    "text": overlap_text + " " + segment_text,
                    "start_time": segment.start_time,
                    "end_time": segment.end_time,
                    "speaker": segment.speaker_id
                }
            else:
                if not current_chunk["text"]:
                    current_chunk["start_time"] = segment.start_time
                current_chunk["text"] += " " + segment_text
                current_chunk["end_time"] = segment.end_time
                current_chunk["speaker"] = segment.speaker_id

        # Add final chunk
        if current_chunk["text"]:
            chunks.append(current_chunk)

        return chunks

    async def _generate_embeddings(
        self,
        texts: List[str],
        batch_size: int = 100
    ) -> List[List[float]]:
        """Generate embeddings for texts in batches."""
        all_embeddings = []

        for i in range(0, len(texts), batch_size):
            batch = texts[i:i + batch_size]

            response = await self.client.embeddings.create(
                model=self.settings.embedding_model,
                input=batch
            )

            batch_embeddings = [item.embedding for item in response.data]
            all_embeddings.extend(batch_embeddings)

        return all_embeddings

    async def search(
        self,
        query: str,
        limit: int = 10,
        meeting_id: Optional[str] = None
    ) -> List[Tuple[EmbeddingChunk, float]]:
        """
        Search meetings by semantic similarity.

        Args:
            query: Natural language search query
            limit: Maximum results to return
            meeting_id: Optional filter to specific meeting

        Returns:
            List of (chunk, score) tuples
        """
        # Generate query embedding
        response = await self.client.embeddings.create(
            model=self.settings.embedding_model,
            input=[query]
        )
        query_embedding = response.data[0].embedding

        # Search vector store
        return self.vector_store.search(query_embedding, limit, meeting_id)
```

```python
                        overview=meeting.summary.overview,
                        key_topics=meeting.summary.key_topics,
                        decisions=meeting.summary.decisions,
                        next_steps=meeting.summary.next_steps,
                        sentiment=SentimentDB(meeting.summary.sentiment.value)
                    ))

            logger.info(f"Saved meeting {meeting.id}")
            return meeting.id

    def get_meeting(self, meeting_id: str) -> Optional[Meeting]:
        """Retrieve a meeting by ID."""
        with get_db() as db:
            meeting_db = db.query(MeetingDB).filter(MeetingDB.id == meeting_id).first()

            if not meeting_db:
                return None

            return self._db_to_model(meeting_db)

    def list_meetings(
        self,
        limit: int = 20,
        offset: int = 0,
        status: Optional[ProcessingStatus] = None,
        from_date: Optional[datetime] = None,
        to_date: Optional[datetime] = None
    ) -> tuple[List[MeetingListItem], int]:
        """List meetings with pagination and filters."""
        with get_db() as db:
            query = db.query(MeetingDB)

            if status:
                query = query.filter(MeetingDB.status == ProcessingStatusDB(status.value))
            if from_date:
                query = query.filter(MeetingDB.date >= from_date)
            if to_date:
                query = query.filter(MeetingDB.date <= to_date)

            total = query.count()

            meetings = query.order_by(MeetingDB.date.desc()).offset(offset).limit(limit).all()

            items = [
                MeetingListItem(
                    id=m.id,
                    title=m.title,
                    date=m.date,
                    duration_seconds=m.duration_seconds,
                    status=ProcessingStatus(m.status.value),
                    action_item_count=len(m.action_items),
                    participant_count=len(m.participants) if m.participants else 0
                )
                for m in meetings
            ]

            return items, total

    def _db_to_model(self, meeting_db: MeetingDB) -> Meeting:
        """Convert database model to Pydantic model."""
        # Convert transcript segments
        transcript = None
        if meeting_db.segments:
            segments = [
                TranscriptSegment(
                    id=s.id,
                    speaker_id=s.speaker_id,
                    speaker_name=s.speaker_name,
                    text=s.text,
                    start_time=s.start_time,
                    end_time=s.end_time,
                    confidence=s.confidence
                )
                for s in sorted(meeting_db.segments, key=lambda x: x.start_time)
            ]
            transcript = Transcript(segments=segments)

        # Convert action items
        action_items = [
            ActionItem(
                id=a.id,
                description=a.description,
                assignee=a.assignee,
                deadline=a.deadline.date() if a.deadline else None,
                priority=Priority(a.priority.value),
                status=ActionItemStatus(a.status.value),
                context=a.context or "",
                created_at=a.created_at
            )
            for a in meeting_db.action_items
        ]

        # Convert summary
        summary = None
        if meeting_db.summary:
            summary = MeetingSummary(
                overview=meeting_db.summary.overview,
                key_topics=meeting_db.summary.key_topics or [],
                decisions=meeting_db.summary.decisions or [],
                next_steps=meeting_db.summary.next_steps or [],
                sentiment=Sentiment(meeting_db.summary.sentiment.value)
            )

        return Meeting(
            id=meeting_db.id,
            title=meeting_db.title,
            date=meeting_db.date,
```

```python
# services/pipeline.py
"""Complete meeting processing pipeline orchestration."""

import logging
import asyncio
from pathlib import Path
from datetime import datetime
from typing import Optional

from config import get_settings
from models.meeting import Meeting, ProcessingStatus
from services.audio_processor import audio_processor
from services.transcriber import transcriber
from services.diarizer import diarizer
from services.action_extractor import action_extractor
from services.summarizer import summarizer
from services.embedder import embedder
from services.archiver import archiver

logger = logging.getLogger(__name__)


class ProcessingError(Exception):
    """Custom exception for pipeline processing failures."""
    pass


class MeetingPipeline:
    """Orchestrates the complete meeting processing pipeline."""

    def __init__(self):
        self.settings = get_settings()
        self._processing_semaphore = asyncio.Semaphore(
            self.settings.max_concurrent_processing
        )

    async def process_meeting(self, meeting: Meeting) -> Meeting:
        """
        Process a meeting through the complete pipeline.

        Pipeline stages:
        1. Audio validation and conversion
        2. Transcription with Whisper
        3. Speaker diarization
        4. Action item extraction
        5. Summary generation
        6. Embedding generation
        7. Archive storage

        Args:
            meeting: Meeting with audio_path set

        Returns:
            Fully processed meeting
        """
        async with self._processing_semaphore:
            try:
                meeting.status = ProcessingStatus.PROCESSING
                archiver.save_meeting(meeting)

                logger.info(f"Starting pipeline for meeting {meeting.id}")
                start_time = datetime.utcnow()

                # Stage 1: Validate and convert audio
                audio_path = Path(meeting.audio_path)
                is_valid, error = audio_processor.validate_file(audio_path)
                if not is_valid:
                    raise ProcessingError(f"Audio validation failed: {error}")

                # Convert to WAV for processing
                wav_path = audio_processor.convert_to_wav(audio_path)
                logger.info(f"Audio converted: {wav_path}")

                # Stage 2: Transcribe audio
                logger.info("Starting transcription...")
                transcript = await transcriber.transcribe(wav_path, meeting.language)
                meeting.duration_seconds = int(transcript.duration_seconds)
                logger.info(f"Transcription complete: {len(transcript.segments)} segments")

                # Stage 3: Speaker diarization
                logger.info("Starting speaker diarization...")
                try:
                    speaker_segments = await diarizer.diarize(
                        wav_path,
                        min_speakers=self.settings.min_speakers,
                        max_speakers=self.settings.max_speakers
                    )
                    transcript = diarizer.merge_with_transcript(transcript, speaker_segments)
                    logger.info(f"Diarization complete: {len(set(s.speaker_id for s in speaker_segments))} speakers")
                except Exception as e:
                    logger.warning(f"Diarization failed, using single speaker: {e}")
                    # Continue with single speaker

                meeting.transcript = transcript

                # Stage 4: Extract action items
                logger.info("Extracting action items...")
                action_items = await action_extractor.extract(transcript)
                meeting.action_items = action_items
                logger.info(f"Extracted {len(action_items)} action items")

                # Stage 5: Generate summary
                logger.info("Generating summary...")
                summary = await summarizer.summarize(
                    transcript,
                    action_items,
                    meeting.title
                )
                meeting.summary = summary
                logger.info("Summary generated")

                # Stage 6: Generate embeddings for search
                logger.info("Generating embeddings...")
                chunk_count = await embedder.embed_meeting(meeting)
                logger.info(f"Generated {chunk_count} embedding chunks")

                # Stage 7: Mark complete and save
                meeting.status = ProcessingStatus.COMPLETED
                meeting.processed_at = datetime.utcnow()
                archiver.save_meeting(meeting)

                processing_time = (datetime.utcnow() - start_time).total_seconds()
                logger.info(f"Pipeline complete for {meeting.id} in {processing_time:.1f}s")

                return meeting

            except Exception as e:
                logger.error(f"Pipeline failed for {meeting.id}: {e}")
                meeting.status = ProcessingStatus.FAILED
                meeting.error_message = str(e)
                archiver.save_meeting(meeting)
                raise ProcessingError(f"Meeting processing failed: {e}")


# Global instance
pipeline = MeetingPipeline()
```

```python
@app.post("/meetings/upload", status_code=202)
async def upload_meeting(
    background_tasks: BackgroundTasks,
    file: UploadFile = File(...),
    title: Optional[str] = Form(None),
    date: Optional[str] = Form(None),
    participants: Optional[str] = Form(None),
    language: str = Form("en")
):
    """
    Upload and process a meeting recording.

    Processing happens in the background. Use GET /meetings/{id} to check status.
    """
    # Validate file type
    if not file.filename:
        raise HTTPException(400, "No filename provided")

    suffix = "." + file.filename.split(".")[-1].lower()
    if suffix not in settings.supported_audio_formats:
        raise HTTPException(
            400,
            f"Unsupported format: {suffix}. Supported: {settings.supported_audio_formats}"
        )

    # Create meeting record
    meeting_id = str(uuid.uuid4())
    meeting_date = datetime.fromisoformat(date) if date else datetime.utcnow()
    participant_list = [p.strip() for p in participants.split(",")] if participants else []

    meeting = Meeting(
        id=meeting_id,
        title=title or file.filename,
        date=meeting_date,
        participants=participant_list,
        language=language,
        status=ProcessingStatus.PENDING
    )

    # Save uploaded file
    file_content = await file.read()
    audio_path = await audio_processor.save_upload(file_content, file.filename, meeting_id)
    meeting.audio_path = str(audio_path)

    # Save initial meeting record
    archiver.save_meeting(meeting)

    # Start background processing
    background_tasks.add_task(pipeline.process_meeting, meeting)

    # Estimate processing time (rough: 1 minute per 10 minutes of audio)
    file_size_mb = len(file_content) / (1024 * 1024)
    estimated_duration = int(file_size_mb * 6)  # Rough estimate

    return {
        "meeting_id": meeting_id,
        "status": "processing",
        "message": "Meeting upload accepted, processing started",
        "estimated_duration_seconds": estimated_duration
    }

@app.post("/meetings/search", response_model=SearchResponse)
async def search_meetings(query: SearchQuery):
    """Search meetings by semantic similarity."""
    results = await embedder.search(
        query=query.query,
        limit=query.limit
    )

    # Group results by meeting
    meeting_results: dict = {}
    for chunk, score in results:
        if chunk.meeting_id not in meeting_results:
            meeting = archiver.get_meeting(chunk.meeting_id)
            if meeting:
                meeting_results[chunk.meeting_id] = SearchResult(
                    meeting_id=chunk.meeting_id,
                    title=meeting.title,
                    date=meeting.date,
                    relevance_score=score,
                    matching_segments=[]
                )

        if chunk.meeting_id in meeting_results and query.include_context:
            meeting_results[chunk.meeting_id].matching_segments.append(
                MatchingSegment(
                    speaker=chunk.speaker,
                    text=chunk.text,
                    timestamp=chunk.start_time
                )
            )

    return SearchResponse(
        results=list(meeting_results.values()),
        query=query.query,
        total_results=len(meeting_results)
    )
```
