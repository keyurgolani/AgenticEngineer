## The Integration Problem

Before MCP, connecting an LLM to external tools was a nightmare. Every model provider had different formats:

- OpenAI: Function calling with JSON schemas
- Anthropic: Tool use with XML
- Google: Function declarations
- Local models: Custom implementations

**MCP solves this** by providing a universal protocol that works across all models and tools.

> **Key Insight**: MCP is to AI tools what USB-C is to devices—write once, connect everywhere.

**Real-World Analogy**: Remember when every phone had a different charger? Apple had its 30-pin connector, Android phones used micro-USB, and laptops needed proprietary adapters. USB-C solved this chaos by creating one universal standard. MCP does the same for AI agents—instead of writing custom integrations for every model provider, you write one MCP server that works with Claude, ChatGPT, Gemini, and any future AI system.

## 11.2 MCP Server Types and Transports

### Transport Methods

| Transport           | Use Case              | Status         | Example                  |
| :------------------ | :-------------------- | :------------- | :----------------------- |
| **stdio**           | Local processes       | ✅ Stable      | CLI tools, local scripts |
| **Streamable HTTP** | Remote servers        | ✅ Recommended | Cloud APIs, web services |
| **SSE**             | Legacy remote servers | ⚠️ Deprecated  | Older implementations    |

<Callout type="warning">
  **Transport Update (May 2025)**: The MCP specification deprecated HTTP+SSE in
  favor of **Streamable HTTP**. This simplifies security by eliminating the need
  for separate endpoints and enables better request-response correlation. New
  implementations should use Streamable HTTP for remote servers.
</Callout>

## 11.4 Building Custom MCP Servers

### Using FastMCP 2.x (Python)

FastMCP has become the standard Python framework for MCP servers. Version 2.x (actively maintained) provides a complete toolkit with decorator-based APIs.

```python
from fastmcp import FastMCP

# Create server with metadata
mcp = FastMCP(
    "My Custom Server",
    version="1.0.0",
    description="A custom MCP server for mortgage calculations"
)

# Define a tool with automatic schema inference
@mcp.tool()
def calculate_mortgage(
    principal: float,
    annual_rate: float,
    years: int
) -> dict:
    """Calculate monthly mortgage payment.

    Args:
        principal: Loan amount in dollars
        annual_rate: Annual interest rate (e.g., 0.065 for 6.5%)
        years: Loan term in years

    Returns:
        Monthly payment and total interest
    """
    monthly_rate = annual_rate / 12
    num_payments = years * 12

    monthly_payment = principal * (
        monthly_rate * (1 + monthly_rate)**num_payments
    ) / ((1 + monthly_rate)**num_payments - 1)

    total_paid = monthly_payment * num_payments
    total_interest = total_paid - principal

    return {
        "monthly_payment": round(monthly_payment, 2),
        "total_interest": round(total_interest, 2),
        "total_paid": round(total_paid, 2)
    }

# Define a resource for read-only data
@mcp.resource("rates://current")
def get_current_rates() -> str:
    """Get current mortgage rates."""
    return """
    Current Mortgage Rates (as of today):
    - 30-year fixed: 6.5%
    - 15-year fixed: 5.8%
    - 5/1 ARM: 6.0%
    """

# Define a prompt template
@mcp.prompt()
def mortgage_advisor(loan_amount: float) -> str:
    """Generate a mortgage advice prompt."""
    return f"""You are a mortgage advisor. The client is considering
    a loan of ${loan_amount:,.2f}. Analyze their options considering:
    1. Different loan terms (15 vs 30 year)
    2. Current interest rates
    3. Monthly payment affordability
    4. Total interest paid over the loan lifetime
    """

# Run the server
if __name__ == "__main__":
    mcp.run()
```

### FastMCP Advanced Features

```python
from fastmcp import FastMCP, Context
from pydantic import BaseModel, Field
from typing import Literal
import asyncio

mcp = FastMCP("Advanced Server")

# Async tools for I/O-bound operations
@mcp.tool()
async def fetch_stock_price(symbol: str) -> dict:
    """Fetch current stock price (async for non-blocking I/O)."""
    # Simulated async API call
    await asyncio.sleep(0.1)
    return {"symbol": symbol, "price": 150.25, "currency": "USD"}

# Using Pydantic models for complex inputs
class TradeOrder(BaseModel):
    symbol: str = Field(description="Stock ticker symbol")
    quantity: int = Field(gt=0, description="Number of shares")
    order_type: Literal["market", "limit"] = Field(description="Order type")
    limit_price: float | None = Field(default=None, description="Limit price (required for limit orders)")

@mcp.tool()
def place_order(order: TradeOrder) -> dict:
    """Place a stock trade order with validated input."""
    if order.order_type == "limit" and order.limit_price is None:
        raise ValueError("Limit price required for limit orders")
    return {
        "order_id": "ORD-12345",
        "status": "submitted",
        "details": order.model_dump()
    }

# Using Context for logging and metadata
@mcp.tool()
async def analyze_portfolio(ctx: Context, user_id: str) -> dict:
    """Analyze user's portfolio with logging."""
    ctx.info(f"Analyzing portfolio for user: {user_id}")

    # Your analysis logic here
    await asyncio.sleep(0.5)

    ctx.info("Analysis complete")
    return {"user_id": user_id, "total_value": 50000, "risk_score": 0.65}
```

### Using TypeScript SDK

```typescript
import { Server } from "@modelcontextprotocol/sdk/server/index.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";

const server = new Server(
  { name: "my-server", version: "1.0.0" },
  { capabilities: { tools: {}, resources: {} } },
);

// Define tools
server.setRequestHandler("tools/list", async () => ({
  tools: [
    {
      name: "get_weather",
      description: "Get current weather for a city",
      inputSchema: {
        type: "object",
        properties: {
          city: { type: "string", description: "City name" },
          units: {
            type: "string",
            enum: ["celsius", "fahrenheit"],
            default: "celsius",
          },
        },
        required: ["city"],
      },
    },
  ],
}));

// Handle tool calls
server.setRequestHandler("tools/call", async (request) => {
  if (request.params.name === "get_weather") {
    const { city, units = "celsius" } = request.params.arguments;
    // Fetch weather from API...
    const temp = units === "celsius" ? "22°C" : "72°F";
    return {
      content: [
        {
          type: "text",
          text: `Weather in ${city}: ${temp}, sunny`,
        },
      ],
    };
  }
  throw new Error(`Unknown tool: ${request.params.name}`);
});

// Start server
const transport = new StdioServerTransport();
await server.connect(transport);
```

## 11.5 Remote MCP Servers

Local MCP servers are great for development, but production deployments often require **remote servers** that can be shared across teams and scale with demand.

### Why Remote MCP Servers?

| Local Servers              | Remote Servers              |
| :------------------------- | :-------------------------- |
| Run on user's machine      | Run in cloud infrastructure |
| No authentication needed   | OAuth 2.1 authentication    |
| Single user access         | Team/organization access    |
| Limited by local resources | Scalable infrastructure     |
| Development/personal use   | Production deployments      |

### Deploying to Cloudflare Workers

Cloudflare was the first platform to support remote MCP servers with built-in OAuth:

```typescript
// src/index.ts - Cloudflare Worker MCP Server
import { McpServer } from "@cloudflare/mcp-server-cloudflare";

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const server = new McpServer({
      name: "cloud-tools",
      version: "1.0.0",
    });

    // Define tools
    server.tool("get_kv_value", {
      description: "Get a value from Cloudflare KV",
      parameters: {
        key: { type: "string", description: "The key to retrieve" },
      },
      handler: async ({ key }) => {
        const value = await env.MY_KV.get(key);
        return { value: value || "Not found" };
      },
    });

    return server.handleRequest(request);
  },
};
```

### The MCP Threat Landscape

<Callout type="danger">
  **Security Alert**: Research by Knostic discovered thousands of
  internet-exposed MCP servers with zero authentication. Tool poisoning attacks
  can hijack LLM behavior through malicious tool descriptions.
</Callout>

| Threat                    | Description                                   | Mitigation                                    |
| :------------------------ | :-------------------------------------------- | :-------------------------------------------- |
| **Tool Poisoning**        | Malicious instructions in tool descriptions   | Validate tool sources, use trusted registries |
| **Prompt Injection**      | Hidden instructions in data accessed by tools | Input sanitization, output filtering          |
| **NeighborJacking**       | Servers bound to 0.0.0.0 exposed on networks  | Bind to localhost, use proper network config  |
| **Credential Exposure**   | API keys in server configs or logs            | Environment variables, secret managers        |
| **Excessive Permissions** | Tools with more access than needed            | Principle of least privilege                  |

### OAuth 2.1 Authentication (MCP 2025-06-18)

The latest MCP spec requires OAuth 2.1 for remote servers:

```python
from fastmcp import FastMCP
from fastmcp.auth import OAuth2Config

mcp = FastMCP("Secure Server")

# Configure OAuth 2.1
oauth_config = OAuth2Config(
    issuer="https://auth.example.com",
    client_id="mcp-server",
    scopes=["tools:read", "tools:execute"],
    # Resource indicators (RFC 8707) prevent token misuse
    resource_indicator="https://mcp.example.com"
)

mcp.configure_auth(oauth_config)

@mcp.tool(scopes=["tools:execute"])  # Require specific scope
def sensitive_operation(data: str) -> dict:
    """Perform a sensitive operation requiring authentication."""
    # Only accessible with valid OAuth token and correct scope
    return {"status": "completed"}
```

### Input Validation with Pydantic

```python
from pydantic import BaseModel, field_validator, Field
from fastmcp import FastMCP

mcp = FastMCP("Validated Server")

    table: str = Field(description="Table to query")
    limit: int = Field(default=100, le=1000, description="Max rows")

    @field_validator("table")
    @classmethod
    def validate_table(cls, v):
        allowed_tables = ["users", "products", "orders"]
        if v not in allowed_tables:
            raise ValueError(f"Table must be one of: {allowed_tables}")
        return v

@mcp.tool()
def query_database(input: QueryInput) -> list:
    """Query the database with validated input."""
    # Safe to use - input is validated
    return [{"id": 1, "name": "Example"}]
```

### Implementing Elicitation

```python
from fastmcp import FastMCP
from fastmcp.elicitation import ElicitationRequest, ElicitationResponse

mcp = FastMCP("Booking Server")

@mcp.tool(supports_elicitation=True)
async def book_flight(
    destination: str,
    date: str | None = None,
    time_preference: str | None = None
) -> dict | ElicitationRequest:
    """Book a flight with multi-turn clarification."""

    # Request missing information
    if date is None:
        return ElicitationRequest(
            message="What date would you like to travel?",
            schema={
                "type": "object",
                "properties": {
                    "date": {"type": "string", "description": "Travel date"}
                },
                "required": ["date"]
            }
        )

    if time_preference is None:
        return ElicitationRequest(
            message="Do you prefer a morning or evening flight?",
            schema={
                "type": "object",
                "properties": {
                    "time_preference": {
                        "type": "string",
                        "enum": ["morning", "evening"]
                    }
                },
                "required": ["time_preference"]
            }
        )

    # All information collected, proceed with booking
    return {
        "flight": "AA123",
        "destination": destination,
        "date": date,
        "time": time_preference,
        "status": "confirmed"
    }
```

### Use Cases for Elicitation

- **Clarifying ambiguous requests**: "Which John Smith?"
- **Collecting required information**: Multi-step forms
- **Confirming high-risk actions**: "Are you sure you want to delete?"
- **Progressive disclosure**: Gathering details as needed

## 11.9 Debugging and Observability

Debugging MCP servers can be challenging because interactions happen between clients and servers with LLMs in the middle. Here are the essential tools and techniques.

### MCP Inspector

The **MCP Inspector** is the primary debugging tool for MCP servers:

```bash
# Install and run MCP Inspector
npx @modelcontextprotocol/inspector python my_server.py

# Or for npm-based servers
npx @modelcontextprotocol/inspector npx -y @modelcontextprotocol/server-filesystem ./data
```

The Inspector provides:

- **Tool discovery**: See all tools, resources, and prompts
- **Schema inspection**: View input/output schemas
- **Live testing**: Call tools with custom arguments
- **Message logging**: See raw JSON-RPC messages

### Logging with FastMCP

```python
from fastmcp import FastMCP, Context
import logging

# Configure logging
logging.basicConfig(level=logging.DEBUG)

mcp = FastMCP("Debug Server")

@mcp.tool()
async def debug_tool(ctx: Context, input: str) -> dict:
    """A tool with comprehensive logging."""
    ctx.debug(f"Received input: {input}")
    ctx.info("Processing request...")

    try:
        result = process(input)
        ctx.info(f"Success: {result}")
        return {"result": result}
    except Exception as e:
        ctx.error(f"Failed: {e}")
        raise
```

### OpenTelemetry Integration

For production observability, integrate with OpenTelemetry:

```python
from fastmcp import FastMCP
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter

# Configure OpenTelemetry
provider = TracerProvider()
processor = BatchSpanProcessor(OTLPSpanExporter())
provider.add_span_processor(processor)
trace.set_tracer_provider(provider)

tracer = trace.get_tracer("mcp-server")

mcp = FastMCP("Observable Server")

@mcp.tool()
async def traced_tool(query: str) -> dict:
    """A tool with distributed tracing."""
    with tracer.start_as_current_span("traced_tool") as span:
        span.set_attribute("query", query)

        # Your tool logic
        result = await execute_query(query)

        span.set_attribute("result_count", len(result))
        return {"data": result}
```

### Common Debugging Scenarios

| Problem            | Symptom                    | Solution                                         |
| :----------------- | :------------------------- | :----------------------------------------------- |
| Tool not appearing | Client doesn't see tool    | Check `tools/list` response in Inspector         |
| Schema mismatch    | "Invalid arguments" error  | Verify type annotations match expected schema    |

| Auth errors        | 401/403 responses          | Verify OAuth config, check token scopes          |
| Slow responses     | Timeouts                   | Add async, check for blocking I/O                |

## 11.11 The Protocol Landscape: MCP, A2A, and Beyond

MCP isn't the only protocol shaping agent interoperability. Understanding the broader ecosystem helps you make informed architectural decisions.

### Agent-to-Agent (A2A) Protocol

In April 2025, Google introduced the **Agent2Agent (A2A) Protocol** with backing from over 50 tech companies including Salesforce, Atlassian, and PayPal. A2A is designed for agent-to-agent communication rather than agent-to-tool communication.

<Callout type="info">

  standardizes how agents access tools (vertical integration); A2A standardizes
  how agents collaborate with each other (horizontal integration). Production
  systems often use both protocols together to create powerful, interoperable
  agent ecosystems.
</Callout>

In December 2025, the Linux Foundation announced the **Agentic AI Foundation**, bringing together:

- **goose** (Block): Open-source AI agent framework

```python
from fastmcp import FastMCP, Context
import ast
import subprocess
from pathlib import Path

mcp = FastMCP("Code Analysis Server")

@mcp.tool()
async def analyze_python_file(ctx: Context, file_path: str) -> dict:
    """Analyze a Python file for complexity and issues.

    Args:
        file_path: Path to the Python file

    Returns:
        Analysis results including complexity metrics
    """
    ctx.info(f"Analyzing: {file_path}")

    path = Path(file_path)
    if not path.exists():
        raise ValueError(f"File not found: {file_path}")

    content = path.read_text()
    tree = ast.parse(content)

    functions = [node.name for node in ast.walk(tree)
                 if isinstance(node, ast.FunctionDef)]
    classes = [node.name for node in ast.walk(tree)
               if isinstance(node, ast.ClassDef)]

    return {
        "file": file_path,
        "lines": len(content.splitlines()),
        "functions": functions,
        "classes": classes,
        "function_count": len(functions),
        "class_count": len(classes)
    }

@mcp.tool()
async def run_tests(test_path: str, verbose: bool = False) -> dict:
    """Run pytest on specified path.

    Args:
        test_path: Path to test file or directory
        verbose: Enable verbose output

    Returns:
        Test results summary
    """
    cmd = ["pytest", test_path, "--tb=short"]
    if verbose:
        cmd.append("-v")

    result = subprocess.run(cmd, capture_output=True, text=True)

    return {
        "exit_code": result.returncode,
        "passed": result.returncode == 0,
        "stdout": result.stdout,
        "stderr": result.stderr
    }

@mcp.resource("project://structure")
def get_project_structure() -> str:
    """Get the project directory structure."""
    # Return tree-like structure of current directory
    return subprocess.run(
        ["find", ".", "-type", "f", "-name", "*.py"],
        capture_output=True, text=True
    ).stdout

if __name__ == "__main__":
    mcp.run()
```

### Option 2: API Integration Server

```python
from fastmcp import FastMCP
from pydantic import BaseModel, Field
import httpx
import os

mcp = FastMCP("GitHub Integration Server")

GITHUB_TOKEN = os.environ.get("GITHUB_TOKEN")

class IssueFilter(BaseModel):
    repo: str = Field(description="Repository in format 'owner/repo'")
    state: str = Field(default="open", description="Issue state")
    labels: list[str] = Field(default=[], description="Filter by labels")

@mcp.tool()
async def fetch_github_issues(filter: IssueFilter) -> list:
    """Fetch issues from a GitHub repository.

    Returns:
        List of issues with title, number, and labels
    """
    async with httpx.AsyncClient() as client:
        headers = {"Authorization": f"token {GITHUB_TOKEN}"} if GITHUB_TOKEN else {}

        params = {"state": filter.state}
        if filter.labels:
            params["labels"] = ",".join(filter.labels)

        response = await client.get(
            f"https://api.github.com/repos/{filter.repo}/issues",
            headers=headers,
            params=params
        )
        response.raise_for_status()

        issues = response.json()
        return [
            {
                "number": issue["number"],
                "title": issue["title"],
                "state": issue["state"],
                "labels": [l["name"] for l in issue["labels"]]
            }
            for issue in issues[:10]  # Limit to 10
        ]

@mcp.tool()
async def create_github_issue(
    repo: str,
    title: str,
    body: str,
    labels: list[str] = []
) -> dict:
    """Create a new GitHub issue.

    Args:
        repo: Repository in format 'owner/repo'
        title: Issue title
        body: Issue description
        labels: Labels to apply

    Returns:
        Created issue details
    """
    if not GITHUB_TOKEN:
        raise ValueError("GITHUB_TOKEN environment variable required")

    async with httpx.AsyncClient() as client:
        response = await client.post(
            f"https://api.github.com/repos/{repo}/issues",
            headers={"Authorization": f"token {GITHUB_TOKEN}"},
            json={"title": title, "body": body, "labels": labels}
        )
        response.raise_for_status()

        issue = response.json()
        return {
            "number": issue["number"],
            "url": issue["html_url"],
            "created": True
        }

if __name__ == "__main__":
    mcp.run()
```
