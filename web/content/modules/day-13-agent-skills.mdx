# Agent Skills Standard

## The Context Window Problem

Every agent faces a fundamental constraint: **context window bloat**. Loading all possible instructions upfront wastes tokens and dilutes focus.

```python
# ❌ Traditional approach: Load everything
system_prompt = """
You are an AI assistant with these capabilities:

1. Code Review (500 words of instructions...)
2. Database Design (400 words of instructions...)
3. API Documentation (350 words of instructions...)
4. Security Audit (450 words of instructions...)
5. Performance Optimization (400 words of instructions...)
... (20 more capabilities)

Total: 15,000 tokens consumed before user even asks a question!
"""
```

**Agent Skills** solves this with **progressive disclosure**: load only what you need, when you need it.

## What is a Skill?

A **skill** is a self-contained directory containing:

1. **SKILL.md**: Instructions and metadata
2. **scripts/**: Executable tools
3. **references/**: Documentation and examples
4. **assets/**: Templates and resources

<ConceptVisual
  src="/images/skill_structure_diagram.png"
  alt="Skill Directory Structure"
  caption="Standard skill anatomy: Core instructions, scripts, docs, and assets."
/>

```
my-skill/
├── SKILL.md              # Core instructions
├── scripts/
│   ├── analyze.py        # Executable script
│   └── validate.sh       # Helper script
├── references/
│   ├── examples.md       # Usage examples
│   └── api-docs.md       # API reference
└── assets/
    └── template.json     # Code template
```

**scripts/analyze.py**:

```python
#!/usr/bin/env python3
"""Static analysis tool for code review"""

import sys
import re
from pathlib import Path

def check_sql_injection(code: str) -> list:
    """Detect potential SQL injection"""
    issues = []

    # Pattern: f"SELECT ... {variable}"
    pattern = r'f["\'].*SELECT.*\{.*\}.*["\']'
    matches = re.finditer(pattern, code, re.IGNORECASE)

    for match in matches:
        issues.append({
            "type": "SQL Injection",
            "severity": "HIGH",
            "line": code[:match.start()].count('\n') + 1,
            "code": match.group()
        })

    return issues

def check_hardcoded_secrets(code: str) -> list:
    """Detect hardcoded secrets"""
    issues = []

    patterns = [
        (r'password\s*=\s*["\'][^"\']+["\']', "Hardcoded Password"),
        (r'api_key\s*=\s*["\'][^"\']+["\']', "Hardcoded API Key"),
        (r'secret\s*=\s*["\'][^"\']+["\']', "Hardcoded Secret"),
    ]

    for pattern, issue_type in patterns:
        matches = re.finditer(pattern, code, re.IGNORECASE)
        for match in matches:
            issues.append({
                "type": issue_type,
                "severity": "HIGH",
                "line": code[:match.start()].count('\n') + 1,
                "code": match.group()
            })

    return issues

def analyze_file(filepath: str):
    """Analyze a single file"""
    with open(filepath, 'r') as f:
        code = f.read()

    issues = []
    issues.extend(check_sql_injection(code))
    issues.extend(check_hardcoded_secrets(code))

    return issues

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: analyze.py <file>")
        sys.exit(1)

    issues = analyze_file(sys.argv[1])

    if issues:
        print(f"Found {len(issues)} issues:")
        for issue in issues:
            print(f"  [{issue['severity']}] {issue['type']} at line {issue['line']}")
    else:
        print("No issues found!")
```

## Using Skills in Agents

### Discovery Phase (Lightweight)

```python
from typing import List, Dict

class SkillRegistry:
    """Manages available skills"""

    def __init__(self, skills_dir: str = "~/.gemini/skills"):
        self.skills_dir = Path(skills_dir).expanduser()
        self.skills = self._discover_skills()

    def _discover_skills(self) -> List[Dict]:
        """Discover all available skills"""
        skills = []

        for skill_path in self.skills_dir.iterdir():
            if not skill_path.is_dir():
                continue

            skill_md = skill_path / "SKILL.md"
            if not skill_md.exists():
                continue

            # Parse frontmatter
            with open(skill_md, 'r') as f:
                content = f.read()

            # Extract metadata (simplified)
            name = self._extract_metadata(content, "name")
            description = self._extract_metadata(content, "description")

            skills.append({
                "name": name,
                "description": description,
                "path": skill_path,
                "tokens": 35  # Approximate token cost for metadata
            })

        return skills

    def _extract_metadata(self, content: str, key: str) -> str:
        """Extract metadata from frontmatter"""
        pattern = f'{key}:\\s*"([^"]+)"'
        match = re.search(pattern, content)
        return match.group(1) if match else ""

    def get_skill_catalog(self) -> str:
        """Get lightweight catalog for agent context"""
        catalog = "## Available Skills\n\n"

        for skill in self.skills:
            catalog += f"- **{skill['name']}**: {skill['description']}\n"

        return catalog  # Only ~200 tokens for 20 skills!

# Usage
registry = SkillRegistry()
catalog = registry.get_skill_catalog()

print(catalog)
# Output:
# ## Available Skills
#
# - **code-reviewer**: Performs comprehensive code reviews
# - **api-designer**: Designs RESTful APIs following best practices
# - **db-optimizer**: Optimizes database queries and schemas
# ...
```

### Activation Phase (Full Context)

```python
class SkillActivator:
    """Activates skills on-demand"""

    def __init__(self, registry: SkillRegistry):
        self.registry = registry

    def activate(self, skill_name: str) -> Dict:
        """Load full skill context"""

        # Find skill
        skill = next(
            (s for s in self.registry.skills if s["name"] == skill_name),
            None
        )

        if not skill:
            raise ValueError(f"Skill '{skill_name}' not found")

        skill_path = skill["path"]

        # Load SKILL.md
        with open(skill_path / "SKILL.md", 'r') as f:
            instructions = f.read()

        # List available resources
        resources = {
            "scripts": list((skill_path / "scripts").glob("*")) if (skill_path / "scripts").exists() else [],
            "references": list((skill_path / "references").glob("*")) if (skill_path / "references").exists() else [],
            "assets": list((skill_path / "assets").glob("*")) if (skill_path / "assets").exists() else [],
        }

        # Build file tree
        file_tree = self._build_file_tree(skill_path)

        return {
            "name": skill_name,
            "instructions": instructions,
            "resources": resources,
            "file_tree": file_tree,
            "path": skill_path
        }

    def _build_file_tree(self, path: Path, prefix: str = "") -> str:
        """Build visual file tree"""
        tree = ""
        items = sorted(path.iterdir(), key=lambda x: (not x.is_dir(), x.name))

        for i, item in enumerate(items):
            is_last = i == len(items) - 1
            connector = "└── " if is_last else "├── "
            tree += f"{prefix}{connector}{item.name}\n"

            if item.is_dir() and item.name in ["scripts", "references", "assets"]:
                extension = "    " if is_last else "│   "
                tree += self._build_file_tree(item, prefix + extension)

        return tree

# Usage
activator = SkillActivator(registry)

# Agent decides to use code-reviewer skill
activated = activator.activate("code-reviewer")

print(f"Activated: {activated['name']}")
print(f"Instructions: {len(activated['instructions'])} chars")
print(f"File tree:\n{activated['file_tree']}")
```

## Building an Agent with Skills

````python
from langchain_openai import ChatOpenAI
from langchain.tools import tool

class SkillAgent:
    """Agent that uses skills on-demand"""

    def __init__(self):
        self.llm = ChatOpenAI(model="gpt-4o", temperature=0)
        self.registry = SkillRegistry()
        self.activator = SkillActivator(self.registry)
        self.active_skills = {}

    @tool
    def activate_skill(self, skill_name: str) -> str:
        """Activate a skill to access its full instructions"""
        activated = self.activator.activate(skill_name)
        self.active_skills[skill_name] = activated
        return f"Activated {skill_name}. Instructions loaded."

    @tool
    def execute_skill_script(self, skill_name: str, script_name: str, args: str = "") -> str:
        """Execute a script from an activated skill"""
        if skill_name not in self.active_skills:
            return f"Error: Skill '{skill_name}' not activated"

        skill = self.active_skills[skill_name]
        script_path = skill["path"] / "scripts" / script_name

        if not script_path.exists():
            return f"Error: Script '{script_name}' not found"

        # Execute script
        import subprocess
        result = subprocess.run(
            [str(script_path)] + args.split(),
            capture_output=True,
            text=True
        )

        return result.stdout if result.returncode == 0 else result.stderr

    def run(self, task: str) -> str:
        """Execute task using skills as needed"""

        # Build initial context with skill catalog
        catalog = self.registry.get_skill_catalog()

        system_prompt = f"""
You are an AI assistant with access to specialized skills.

{catalog}

To use a skill:
1. Call activate_skill(skill_name) to load full instructions
2. Follow the skill's instructions
3. Use execute_skill_script() if needed

Current task: {task}
"""

        # Create agent with tools
        from langchain.agents import create_openai_functions_agent, AgentExecutor
        from langchain.prompts import ChatPromptTemplate, MessagesPlaceholder

        prompt = ChatPromptTemplate.from_messages([
            ("system", system_prompt),
            MessagesPlaceholder(variable_name="chat_history", optional=True),
            ("human", "{input}"),
            MessagesPlaceholder(variable_name="agent_scratchpad"),
        ])

        tools = [self.activate_skill, self.execute_skill_script]
        agent = create_openai_functions_agent(self.llm, tools, prompt)
        executor = AgentExecutor(agent=agent, tools=tools, verbose=True)

        result = executor.invoke({"input": task})
        return result["output"]

# Usage
agent = SkillAgent()

result = agent.run("""
Review this Python code for security issues:

```python
def login(username, password):
    query = f"SELECT * FROM users WHERE username='{username}' AND password='{password}'"
    return db.execute(query)
````

""")

print(result)


````
## Skill Precedence

Skills can be installed at three levels:

```python
class SkillRegistry:
    def __init__(self):
        self.skill_paths = [
            Path("~/.gemini/skills").expanduser(),      # User skills
            Path(".gemini/skills"),                      # Workspace skills
            Path("/usr/share/gemini/skills"),           # Extension skills
        ]

    def _discover_skills(self) -> List[Dict]:
        """Discover skills with precedence: Workspace > User > Extension"""
        skills_by_name = {}

        # Reverse order so workspace overrides user overrides extension
        for skills_dir in reversed(self.skill_paths):
            if not skills_dir.exists():
                continue

            for skill_path in skills_dir.iterdir():
                if not skill_path.is_dir():
                    continue

                skill_md = skill_path / "SKILL.md"
                if not skill_md.exists():
                    continue

                name = self._extract_name(skill_md)
                skills_by_name[name] = {
                    "name": name,
                    "path": skill_path,
                    "source": skills_dir.name
                }

        return list(skills_by_name.values())
````
