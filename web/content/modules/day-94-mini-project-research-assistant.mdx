A personal research assistant acts like having a dedicated analyst on staff—monitoring your interests 24/7, filtering noise, and delivering only what matters in a digestible format.

## Module Summary

In this project, we build a comprehensive personal research assistant:

1. **Topic Configuration**: We define research topics with keywords, RSS feeds, and target websites, allowing flexible monitoring of any subject area.

2. **Content Acquisition**: We implement RSS parsing and respectful web scraping with rate limiting, gathering fresh content from configured sources.

3. **Credibility Scoring**: We score each source based on domain authority, publication patterns, and cross-reference verification to filter out unreliable information.

4. **RAG-Based Synthesis**: We use ChromaDB for vector storage and LangChain for retrieval-augmented generation, synthesizing multiple documents into coherent summaries.

5. **Briefing Generation**: We create professional, executive-style briefings with proper citations, key takeaways, and actionable insights.

6. **Automated Delivery**: We schedule daily/weekly briefings and deliver them via email (SMTP) or Slack webhooks.

```python
        weights = {
            "domain_authority": 0.35,
            "publication_frequency": 0.15,
            "cross_reference_score": 0.25,
            "content_quality": 0.25
        }
        return (
            self.domain_authority * weights["domain_authority"] +
            self.publication_frequency * weights["publication_frequency"] +
            self.cross_reference_score * weights["cross_reference_score"] +
            self.content_quality * weights["content_quality"]
```

```python
# acquisition/rss_parser.py
import feedparser
import httpx
from typing import List, Optional
from datetime import datetime
from urllib.parse import urlparse
import logging

from models import SourceDocument, SourceType
from config import get_settings

logger = logging.getLogger(__name__)

class RSSParser:
    """Parse RSS feeds to extract articles."""

    def __init__(self):
        self.settings = get_settings()

    async def parse_feed(self, feed_url: str, topic_id: str) -> List[SourceDocument]:
        """
        Parse an RSS feed and return source documents.

        Args:
            feed_url: URL of the RSS feed
            topic_id: ID of the research topic

        Returns:
            List of SourceDocument objects
        """
        documents = []

        try:
            async with httpx.AsyncClient(timeout=self.settings.scrape_timeout_seconds) as client:
                response = await client.get(
                    feed_url,
                    headers={"User-Agent": self.settings.user_agent}
                )
                response.raise_for_status()

            feed = feedparser.parse(response.text)

            for entry in feed.entries:
                doc = self._entry_to_document(entry, feed_url, topic_id)
                if doc:
                    documents.append(doc)

            logger.info(f"Parsed {len(documents)} entries from {feed_url}")

        except httpx.HTTPError as e:
            logger.error(f"HTTP error fetching RSS feed {feed_url}: {e}")
        except Exception as e:
            logger.error(f"Error parsing RSS feed {feed_url}: {e}")

        return documents

    def _entry_to_document(
        self,
        entry: feedparser.FeedParserDict,
        feed_url: str,
        topic_id: str
    ) -> Optional[SourceDocument]:
        """Convert a feed entry to a SourceDocument."""
        try:
            # Extract URL
            url = entry.get("link", "")
            if not url:
                return None

            # Extract title
            title = entry.get("title", "Untitled")

            # Extract content (prefer full content, fall back to summary)
            content = ""
            if "content" in entry and entry.content:
                content = entry.content[0].get("value", "")
            elif "summary" in entry:
                content = entry.summary

            # Clean HTML from content
            content = self._clean_html(content)

            if not content:
                return None

            # Extract publication date
            published_at = None
            if "published_parsed" in entry and entry.published_parsed:
                published_at = datetime(*entry.published_parsed[:6])
            elif "updated_parsed" in entry and entry.updated_parsed:
                published_at = datetime(*entry.updated_parsed[:6])

            # Extract domain
            domain = urlparse(url).netloc

            return SourceDocument(
                topic_id=topic_id,
                url=url,
                title=title,
                content=content[:self.settings.max_content_length],
                published_at=published_at,
                source_type=SourceType.RSS,
                domain=domain
            )

        except Exception as e:
            logger.warning(f"Error converting feed entry: {e}")
            return None

    def _clean_html(self, html_content: str) -> str:
        """Remove HTML tags and clean text."""
        from bs4 import BeautifulSoup

        if not html_content:
            return ""

        soup = BeautifulSoup(html_content, "html.parser")

        # Remove script and style elements
        for element in soup(["script", "style", "nav", "footer", "header"]):
            element.decompose()

        # Get text and clean whitespace
        text = soup.get_text(separator=" ")
        lines = (line.strip() for line in text.splitlines())
        chunks = (phrase.strip() for line in lines for phrase in line.split("  "))
        text = " ".join(chunk for chunk in chunks if chunk)

        return text
```

```python
import httpx
from bs4 import BeautifulSoup
from typing import Optional, List

import asyncio
import logging

logger = logging.getLogger(__name__)

class RateLimiter:
    """Simple rate limiter for respectful scraping."""

    def __init__(self, delay_seconds: float = 2.0):
        self.delay_seconds = delay_seconds
        self._last_request: dict[str, float] = {}
        self._lock = asyncio.Lock()

    async def wait(self, domain: str) -> None:
        """Wait if necessary to respect rate limits for a domain."""
        async with self._lock:
            now = asyncio.get_event_loop().time()
            last = self._last_request.get(domain, 0)
            wait_time = self.delay_seconds - (now - last)

            if wait_time > 0:
                await asyncio.sleep(wait_time)

            self._last_request[domain] = asyncio.get_event_loop().time()

class WebScraper:
    """Respectful web scraper with robots.txt compliance."""

    def __init__(self):
        self.settings = get_settings()
        self.rate_limiter = RateLimiter(self.settings.scrape_delay_seconds)
        self._robots_cache: dict[str, RobotFileParser] = {}

    async def scrape_url(self, url: str, topic_id: str) -> Optional[SourceDocument]:
        """
        Scrape a single URL and return a source document.

        Args:
            url: URL to scrape
            topic_id: ID of the research topic

        Returns:
            SourceDocument if successful, None otherwise
        """
        domain = urlparse(url).netloc

        # Check robots.txt
        if not await self._can_fetch(url):
            logger.info(f"Robots.txt disallows scraping: {url}")
            return None

        # Respect rate limits
        await self.rate_limiter.wait(domain)

        try:
            async with httpx.AsyncClient(
                timeout=self.settings.scrape_timeout_seconds,
                follow_redirects=True
            ) as client:
                response = await client.get(
                    url,
                    headers={
                        "User-Agent": self.settings.user_agent,
                        "Accept": "text/html,application/xhtml+xml",
                        "Accept-Language": "en-US,en;q=0.9"
                    }
                )
                response.raise_for_status()

            # Parse HTML
            soup = BeautifulSoup(response.text, "html.parser")

            # Extract title
            title = self._extract_title(soup)

            # Extract main content
            content = self._extract_content(soup)

            if not content or len(content) < 100:
                logger.warning(f"Insufficient content from {url}")
                return None

            # Try to extract publication date
            published_at = self._extract_date(soup)

            return SourceDocument(
                topic_id=topic_id,
                url=url,
                title=title,
                content=content[:self.settings.max_content_length],
                published_at=published_at,
                source_type=SourceType.SCRAPED,
                domain=domain
            )

        except httpx.HTTPError as e:
            logger.error(f"HTTP error scraping {url}: {e}")
        except Exception as e:
            logger.error(f"Error scraping {url}: {e}")

        return None

    async def _can_fetch(self, url: str) -> bool:
        """Check if robots.txt allows fetching this URL."""
        parsed = urlparse(url)
        robots_url = f"{parsed.scheme}://{parsed.netloc}/robots.txt"

        if robots_url not in self._robots_cache:
            rp = RobotFileParser()
            try:
                async with httpx.AsyncClient(timeout=10) as client:
                    response = await client.get(robots_url)
                    if response.status_code == 200:
                        rp.parse(response.text.splitlines())
                    else:
                        # No robots.txt, assume allowed
                        return True
            except Exception:
                # Can't fetch robots.txt, assume allowed
                return True

            self._robots_cache[robots_url] = rp

        return self._robots_cache[robots_url].can_fetch(self.settings.user_agent, url)

    def _extract_title(self, soup: BeautifulSoup) -> str:
        """Extract page title."""
        # Try og:title first
        og_title = soup.find("meta", property="og:title")
        if og_title and og_title.get("content"):
            return og_title["content"]

        # Try <title> tag
        if soup.title and soup.title.string:
            return soup.title.string.strip()

        # Try h1
        h1 = soup.find("h1")
        if h1:
            return h1.get_text(strip=True)

        return "Untitled"

    def _extract_content(self, soup: BeautifulSoup) -> str:
        """Extract main content from page."""
        # Remove unwanted elements
        for element in soup(["script", "style", "nav", "footer", "header", "aside", "form"]):
            element.decompose()

        # Try to find main content area
        main_content = None

        # Look for article tag
        article = soup.find("article")
        if article:
            main_content = article

        # Look for main tag
        if not main_content:
            main = soup.find("main")
            if main:
                main_content = main

        # Look for common content class names
        if not main_content:
            for class_name in ["content", "post-content", "article-content", "entry-content"]:
                content_div = soup.find(class_=class_name)
                if content_div:
                    main_content = content_div
                    break

        # Fall back to body
        if not main_content:
            main_content = soup.body if soup.body else soup

        # Extract text
        text = main_content.get_text(separator=" ")

        # Clean whitespace
        lines = (line.strip() for line in text.splitlines())
        chunks = (phrase.strip() for line in lines for phrase in line.split("  "))
        text = " ".join(chunk for chunk in chunks if chunk)

        return text

    def _extract_date(self, soup: BeautifulSoup) -> Optional[datetime]:
        """Try to extract publication date from page."""
        from dateutil import parser as date_parser

        # Try meta tags
        date_metas = [
            ("meta", {"property": "article:published_time"}),
            ("meta", {"name": "publication_date"}),
            ("meta", {"name": "date"}),
            ("time", {"datetime": True}),
        ]

        for tag, attrs in date_metas:
            element = soup.find(tag, attrs)
            if element:
                date_str = element.get("content") or element.get("datetime")
                if date_str:
                    try:
                        return date_parser.parse(date_str)
                    except Exception:
                        continue

        return None
```

```python
# analysis/credibility_scorer.py
from typing import Dict, List, Optional
from urllib.parse import urlparse
import re
import logging

from models import SourceDocument, CredibilityScore

logger = logging.getLogger(__name__)

# Known high-authority domains (simplified - in production, use a proper database)
HIGH_AUTHORITY_DOMAINS = {
    # News organizations
    "reuters.com": 0.95,
    "apnews.com": 0.95,
    "bbc.com": 0.90,
    "bbc.co.uk": 0.90,
    "nytimes.com": 0.88,
    "washingtonpost.com": 0.85,
    "theguardian.com": 0.85,
    "economist.com": 0.90,
    # Academic/Research
    "nature.com": 0.95,
    "science.org": 0.95,
    "arxiv.org": 0.85,
    "pubmed.ncbi.nlm.nih.gov": 0.95,
    "scholar.google.com": 0.80,
    # Tech
    "techcrunch.com": 0.75,
    "wired.com": 0.78,
    "arstechnica.com": 0.80,
    "theverge.com": 0.72,
    # Government/Official
    "gov": 0.85,  # TLD match
    "edu": 0.80,  # TLD match
    "org": 0.65,  # TLD match (lower, varies widely)
}

# Known low-credibility patterns
LOW_CREDIBILITY_PATTERNS = [
    r"blog\..*\.com",  # Generic blog subdomains
    r"medium\.com/@",  # Medium personal blogs (not publications)
    r"wordpress\.com",
    r"blogspot\.com",
    r"tumblr\.com",
]

class CredibilityScorer:
    """Score source credibility based on multiple factors."""

    def __init__(self):
        self._domain_cache: Dict[str, float] = {}

    def score_document(
        self,
        document: SourceDocument,
        all_documents: Optional[List[SourceDocument]] = None
    ) -> CredibilityScore:
        """
        Calculate credibility score for a document.

        Args:
            document: The document to score
            all_documents: Other documents for cross-reference scoring

        Returns:
            CredibilityScore with component scores
        """
        domain_authority = self._score_domain_authority(document.domain)
        publication_frequency = self._score_publication_frequency(document)
        cross_reference = self._score_cross_reference(document, all_documents or [])
        content_quality = self._score_content_quality(document)

        return CredibilityScore(
            domain_authority=domain_authority,
            publication_frequency=publication_frequency,
            cross_reference_score=cross_reference,
            content_quality=content_quality
        )

    def _score_domain_authority(self, domain: str) -> float:
        """Score based on domain reputation."""
        if domain in self._domain_cache:
            return self._domain_cache[domain]

        score = 0.5  # Default neutral score

        # Check exact domain match
        if domain in HIGH_AUTHORITY_DOMAINS:
            score = HIGH_AUTHORITY_DOMAINS[domain]
        else:
            # Check TLD
            tld = domain.split(".")[-1]
            if tld in HIGH_AUTHORITY_DOMAINS:
                score = HIGH_AUTHORITY_DOMAINS[tld]

            # Check for low-credibility patterns
            for pattern in LOW_CREDIBILITY_PATTERNS:
                if re.search(pattern, domain):
                    score = min(score, 0.35)
                    break

            # Boost for HTTPS (implied by modern scraping)
            # Penalize very new or obscure domains (simplified heuristic)
            if len(domain.split(".")) > 3:
                score *= 0.9  # Subdomains slightly less trusted

        self._domain_cache[domain] = score
        return score

    def _score_publication_frequency(self, document: SourceDocument) -> float:
        """Score based on publication recency and patterns."""
        from datetime import datetime, timedelta

        if not document.published_at:
            return 0.5  # Neutral if no date

        age = datetime.utcnow() - document.published_at

        # Recent content scores higher
        if age < timedelta(hours=24):
            return 0.95
        elif age < timedelta(days=7):
            return 0.85
        elif age < timedelta(days=30):
            return 0.70
        elif age < timedelta(days=90):
            return 0.55
        else:
            return 0.40

    def _score_cross_reference(
        self,
        document: SourceDocument,
        all_documents: List[SourceDocument]
    ) -> float:
        """Score based on how many other sources cover similar content."""
        if not all_documents:
            return 0.5

        # Extract key terms from document
        doc_terms = set(self._extract_key_terms(document.content))

        if not doc_terms:
            return 0.5

        # Count documents with overlapping terms
        overlap_count = 0
        for other_doc in all_documents:
            if other_doc.id == document.id:
                continue

            other_terms = set(self._extract_key_terms(other_doc.content))
            overlap = len(doc_terms & other_terms) / len(doc_terms)

            if overlap > 0.3:  # 30% term overlap threshold
                overlap_count += 1

        # More cross-references = higher credibility
        if overlap_count >= 3:
            return 0.90
        elif overlap_count >= 2:
            return 0.75
        elif overlap_count >= 1:
            return 0.60
        else:
            return 0.45  # Unique content, slightly lower

    def _score_content_quality(self, document: SourceDocument) -> float:
        """Score based on content quality indicators."""
        content = document.content
        score = 0.5

        # Length check (too short or too long is suspicious)
        length = len(content)
        if 500 <= length <= 10000:
            score += 0.15
        elif length < 200:
            score -= 0.20

        # Check for quality indicators
        quality_indicators = [
            r"\b(according to|research shows|study finds|data indicates)\b",
            r"\b(professor|researcher|expert|analyst)\b",
            r"\b(university|institute|organization)\b",
            r"\d{4}",  # Years (citations)
            r"https?://",  # Links (references)
        ]

        for pattern in quality_indicators:
            if re.search(pattern, content, re.IGNORECASE):
                score += 0.05

        # Check for low-quality indicators
        low_quality_indicators = [
            r"(click here|subscribe now|buy now|limited time)",
            r"(!!+|\?\?+)",  # Excessive punctuation
            r"(BREAKING|SHOCKING|YOU WON'T BELIEVE)",
        ]

        for pattern in low_quality_indicators:
            if re.search(pattern, content, re.IGNORECASE):
                score -= 0.15

        return max(0.0, min(1.0, score))

    def _extract_key_terms(self, content: str, max_terms: int = 50) -> List[str]:
        """Extract key terms from content for comparison."""
        # Simple term extraction (in production, use NLP)
        words = re.findall(r'\b[a-zA-Z]{4,}\b', content.lower())

        # Filter common words
        stopwords = {
            "that", "this", "with", "from", "have", "been", "were", "they",
            "their", "which", "would", "could", "should", "about", "into",
            "more", "some", "than", "when", "what", "there", "other"
        }

        terms = [w for w in words if w not in stopwords]

        # Return most frequent terms
        from collections import Counter
        term_counts = Counter(terms)
        return [term for term, _ in term_counts.most_common(max_terms)]
```

```python
import chromadb
from chromadb.config import Settings as ChromaSettings
from typing import List, Dict, Optional, Any
from langchain_openai import OpenAIEmbeddings
import logging

from models import SourceDocument
from config import get_settings

logger = logging.getLogger(__name__)

class VectorStore:
    """ChromaDB-based vector store for document retrieval."""

    def __init__(self):
        self.settings = get_settings()

        # Initialize ChromaDB client with persistence
        self.client = chromadb.PersistentClient(
            path=self.settings.chroma_persist_dir,
            settings=ChromaSettings(anonymized_telemetry=False)
        )

        # Initialize embeddings
        self.embeddings = OpenAIEmbeddings(
            model=self.settings.embedding_model,
            openai_api_key=self.settings.openai_api_key
        )

    def _get_collection_name(self, topic_id: str) -> str:
        """Get collection name for a topic."""
        return f"{self.settings.chroma_collection_prefix}{topic_id}"

    def _get_or_create_collection(self, topic_id: str) -> chromadb.Collection:
        """Get or create a collection for a topic."""
        return self.client.get_or_create_collection(
            name=self._get_collection_name(topic_id),
            metadata={"hnsw:space": "cosine"}
        )

    async def add_documents(
        self,
        documents: List[SourceDocument]
    ) -> List[str]:
        """
        Add documents to the vector store.

        Args:
            documents: List of SourceDocument objects

        Returns:
            List of embedding IDs
        """
        if not documents:
            return []

        # Group by topic
        by_topic: Dict[str, List[SourceDocument]] = {}
        for doc in documents:
            if doc.topic_id not in by_topic:
                by_topic[doc.topic_id] = []
            by_topic[doc.topic_id].append(doc)

        embedding_ids = []

        for topic_id, topic_docs in by_topic.items():
            collection = self._get_or_create_collection(topic_id)

            # Prepare data for ChromaDB
            ids = [doc.id for doc in topic_docs]
            texts = [f"{doc.title}\n\n{doc.content}" for doc in topic_docs]
            metadatas = [
                {
                    "url": doc.url,
                    "title": doc.title,
                    "domain": doc.domain,
                    "source_type": doc.source_type.value,
                    "credibility_score": doc.credibility.overall_score,
                    "published_at": doc.published_at.isoformat() if doc.published_at else "",
                    "fetched_at": doc.fetched_at.isoformat()
                }
                for doc in topic_docs
            ]

            # Generate embeddings
            try:
                embeddings = self.embeddings.embed_documents(texts)

                # Add to collection
                collection.add(
                    ids=ids,
                    embeddings=embeddings,
                    documents=texts,
                    metadatas=metadatas
                )

                embedding_ids.extend(ids)
                logger.info(f"Added {len(ids)} documents to topic {topic_id}")

            except Exception as e:
                logger.error(f"Error adding documents to vector store: {e}")
                raise

        return embedding_ids

    async def search(
        self,
        topic_id: str,
        query: str,
        top_k: int = 10,
        min_credibility: float = 0.4
    ) -> List[Dict[str, Any]]:
        """
        Search for relevant documents.

        Args:
            topic_id: Topic to search within
            query: Search query
            top_k: Number of results to return
            min_credibility: Minimum credibility score filter

        Returns:
            List of document results with metadata
        """
        collection = self._get_or_create_collection(topic_id)

        # Generate query embedding
        query_embedding = self.embeddings.embed_query(query)

        # Search with credibility filter
        results = collection.query(
            query_embeddings=[query_embedding],
            n_results=top_k * 2,  # Fetch extra for filtering
            where={"credibility_score": {"$gte": min_credibility}},
            include=["documents", "metadatas", "distances"]
        )

        # Format results
        formatted = []
        if results["ids"] and results["ids"][0]:
            for i, doc_id in enumerate(results["ids"][0][:top_k]):
                formatted.append({
                    "id": doc_id,
                    "content": results["documents"][0][i] if results["documents"] else "",
                    "metadata": results["metadatas"][0][i] if results["metadatas"] else {},
                    "distance": results["distances"][0][i] if results["distances"] else 0
                })

        return formatted

    async def get_recent_documents(
        self,
        topic_id: str,
        hours: int = 24,
        limit: int = 100
    ) -> List[Dict[str, Any]]:
        """Get recently added documents for a topic."""
        from datetime import datetime, timedelta

        collection = self._get_or_create_collection(topic_id)
        cutoff = (datetime.utcnow() - timedelta(hours=hours)).isoformat()

        # Get all documents and filter by date
        # Note: ChromaDB doesn't support date range queries well, so we fetch and filter
        results = collection.get(
            include=["documents", "metadatas"],
            limit=limit * 2
        )

        recent = []
        if results["ids"]:
            for i, doc_id in enumerate(results["ids"]):
                metadata = results["metadatas"][i] if results["metadatas"] else {}
                fetched_at = metadata.get("fetched_at", "")

                if fetched_at >= cutoff:
                    recent.append({
                        "id": doc_id,
                        "content": results["documents"][i] if results["documents"] else "",
                        "metadata": metadata
                    })

                if len(recent) >= limit:
                    break

        return recent

    def delete_topic(self, topic_id: str) -> bool:
        """Delete all documents for a topic."""
        try:
            self.client.delete_collection(self._get_collection_name(topic_id))
            logger.info(f"Deleted collection for topic {topic_id}")
            return True
        except Exception as e:
            logger.error(f"Error deleting collection: {e}")
            return False

    def get_document_count(self, topic_id: str) -> int:
        """Get the number of documents in a topic's collection."""
        try:
            collection = self._get_or_create_collection(topic_id)
            return collection.count()
        except Exception:
            return 0
```

### Step 5: Multi-Document Synthesis with LLM

```python
# rag/synthesizer.py
from typing import List, Dict, Any, Optional
from langchain_openai import ChatOpenAI
from langchain_core.messages import SystemMessage, HumanMessage
import logging
import re

from config import get_settings

logger = logging.getLogger(__name__)

SYNTHESIS_SYSTEM_PROMPT = """You are a professional research analyst creating executive briefings.

Your task is to synthesize information from multiple sources into a coherent, well-structured summary.

Guidelines:
1. Focus on the most important and actionable information
2. Identify key themes and trends across sources
3. Note any contradictions or disagreements between sources
4. Always cite sources using [1], [2], etc. format
5. Be concise but comprehensive
6. Highlight implications and potential impacts
7. Use professional, objective language

Output Format:
- Start with a 2-3 sentence executive summary
- Organize findings into logical sections
- End with 3-5 key takeaways as bullet points
- Every factual claim must have a citation"""

class Synthesizer:
    """LLM-powered multi-document synthesizer."""

    def __init__(self):
        self.settings = get_settings()
        self.llm = ChatOpenAI(
            model=self.settings.openai_model,
            temperature=0.3,
            max_tokens=self.settings.synthesis_max_tokens,
            openai_api_key=self.settings.openai_api_key
        )

    async def synthesize(
        self,
        query: str,
        documents: List[Dict[str, Any]],
        topic_name: str
    ) -> Dict[str, Any]:
        """
        Synthesize multiple documents into a coherent summary.

        Args:
            query: The synthesis query/focus
            documents: Retrieved documents with metadata
            topic_name: Name of the research topic

        Returns:
            Dictionary with synthesis results and citations
        """
        if not documents:
            return {
                "summary": "No relevant documents found for this period.",
                "sections": [],
                "key_takeaways": [],
                "citations": []
            }

        # Build source context with numbered citations
        source_context, citations = self._build_source_context(documents)

        # Create synthesis prompt
        prompt = f"""Topic: {topic_name}
Focus: {query}

Sources:
{source_context}

Please synthesize these sources into a comprehensive briefing following the guidelines."""

        try:
            response = await self.llm.ainvoke([
                SystemMessage(content=SYNTHESIS_SYSTEM_PROMPT),
                HumanMessage(content=prompt)
            ])

            content = response.content

            # Parse the response into sections
            sections = self._parse_sections(content)
            key_takeaways = self._extract_takeaways(content)
            executive_summary = self._extract_executive_summary(content)

            # Validate citations used
            used_citations = self._extract_used_citations(content, citations)

            return {
                "summary": executive_summary,
                "full_content": content,
                "sections": sections,
                "key_takeaways": key_takeaways,
                "citations": used_citations
            }

        except Exception as e:
            logger.error(f"Synthesis failed: {e}")
            raise

    def _build_source_context(
        self,
        documents: List[Dict[str, Any]]
    ) -> tuple[str, List[Dict[str, Any]]]:
        """Build numbered source context for the LLM."""
        context_parts = []
        citations = []

        for i, doc in enumerate(documents, 1):
            metadata = doc.get("metadata", {})
            content = doc.get("content", "")

            # Truncate content if too long
            if len(content) > 2000:
                content = content[:2000] + "..."

            context_parts.append(f"""[{i}] {metadata.get('title', 'Untitled')}
Source: {metadata.get('domain', 'Unknown')}
URL: {metadata.get('url', '')}
Credibility: {metadata.get('credibility_score', 0.5):.2f}

{content}
---""")

            citations.append({
                "id": i,
                "title": metadata.get("title", "Untitled"),
                "url": metadata.get("url", ""),
                "domain": metadata.get("domain", "Unknown"),
                "credibility_score": metadata.get("credibility_score", 0.5)
            })

        return "\n\n".join(context_parts), citations

    def _parse_sections(self, content: str) -> List[Dict[str, Any]]:
        """Parse content into sections based on headers."""
        sections = []

        # Split by markdown headers
        lines = content.split("\n")
        current_section = None
        current_content = []

        for line in lines:
            if line.startswith("## ") or line.startswith("### "):
                # Save previous section
                if current_section:
                    sections.append({
                        "title": current_section,
                        "content": "\n".join(current_content).strip(),
                        "citation_ids": self._find_citations_in_text("\n".join(current_content))
                    })

                current_section = line.lstrip("#").strip()
                current_content = []
            elif current_section:
                current_content.append(line)

        # Save last section
        if current_section:
            sections.append({
                "title": current_section,
                "content": "\n".join(current_content).strip(),
                "citation_ids": self._find_citations_in_text("\n".join(current_content))
            })

        return sections

    def _extract_takeaways(self, content: str) -> List[str]:
        """Extract key takeaways from the content."""
        takeaways = []

        # Look for takeaways section
        takeaway_patterns = [
            r"(?:Key Takeaways|Takeaways|Key Points|Summary Points)[\s:]*\n((?:[-•*]\s*.+\n?)+)",
            r"(?:^|\n)((?:[-•*]\s*.+\n?){3,})"  # Fallback: any bullet list with 3+ items
        ]

        for pattern in takeaway_patterns:
            match = re.search(pattern, content, re.IGNORECASE | re.MULTILINE)
            if match:
                bullet_text = match.group(1)
                items = re.findall(r"[-•*]\s*(.+)", bullet_text)
                takeaways = [item.strip() for item in items[:5]]
                break

        return takeaways

    def _extract_executive_summary(self, content: str) -> str:
        """Extract the executive summary from content."""
        # Take first paragraph or first 3 sentences
        paragraphs = content.split("\n\n")

        for para in paragraphs:
            para = para.strip()
            # Skip headers and short lines
            if para and not para.startswith("#") and len(para) > 50:
                # Limit to ~3 sentences
                sentences = re.split(r'(?<=[.!?])\s+', para)
                return " ".join(sentences[:3])

        return content[:500] if content else ""

    def _find_citations_in_text(self, text: str) -> List[int]:
        """Find citation numbers used in text."""
        citations = re.findall(r'\[(\d+)\]', text)
        return list(set(int(c) for c in citations))

    def _extract_used_citations(
        self,
        content: str,
        all_citations: List[Dict[str, Any]]
    ) -> List[Dict[str, Any]]:
        """Extract only the citations that were actually used."""
        used_ids = set(self._find_citations_in_text(content))
        return [c for c in all_citations if c["id"] in used_ids]
```

```python
from typing import Optional
from datetime import datetime, timedelta
import logging

from models import Briefing, BriefingSection, Citation, ResearchTopic
from rag.vector_store import VectorStore
from rag.synthesizer import Synthesizer
from config import get_settings

logger = logging.getLogger(__name__)

class BriefingGenerator:
    """Generate research briefings from stored documents."""

    def __init__(self):
        self.settings = get_settings()
        self.vector_store = VectorStore()
        self.synthesizer = Synthesizer()

    async def generate_briefing(
        self,
        topic: ResearchTopic,
        hours: int = 24
    ) -> Briefing:
        """
        Generate a briefing for a topic.

        Args:
            topic: The research topic
            hours: Number of hours to look back

        Returns:
            Generated Briefing object
        """
        period_end = datetime.utcnow()
        period_start = period_end - timedelta(hours=hours)

        # Retrieve relevant documents
        query = f"Latest developments and news about {topic.name}: {', '.join(topic.keywords)}"

        documents = await self.vector_store.search(
            topic_id=topic.id,
            query=query,
            top_k=self.settings.retrieval_top_k,
            min_credibility=0.4
        )

        # Filter to recent documents
        recent_docs = [
            doc for doc in documents
            if self._is_recent(doc, hours)
        ]

        if not recent_docs:
            # Fall back to all retrieved docs if no recent ones
            recent_docs = documents[:self.settings.retrieval_top_k]

        # Synthesize content
        synthesis = await self.synthesizer.synthesize(
            query=f"Summarize the latest developments in {topic.name}",
            documents=recent_docs,
            topic_name=topic.name
        )

        # Build briefing sections
        sections = [
            BriefingSection(
                title=s["title"],
                content=s["content"],
                citation_ids=s.get("citation_ids", [])
            )
            for s in synthesis.get("sections", [])
        ]

        # Build citations
        citations = [
            Citation(
                id=c["id"],
                title=c["title"],
                url=c["url"],
                domain=c["domain"],
                credibility_score=c["credibility_score"]
            )
            for c in synthesis.get("citations", [])
        ]

        return Briefing(
            topic_id=topic.id,
            topic_name=topic.name,
            period_start=period_start,
            period_end=period_end,
            executive_summary=synthesis.get("summary", ""),
            sections=sections,
            key_takeaways=synthesis.get("key_takeaways", []),
            citations=citations,
            sources_analyzed=len(recent_docs)
        )

    def _is_recent(self, doc: dict, hours: int) -> bool:
        """Check if a document is within the time window."""
        metadata = doc.get("metadata", {})
        fetched_at = metadata.get("fetched_at", "")

        if not fetched_at:
            return True  # Include if no date

        try:
            fetched = datetime.fromisoformat(fetched_at)
            cutoff = datetime.utcnow() - timedelta(hours=hours)
            return fetched >= cutoff
        except Exception:
            return True

    def format_briefing_markdown(self, briefing: Briefing) -> str:
        """Format briefing as Markdown."""
        lines = []

        # Header
        lines.append(f"# Research Briefing: {briefing.topic_name}")
        lines.append("")
        lines.append(f"**Generated:** {briefing.generated_at.strftime('%Y-%m-%d %H:%M UTC')}")
        lines.append(f"**Period:** {briefing.period_start.strftime('%Y-%m-%d')} to {briefing.period_end.strftime('%Y-%m-%d')}")
        lines.append(f"**Sources Analyzed:** {briefing.sources_analyzed}")
        lines.append("")

        # Executive Summary
        lines.append("## Executive Summary")
        lines.append("")
        lines.append(briefing.executive_summary)
        lines.append("")

        # Sections
        for section in briefing.sections:
            lines.append(f"## {section.title}")
            lines.append("")
            lines.append(section.content)
            lines.append("")

        # Key Takeaways
        if briefing.key_takeaways:
            lines.append("## Key Takeaways")
            lines.append("")
            for takeaway in briefing.key_takeaways:
                lines.append(f"- {takeaway}")
            lines.append("")

        # Citations
        if briefing.citations:
            lines.append("## Sources")
            lines.append("")
            for citation in sorted(briefing.citations, key=lambda c: c.id):
                credibility = "⭐" * int(citation.credibility_score * 5)
                lines.append(f"[{citation.id}] [{citation.title}]({citation.url}) - {citation.domain} {credibility}")
            lines.append("")

        return "\n".join(lines)

    def format_briefing_html(self, briefing: Briefing) -> str:
        """Format briefing as HTML for email."""
        import html

        def escape(text: str) -> str:
            return html.escape(text)

        parts = []

        # Header
        parts.append(f"""
<div style="font-family: Arial, sans-serif; max-width: 800px; margin: 0 auto; padding: 20px;">
    <h1 style="color: #1a1a1a; border-bottom: 2px solid #0066cc; padding-bottom: 10px;">
        Research Briefing: {escape(briefing.topic_name)}
    </h1>
    <p style="color: #666; font-size: 14px;">
        <strong>Generated:</strong> {briefing.generated_at.strftime('%Y-%m-%d %H:%M UTC')} |
        <strong>Sources:</strong> {briefing.sources_analyzed}
    </p>
""")

        # Executive Summary
        parts.append(f"""
    <div style="background: #f5f5f5; padding: 15px; border-radius: 5px; margin: 20px 0;">
        <h2 style="color: #333; margin-top: 0;">Executive Summary</h2>
        <p style="color: #444; line-height: 1.6;">{escape(briefing.executive_summary)}</p>
    </div>
""")

        # Sections
        for section in briefing.sections:
            content_html = escape(section.content).replace('\n', '<br>')
            parts.append(f"""
    <div style="margin: 20px 0;">
        <h2 style="color: #333;">{escape(section.title)}</h2>
        <p style="color: #444; line-height: 1.6;">{content_html}</p>
    </div>
""")

        # Key Takeaways
        if briefing.key_takeaways:
            takeaways_html = "".join(
                f"<li style='margin: 8px 0;'>{escape(t)}</li>"
                for t in briefing.key_takeaways
            )
            parts.append(f"""

        <ul style="color: #444; line-height: 1.6;">{takeaways_html}</ul>
    </div>
""")

        # Citations
        if briefing.citations:
            citations_html = ""
            for c in sorted(briefing.citations, key=lambda x: x.id):
                stars = "★" * int(c.credibility_score * 5) + "☆" * (5 - int(c.credibility_score * 5))
                citations_html += f"""
                <li style="margin: 5px 0;">
                    [{c.id}] <a href="{escape(c.url)}" style="color: #0066cc;">{escape(c.title)}</a>

        <ul style="font-size: 14px; color: #444;">{citations_html}</ul>
```

```python
from email.mime.text import MIMEText

import logging

class EmailSender:
    """Send briefings via SMTP email."""

    def __init__(self):
        self.settings = get_settings()
        self.briefing_generator = BriefingGenerator()

    async def send_briefing(
        self,
        briefing: Briefing,
        to_email: str,
        subject: Optional[str] = None
    ) -> DeliveryStatus:
        """
        Send a briefing via email.

        Args:
            briefing: The briefing to send
            to_email: Recipient email address
            subject: Optional custom subject line

        Returns:
            DeliveryStatus with result
        """
        status = DeliveryStatus()

        if not self.settings.smtp_host:
            status.email = "not_configured"
            return status

        try:
            # Create message
            msg = MIMEMultipart("alternative")
            msg["Subject"] = subject or f"Research Briefing: {briefing.topic_name}"
            msg["From"] = self.settings.smtp_from_email
            msg["To"] = to_email

            # Plain text version
            text_content = self.briefing_generator.format_briefing_markdown(briefing)
            msg.attach(MIMEText(text_content, "plain"))

            # HTML version
            html_content = self.briefing_generator.format_briefing_html(briefing)
            msg.attach(MIMEText(html_content, "html"))

            # Send email
            with smtplib.SMTP(self.settings.smtp_host, self.settings.smtp_port) as server:
                server.starttls()
                if self.settings.smtp_username and self.settings.smtp_password:
                    server.login(self.settings.smtp_username, self.settings.smtp_password)
                server.send_message(msg)

            status.email = "sent"
            logger.info(f"Email sent to {to_email} for briefing {briefing.id}")

        except smtplib.SMTPException as e:
            status.email = "failed"
            status.email_error = str(e)
            logger.error(f"SMTP error sending to {to_email}: {e}")
        except Exception as e:
            status.email = "failed"
            status.email_error = str(e)
            logger.error(f"Error sending email to {to_email}: {e}")

        return status
```

```python
import logging
```

```python
class ResearchScheduler:
    """Schedule research monitoring and briefing tasks."""

    def __init__(self):
        self.settings = get_settings()
        self.scheduler = AsyncIOScheduler(timezone=self.settings.timezone)
        self._topic_jobs: Dict[str, str] = {}  # topic_id -> job_id

        """Start the scheduler."""
        if not self.scheduler.running:
            self.scheduler.start()
            logger.info("Research scheduler started")

    def shutdown(self):
        """Shutdown the scheduler."""
        if self.scheduler.running:
            self.scheduler.shutdown()
            logger.info("Research scheduler stopped")

    def schedule_topic(self, topic: ResearchTopic) -> Optional[str]:
        """
        Schedule monitoring and briefing for a topic.

        Args:
            topic: The research topic to schedule

        Returns:
            Job ID if scheduled successfully
        """
        from scheduling.tasks import run_topic_monitoring, run_topic_briefing

        # Remove existing job if any
        self.unschedule_topic(topic.id)

        # Parse schedule time
        hour, minute = map(int, topic.schedule.time.split(":"))

        # Create trigger based on frequency
        if topic.schedule.frequency == BriefingFrequency.DAILY:
            trigger = CronTrigger(hour=hour, minute=minute, timezone=topic.schedule.timezone)
        else:  # Weekly
            trigger = CronTrigger(
                day_of_week="mon",
                hour=hour,
                minute=minute,
                timezone=topic.schedule.timezone
            )

        # Schedule briefing job
        job = self.scheduler.add_job(
            run_topic_briefing,
            trigger=trigger,
            args=[topic.id],
            id=f"briefing_{topic.id}",
            name=f"Briefing: {topic.name}",
            replace_existing=True
        )

        # Schedule monitoring job (every 2 hours)
        self.scheduler.add_job(
            run_topic_monitoring,
            trigger=CronTrigger(hour="*/2", timezone=topic.schedule.timezone),
            args=[topic.id],
            id=f"monitor_{topic.id}",
            name=f"Monitor: {topic.name}",
            replace_existing=True
        )

        self._topic_jobs[topic.id] = job.id
        logger.info(f"Scheduled topic {topic.name} for {topic.schedule.frequency.value} briefings at {topic.schedule.time}")

        return job.id

    def unschedule_topic(self, topic_id: str):
        """Remove scheduled jobs for a topic."""
        for prefix in ["briefing_", "monitor_"]:
            job_id = f"{prefix}{topic_id}"
            try:
                self.scheduler.remove_job(job_id)
            except Exception:
                pass  # Job might not exist

        if topic_id in self._topic_jobs:
            del self._topic_jobs[topic_id]

    def get_next_run(self, topic_id: str) -> Optional[datetime]:
        """Get the next scheduled run time for a topic."""
        job_id = f"briefing_{topic_id}"
        job = self.scheduler.get_job(job_id)
        return job.next_run_time if job else None

    def list_jobs(self) -> list:
        """List all scheduled jobs."""
        return [
            {
                "id": job.id,
                "name": job.name,
                "next_run": job.next_run_time.isoformat() if job.next_run_time else None
            }
            for job in self.scheduler.get_jobs()
        ]
```

```python
# scheduling/tasks.py
import asyncio
import logging
from typing import Optional

from models import ResearchTopic
from acquisition.rss_parser import RSSParser
from acquisition.web_scraper import WebScraper
from analysis.credibility_scorer import CredibilityScorer
from rag.vector_store import VectorStore
from briefing.generator import BriefingGenerator
from delivery.email_sender import EmailSender
from delivery.slack_sender import SlackSender

logger = logging.getLogger(__name__)

# In-memory topic storage (use database in production)
_topics: dict[str, ResearchTopic] = {}

def get_topic(topic_id: str) -> Optional[ResearchTopic]:
    """Get a topic by ID."""
    return _topics.get(topic_id)

def save_topic(topic: ResearchTopic):
    """Save a topic."""
    _topics[topic.id] = topic

async def run_topic_monitoring(topic_id: str):
    """
    Run content monitoring for a topic.

    Fetches new content from RSS feeds and websites,
    scores credibility, and stores in vector database.
    """
    topic = get_topic(topic_id)
    if not topic:
        logger.error(f"Topic not found: {topic_id}")
        return

    logger.info(f"Starting monitoring for topic: {topic.name}")

    rss_parser = RSSParser()
    web_scraper = WebScraper()
    credibility_scorer = CredibilityScorer()
    vector_store = VectorStore()

    all_documents = []

    # Fetch from RSS feeds
    for feed_url in topic.rss_feeds:
        try:
            docs = await rss_parser.parse_feed(feed_url, topic.id)
            all_documents.extend(docs)
        except Exception as e:
            logger.error(f"Error parsing RSS feed {feed_url}: {e}")

    # Fetch from websites
    for website_url in topic.websites:
        try:
            doc = await web_scraper.scrape_url(website_url, topic.id)
            if doc:
                all_documents.append(doc)
        except Exception as e:
            logger.error(f"Error scraping {website_url}: {e}")

    if not all_documents:
        logger.info(f"No new documents found for topic: {topic.name}")
        return

    # Score credibility
    for doc in all_documents:
        doc.credibility = credibility_scorer.score_document(doc, all_documents)

    # Filter low-credibility documents
    credible_docs = [
        doc for doc in all_documents
        if doc.credibility.overall_score >= 0.4
    ]

    logger.info(f"Filtered {len(all_documents)} -> {len(credible_docs)} credible documents")

    # Store in vector database
    if credible_docs:
        await vector_store.add_documents(credible_docs)

    logger.info(f"Monitoring complete for topic: {topic.name}")

async def run_topic_briefing(topic_id: str):
    """
    Generate and deliver a briefing for a topic.
    """
    topic = get_topic(topic_id)
    if not topic:
        logger.error(f"Topic not found: {topic_id}")
        return

    logger.info(f"Generating briefing for topic: {topic.name}")

    briefing_generator = BriefingGenerator()
    email_sender = EmailSender()
    slack_sender = SlackSender()

    # Determine time window based on frequency
    hours = 24 if topic.schedule.frequency == "daily" else 168  # 7 days

    # Generate briefing
    try:
        briefing = await briefing_generator.generate_briefing(topic, hours=hours)
    except Exception as e:
        logger.error(f"Error generating briefing for {topic.name}: {e}")
        return

    # Deliver via configured channels
    if topic.delivery.email:
        await email_sender.send_briefing(briefing, topic.delivery.email)

    if topic.delivery.slack_webhook:
        await slack_sender.send_briefing(briefing, topic.delivery.slack_webhook)

    logger.info(f"Briefing delivered for topic: {topic.name}")
```

```python
    websites: List[str] = []
    schedule: Optional[ScheduleConfig] = None
    delivery: Optional[DeliveryConfig] = None

class TopicResponse(BaseModel):
    id: str
    name: str
    status: str
    documents_count: int
    last_updated: Optional[str] = None

class TopicListResponse(BaseModel):
    topics: List[TopicResponse]

class BriefingResponse(BaseModel):
    briefing_id: str
    topic: str
    generated_at: str
    summary: str
    sections: List[dict]
    key_takeaways: List[str]
    sources_analyzed: int
    citations: List[dict]

# In-memory briefing storage
_briefings: dict[str, Briefing] = {}

# API Endpoints

@app.get("/health")
async def health_check():
    """Health check endpoint."""
    return {"status": "healthy", "service": "research-assistant"}

@app.post("/topics", response_model=TopicResponse, status_code=201)
async def create_topic(topic_data: TopicCreate, background_tasks: BackgroundTasks):
    """Create a new research topic."""
    topic = ResearchTopic(
        name=topic_data.name,
        description=topic_data.description,
        keywords=topic_data.keywords,
        rss_feeds=topic_data.rss_feeds,
        websites=topic_data.websites,
        schedule=topic_data.schedule or ScheduleConfig(),
        delivery=topic_data.delivery or DeliveryConfig()
    )

    save_topic(topic)

    # Schedule monitoring and briefings
    scheduler.schedule_topic(topic)

    # Run initial monitoring in background
    background_tasks.add_task(run_topic_monitoring, topic.id)

    logger.info(f"Created topic: {topic.name} ({topic.id})")

    return TopicResponse(
        id=topic.id,
        name=topic.name,
        status=topic.status.value,
        documents_count=0
    )

@app.get("/topics", response_model=TopicListResponse)
async def list_topics():
    """List all research topics."""
    from scheduling.tasks import _topics

    vector_store = VectorStore()
    topics = []

    for topic in _topics.values():
        doc_count = vector_store.get_document_count(topic.id)
        next_run = scheduler.get_next_run(topic.id)

        topics.append(TopicResponse(
            id=topic.id,
            name=topic.name,
            status=topic.status.value,
            documents_count=doc_count,
            last_updated=next_run.isoformat() if next_run else None
        ))

    return TopicListResponse(topics=topics)

@app.get("/topics/{topic_id}")
async def get_topic_details(topic_id: str):
    """Get details for a specific topic."""
    topic = get_topic(topic_id)
    if not topic:
        raise HTTPException(status_code=404, detail="Topic not found")

    vector_store = VectorStore()
    doc_count = vector_store.get_document_count(topic_id)
    next_run = scheduler.get_next_run(topic_id)

    return {
        "id": topic.id,
        "name": topic.name,
        "description": topic.description,
        "keywords": topic.keywords,
        "rss_feeds": topic.rss_feeds,
        "websites": topic.websites,
        "schedule": topic.schedule.model_dump(),
        "delivery": {
            "email": topic.delivery.email,
            "slack_webhook": "configured" if topic.delivery.slack_webhook else None
        },
        "status": topic.status.value,
        "documents_count": doc_count,
        "next_briefing": next_run.isoformat() if next_run else None
    }

@app.delete("/topics/{topic_id}")
async def delete_topic(topic_id: str):
    """Delete a research topic."""
    from scheduling.tasks import _topics

    topic = get_topic(topic_id)
    if not topic:
        raise HTTPException(status_code=404, detail="Topic not found")

    # Unschedule jobs
    scheduler.unschedule_topic(topic_id)

    # Delete vector store collection
    vector_store = VectorStore()
    vector_store.delete_topic(topic_id)

    # Remove from storage
    del _topics[topic_id]

    logger.info(f"Deleted topic: {topic.name} ({topic_id})")

    return {"status": "deleted", "topic_id": topic_id}

@app.post("/topics/{topic_id}/monitor")
async def trigger_monitoring(topic_id: str, background_tasks: BackgroundTasks):
    """Manually trigger monitoring for a topic."""
    topic = get_topic(topic_id)
    if not topic:
        raise HTTPException(status_code=404, detail="Topic not found")

    background_tasks.add_task(run_topic_monitoring, topic_id)

    return {"status": "monitoring_started", "topic_id": topic_id}

@app.post("/topics/{topic_id}/briefing", response_model=BriefingResponse)
async def generate_briefing(topic_id: str, hours: int = 24):
    """Generate a briefing on-demand."""
    topic = get_topic(topic_id)
    if not topic:
        raise HTTPException(status_code=404, detail="Topic not found")

    briefing_generator = BriefingGenerator()

    try:
        briefing = await briefing_generator.generate_briefing(topic, hours=hours)
    except Exception as e:
        logger.error(f"Briefing generation failed: {e}")
        raise HTTPException(status_code=500, detail=f"Briefing generation failed: {str(e)}")

    # Store briefing
    _briefings[briefing.id] = briefing

    return BriefingResponse(
        briefing_id=briefing.id,
        topic=briefing.topic_name,
        generated_at=briefing.generated_at.isoformat(),
        summary=briefing.executive_summary,
        sections=[s.model_dump() for s in briefing.sections],
        key_takeaways=briefing.key_takeaways,
        sources_analyzed=briefing.sources_analyzed,
        citations=[c.model_dump() for c in briefing.citations]
    )

@app.get("/briefings/{briefing_id}")
async def get_briefing(briefing_id: str, format: str = "json"):
    """Get a generated briefing."""
    briefing = _briefings.get(briefing_id)
    if not briefing:
        raise HTTPException(status_code=404, detail="Briefing not found")

    if format == "markdown":
        generator = BriefingGenerator()
        return {"content": generator.format_briefing_markdown(briefing)}
    elif format == "html":
        generator = BriefingGenerator()
        return {"content": generator.format_briefing_html(briefing)}
    else:
        return {
            "briefing_id": briefing.id,
            "topic": briefing.topic_name,
            "generated_at": briefing.generated_at.isoformat(),
            "period_start": briefing.period_start.isoformat(),
            "period_end": briefing.period_end.isoformat(),
            "executive_summary": briefing.executive_summary,
            "sections": [s.model_dump() for s in briefing.sections],
            "key_takeaways": briefing.key_takeaways,
            "citations": [c.model_dump() for c in briefing.citations],
            "sources_analyzed": briefing.sources_analyzed
        }

@app.get("/scheduler/jobs")
async def list_scheduled_jobs():
    """List all scheduled jobs."""
    return {"jobs": scheduler.list_jobs()}
```

```python
        """Old content should have lower publication frequency score."""

        """Content with quality indicators should score higher."""
        sample_document.content = """
        According to research published by Harvard University, Professor Smith
        found that the data indicates significant improvements. The study from 2024
        references https://example.com/source for verification.
        """ * 10
        score = scorer.score_document(sample_document)
        assert score.content_quality >= 0.6

    def test_low_quality_indicators_reduce_score(self, scorer, sample_document):
        """Content with low-quality indicators should score lower."""
        sample_document.content = """
        BREAKING!!! YOU WON'T BELIEVE what happened!! Click here to subscribe now!!
        Limited time offer!!! SHOCKING news!!!
        """
        score = scorer.score_document(sample_document)
        assert score.content_quality <= 0.4

                content="According to research, this study finds important results. " * 20,
                source_type=SourceType.RSS,
                domain=f"example{i}.com"
            )
            for i in range(3)
```

## Summary

In this project, you built a comprehensive personal research assistant that:

1. **Monitors Multiple Sources**: Parses RSS feeds and scrapes websites with respectful rate limiting and robots.txt compliance.

2. **Scores Source Credibility**: Evaluates sources based on domain authority, publication patterns, cross-references, and content quality.

3. **Stores and Retrieves with RAG**: Uses ChromaDB for vector storage and retrieval-augmented generation for intelligent document retrieval.

4. **Synthesizes Multi-Document Summaries**: Leverages LLMs to create coherent summaries from multiple sources with proper citation tracking.

5. **Generates Professional Briefings**: Creates executive-style briefings with key takeaways, structured sections, and verified citations.

6. **Delivers Automatically**: Sends briefings via email (SMTP) and Slack webhooks on configurable schedules.

7. **Runs on Schedule**: Uses APScheduler for automated monitoring and briefing generation.
